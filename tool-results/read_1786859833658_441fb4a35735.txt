    31→/** Calculate stay duration in days (checkout - checkin). Uses safeDate to avoid UTC drift. */
    32→const calcDias = (checkin: string, checkout: string): number => {
    33→  const c = safeDate(checkin);
    34→  const o = safeDate(checkout);
    35→  const diff = Math.round((o.getTime() - c.getTime()) / (1000 * 60 * 60 * 24));
    36→  return diff > 0 ? diff : 0;
    37→};
    38→
    39→/** Get initials from a full name (first letter of first word + first letter of last word) */
    40→const getInitials = (name: string): string => {
    41→  const parts = name.trim().split(/\s+/).filter(Boolean);
    42→  if (parts.length === 0) return '?';
    43→  if (parts.length === 1) return parts[0][0].toUpperCase();
    44→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
    45→};
    46→
    47→/** Loyalty level based on stay count */
    48→type LoyaltyLevel = 'Nuevo' | 'Habitual' | 'Frecuente' | 'VIP';
    49→
    50→interface LoyaltyInfo {
    51→  level: LoyaltyLevel;
    52→  color: string;       // badge bg class
    53→  textColor: string;   // badge text class
    54→  borderColor: string; // left border color
    55→  icon?: React.ReactNode;
    56→}
    57→
    58→const getLoyaltyInfo = (stayCount: number): LoyaltyInfo => {
    59→  if (stayCount >= 7) return {
    60→    level: 'VIP',
    61→    color: 'bg-primary/10',
    62→    textColor: 'text-primary',
    63→    borderColor: 'border-l-primary',
    64→    icon: <Star className="w-3 h-3 fill-primary text-primary" />,
    65→  };
    66→  if (stayCount >= 4) return {
    67→    level: 'Frecuente',
    68→    color: 'bg-warning/15',
    69→    textColor: 'text-warning',
    70→    borderColor: 'border-l-amber-400',
    71→  };
    72→  if (stayCount >= 2) return {
    73→    level: 'Habitual',
    74→    color: 'bg-sky-100',
    75→    textColor: 'text-sky-700',
    76→    borderColor: 'border-l-sky-400',
    77→  };
    78→  return {
    79→    level: 'Nuevo',
    80→    color: 'bg-muted',
    81→    textColor: 'text-muted-foreground',
    82→    borderColor: 'border-l-gray-300',
    83→  };
    84→};
    85→
    86→/** Format a date as relative time in Spanish */
    87→const formatRelativeTime = (dateStr: string): string => {
    88→  if (!dateStr) return '';
    89→  const d = safeDate(dateStr);
    90→  const now = new Date();
    91→  const diffMs = now.getTime() - d.getTime();
    92→  const diffDays = Math.floor(diffMs / (1000 * 60 * 60 * 24));
    93→
    94→  if (diffDays < 0) return 'próximo';
    95→  if (diffDays === 0) return 'hoy';
    96→  if (diffDays === 1) return 'ayer';
    97→  if (diffDays < 7) return `hace ${diffDays} días`;
    98→  if (diffDays < 14) return 'hace 1 semana';
    99→  if (diffDays < 30) return `hace ${Math.floor(diffDays / 7)} semanas`;
   100→  if (diffDays < 60) return 'hace 1 mes';
   101→  if (diffDays < 365) return `hace ${Math.floor(diffDays / 30)} meses`;
   102→  if (diffDays < 730) return 'hace 1 año';
   103→  return `hace ${Math.floor(diffDays / 365)} años`;
   104→};
