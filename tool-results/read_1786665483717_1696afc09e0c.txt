   105→  // Etiqueta legible del rango: "1 persona", "2 personas", "3+ personas", "2-4 personas"

   107→    ? `${rango.minPersonas}+ persona${rango.minPersonas > 1 ? 's' : ''}`
   108→    : rango.minPersonas === rango.maxPersonas

   149→function formatoRango(r: RangoPrecio): string {
   150→  if (r.maxPersonas === null) return `${r.minPersonas}+`;
   151→  if (r.minPersonas === r.maxPersonas) return `${r.minPersonas}`;
   152→  return `${r.minPersonas}-${r.maxPersonas}`;
   153→}

   195→// Count active promotions on a tariff
   196→function countPromos(promos?: PromocionesTarifa | null): number {
   197→  if (!promos) return 0;
   198→  let n = 0;
   199→  if (promos.acompananteSinCargo?.activo) n++;
   200→  if (promos.ninosDiferenciado?.activo) n++;
   201→  if (promos.nochesCortesia?.activo) n++;
   202→  return n;
   203→}

   205→// Compute "Desde" price (min positive price, fallback to first range)
   206→function precioDesde(rangos: RangoPrecio[]): number {
   207→  const preciosPositivos = rangos.map(r => r.precio).filter(p => p > 0);
   208→  return preciosPositivos.length > 0 ? Math.min(...preciosPositivos) : (rangos[0]?.precio || 0);
   209→}

   211→// Build a label describing a noches cortesía modalidad
   212→function describeNochesCortesia(mod: ModalidadNochesCortesia): string {
   213→  if (mod.tipo === 'cadaX') return `Cada ${mod.cada} noches, 1 gratis`;
   214→  if (mod.tipo === 'aPartirDe') return `Desde ${mod.minNoches} noches, ${mod.nochesGratis} gratis`;
   215→  const dias = ['Domingo', 'Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado'];
   216→  return `${dias[mod.dia]} gratis`;
   217→}

   219→// Export a single tariff as CSV (client-side download)
   220→function exportTariffCSV(tipo: string, t: TarifaPrecios) {
   221→  const rows: (string | number)[][] = [];
   222→  rows.push(['Campo', 'Valor']);
   223→  rows.push(['Nombre', tipo]);
   224→  rows.push(['Modo de cobro', t.modoCobro || 'porGrupo']);
   225→  const rangos = t.rangos || [];
   226→  rows.push(['Cantidad de rangos', rangos.length]);
   227→  rangos.forEach((r, i) => {
   228→    rows.push([`Rango ${i + 1} (${formatoRango(r)} pers.)`, r.precio]);
   229→  });
   230→  rows.push(['Acompañante sin cargo', t.promociones?.acompananteSinCargo?.activo ? 'Sí' : 'No']);
   231→  if (t.promociones?.acompananteSinCargo?.activo) {
   232→    rows.push(['  Etiqueta', t.promociones.acompananteSinCargo.etiqueta || '—']);
   233→    rows.push(['  Cantidad', t.promociones.acompananteSinCargo.cantidad ?? 1]);
   234→  }
   235→  rows.push(['Niños diferenciado', t.promociones?.ninosDiferenciado?.activo ? 'Sí' : 'No']);
   236→  if (t.promociones?.ninosDiferenciado?.activo) {
   237→    rows.push(['  Precio niño/noche', t.promociones.ninosDiferenciado.precioNino || 0]);
   238→  }
   239→  rows.push(['Noches cortesía', t.promociones?.nochesCortesia?.activo ? 'Sí' : 'No']);
   240→  if (t.promociones?.nochesCortesia?.activo) {
   241→    rows.push(['  Modalidad', describeNochesCortesia(t.promociones.nochesCortesia.modalidad)]);
   242→  }
   243→  const campos = t.camposPersonalizados || [];
   244→  rows.push(['Cantidad de campos personalizados', campos.length]);
   245→  campos.forEach((c, i) => {
   246→    rows.push([`  Campo ${i + 1}`, `${c.nombre} (${c.tipo}${c.requerido ? ', requerido' : ''})`]);
   247→  });
   248→
   249→  const csv = rows.map(r => r.map(cell => {
   250→    const s = String(cell);
   251→    return s.includes(',') || s.includes('"') ? `"${s.replace(/"/g, '""')}"` : s;
   252→  }).join(',')).join('\n');
   253→
   254→  const blob = new Blob([`\uFEFF${csv}`], { type: 'text/csv;charset=utf-8;' });
   255→  const url = URL.createObjectURL(blob);
   256→  const a = document.createElement('a');
   257→  a.href = url;
   258→  a.download = `tarifa_${tipo.toLowerCase().replace(/\s+/g, '_')}.csv`;
   259→  document.body.appendChild(a);
   260→  a.click();
   261→  document.body.removeChild(a);
   262→  URL.revokeObjectURL(url);
   263→}

   267→function WizardStepper({ current, onSelect }: { current: 1 | 2 | 3; onSelect: (s: 1 | 2 | 3) => void }) {
   268→  const steps: { n: 1 | 2 | 3; label: string; icon: typeof Info }[] = [
   269→    { n: 1, label: 'Información básica', icon: Info },
   270→    { n: 2, label: 'Rangos de precios', icon: Tags },
   271→    { n: 3, label: 'Promociones', icon: Sparkles },
   272→  ];
   273→  return (
   274→    <div className="space-y-2">
   275→      <div className="flex items-center gap-1">
   276→        {steps.map((s, i) => {
   277→          const Icon = s.icon;
   278→          const isDone = current > s.n;
   279→          const isActive = current === s.n;
   280→          return (
   281→            <div key={s.n} className="flex items-center flex-1 last:flex-none">
   282→              <button
   283→                type="button"
   284→                onClick={() => onSelect(s.n)}
   285→                className={`flex items-center gap-2 px-3 py-1.5 rounded-md text-xs font-semibold transition-all whitespace-nowrap ${
   286→                  isActive
   287→                    ? 'bg-[#0F2B28] text-white shadow-sm'
   288→                    : isDone
   289→                      ? 'bg-[#DCFCE7] text-[#166534] hover:bg-[#BBF7D0]'
   290→                      : 'bg-muted text-muted-foreground hover:bg-muted/80'
   291→                }`}
   292→                aria-current={isActive ? 'step' : undefined}
   293→              >
   294→                <span className={`w-5 h-5 rounded-full flex items-center justify-center text-[10px] ${
   295→                  isActive ? 'bg-white/20' : isDone ? 'bg-[#10B981] text-white' : 'bg-background'
   296→                }`}>
   297→                  {isDone ? <Check className="w-3 h-3" /> : s.n}
   298→                </span>
   299→                <span className="hidden sm:inline">{s.label}</span>
   300→                <Icon className="w-3.5 h-3.5 sm:hidden" />
   301→              </button>
   302→              {i < steps.length - 1 && (
   303→                <div className={`flex-1 h-0.5 mx-1 rounded ${isDone ? 'bg-[#10B981]' : 'bg-muted'}`} />
   304→              )}
   305→            </div>
   306→          );
   307→        })}
   308→      </div>
   309→      <Progress value={(current / 3) * 100} className="h-1 bg-muted" />
   310→    </div>
   311→  );
   312→}

   417→function ComparisonModal({ tariffs, onClose }: {
   418→  tariffs: { tipo: string; t: TarifaPrecios }[];
   419→  onClose: () => void;
   420→}) {
   421→  // Compute base price per person for each tariff (min positive price)
   422→  const basePrices = tariffs.map(({ t }) => {
   423→    const rangos = t.rangos || [];
   424→    const pos = rangos.map(r => r.precio).filter(p => p > 0);
   425→    return pos.length > 0 ? Math.min(...pos) : (rangos[0]?.precio || 0);
   426→  });
   427→
   428→  // Determine which rows have differing values for highlighting
