     1→'use client';
     2→
     3→import { useState, useMemo, useCallback } from 'react';
     4→import {
     5→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent, AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
     6→} from '@/components/ui/alert-dialog';
     7→import { useHotelStore } from '@/lib/store';
     8→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
     9→import { Button } from '@/components/ui/button';
    10→import { Input } from '@/components/ui/input';
    11→import { Label } from '@/components/ui/label';
    12→import { Badge } from '@/components/ui/badge';
    13→import {
    14→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter, DialogClose,
    15→} from '@/components/ui/dialog';
    16→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table';
    17→import {
    18→  Wallet, Lock, Unlock, Plus, Minus, Loader2, Pencil, Trash2, AlertTriangle, Tag,
    19→} from 'lucide-react';
    20→import ModuleHeader from '@/components/layout/ModuleHeader';
    21→import { toast } from 'sonner';
    22→import { DialogTrigger } from '@/components/ui/dialog';
    23→import { BILLETES } from '@/lib/types';
    24→
    25→const formatFechaHora = (f: string) => {
    26→  if (!f) return '—';
    27→  const d = new Date(f.length === 10 ? f + 'T12:00:00' : f);
    28→  return d.toLocaleDateString('es-AR') + ' ' + d.toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit' });
    29→};
    30→
    31→const formatHora = (f: string) => {
    32→  const d = new Date(f);
    33→  return d.toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit' });
    34→};
    35→
    36→const formatMoney = (n: number) => '$' + n.toLocaleString('es-AR', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    37→
    38→const METODOS = ['Efectivo', 'Transferencia', 'Tarjeta de Credito', 'Tarjeta de Debito', 'Mercado Pago', 'Otro'];
    39→
    40→export default function CajaModule() {
    41→  const {
    42→    caja, abrirCaja, registrarMovimientoCaja, cerrarCaja, saldoActualCaja,
    43→    editarMovimientoCaja, eliminarMovimientoCaja, usuarioActual,
    44→    metodosPago, categoriasGastos,
    45→  } = useHotelStore();
    46→
    47→  // Loading states
    48→  const [loadingAbrir, setLoadingAbrir] = useState(false);
    49→  const [loadingMovimiento, setLoadingMovimiento] = useState(false);
    50→  const [loadingCerrar, setLoadingCerrar] = useState(false);
    51→
    52→  // Form states
    53→  const [montoInicial, setMontoInicial] = useState('');
    54→  const [showApertura, setShowApertura] = useState(false);
    55→  const [showMovForm, setShowMovForm] = useState<'ingreso' | 'egreso' | null>(null);
    56→  const [movMonto, setMovMonto] = useState('');
    57→  const [movDesc, setMovDesc] = useState('');
    58→  const [movMetodo, setMovMetodo] = useState('Efectivo');
    59→  const [movCategoria, setMovCategoria] = useState('');
    60→
    61→  // Close dialog
    62→  const [showCierre, setShowCierre] = useState(false);
    63→  const [billetes, setBilletes] = useState<Record<number, number>>(() => Object.fromEntries(BILLETES.map(b => [b, 0])));
    64→
    65→  // Edit dialog
    66→  const [editingMov, setEditingMov] = useState<{ id: string; monto: number; descripcion: string } | null>(null);
    67→  const [editMonto, setEditMonto] = useState('');
    68→  const [editDesc, setEditDesc] = useState('');
    69→  const [loadingEdit, setLoadingEdit] = useState(false);
    70→  const [loadingDelete, setLoadingDelete] = useState(false);
    71→  const [deleteConfirmId, setDeleteConfirmId] = useState<string | null>(null);
    72→
    73→  const saldo = useMemo(() => saldoActualCaja(), [caja, caja.movimientos]);
    74→  const movimientos = caja.movimientos || [];
    75→
    76→  const isAdminOrOwner = usuarioActual?.rol === 'owner' || usuarioActual?.rol === 'admin';
    77→
    78→  // Summary calculations
    79→  const resumenOtros = useMemo(() => {
    80→    const res: Record<string, { ingresos: number; egresos: number }> = {};
    81→    movimientos.forEach(m => {
    82→      if (m.metodo !== 'Efectivo') {
    83→        if (!res[m.metodo]) res[m.metodo] = { ingresos: 0, egresos: 0 };
    84→        res[m.metodo][m.tipo === 'ingreso' ? 'ingresos' : 'egresos'] += m.monto;
    85→      }
    86→    });
    87→    return res;
    88→  }, [movimientos]);
    89→
    90→  const totalIngresosPorMetodo = useMemo(() => {
    91→    const res: Record<string, number> = {};
    92→    movimientos.forEach(m => {
    93→      if (m.tipo === 'ingreso') res[m.metodo] = (res[m.metodo] || 0) + m.monto;
    94→    });
    95→    return res;
    96→  }, [movimientos]);
    97→
    98→  const totalOtros = Object.values(resumenOtros).reduce((s, v) => s + v.ingresos - v.egresos, 0);
    99→  const totalEfectivo = useMemo(() => BILLETES.reduce((s, b) => s + b * (billetes[b] || 0), 0), [billetes]);
   100→  const totalTurno = totalEfectivo + totalOtros;
   101→
   102→  // Handlers
   103→  const handleAbrir = async () => {
   104→    const m = parseFloat(montoInicial);
   105→    if (isNaN(m) || m < 0) return toast.error('Monto invalido');
   106→    setLoadingAbrir(true);
   107→    try {
   108→      const ok = await abrirCaja(m);
   109→      if (ok) {
   110→        toast.success('Caja abierta', { description: 'Turno iniciado' });
   111→        setShowApertura(false);
   112→        setMontoInicial('');
   113→      } else {
   114→        toast.error('Error al abrir la caja');
   115→      }
   116→    } catch (e) {
   117→      toast.error('Error al abrir la caja');
   118→    }
   119→    setLoadingAbrir(false);
   120→  };
   121→
   122→  const handleMovimiento = async () => {
   123→    if (!showMovForm) return;
   124→    const m = parseFloat(movMonto);
   125→    if (isNaN(m) || m <= 0) return toast.error('Monto invalido');
   126→    if (showMovForm === 'egreso' && !movCategoria) return toast.error('Selecciona una categoria de gasto');
   127→    const desc = movDesc.trim() || (showMovForm === 'ingreso' ? 'Ingreso manual' : `Egreso: ${movCategoria}`);
   128→    setLoadingMovimiento(true);
   129→    try {
   130→      const ok = await registrarMovimientoCaja(showMovForm, m, desc, movMetodo, showMovForm === 'egreso' ? movCategoria : undefined);
   131→      if (ok) {
   132→        toast.success(`${showMovForm === 'ingreso' ? 'Ingreso' : 'Egreso'} registrado`, { description: formatMoney(m) });
   133→        setShowMovForm(null);
   134→        setMovMonto('');
   135→        setMovDesc('');
   136→        setMovMetodo('Efectivo');
   137→        setMovCategoria('');
   138→      } else {
   139→        toast.error(`Error al registrar ${showMovForm}`);
   140→      }
   141→    } catch (e) {
   142→      toast.error(`Error al registrar ${showMovForm}`);
   143→    }
   144→    setLoadingMovimiento(false);
   145→  };
   146→
   147→  const handleCerrar = async () => {
   148→    setLoadingCerrar(true);
   149→    try {
   150→      const cierre = await cerrarCaja(billetes, totalOtros);
   151→      if (cierre) {
   152→        toast.success('Caja cerrada', { description: 'Turno finalizado' });
   153→        setShowCierre(false);
   154→        setBilletes(Object.fromEntries(BILLETES.map(b => [b, 0])));
   155→      } else {
   156→        toast.error('Error al cerrar la caja');
   157→      }
   158→    } catch (e) {
   159→      toast.error('Error al cerrar la caja');
   160→    }
   161→    setLoadingCerrar(false);
   162→  };
   163→
   164→  const handleEditOpen = (mov: typeof movimientos[0]) => {
   165→    setEditingMov({ id: mov.id, monto: mov.monto, descripcion: mov.descripcion });
   166→    setEditMonto(String(mov.monto));
   167→    setEditDesc(mov.descripcion);
   168→  };
   169→
   170→  const handleEditSave = async () => {
   171→    if (!editingMov) return;
   172→    const m = parseFloat(editMonto);
   173→    if (isNaN(m) || m <= 0) return toast.error('Monto invalido');
   174→    if (!editDesc.trim()) return toast.error('Descripcion requerida');
   175→    setLoadingEdit(true);
   176→    try {
   177→      const ok = await editarMovimientoCaja(editingMov.id, { monto: m, descripcion: editDesc.trim() });
   178→      if (ok) {
   179→        toast.success('Movimiento actualizado');
   180→        setEditingMov(null);
   181→      } else {
   182→        toast.error('Error al editar');
   183→      }
   184→    } catch (e) {
   185→      toast.error('Error al editar movimiento');
   186→    }
   187→    setLoadingEdit(false);
   188→  };
   189→
   190→  const handleDelete = async (movId: string) => {
   191→    setDeleteConfirmId(movId);
   192→  };
   193→
   194→  const confirmDelete = async () => {
   195→    if (!deleteConfirmId) return;
   196→    setLoadingDelete(true);
   197→    try {
   198→      const ok = await eliminarMovimientoCaja(deleteConfirmId);
   199→      if (ok) {
   200→        toast.success('Movimiento eliminado');
   201→      } else {
   202→        toast.error('Error al eliminar');
   203→      }
   204→    } catch (e) {
   205→      toast.error('Error al eliminar movimiento');
   206→    }
   207→    setLoadingDelete(false);
   208→    setDeleteConfirmId(null);
   209→  };
   210→
   211→  // Saldo esperado (efectivo only)
   212→  const saldoEsperadoEfectivo = useMemo(() => {
   213→    if (!caja.apertura) return 0;
   214→    let s = caja.apertura.montoInicial;
   215→    movimientos.forEach(m => {
   216→      if (m.metodo === 'Efectivo') s += m.tipo === 'ingreso' ? m.monto : -m.monto;
   217→    });
   218→    return s;
   219→  }, [caja.apertura, movimientos]);
   220→
   221→  const diferencia = totalEfectivo - saldoEsperadoEfectivo;
   222→
   223→  const cancelarForm = () => { setShowMovForm(null); setMovMonto(''); setMovDesc(''); setMovMetodo('Efectivo'); setMovCategoria(''); };
   224→
   225→  return (
   226→    <div className="space-y-6">
   227→      <ModuleHeader icon={Wallet} title="Caja" subtitle="Controla los movimientos de dinero del dia" />
   228→
   229→      {caja.estado === 'cerrada' ? (
   230→        /* ═══════ CAJA CERRADA ═══════ */
   231→        <Card>
   232→          <CardContent className="text-center py-10 space-y-4">
   233→            <Lock className="w-12 h-12 mx-auto text-destructive" />
   234→            <div>
   235→              <h3 className="text-lg font-semibold">Caja cerrada</h3>
   236→              <p className="text-sm text-muted-foreground">Inicie un nuevo turno para comenzar a operar.</p>
   237→            </div>
   238→            {!showApertura ? (
   239→              <Button size="lg" onClick={() => setShowApertura(true)}><Unlock className="w-4 h-4 mr-1" />Abrir caja</Button>
   240→            ) : (
   241→              <div className="max-w-xs mx-auto space-y-2">
   242→                <Label className="text-sm text-muted-foreground">Monto inicial en efectivo</Label>
   243→                <Input type="number" placeholder="0.00" step="0.01" min="0" value={montoInicial} onChange={e => setMontoInicial(e.target.value)} autoFocus />
   244→                <div className="flex gap-2">
   245→                  <Button onClick={handleAbrir} className="flex-1" disabled={loadingAbrir}>
   246→                    {loadingAbrir ? <Loader2 className="w-4 h-4 animate-spin mr-1" /> : null}Confirmar apertura
   247→                  </Button>
   248→                  <Button variant="secondary" onClick={() => setShowApertura(false)} disabled={loadingAbrir}>Cancelar</Button>
   249→                </div>
   250→              </div>
   251→            )}
   252→            {caja.historial && caja.historial.length > 0 && (
   253→              <p className="text-xs text-muted-foreground">Ultimo cierre: {formatFechaHora(caja.historial[caja.historial.length - 1].cierre.fecha)}</p>
   254→            )}
   255→          </CardContent>
   256→        </Card>
   257→      ) : (
   258→        /* ═══════ CAJA ABIERTA ═══════ */
   259→        <div className="space-y-4">
   260→          {/* ── Mobile: compact status bar ── */}
   261→          <Card className="lg:hidden">
   262→            <CardContent className="p-3 space-y-3">
   263→              <div className="flex items-center justify-between">
   264→                <h3 className="font-semibold flex items-center gap-2 text-sm"><Unlock className="w-4 h-4 text-[#166534]" /> Caja abierta</h3>
   265→                <Dialog open={showCierre} onOpenChange={setShowCierre}>
   266→                  <DialogTrigger asChild>
   267→                    <Button variant="destructive" size="sm" className="h-7 text-xs"><Lock className="w-3.5 h-3.5 mr-1" />Cerrar</Button>
   268→                  </DialogTrigger>
   269→                  <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
   270→                    <ClosingDialogContent
   271→                      saldoEsperado={saldoEsperadoEfectivo}
   272→                      totalEfectivo={totalEfectivo}
   273→                      diferencia={diferencia}
   274→                      totalOtros={totalOtros}
   275→                      resumenOtros={resumenOtros}
   276→                      totalIngresosPorMetodo={totalIngresosPorMetodo}
   277→                      billetes={billetes}
   278→                      setBilletes={setBilletes}
   279→                      loadingCerrar={loadingCerrar}
   280→                      handleCerrar={handleCerrar}
   281→                      setShowCierre={setShowCierre}
   282→                    />
   283→                  </DialogContent>
   284→                </Dialog>
   285→              </div>
   286→              {/* Saldo + info */}
   287→              <div className="grid grid-cols-3 gap-2 text-center rounded-lg border p-2.5 bg-muted/30">
   288→                <div>
   289→                  <p className="text-[10px] text-muted-foreground leading-tight">Saldo</p>
   290→                  <p className="text-sm font-bold">{formatMoney(saldo)}</p>
   291→                </div>
   292→                <div>
   293→                  <p className="text-[10px] text-muted-foreground leading-tight">Movimientos</p>
   294→                  <p className="text-sm font-semibold">{movimientos.length}</p>
   295→                </div>
   296→                <div>
   297→                  <p className="text-[10px] text-muted-foreground leading-tight">Apertura</p>
   298→                  <p className="text-sm font-medium">{caja.apertura ? formatHora(caja.apertura.fecha) : '--'}</p>
   299→                </div>
   300→              </div>
   301→              {/* Action buttons */}
   302→              <div className="grid grid-cols-2 gap-2">
   303→                <Button variant="outline" className="h-9 text-sm" onClick={() => setShowMovForm('ingreso')}><Plus className="w-4 h-4 mr-1" />Ingreso</Button>
   304→                <Button variant="outline" className="h-9 text-sm" onClick={() => setShowMovForm('egreso')}><Minus className="w-4 h-4 mr-1" />Egreso</Button>
   305→              </div>
   306→              {/* Movement form inline */}
   307→              {showMovForm && (
   308→                <MovFormInline
   309→                  tipo={showMovForm}
   310→                  movMonto={movMonto}
   311→                  setMovMonto={setMovMonto}
   312→                  movDesc={movDesc}
   313→                  setMovDesc={setMovDesc}
   314→                  movMetodo={movMetodo}
   315→                  setMovMetodo={setMovMetodo}
   316→                  movCategoria={movCategoria}
   317→                  setMovCategoria={setMovCategoria}
   318→                  loading={loadingMovimiento}
   319→                  onGuardar={handleMovimiento}
   320→                  onCancelar={cancelarForm}
   321→                  metodosPago={metodosPago}
   322→                  categoriasGastos={categoriasGastos}
   323→                />
   324→              )}
   325→            </CardContent>
   326→          </Card>
   327→
   328→          {/* ── Mobile: movement cards ── */}
   329→          <Card className="lg:hidden">
   330→            <CardHeader className="pb-3"><CardTitle className="text-base">Movimientos del turno</CardTitle></CardHeader>
   331→            <CardContent className="p-0">
   332→              {movimientos.length === 0 ? (
   333→                <div className="text-center py-10 text-muted-foreground">Sin movimientos.</div>
   334→              ) : (
   335→                <div className="divide-y">
   336→                  {[...movimientos].reverse().map((m) => (
   337→                    <div key={m.id} className="p-3.5 space-y-1.5">
   338→                      <div className="flex items-start justify-between gap-2">
   339→                        <div className="flex items-center gap-2">
   340→                          {m.tipo === 'ingreso' ? (
   341→                            <Badge className="bg-[#DCFCE7] text-[#166534]"><Plus className="w-3 h-3 mr-0.5" />Ingreso</Badge>
   342→                          ) : (
   343→                            <Badge className="bg-[#FEE2E2] text-[#991B1B]"><Minus className="w-3 h-3 mr-0.5" />Egreso</Badge>
   344→                          )}
   345→                          <Badge variant="secondary">{m.metodo}</Badge>
   346→                        </div>
   347→                        <p className={`text-sm font-bold shrink-0 ${m.tipo === 'ingreso' ? 'text-[#166534]' : 'text-[#991B1B]'}`}>
   348→                          {m.tipo === 'ingreso' ? '+' : '-'}{formatMoney(m.monto)}
   349→                        </p>
   350→                      </div>
   351→                      {m.descripcion && (
   352→                        <p className="text-xs text-muted-foreground leading-relaxed">{m.descripcion}</p>
   353→                      )}
   354→                      <div className="flex items-center justify-between">
   355→                        <p className="text-[10px] text-muted-foreground">{formatFechaHora(m.fecha)}</p>
   356→                        {isAdminOrOwner && (
   357→                          <div className="flex gap-1">
   358→                            <Button size="icon" variant="ghost" className="h-6 w-6" onClick={() => handleEditOpen(m)}><Pencil className="w-3 h-3" /></Button>
   359→                            <Button size="icon" variant="ghost" className="h-6 w-6 text-destructive" onClick={() => handleDelete(m.id)} disabled={loadingDelete}>
   360→                              {loadingDelete ? <Loader2 className="w-3 h-3 animate-spin" /> : <Trash2 className="w-3 h-3" />}
   361→                            </Button>
   362→                          </div>
   363→                        )}
   364→                      </div>
   365→                    </div>
   366→                  ))}
   367→                </div>
   368→              )}
   369→            </CardContent>
   370→          </Card>
   371→
   372→          {/* ═══════ DESKTOP ═══════ */}
   373→          <div className="hidden lg:grid lg:grid-cols-3 gap-4">
   374→            <div className="lg:col-span-2 space-y-4">
   375→              {/* Status bar */}
   376→              <Card>
   377→                <CardContent className="flex items-center justify-between py-3">
   378→                  <h3 className="font-semibold flex items-center gap-2"><Unlock className="w-5 h-5 text-[#166534]" /> Caja abierta</h3>
   379→                  <Dialog open={showCierre} onOpenChange={setShowCierre}>
   380→                    <DialogTrigger asChild>
   381→                      <Button variant="destructive" disabled={loadingCerrar}><Lock className="w-4 h-4 mr-1" />Cerrar caja</Button>
   382→                    </DialogTrigger>
   383→                    <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
   384→                      <ClosingDialogContent
   385→                        saldoEsperado={saldoEsperadoEfectivo}
   386→                        totalEfectivo={totalEfectivo}
   387→                        diferencia={diferencia}
   388→                        totalOtros={totalOtros}
   389→                        resumenOtros={resumenOtros}
   390→                        totalIngresosPorMetodo={totalIngresosPorMetodo}
   391→                        billetes={billetes}
   392→                        setBilletes={setBilletes}
   393→                        loadingCerrar={loadingCerrar}
   394→                        handleCerrar={handleCerrar}
   395→                        setShowCierre={setShowCierre}
   396→                      />
   397→                    </DialogContent>
   398→                  </Dialog>
   399→                </CardContent>
   400→              </Card>
   401→
   402→              {/* Movements table */}
   403→              <Card>
   404→                <CardHeader className="pb-3"><CardTitle className="text-base">Movimientos del turno</CardTitle></CardHeader>
   405→                <CardContent>
   406→                  {movimientos.length === 0 ? <p className="text-sm text-muted-foreground">Sin movimientos.</p> : (
   407→                    <Table>
   408→                      <TableHeader>
   409→                        <TableRow>
   410→                          <TableHead>Hora</TableHead>
   411→                          <TableHead>Tipo</TableHead>
   412→                          <TableHead>Monto</TableHead>
   413→                          <TableHead>Metodo</TableHead>
   414→                          <TableHead>Descripcion</TableHead>
   415→                          {isAdminOrOwner && <TableHead className="w-[80px]">Acciones</TableHead>}
   416→                        </TableRow>
   417→                      </TableHeader>
   418→                      <TableBody>
   419→                        {movimientos.map((m) => (
   420→                          <TableRow key={m.id}>
   421→                            <TableCell className="text-xs">{formatFechaHora(m.fecha)}</TableCell>
   422→                            <TableCell>
   423→                              {m.tipo === 'ingreso' ? (
   424→                                <Badge className="bg-[#DCFCE7] text-[#166534]"><Plus className="w-3 h-3 mr-1" />Ingreso</Badge>
   425→                              ) : (
   426→                                <Badge className="bg-[#FEE2E2] text-[#991B1B]"><Minus className="w-3 h-3 mr-1" />Egreso</Badge>
   427→                              )}
   428→                            </TableCell>
   429→                            <TableCell className={`font-medium ${m.tipo === 'ingreso' ? 'text-[#166534]' : 'text-[#991B1B]'}`}>
   430→                              {m.tipo === 'ingreso' ? '+' : '-'}{formatMoney(m.monto)}
   431→                            </TableCell>
   432→                            <TableCell>{m.metodo}</TableCell>
   433→                            <TableCell className="text-sm text-muted-foreground">{m.descripcion}</TableCell>
   434→                            {isAdminOrOwner && (
   435→                              <TableCell>
   436→                                <div className="flex gap-1">
   437→                                  <Button size="icon" variant="ghost" className="h-7 w-7" onClick={() => handleEditOpen(m)}><Pencil className="w-3.5 h-3.5" /></Button>
   438→                                  <Button size="icon" variant="ghost" className="h-7 w-7 text-destructive" onClick={() => handleDelete(m.id)} disabled={loadingDelete}>
   439→                                    {loadingDelete ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Trash2 className="w-3.5 h-3.5" />}
   440→                                  </Button>
   441→                                </div>
   442→                              </TableCell>
   443→                            )}
   444→                          </TableRow>
   445→                        ))}
   446→                      </TableBody>
   447→                    </Table>
   448→                  )}
   449→                </CardContent>

   621→function MovFormInline({
   622→  tipo, movMonto, setMovMonto, movDesc, setMovDesc,
   623→  movMetodo, setMovMetodo, movCategoria, setMovCategoria,

   628→  movDesc: string; setMovDesc: (v: string) => void;

   630→  movCategoria: string; setMovCategoria: (v: string) => void;
