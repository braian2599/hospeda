    36→/** Get initials from a full name (first letter of first word + first letter of last word) */
    37→const getInitials = (name: string): string => {
    38→  const parts = name.trim().split(/\s+/).filter(Boolean);
    39→  if (parts.length === 0) return '?';
    40→  if (parts.length === 1) return parts[0][0].toUpperCase();
    41→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
    42→};
    43→
    44→/** Format a date string as relative time in Spanish (short form) */
    45→const formatRelativeTime = (dateStr: string): string => {
    46→  if (!dateStr) return '';
    47→  const d = new Date(dateStr);
    48→  const now = new Date();
    49→  const diffMs = now.getTime() - d.getTime();
    50→  const diffMins = Math.floor(diffMs / 60000);
    51→  const diffHrs = Math.floor(diffMs / 3600000);
    52→  const diffDays = Math.floor(diffMs / 86400000);
    53→
    54→  if (diffMins < 1) return 'recién';
    55→  if (diffMins < 60) return `hace ${diffMins} min`;
    56→  if (diffHrs < 24) return `hace ${diffHrs} h`;
    57→  if (diffDays === 1) return 'ayer';
    58→  if (diffDays < 7) return `hace ${diffDays} días`;
    59→  if (diffDays < 30) return `hace ${Math.floor(diffDays / 7)} sem`;
    60→  if (diffDays < 365) return `hace ${Math.floor(diffDays / 30)} meses`;
    61→  return `hace ${Math.floor(diffDays / 365)} años`;
    62→};
    63→
    64→/** Generate a random password: 10 chars, at least one uppercase + one digit */
    65→const randomPassword = (): string => {
    66→  const upper = 'ABCDEFGHJKLMNPQRSTUVWXYZ';
    67→  const lower = 'abcdefghjkmnpqrstuvwxyz';
    68→  const digits = '23456789';
    69→  const all = upper + lower + digits;
    70→  const pick = (s: string) => s[Math.floor(Math.random() * s.length)];
    71→  const chars: string[] = [pick(upper), pick(digits)];
    72→  for (let i = 2; i < 10; i++) chars.push(pick(all));
    73→  // Fisher-Yates shuffle to avoid predictable positions
    74→  for (let i = chars.length - 1; i > 0; i--) {
    75→    const j = Math.floor(Math.random() * (i + 1));
    76→    [chars[i], chars[j]] = [chars[j], chars[i]];
    77→  }
    78→  return chars.join('');
    79→};
