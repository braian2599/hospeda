    65→interface KpiProps {
    66→  label: string;
    67→  value: string | number;
    68→  icon: React.ReactNode;
    69→  color: string;
    70→  bgColor: string;
    71→  tooltip?: string;
    72→  subtext?: string;
    73→  trend?: { value: number; label: string };
    74→  onClick?: () => void;
    75→}
    76→
    77→function KpiCard({ label, value, icon, color, bgColor, tooltip, subtext, trend, onClick }: KpiProps) {
    78→  const card = (
    79→    <Card
    80→      className={`relative overflow-hidden transition-all duration-200 hover:shadow-md ${onClick ? 'cursor-pointer group' : ''}`}
    81→      onClick={onClick}
    82→    >
    83→      <CardContent className="p-3 sm:p-4">
    84→        <div className="flex items-start justify-between gap-2 sm:gap-3">
    85→          <div className="flex-1 min-w-0 text-center">
    86→            <p className="text-[10px] sm:text-[11px] font-medium uppercase tracking-wider text-muted-foreground mb-1 sm:mb-1.5">{label}</p>
    87→            <p className={`text-lg sm:text-2xl font-bold tracking-tight ${color} truncate`}>{value}</p>
    88→            {trend && (
    89→              <div className="flex items-center justify-center gap-1 mt-1 sm:mt-1.5">
    90→                {trend.value > 0 ? (
    91→                  <ArrowUpRight className="w-3 h-3 sm:w-3.5 sm:h-3.5 text-[#059669]" />
    92→                ) : trend.value < 0 ? (
    93→                  <ArrowDownRight className="w-3 h-3 sm:w-3.5 sm:h-3.5 text-[#EF4444]" />
    94→                ) : (
    95→                  <Minus className="w-3 h-3 sm:w-3.5 sm:h-3.5 text-muted-foreground" />
    96→                )}
    97→                <span className={`text-[11px] sm:text-xs font-medium ${trend.value > 0 ? 'text-[#059669]' : trend.value < 0 ? 'text-[#EF4444]' : 'text-muted-foreground'}`}>
    98→                  {Math.abs(trend.value)}%
    99→                </span>
   100→                <span className="text-[11px] sm:text-xs text-muted-foreground hidden sm:inline">{trend.label}</span>
   101→              </div>
   102→            )}
   103→            {subtext && !trend && <p className="text-[11px] sm:text-xs text-muted-foreground mt-1 truncate hidden sm:block">{subtext}</p>}
   104→          </div>
   105→          <div className={`shrink-0 w-8 h-8 sm:w-10 sm:h-10 rounded-lg sm:rounded-xl flex items-center justify-center ${bgColor}`}>
   106→            <div className="sm:scale-100 scale-90">{icon}</div>
   107→          </div>
   108→        </div>
   109→      </CardContent>
   110→      {onClick && (
   111→        <div className="absolute inset-x-0 bottom-0 h-0.5 bg-primary opacity-0 group-hover:opacity-100 transition-opacity" />
   112→      )}
   113→    </Card>
   114→  );
   115→
   116→  if (tooltip) {
   117→    return (
   118→      <TooltipProvider delayDuration={300}>
   119→        <Tooltip>
   120→          <TooltipTrigger asChild>{card}</TooltipTrigger>
   121→          <TooltipContent side="bottom" className="text-xs max-w-[220px]">{tooltip}</TooltipContent>
   122→        </Tooltip>
   123→      </TooltipProvider>
   124→    );
   125→  }
   126→  return card;
   127→}

   141→function ProgressKpi({ label, value, max, color = 'bg-primary', suffix = '%', description }: { label: string; value: number; max: number; color?: string; suffix?: string; description?: string }) {
   142→  const pct = max > 0 ? Math.round((value / max) * 100) : 0;
   143→  return (
   144→    <Card>
   145→      <CardContent className="p-3 sm:p-4">
   146→        <div className="flex items-center justify-between mb-1.5 sm:mb-2">
   147→          <p className="text-[10px] sm:text-[11px] font-medium uppercase tracking-wider text-muted-foreground">{label}</p>
   148→          <span className="text-base sm:text-lg font-bold">{value}{suffix}</span>
   149→        </div>
   150→        <Progress value={pct} className="h-2 sm:h-2.5" />
   151→        <p className="text-[11px] sm:text-xs text-muted-foreground mt-1 sm:mt-1.5 text-center">{description || `${pct}%`}</p>
   152→      </CardContent>
   153→    </Card>
   154→  );
   155→}
