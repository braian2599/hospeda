    17→    17→/* ─── Intersection Observer hook ─── */
    18→    18→function useInView(threshold = 0.12) {
    19→    19→  const ref = useRef<HTMLDivElement>(null);
    20→    20→  const [inView, setInView] = useState(false);
    21→    21→  useEffect(() => {
    22→    22→    const el = ref.current;
    23→    23→    if (!el) return;
    24→    24→    const obs = new IntersectionObserver(
    25→    25→      ([e]) => { if (e.isIntersecting) { setInView(true); obs.disconnect(); } },
    26→    26→      { threshold }
    27→    27→    );
    28→    28→    obs.observe(el);
    29→    29→    return () => obs.disconnect();
    30→    30→  }, [threshold]);
    31→    31→  return { ref, inView };
    32→    32→}

   169→   169→/* ─── Fade-in wrapper ─── */
   170→   170→function FadeIn({ children, className = '', delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   171→   171→  const { ref, inView } = useInView();
   172→   172→  return (
   173→   173→    <div
   174→   174→      ref={ref}
   175→   175→      className={`transition-all duration-700 ease-out ${className}`}
   176→   176→      style={{
   177→   177→        opacity: inView ? 1 : 0,
   178→   178→        transform: inView ? 'translateY(0)' : 'translateY(32px)',
   179→   179→        transitionDelay: `${delay}ms`,
   180→   180→      }}
   181→   181→    >
   182→   182→      {children}
   183→   183→    </div>
   184→   184→  );
   185→   185→}
