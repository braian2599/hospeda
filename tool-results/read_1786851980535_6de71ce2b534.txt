    78→function KPIAnimated({ icon: Icon, label, value, sub, color, bgGradient, borderColor, iconBg, labelColor, valueColor, subColor, trend, numericValue, suffix, sparkData, sparkColor }: {
    79→  icon: ComponentType<{ className?: string }>;

    83→  color: string;

    85→  borderColor?: string;

   139→function WeatherIcon({ code }: { code: number }) {
   140→  if (code === 0) return <Sun className="w-5 h-5 text-brand-amber" />;
   141→  if (code >= 1 && code <= 3) return <CloudSun className="w-5 h-5 text-slate-400" />;
   142→  if (code >= 45 && code <= 48) return <CloudFog className="w-5 h-5 text-slate-400" />;
   143→  if (code >= 51 && code <= 55) return <CloudDrizzle className="w-5 h-5 text-info" />;
   144→  if (code >= 56 && code <= 57) return <CloudDrizzle className="w-5 h-5 text-info" />;
   145→  if (code >= 61 && code <= 67) return <CloudRain className="w-5 h-5 text-info" />;
   146→  if (code >= 71 && code <= 77) return <CloudSnow className="w-5 h-5 text-info" />;
   147→  if (code >= 80 && code <= 82) return <CloudLightning className="w-5 h-5 text-amber-300" />;
   148→  if (code >= 95) return <CloudLightning className="w-5 h-5 text-violet-300" />;
   149→  return <Thermometer className="w-5 h-5 text-slate-400" />;
   150→}

   339→// ==================== CALENDARIO GANTT (con navegación y popover mejorado) ====================
   340→
   341→interface GanttReserva {
   342→  tipo: string;
   343→  checkin: string;
   344→  checkout: string;
   345→  huesped: string;
   346→  horaCheckin: Date;
   347→  horaCheckout: Date;
   348→  tarifa?: string;
   349→  monto?: number;
   350→  estadoPago?: string;
   351→  ninos?: number;
   352→}
   353→
   354→function CalendarioGantt({ habitaciones, reservas, fechaInicioBase }: {
   355→  habitaciones: Record<string, { tipo: string; estado: string; problema?: string }>;
   356→  reservas: { habitacion: string; estado: string; checkin: string; checkout: string; huesped: string; horaCheckin?: string; horaCheckout?: string; tipoTarifa?: string; total?: number; estadoPago?: string; ninos?: number }[];
   357→  fechaInicioBase: Date;
   358→}) {
   359→  const [offset, setOffset] = useState(0);
   360→  const [ganttDays, setGanttDays] = useState(14);
   361→  const [mostrarHistorial, setMostrarHistorial] = useState(false);
   362→  const [popoverData, setPopoverData] = useState<PopoverData | null>(null);
   363→  const [popoverPos, setPopoverPos] = useState({ top: 0, left: 0 });
   364→
   365→  const fechaInicio = useMemo(() => {
   366→    const d = new Date(fechaInicioBase);
   367→    d.setDate(d.getDate() + offset * 7);
   368→    return d;
   369→  }, [fechaInicioBase, offset]);
   370→
   371→  const columnas = useMemo(() => {
   372→    const cols: string[] = [];
   373→    for (let i = 0; i < ganttDays; i++) {
   374→      const f = new Date(fechaInicio);
   375→      f.setDate(f.getDate() + i);
   376→      cols.push(toLocalDateStr(f));
   377→    }
   378→    return cols;
   379→  }, [fechaInicio, ganttDays]);
   380→
   381→  const colIdx = useMemo(() => {
   382→    const idx: Record<string, number> = {};
   383→    columnas.forEach((c, i) => idx[c] = i);
   384→    return idx;
   385→  }, [columnas]);
   386→
   387→  const hoyStr = todayLocal();
   388→
   389→  const handleBarClick = useCallback((e: React.MouseEvent, res: GanttReserva, num: string) => {
   390→    e.stopPropagation();
   391→    const rect = (e.currentTarget as HTMLElement).getBoundingClientRect();
   392→    let left = rect.left;
   393→    if (left + 280 > window.innerWidth) left = window.innerWidth - 290;
   394→    setPopoverPos({ top: rect.bottom + 6, left: Math.max(left, 10) });
   395→    setPopoverData({
   396→      estado: res.tipo,
   397→      habitacion: num,
   398→      huesped: res.huesped,
   399→      checkin: res.checkin,
   400→      checkout: res.checkout,
   401→      problema: res.tipo === 'Mantenimiento' ? res.huesped : undefined,
   402→      tarifa: res.tarifa,
   403→      monto: res.monto,
   404→      estadoPago: res.estadoPago,
   405→      ninos: res.ninos,
   406→    });
   407→  }, []);
   408→
   409→  const rows = useMemo(() => {
   410→    const habNumbers = Object.keys(habitaciones).sort((a, b) => parseInt(a) - parseInt(b));
   411→    const result: React.ReactNode[] = [];
   412→
   413→    habNumbers.forEach((num, rowIndex) => {
   414→      const hab = habitaciones[num];
   415→      const reservasHab: GanttReserva[] = [];
   416→
   417→      reservas.forEach(r => {
   418→        if (r.habitacion !== num || r.estado === 'Cancelada') return;
   419→        if (r.checkin > columnas[ganttDays - 1] || r.checkout < columnas[0]) return;
   420→
   421→        // Usar tanto el formato del store como el de la BD para comparar
   422→        const esCheckout = r.estado === 'Checkout_realizado' || r.estado === 'Check-Out realizado';
   423→        const esCheckin = r.estado === 'CheckIn_realizado' || r.estado === 'Check-In realizado';
   424→
   425→        const esHistorica = esCheckout && !mostrarHistorial;
   426→        if (esHistorica) return;
   427→
   428→        const horaCheckin = r.horaCheckin ? new Date(r.horaCheckin) : new Date(r.checkin + 'T14:00:00');
   429→        const horaCheckout = r.horaCheckout ? new Date(r.horaCheckout) : new Date(r.checkout + 'T09:00:00');
   430→
   431→        let estado: string;
   432→        if (esCheckout) {
   433→          estado = 'Finalizada';
   434→        } else if (esCheckin) {
   435→          estado = 'Ocupada';
   436→        } else {
   437→          estado = 'Reservada';
   438→        }
   439→        reservasHab.push({
   440→          tipo: estado, checkin: r.checkin, checkout: r.checkout, huesped: r.huesped,
   441→          horaCheckin, horaCheckout, tarifa: r.tipoTarifa, monto: r.total, estadoPago: r.estadoPago, ninos: r.ninos,
   442→        });
   443→      });
   444→
   445→      if (hab.estado === 'Limpieza' && columnas.includes(hoyStr)) {
   446→        reservasHab.push({ tipo: 'Limpieza', checkin: hoyStr, checkout: hoyStr, huesped: 'Limpieza', horaCheckin: new Date(hoyStr + 'T00:00:00'), horaCheckout: new Date(hoyStr + 'T23:59:59') });
   447→      }
   448→
   449→      if (hab.estado === 'Mantenimiento') {
   450→        reservasHab.push({ tipo: 'Mantenimiento', checkin: columnas[0], checkout: columnas[ganttDays - 1], huesped: hab.problema || 'Mantenimiento', horaCheckin: new Date(columnas[0] + 'T00:00:00'), horaCheckout: new Date(columnas[ganttDays - 1] + 'T23:59:59') });
   451→      }
   452→
   453→      // Habitaciones compartidas: carriles múltiples
   454→      if (hab.tipo === 'Compartida' && reservasHab.length > 0) {
   455→        const reservasActivas = reservasHab.filter(r => r.tipo !== 'Limpieza' && r.tipo !== 'Mantenimiento');
   456→        const numCarriles = reservasActivas.length || 1;
   457→        const FILA_H = Math.max(ROW_H, numCarriles * (BAR_H + 4) + 8);
   458→
   459→        const bgCells = columnas.map((col, ci) => {

   643→function getBarColorClass(tipo: string): string {
   644→  const map: Record<string, string> = {
   645→    Reservada: 'bg-status-reserved shadow-sm',
   646→    Ocupada: 'bg-status-available shadow-sm',
   647→    Finalizada: 'bg-status-finalized opacity-50 border border-dashed border-slate-600/40',
   648→    Limpieza: 'bg-status-cleaning shadow-sm',
   649→    Mantenimiento: 'bg-status-maintenance shadow-sm',
   650→  };
   651→  return map[tipo] || map.Reservada;

   654→function calcularBarra(res: GanttReserva, colIdx: Record<string, number>, columnas: string[], DIAS: number) {
   655→  const COL_PCT = 100 / DIAS;
   656→  const MITAD_COL_PCT = COL_PCT / 2;
   657→  let startCol = Math.max(colIdx[res.checkin] ?? 0, 0);
   658→  let endCol = Math.min(colIdx[res.checkout] !== undefined ? colIdx[res.checkout] : DIAS - 1, DIAS - 1);
   659→
   660→  const esCheckinTarde = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckin && res.horaCheckin.getHours() >= 12;
   661→  const esCheckoutManana = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckout && res.horaCheckout.getHours() < 12;
   662→
   663→  let leftPct: number, widthPct: number;
   664→
   665→  if (esCheckinTarde && startCol >= 0 && !(res.checkin < columnas[0])) {
   666→    leftPct = startCol * COL_PCT + MITAD_COL_PCT;
   667→  } else {
   668→    leftPct = startCol * COL_PCT;
   669→  }
   670→
   671→  if (esCheckoutManana && endCol >= 0 && !(res.checkout > columnas[DIAS - 1])) {
   672→    widthPct = ((endCol - startCol) * COL_PCT) + (esCheckinTarde ? MITAD_COL_PCT : COL_PCT) - (esCheckoutManana ? MITAD_COL_PCT : 0);
   673→  } else {
   674→    widthPct = (endCol - startCol + 1) * COL_PCT;
   675→  }
   676→
   677→  widthPct = Math.max(widthPct, MITAD_COL_PCT);
   678→  return { leftPct, widthPct };
   679→}

   683→function RoomHeatmap({ habitaciones, reservas }: {
   684→  habitaciones: Record<string, { tipo: string; estado: string; capacidad: number; problema?: string }>;
   685→  reservas: { habitacion: string; estado: string; huesped: string; checkin: string; checkout: string }[];

   691→    reservas.forEach(r => {
   692→      if ((r.estado === 'Check-In realizado' || r.estado === 'Confirmada') && r.checkin <= hoyStr && r.checkout >= hoyStr) {
   693→        map[r.habitacion] = { huesped: r.huesped, estado: r.estado === 'Check-In realizado' ? 'Ocupada' : 'Reservada' };

   697→  }, [reservas, hoyStr]);

   699→  const colors: Record<string, string> = {
   700→    Disponible: 'bg-emerald-900/60 text-emerald-300',
   701→    Ocupada: 'bg-red-900/60 text-red-300',
   702→    Limpieza: 'bg-amber-900/60 text-amber-300',
   703→    Mantenimiento: 'bg-muted/30 text-slate-400',

   707→    Disponible: 'bg-status-available', Ocupada: 'bg-status-occupied', Limpieza: 'bg-status-cleaning', Mantenimiento: 'bg-status-maintenance', Reservada: 'bg-status-reserved',
   708→  };

   710→  return (
   711→    <Card>
   712→      <CardHeader className="pb-3">
   713→        <CardTitle className="text-base">Estado de habitaciones</CardTitle>

   715→      <CardContent>

   717→          {Object.entries(habitaciones).sort(([a], [b]) => a.localeCompare(b)).map(([num, hab]) => {
   718→            const info = habInfo[num];
   719→            const tooltip = info
   720→              ? `${num} - ${info.huesped} (${info.estado})`
   721→              : hab.estado === 'Mantenimiento'
   722→                ? `${num} - ${hab.problema || 'Mantenimiento'}`
   723→                : `${num} - ${hab.estado} · Cap. ${hab.capacidad} · ${hab.tipo}`;
   724→            return (
   725→              <Tooltip key={num} text={tooltip}>
   726→                <div className={`rounded-lg border-2 p-2 text-center text-xs font-semibold transition-all duration-200 hover:-translate-y-0.5 hover:scale-[1.04] hover:shadow-lg cursor-default ${colors[hab.estado] || colors.Disponible}`}>
   727→                  <div className={`w-2 h-2 rounded-full ${dots[hab.estado] || dots.Disponible} ${hab.estado === 'Ocupada' || hab.estado === 'Limpieza' ? 'heat-pulse' : ''} mx-auto mb-1`} />
   728→                  <div>{num}</div>
   729→                  {info && <div className="text-[9px] font-normal opacity-70 truncate mt-0.5">{info.huesped}</div>}
   730→                </div>
   731→              </Tooltip>
   732→            );
   733→          })}
   734→        </div>
   735→        <div className="flex flex-wrap gap-4 mt-4 text-xs text-muted-foreground">
