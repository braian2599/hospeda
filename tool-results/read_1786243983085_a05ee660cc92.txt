   101→function DateRangePickerInline({
   102→  checkin,
   103→  checkout,
   104→  onChangeCheckin,
   105→  onChangeCheckout,
   106→  label,
   107→  disabled,
   108→}: {
   109→  checkin: string;
   110→  checkout: string;
   111→  onChangeCheckin: (val: string) => void;
   112→  onChangeCheckout: (val: string) => void;
   113→  label?: string;
   114→  disabled?: boolean;
   115→}) {
   116→  const [open, setOpen] = useState(false);
   117→  const [localRange, setLocalRange] = useState<{ from?: Date; to?: Date }>({});
   118→  const syncedRef = useRef(false);
   119→
   120→  // Sync local state with the form when the popover opens (event-driven, not effect-driven).
   121→  const handleOpenChange = (next: boolean) => {
   122→    setOpen(next);
   123→    if (next && !syncedRef.current) {
   124→      syncedRef.current = true;
   125→      const f = checkin ? new Date(checkin + 'T12:00:00') : undefined;
   126→      const t = checkout ? new Date(checkout + 'T12:00:00') : undefined;
   127→      setLocalRange(f ? { from: f, to: t } : {});
   128→    }
   129→    if (!next) syncedRef.current = false;
   130→  };
   131→
   132→  const handleSelect = (range: { from?: Date; to?: Date } | undefined) => {
   133→    if (!range?.from) {
   134→      setLocalRange({});
   135→      return;
   136→    }
   137→    setLocalRange(range);
   138→  };
   139→
   140→  const fmt = (d: Date) => {
   141→    const yyyy = d.getFullYear();
   142→    const mm = String(d.getMonth() + 1).padStart(2, '0');
   143→    const dd = String(d.getDate()).padStart(2, '0');
   144→    return `${yyyy}-${mm}-${dd}`;
   145→  };
   146→
   147→  const handleConfirm = () => {
   148→    if (!localRange.from) return;
   149→    onChangeCheckin(fmt(localRange.from));
   150→    onChangeCheckout(localRange.to ? fmt(localRange.to) : '');
   151→    handleOpenChange(false);
   152→  };
   153→
   154→  const from = checkin ? new Date(checkin + 'T12:00:00') : undefined;
   155→  const to = checkout ? new Date(checkout + 'T12:00:00') : undefined;
   156→
   157→  const displayText = from && to
   158→    ? `${format(from, 'dd/MM/yyyy')} → ${format(to, 'dd/MM/yyyy')}`
   159→    : from
   160→      ? `${format(from, 'dd/MM/yyyy')} → ...`
   161→      : 'Seleccionar fechas';
   162→
   163→  return (
   164→    <div className="grid gap-1.5">
   165→      {label && <Label>{label} *</Label>}
   166→      <Button
   167→        variant="outline"
   168→        className={cn(
   169→          'w-full justify-start text-left font-normal h-9',
   170→          !checkin && 'text-muted-foreground'
   171→        )}
   172→        disabled={disabled}
   173→        onClick={() => handleOpenChange(true)}
   174→      >
   175→        <CalendarDays className="w-4 h-4 mr-2 text-muted-foreground" />
   176→        {displayText}
   177→      </Button>
   178→      {open
   179→        ? createPortal(
   180→            <div className="fixed inset-0 z-[200] flex items-center justify-center bg-black/40" onClick={() => handleOpenChange(false)}>
   181→              <div
   182→                className="bg-white rounded-xl shadow-2xl border border-[#E2E8F0] pointer-events-auto"
   183→                onClick={e => e.stopPropagation()}
   184→                onMouseDown={e => e.stopPropagation()}
   185→              >
   186→                <Calendar
   187→                  mode="range"
   188→                  selected={localRange.from ? { from: localRange.from, to: localRange.to } as { from: Date; to: Date } : undefined}
   189→                  onSelect={handleSelect}
   190→                  numberOfMonths={2}
   191→                  autoFocus
   192→                  classNames={{
   193→                    range_start: 'rounded-l-md bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   194→                    range_end: 'rounded-r-md bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   195→                    range_middle: 'rounded-none bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   196→                    day: 'relative w-full h-full p-0 text-center [&:first-child[data-selected=true]_button]:rounded-l-md [&:last-child[data-selected=true]_button]:rounded-r-md group/day aspect-square select-none',
   197→                  }}
   198→                />
   199→                <div className="flex justify-end border-t border-[#E2E8F0] p-3">
   200→                  <Button
   201→                    size="sm"
   202→                    onClick={handleConfirm}
   203→                    disabled={!localRange.from}
   204→                    className="bg-[#059669] hover:bg-[#047857] text-white text-xs h-8"
   205→                  >
   206→                    Confirmar fechas
   207→                  </Button>
   208→                </div>
   209→              </div>
   210→            </div>,
   211→            document.body
   212→          )
   213→        : null}
   214→    </div>
   215→  );
   216→}

   246→// ==================== DESGLOSE PRECIO COMPONENT ====================
   247→function DesglosePrecio({ form, computed, formatMoney, s }: {
   248→  form: { tipoTarifa: string; reservaMultiple: boolean; habitacion: string; habitacion2?: string; personas: string; personas2?: string };
   249→  computed: { noches: number; subtotal: number; subtotal2: number; desglose: any };
   250→  formatMoney: (n: number) => string;
   251→  s: (n: number) => string;
   252→}) {
   253→  const d = computed.desglose;
   254→
   255→  const adultoLinea = (d: any) => {
   256→    if (!d || d.totalAdultos === 0) return null;
   257→    const modo = d.modoCobro;
   258→    if (modo === 'porCama') {
   259→      return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''} × ${formatMoney(d.precioRango)}/cama/noche`;
   260→    }
   261→    if (modo === 'porHabitacion') return 'Habitación';
   262→    if (modo === 'porPersona') {
   263→      return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''} × ${formatMoney(d.precioRango)}/noche`;
   264→    }
   265→    // porGrupo
   266→    return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''}`;
   267→  };
   268→
   269→  const renderLines = (sub: number, dLocal: any) => {
   270→    if (!dLocal) return (
   271→      <div className="space-y-1">
   272→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Subtotal</span><span className="font-semibold text-[#1E293B]">{formatMoney(sub)}</span></div>
   273→      </div>
   274→    );
   275→
   276→    const lines: { key: string; label: string; value: string; dim?: boolean }[] = [];
   277→
   278→    const aLabel = adultoLinea(dLocal);
   279→    if (aLabel) {
   280→      lines.push({ key: 'adultos', label: aLabel, value: formatMoney(dLocal.totalAdultos) });
   281→    }
   282→
   283→    if (dLocal.ninosCount > 0 && dLocal.totalNinos > 0) {
   284→      lines.push({
   285→        key: 'ninos',
   286→        label: `${dLocal.ninosCount} niño${dLocal.ninosCount > 1 ? 's' : ''} × ${formatMoney(dLocal.precioNino)}/noche`,
   287→        value: formatMoney(dLocal.totalNinos),
   288→      });
   289→    }
   290→
   291→    if (dLocal.nochesGratis > 0) {
   292→      const totalBase = dLocal.totalAdultos + dLocal.totalNinos;
   293→      const ahorro = totalBase > 0 && dLocal.nochesCobrables > 0
   294→        ? Math.round(totalBase / dLocal.nochesCobrables * dLocal.nochesGratis)
   295→        : 0;
   296→      lines.push({
   297→        key: 'cortesia',
   298→        label: `${dLocal.nochesGratis} noche${dLocal.nochesGratis > 1 ? 's' : ''} de cortesía`,
   299→        value: `- ${formatMoney(ahorro)}`,
   300→        dim: true,
   301→      });
   302→    }
   303→
   304→    return (
   305→      <div className="space-y-1">
   306→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Tarifa</span><span className="font-semibold text-[#1E293B] capitalize">{form.tipoTarifa}</span></div>
   307→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Noches</span><span className="font-semibold text-[#1E293B]">{computed.noches}{dLocal?.nochesGratis > 0 ? ` (${dLocal.nochesCobrables} cobrables)` : ''}</span></div>
   308→        {lines.map(l => (
   309→          <div key={l.key} className="flex justify-between items-center py-1 text-[13px]">
   310→            <span className={l.dim ? 'text-[#94A3B8] line-through' : 'text-[#64748B]'}>{l.label}</span>
   311→            <span className={l.dim ? 'font-medium text-[#94A3B8]' : 'font-semibold text-[#1E293B]'}>{l.value}</span>
   312→          </div>
   313→        ))}
   314→        <div className="flex justify-between items-center py-1.5 mt-1 border-t border-[#E2E8F0] text-[13px]">
   315→          <span className="font-medium text-[#475569]">Subtotal</span>
   316→          <span className="font-bold text-[#1E293B]">{formatMoney(sub)}</span>
   317→        </div>
   318→      </div>
   319→    );
   320→  };
   321→
   322→  return (
   323→    <>
   324→      {form.reservaMultiple ? (
   325→        <>
   326→          <p className="text-[11px] font-semibold text-[#64748B] uppercase tracking-wider">Habitación {form.habitacion}</p>
   327→          {renderLines(computed.subtotal, d)}
   328→          {computed.subtotal2 > 0 && (
   329→            <>
   330→              <p className="text-[11px] font-semibold text-[#64748B] uppercase tracking-wider pt-2">Habitación {form.habitacion2}</p>
   331→              {renderLines(computed.subtotal2, null)}
   332→            </>
   333→          )}
   334→        </>
   335→      ) : (
   336→        renderLines(computed.subtotal, d)
   337→      )}
   338→    </>
   339→  );
   340→}
