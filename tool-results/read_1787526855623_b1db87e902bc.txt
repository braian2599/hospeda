    18→    18→/* ─── Intersection Observer hook ─── */
    19→    19→function useInView(threshold = 0.12) {
    20→    20→  const ref = useRef<HTMLDivElement>(null);
    21→    21→  const [inView, setInView] = useState(false);
    22→    22→  useEffect(() => {
    23→    23→    const el = ref.current;
    24→    24→    if (!el) return;
    25→    25→    const obs = new IntersectionObserver(
    26→    26→      ([e]) => { if (e.isIntersecting) { setInView(true); obs.disconnect(); } },
    27→    27→      { threshold }
    28→    28→    );
    29→    29→    obs.observe(el);
    30→    30→    return () => obs.disconnect();
    31→    31→  }, [threshold]);
    32→    32→  return { ref, inView };
    33→    33→}

    41→    41→function ScrollProgress() {
    42→    42→  const [progress, setProgress] = useState(0);
    43→    43→  useEffect(() => {
    44→    44→    const onScroll = () => {
    45→    45→      const scrollTop = window.scrollY;
    46→    46→      const docHeight = document.documentElement.scrollHeight - window.innerHeight;
    47→    47→      setProgress(docHeight > 0 ? Math.min(100, (scrollTop / docHeight) * 100) : 0);
    48→    48→    };
    49→    49→    onScroll();
    50→    50→    window.addEventListener('scroll', onScroll, { passive: true });
    51→    51→    return () => window.removeEventListener('scroll', onScroll);
    52→    52→  }, []);
    53→    53→  return (
    54→    54→    <div
    55→    55→      className="scroll-progress"
    56→    56→      style={{ width: `${progress}%`, opacity: progress > 0.5 ? 1 : 0 }}
    57→    57→      aria-hidden="true"
    58→    58→    />
    59→    59→  );
    60→    60→}

   171→   171→function FadeIn({ children, className = '', delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   172→   172→  const { ref, inView } = useInView();

   177→   177→      style={{
   178→   178→        opacity: inView ? 1 : 0,
   179→   179→        transform: inView ? 'translateY(0)' : 'translateY(32px)',
   180→   180→        transitionDelay: `${delay}ms`,
   181→   181→      }}
   182→   182→    >
   183→   183→      {children}
   184→   184→    </div>
   185→   185→  );
   186→   186→}

   188→   188→/* ─── Typewriter text effect ─── */
   189→   189→function TypewriterText({ text, className = '' }: { text: string; className?: string }) {
   190→   190→  const [displayed, setDisplayed] = useState('');
   191→   191→  const [done, setDone] = useState(false);
   192→   192→
   193→   193→  useEffect(() => {
   194→   194→    let i = 0;
   195→   195→    const interval = setInterval(() => {
   196→   196→      i++;
   197→   197→      if (i <= text.length) {
   198→   198→        setDisplayed(text.slice(0, i));
   199→   199→      } else {
   200→   200→        setDone(true);
   201→   201→        clearInterval(interval);
   202→   202→      }
   203→   203→    }, 65);
   204→   204→    return () => clearInterval(interval);
   205→   205→  }, [text]);
   206→   206→
   207→   207→  return (
   208→   208→    <span className={className}>
   209→   209→      {displayed}
   210→   210→      {!done && <span className="typewriter-cursor" aria-hidden="true" />}
   211→   211→    </span>
   212→   212→  );
   213→   213→}

   216→   216→function BackToTop() {

   218→   218→

   223→   223→      setVisible(docHeight > 0 && scrollTop / docHeight > 0.5);
   224→   224→    };
   225→   225→    onScroll();
   226→   226→    window.addEventListener('scroll', onScroll, { passive: true });
   227→   227→    return () => window.removeEventListener('scroll', onScroll);
   228→   228→  }, []);
   229→   229→
   230→   230→  if (!visible) return null;
   231→   231→
   232→   232→  return (
   233→   233→    <button
   234→   234→      className="back-to-top-btn"
   235→   235→      onClick={() => window.scrollTo({ top: 0, behavior: 'smooth' })}
