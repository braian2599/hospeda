   154→function WeatherIcon({ code }: { code: number }) {
   155→  if (code === 0) return <Sun className="w-5 h-5 text-[#F59E0B]" />;
   156→  if (code >= 1 && code <= 3) return <CloudSun className="w-5 h-5 text-[#94A3B8]" />;
   157→  if (code >= 45 && code <= 48) return <CloudFog className="w-5 h-5 text-[#94A3B8]" />;
   158→  if (code >= 51 && code <= 55) return <CloudDrizzle className="w-5 h-5 text-[#3B82F6]" />;
   159→  if (code >= 56 && code <= 57) return <CloudDrizzle className="w-5 h-5 text-[#3B82F6]" />;
   160→  if (code >= 61 && code <= 67) return <CloudRain className="w-5 h-5 text-[#3B82F6]" />;
   161→  if (code >= 71 && code <= 77) return <CloudSnow className="w-5 h-5 text-[#3B82F6]" />;
   162→  if (code >= 80 && code <= 82) return <CloudLightning className="w-5 h-5 text-[#92400E]" />;
   163→  if (code >= 95) return <CloudLightning className="w-5 h-5 text-[#5B21B6]" />;
   164→  return <Thermometer className="w-5 h-5 text-[#94A3B8]" />;
   165→}

   169→function LiveClockWeather() {

   171→  const [weather, setWeather] = useState<{ temp: number; code: number } | null>(null);
   172→
   173→  useEffect(() => {
   174→    const update = () => {
   175→      setTime(new Date().toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit', second: '2-digit' }));
   176→    };
   177→    update();
   178→    const id = setInterval(update, 1000);
   179→    return () => clearInterval(id);
   180→  }, []);
   181→
   182→  // TODO: Hacer coordenadas configurables desde la configuración del hotel
   183→  useEffect(() => {
   184→    const fetchWeather = async () => {
   185→      try {
   186→        const res = await fetch('https://api.open-meteo.com/v1/forecast?latitude=-27.65&longitude=-67.03&current_weather=true&timezone=America/Argentina/Catamarca');
   187→        if (!res.ok) return;
   188→        const data = await res.json();
   189→        setWeather({ temp: data.current_weather.temperature, code: data.current_weather.weathercode });
   190→      } catch { /* ignore */ }
   191→    };

   194→    return () => clearInterval(id);

   198→    <div className="flex items-center gap-4">
   199→      {weather && (

   362→    const dias: string[] = [];

   364→      const d = new Date(hoy);
   365→      d.setDate(d.getDate() - i);
   366→      dias.push(toLocalDateStr(d));
   367→    }

   369→      const ing = pagos.filter(p => p.fecha === dia).reduce((s, p) => s + p.monto, 0);
   370→      const egr = gastos.filter(g => g.fecha === dia).reduce((s, g) => s + g.monto, 0);
   371→      const d = new Date(dia + 'T12:00:00');
   372→      return { dia: d.toLocaleDateString('es-AR', { weekday: 'short', day: 'numeric' }).replace(/^\w/, c => c.toUpperCase()), ing, egr, fullDate: formatearFecha(dia) };
   373→    });

   376→  const maxVal = Math.max(...datos.map(d => Math.max(d.ing, d.egr)), 1);

   455→    const out: OccupancyDayData[] = [];
   456→    for (let i = 13; i >= 0; i--) {
   457→      const d = new Date(hoy);
   458→      d.setDate(d.getDate() - i);
   459→      const diaStr = toLocalDateStr(d);
   460→      // Count unique rooms occupied on this day (checkin <= day < checkout, excluding cancelled)
   461→      const ocupadasSet = new Set<string>();
   462→      reservas.forEach(r => {
   463→        if (r.estado === 'Cancelada') return;
   464→        if (r.checkin <= diaStr && r.checkout > diaStr) {
   465→          ocupadasSet.add(r.habitacion);
   466→        }
   467→      });
   468→      const ocupadas = ocupadasSet.size;
   469→      const porcentaje = totalHabitaciones > 0 ? Math.round((ocupadas / totalHabitaciones) * 100) : 0;
   470→      const weekday = NOMBRES_DIAS[d.getDay()];
   471→      out.push({
   472→        fecha: diaStr,
   473→        ocupadas,
   474→        porcentaje,
   475→        label: `${weekday} ${d.getDate()}`,
   476→      });
   477→    }
   478→    return out;
   479→  }, [reservas, totalHabitaciones]);
   480→
   481→  const hoyData = datos[datos.length - 1];
   482→  const ayerData = datos[datos.length - 2];
   483→  const ultimos7 = datos.slice(-7);
   484→  const promedio7d = ultimos7.length > 0 ? ultimos7.reduce((s, d) => s + d.porcentaje, 0) / ultimos7.length : 0;
   485→
   486→  const diffVsAyer = hoyData && ayerData ? hoyData.porcentaje - ayerData.porcentaje : 0;
   487→  const trendUp = diffVsAyer > 0;
   488→  const trendDown = diffVsAyer < 0;
   489→  const trendIcon = trendUp ? '\u2191' : trendDown ? '\u2193' : '\u2192';
   490→  const trendColor = trendUp ? 'text-[#166534]' : trendDown ? 'text-[#991B1B]' : 'text-muted-foreground';
   491→
   492→  const renderTooltip = ({ active, payload }: TooltipProps<number, string>) => {
   493→    if (!active || !payload || !payload.length) return null;
   494→    const d = payload[0]?.payload as OccupancyDayData | undefined;
   495→    if (!d) return null;
   496→    return (
   497→      <div className="bg-[#0F2B28] text-white px-3 py-2 rounded-lg shadow-xl text-xs border border-[#059669]/30">
   498→        <p className="font-semibold text-[#34d399]">{formatearFecha(d.fecha)}</p>
   499→        <p className="mt-0.5">Ocupación: <span className="font-bold">{d.porcentaje}%</span></p>
   500→        <p className="text-white/70">{d.ocupadas} de {totalHabitaciones} hab.</p>
   501→      </div>
   502→    );
   503→  };
   504→
   505→  return (
   506→    <Card className={cn('transition-opacity duration-500', mounted ? 'opacity-100' : 'opacity-0')}>
   507→      <CardHeader className="pb-3">
   508→        <div className="flex items-start justify-between gap-3">
   509→          <div className="min-w-0">
   510→            <CardTitle className="text-base flex items-center gap-2">
   511→              <TrendingUp className="w-4 h-4 text-[#059669]" />
   512→              Tendencia de Ocupación (14 días)
   513→            </CardTitle>
   514→            <p className="text-xs text-muted-foreground mt-1">
   515→              Hoy: <span className="font-semibold text-[#0F2B28]">{hoyData?.porcentaje ?? 0}%</span>
   516→              <span className="mx-1.5 text-muted-foreground/50">·</span>
   517→              Promedio 7d: <span className="font-semibold text-[#0F2B28]">{Math.round(promedio7d)}%</span>
   518→            </p>
   519→          </div>
   520→          {ayerData && (
   521→            <div className={cn('flex items-center gap-1 text-xs font-semibold px-2 py-1 rounded-md bg-muted/50 whitespace-nowrap', trendColor)}>
   522→              <span>{trendIcon}</span>
   523→              <span>{Math.abs(diffVsAyer)}%</span>
   524→              <span className="text-muted-foreground font-normal text-[10px]">vs ayer</span>
   525→            </div>
   526→          )}
   527→        </div>
   528→      </CardHeader>
   529→      <CardContent>
   530→        <div style={{ height: 220, minHeight: 200 }} className="w-full">
   531→          <ResponsiveContainer width="100%" height="100%">
   532→            <AreaChart data={datos} margin={{ top: 5, right: 10, bottom: 0, left: -10 }}>
   533→              <defs>
   534→                <linearGradient id="ocupGradient" x1="0" y1="0" x2="0" y2="1">
   535→                  <stop offset="0%" stopColor="#34d399" stopOpacity={0.85} />
   536→                  <stop offset="100%" stopColor="#059669" stopOpacity={0.15} />
   537→                </linearGradient>
   538→              </defs>
   539→              <XAxis
   540→                dataKey="label"
   541→                tick={{ fontSize: 10, fill: '#94A3B8' }}
   542→                tickLine={false}
   543→                axisLine={{ stroke: '#E2E8F0' }}
   544→                interval={0}
   545→              />
   546→              <YAxis
   547→                domain={[0, 100]}
   548→                tick={{ fontSize: 10, fill: '#94A3B8' }}
   549→                tickLine={false}
   550→                axisLine={false}
   551→                tickFormatter={(v) => `${v}%`}
   552→                width={36}
   553→              />
   554→              <RechartsTooltip
   555→                content={renderTooltip}
   556→                cursor={{ stroke: '#059669', strokeOpacity: 0.3, strokeWidth: 1 }}
   557→              />
   558→              <Area
   559→                type="monotone"
   560→                dataKey="porcentaje"
   561→                stroke="#059669"
   562→                strokeWidth={2.5}
   563→                fill="url(#ocupGradient)"
   564→                dot={{ r: 2.5, fill: '#059669', strokeWidth: 0 }}
   565→                activeDot={{ r: 5, fill: '#0F2B28', stroke: '#34d399', strokeWidth: 2 }}
   566→                isAnimationActive={mounted}
   567→                animationDuration={800}
   568→              />
   569→            </AreaChart>
   570→          </ResponsiveContainer>
   571→        </div>
   572→      </CardContent>
   573→    </Card>
   574→  );
   575→}

   892→function calcularBarra(res: GanttReserva, colIdx: Record<string, number>, columnas: string[], DIAS: number) {
   893→  const COL_PCT = 100 / DIAS;
   894→  const MITAD_COL_PCT = COL_PCT / 2;
   895→  let startCol = Math.max(colIdx[res.checkin] ?? 0, 0);
   896→  let endCol = Math.min(colIdx[res.checkout] !== undefined ? colIdx[res.checkout] : DIAS - 1, DIAS - 1);
   897→
   898→  const esCheckinTarde = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckin && res.horaCheckin.getHours() >= 12;
   899→  const esCheckoutManana = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckout && res.horaCheckout.getHours() < 12;
   900→
   901→  let leftPct: number, widthPct: number;
   902→
   903→  if (esCheckinTarde && startCol >= 0 && !(res.checkin < columnas[0])) {
   904→    leftPct = startCol * COL_PCT + MITAD_COL_PCT;
   905→  } else {
   906→    leftPct = startCol * COL_PCT;
   907→  }
   908→
   909→  if (esCheckoutManana && endCol >= 0 && !(res.checkout > columnas[DIAS - 1])) {
   910→    widthPct = ((endCol - startCol) * COL_PCT) + (esCheckinTarde ? MITAD_COL_PCT : COL_PCT) - (esCheckoutManana ? MITAD_COL_PCT : 0);
   911→  } else {
   912→    widthPct = (endCol - startCol + 1) * COL_PCT;
   913→  }
   914→
   915→  widthPct = Math.max(widthPct, MITAD_COL_PCT);
   916→  return { leftPct, widthPct };
   917→}
