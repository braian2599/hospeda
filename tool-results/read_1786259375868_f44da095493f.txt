   108→   107→/** Priority of a cleaning task based on how long ago the room was checked out. */
   109→   108→type Priority = 'high' | 'medium' | 'low';
   110→   109→
   111→   110→function computePriority(habitacion: string, reservas: Reserva[], now: number): Priority {
   112→   111→  // Find the most recent checkout for this room (any time)
   113→   112→  const checkouts = reservas
   114→   113→    .filter(r => r.habitacion === habitacion && (r.estado === 'Check-Out realizado' || r.estado === 'Check-In realizado'))
   115→   114→    .map(r => {
   116→   115→      // Combine checkout date + horaCheckout if available
   117→   116→      const base = r.checkout ? new Date(r.checkout + 'T12:00:00') : null;
   118→   117→      if (base && r.horaCheckout) {
   119→   118→        const h, mm] = r.horaCheckout.split(':').map(n => parseInt(n, 10));
   120→   119→        if (!isNaN(hh)) base.setHours(hh, isNaN(mm) ? 0 : mm, 0, 0);
   121→   120→      }
   122→   121→      return base ? base.getTime() : 0;
   123→   122→    })
   124→   123→    .filter(t => t > 0)
   125→   124→    .sort((a, b) => b - a);
   126→   125→  if (checkouts.length === 0) return 'low';
   127→   126→  const lastCheckout = checkouts[0];
   128→   127→  const diffHours = (now - lastCheckout) / 3_600_000;
   129→   128→  if (diffHours >= 2) return 'high';
   130→   129→  if (diffHours >= 1) return 'medium';
   131→   130→  return 'low';
   132→   131→}
