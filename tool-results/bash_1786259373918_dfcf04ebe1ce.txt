   107→/** Priority of a cleaning task based on how long ago the room was checked out. */
   108→type Priority = 'high' | 'medium' | 'low';
   109→
   110→function computePriority(habitacion: string, reservas: Reserva[], now: number): Priority {
   111→  // Find the most recent checkout for this room (any time)
   112→  const checkouts = reservas
   113→    .filter(r => r.habitacion === habitacion && (r.estado === 'Check-Out realizado' || r.estado === 'Check-In realizado'))
   114→    .map(r => {
   115→      // Combine checkout date + horaCheckout if available
   116→      const base = r.checkout ? new Date(r.checkout + 'T12:00:00') : null;
   117→      if (base && r.horaCheckout) {
   118→        const h, mm] = r.horaCheckout.split(':').map(n => parseInt(n, 10));
   119→        if (!isNaN(hh)) base.setHours(hh, isNaN(mm) ? 0 : mm, 0, 0);
   120→      }
   121→      return base ? base.getTime() : 0;
   122→    })
   123→    .filter(t => t > 0)
   124→    .sort((a, b) => b - a);
   125→  if (checkouts.length === 0) return 'low';
   126→  const lastCheckout = checkouts[0];
   127→  const diffHours = (now - lastCheckout) / 3_600_000;
   128→  if (diffHours >= 2) return 'high';
   129→  if (diffHours >= 1) return 'medium';
   130→  return 'low';
   131→}
