   355→   355→function GraficoIngresosEgresos({ pagos, gastos }: { pagos: { fecha: string; monto: number }[]; gastos: { fecha: string; monto: number }[] }) {
   356→   356→  const [mounted, setMounted] = useState(false);
   357→   357→  useEffect(() => { const t = setTimeout(() => setMounted(true), 50); return () => clearTimeout(t); }, []);
   358→   358→
   359→   359→  const datos = useMemo(() => {
   360→   360→    const hoy = new Date();
   361→   361→    const dias: string[] = [];
   362→   362→    for (let i = 6; i >= 0; i--) {
   363→   363→      const d = new Date(hoy);
   364→   364→      d.setDate(d.getDate() - i);
   365→   365→      dias.push(toLocalDateStr(d));
   366→   366→    }
   367→   367→    return dias.map(dia => {
   368→   368→      const ing = pagos.filter(p => p.fecha === dia).reduce((s, p) => s + p.monto, 0);
   369→   369→      const egr = gastos.filter(g => g.fecha === dia).reduce((s, g) => s + g.monto, 0);
   370→   370→      const d = new Date(dia + 'T12:00:00');
   371→   371→      return { dia: d.toLocaleDateString('es-AR', { weekday: 'short', day: 'numeric' }).replace(/^\w/, c => c.toUpperCase()), ing, egr, fullDate: formatearFecha(dia) };
   372→   372→    });
   373→   373→  }, [pagos, gastos]);
   374→   374→
   375→   375→  const maxVal = Math.max(...datos.map(d => Math.max(d.ing, d.egr)), 1);
   376→   376→
   377→   377→  return (
   378→   378→    <Card>
   379→   379→      <CardHeader className="pb-3">
   380→   380→        <CardTitle className="text-base flex items-center gap-2">
   381→   381→          <BarChart3 className="w-4 h-4 text-[#3B82F6]" />
   382→   382→          Ingresos vs Egresos (7 días)
   383→   383→        </CardTitle>
   384→   384→      </CardHeader>

   891→   891→function calcularBarra(res: GanttReserva, colIdx: Record<string, number>, columnas: string[], DIAS: number) {
   892→   892→  const COL_PCT = 100 / DIAS;
   893→   893→  const MITAD_COL_PCT = COL_PCT / 2;
   894→   894→  let startCol = Math.max(colIdx[res.checkin] ?? 0, 0);
   895→   895→  let endCol = Math.min(colIdx[res.checkout] !== undefined ? colIdx[res.checkout] : DIAS - 1, DIAS - 1);
   896→   896→
   897→   897→  const esCheckinTarde = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckin && res.horaCheckin.getHours() >= 12;
   898→   898→  const esCheckoutManana = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckout && res.horaCheckout.getHours() < 12;
   899→   899→
   900→   900→  let leftPct: number, widthPct: number;
   901→   901→
   902→   902→  if (esCheckinTarde && startCol >= 0 && !(res.checkin < columnas[0])) {
   903→   903→    leftPct = startCol * COL_PCT + MITAD_COL_PCT;
   904→   904→  } else {
   905→   905→    leftPct = startCol * COL_PCT;
   906→   906→  }
   907→   907→
   908→   908→  if (esCheckoutManana && endCol >= 0 && !(res.checkout > columnas[DIAS - 1])) {
   909→   909→    widthPct = ((endCol - startCol) * COL_PCT) + (esCheckinTarde ? MITAD_COL_PCT : COL_PCT) - (esCheckoutManana ? MITAD_COL_PCT : 0);
   910→   910→  } else {
   911→   911→    widthPct = (endCol - startCol + 1) * COL_PCT;
   912→   912→  }
   913→   913→
   914→   914→  widthPct = Math.max(widthPct, MITAD_COL_PCT);
   915→   915→  return { leftPct, widthPct };
   916→   916→}
