    37→    37→/** Get initials from a full name (first letter of first word + first letter of last word) */
    38→    38→const getInitials = (name: string): string => {
    39→    39→  const parts = name.trim().split(/\s+/).filter(Boolean);
    40→    40→  if (parts.length === 0) return '?';
    41→    41→  if (parts.length === 1) return parts[0][0].toUpperCase();
    42→    42→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
    43→    43→};
    44→    44→
    45→    45→/** Format a date string as relative time in Spanish (short form) */
    46→    46→const formatRelativeTime = (dateStr: string): string => {
    47→    47→  if (!dateStr) return '';
    48→    48→  const d = new Date(dateStr);
    49→    49→  const now = new Date();
    50→    50→  const diffMs = now.getTime() - d.getTime();
    51→    51→  const diffMins = Math.floor(diffMs / 60000);
    52→    52→  const diffHrs = Math.floor(diffMs / 3600000);
    53→    53→  const diffDays = Math.floor(diffMs / 86400000);
    54→    54→
    55→    55→  if (diffMins < 1) return 'recién';
    56→    56→  if (diffMins < 60) return `hace ${diffMins} min`;
    57→    57→  if (diffHrs < 24) return `hace ${diffHrs} h`;
    58→    58→  if (diffDays === 1) return 'ayer';
    59→    59→  if (diffDays < 7) return `hace ${diffDays} días`;
    60→    60→  if (diffDays < 30) return `hace ${Math.floor(diffDays / 7)} sem`;
    61→    61→  if (diffDays < 365) return `hace ${Math.floor(diffDays / 30)} meses`;
    62→    62→  return `hace ${Math.floor(diffDays / 365)} años`;
    63→    63→};
    64→    64→
    65→    65→/** Generate a random password: 10 chars, at least one uppercase + one digit */
    66→    66→const randomPassword = (): string => {
    67→    67→  const upper = 'ABCDEFGHJKLMNPQRSTUVWXYZ';
    68→    68→  const lower = 'abcdefghjkmnpqrstuvwxyz';
    69→    69→  const digits = '23456789';
    70→    70→  const all = upper + lower + digits;
    71→    71→  const pick = (s: string) => s[Math.floor(Math.random() * s.length)];
    72→    72→  const chars: string[] = [pick(upper), pick(digits)];
    73→    73→  for (let i = 2; i < 10; i++) chars.push(pick(all));
    74→    74→  // Fisher-Yates shuffle to avoid predictable positions
    75→    75→  for (let i = chars.length - 1; i > 0; i--) {
    76→    76→    const j = Math.floor(Math.random() * (i + 1));
    77→    77→    [chars[i], chars[j]] = [chars[j], chars[i]];
    78→    78→  }
    79→    79→  return chars.join('');
    80→    80→};
