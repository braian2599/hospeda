   114→/** Priority of a cleaning task based on how long ago the room was checked out. */
   115→type Priority = 'high' | 'medium' | 'low';
   116→
   117→function getLastCheckoutMs(habitacion: string, reservas: Reserva[]): number {
   118→  const checkouts = reservas
   119→    .filter(r => r.habitacion === habitacion && (r.estado === 'Check-Out realizado' || r.estado === 'Check-In realizado'))
   120→    .map(r => {
   121→      const base = r.checkout ? new Date(r.checkout + 'T12:00:00') : null;
   122→      if (base && r.horaCheckout) {
   123→        const [hh, mm] = r.horaCheckout.split(':').map(n => parseInt(n, 10));
   124→        if (!isNaN(hh)) base.setHours(hh, isNaN(mm) ? 0 : mm, 0, 0);
   125→      }
   126→      return base ? base.getTime() : 0;
   127→    })
   128→    .filter(t => t > 0)
   129→    .sort((a, b) => b - a);
   130→  return checkouts[0] || 0;
   131→}
   132→
   133→function computePriority(habitacion: string, reservas: Reserva[], now: number): Priority {
   134→  const lastCheckout = getLastCheckoutMs(habitacion, reservas);
   135→  if (lastCheckout === 0) return 'low';
   136→  const diffHours = (now - lastCheckout) / 3_600_000;
   137→  if (diffHours >= 2) return 'high';
   138→  if (diffHours >= 1) return 'medium';
   139→  return 'low';
   140→}

   180→/** Format a duration in ms as "12m 30s" / "1h 05m" / "45s". */
   181→function formatDuration(ms: number): string {
   182→  if (ms < 0) ms = 0;
   183→  const totalSec = Math.floor(ms / 1000);
   184→  const h = Math.floor(totalSec / 3600);
   185→  const m = Math.floor((totalSec % 3600) / 60);
   186→  const s = totalSec % 60;
   187→  if (h > 0) return `${h}h ${String(m).padStart(2, '0')}m`;
   188→  if (m > 0) return `${m}m ${String(s).padStart(2, '0')}s`;
   189→  return `${s}s`;
   190→}

   192→/** Get initials from a full name (max 2 chars). */
   193→function getInitials(name: string): string {
   194→  const parts = name.trim().split(/\s+/).filter(Boolean);
   195→  if (parts.length === 0) return '?';
   196→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   197→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   198→}
