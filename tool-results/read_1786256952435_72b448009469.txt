    35→const formatFecha = (f: string) => {
    36→  if (!f) return '—';
    37→  const d = new Date(f + 'T12:00:00');
    38→  return d.toLocaleDateString('es-AR');
    39→};

    41→const formatFechaHora = (f: string) => {
    42→  if (!f) return '—';
    43→  // Si la cadena no incluye hora, usar mediodía local para evitar desfase UTC
    44→  const d = new Date(f.includes('T') || f.includes(' ') ? f : f + 'T12:00:00');
    45→  return d.toLocaleDateString('es-AR') + ' ' + d.toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit' });
    46→};

    58→interface KpiProps {
    59→  label: string;
    60→  value: string | number;
    61→  icon: React.ReactNode;
    62→  color: string;
    63→  bgColor: string;
    64→  tooltip?: string;
    65→  subtext?: string;
    66→  trend?: { value: number; label: string };
    67→  onClick?: () => void;
    68→}
    69→
    70→function KpiCard({ label, value, icon, color, bgColor, tooltip, subtext, trend, onClick }: KpiProps) {
    71→  const card = (
    72→    <Card
    73→      className={`relative overflow-hidden transition-all duration-200 hover:shadow-md ${onClick ? 'cursor-pointer group' : ''}`}
    74→      onClick={onClick}
    75→    >
    76→      <CardContent className="p-3 sm:p-4">
    77→        <div className="flex items-start justify-between gap-2 sm:gap-3">
    78→          <div className="flex-1 min-w-0 text-center">
    79→            <p className="text-[10px] sm:text-[11px] font-medium uppercase tracking-wider text-muted-foreground mb-1 sm:mb-1.5">{label}</p>
    80→            <p className={`text-lg sm:text-2xl font-bold tracking-tight ${color} truncate`}>{value}</p>
    81→            {trend && (
    82→              <div className="flex items-center justify-center gap-1 mt-1 sm:mt-1.5">
    83→                {trend.value > 0 ? (
    84→                  <ArrowUpRight className="w-3 h-3 sm:w-3.5 sm:h-3.5 text-[#059669]" />
    85→                ) : trend.value < 0 ? (
    86→                  <ArrowDownRight className="w-3 h-3 sm:w-3.5 sm:h-3.5 text-[#EF4444]" />
    87→                ) : (
    88→                  <Minus className="w-3 h-3 sm:w-3.5 sm:h-3.5 text-muted-foreground" />
    89→                )}
    90→                <span className={`text-[11px] sm:text-xs font-medium ${trend.value > 0 ? 'text-[#059669]' : trend.value < 0 ? 'text-[#EF4444]' : 'text-muted-foreground'}`}>
    91→                  {Math.abs(trend.value)}%
    92→                </span>
    93→                <span className="text-[11px] sm:text-xs text-muted-foreground hidden sm:inline">{trend.label}</span>
    94→              </div>
    95→            )}
    96→            {subtext && !trend && <p className="text-[11px] sm:text-xs text-muted-foreground mt-1 truncate hidden sm:block">{subtext}</p>}
    97→          </div>
    98→          <div className={`shrink-0 w-8 h-8 sm:w-10 sm:h-10 rounded-lg sm:rounded-xl flex items-center justify-center ${bgColor}`}>
    99→            <div className="sm:scale-100 scale-90">{icon}</div>
   100→          </div>
   101→        </div>
   102→      </CardContent>
   103→      {onClick && (
   104→        <div className="absolute inset-x-0 bottom-0 h-0.5 bg-primary opacity-0 group-hover:opacity-100 transition-opacity" />
   105→      )}
   106→    </Card>
   107→  );
   108→
   109→  if (tooltip) {
   110→    return (
   111→      <TooltipProvider delayDuration={300}>
   112→        <Tooltip>
   113→          <TooltipTrigger asChild>{card}</TooltipTrigger>
   114→          <TooltipContent side="bottom" className="text-xs max-w-[220px]">{tooltip}</TooltipContent>
   115→        </Tooltip>
   116→      </TooltipProvider>
   117→    );
   118→  }
   119→  return card;
   120→}

   134→function ProgressKpi({ label, value, max, color = 'bg-primary', suffix = '%', description }: { label: string; value: number; max: number; color?: string; suffix?: string; description?: string }) {
   135→  const pct = max > 0 ? Math.round((value / max) * 100) : 0;
   136→  return (
   137→    <Card>
   138→      <CardContent className="p-3 sm:p-4">
   139→        <div className="flex items-center justify-between mb-1.5 sm:mb-2">
   140→          <p className="text-[10px] sm:text-[11px] font-medium uppercase tracking-wider text-muted-foreground">{label}</p>
   141→          <span className="text-base sm:text-lg font-bold">{value}{suffix}</span>
   142→        </div>
   143→        <Progress value={pct} className="h-2 sm:h-2.5" />
   144→        <p className="text-[11px] sm:text-xs text-muted-foreground mt-1 sm:mt-1.5 text-center">{description || `${pct}%`}</p>
   145→      </CardContent>
   146→    </Card>
   147→  );
   148→}

   192→          </div>

   195→          <div className="flex items-center gap-1 shrink-0">

   197→              <ArrowUpRight className="w-4 h-4 text-[#059669]" />

   204→              {trend.pct}%
   205→            </span>
   206→          </div>
   207→        )}
   208→      </CardContent>
   209→    </Card>
   210→  );
   211→}

   305→  const parseDateRange = useMemo(() => ({
   306→    from: new Date(desde + 'T00:00:00'),

   313→    const d = new Date(hasta + 'T12:00:00');

   350→      const fg = new Date(g.fecha + 'T12:00:00');

   352→    });
   353→  }, [gastos, parseDateRange]);
   354→
   355→  const gastosPrevPeriodo = useMemo(() => {
   356→    const d = new Date(prevDesde + 'T00:00:00');
   357→    const h = new Date(prevHasta + 'T23:59:59');

   375→  // Reservas que se superponen con el periodo (para noches vendidas y ocupación)

   414→    if (previous === 0) return current > 0 ? 100 : 0;
   415→    return Math.round(((current - previous) / previous) * 100);
   416→  };
   417→
   418→  // Noches vendidas en el periodo (conteo basado en noches, no en días calendario)

   420→  // para que el recorte de reservas que se extienden más allá del periodo
   421→  // no produzca subconteo por redondeo.
   422→  const nochesVendidas = useMemo(() => {
   423→    return reservasSuperpuestas.reduce((s, r) => {
   424→      const ci = new Date(r.checkin + 'T12:00:00');
   425→      const co = new Date(r.checkout + 'T12:00:00');
   426→      const inicio = ci < parseDateRange.from ? parseDateRange.from : ci;
   427→      const fin = co > toExclusive ? toExclusive : co;
   428→      return s + Math.max(0, Math.round((fin.getTime() - inicio.getTime()) / (1000 * 60 * 60 * 24)));
   429→    }, 0);
   430→  }, [reservasSuperpuestas, parseDateRange, toExclusive]);
   431→
