   126→   126→/** Busca el rango que corresponde a la cantidad de personas dada. */
   127→   127→function encontrarRango(rangos: RangoPrecio[], personas: number): RangoPrecio | undefined {
   128→   128→  // Buscar rango donde minPersonas <= personas <= maxPersonas
   129→   129→  for (const r of rangos) {
   130→   130→    if (personas >= r.minPersonas && (r.maxPersonas === null || personas <= r.maxPersonas)) {
   131→   131→      return r;
   132→   132→    }
   133→   133→  }
   134→   134→  // Fallback: usar el último rango (el que tiene maxPersonas null, o el de mayor cobertura)
   135→   135→  return rangos.length > 0 ? rangos[rangos.length - 1] : undefined;
   136→   136→}
   137→   137→
   138→   138→/**
   139→   139→ * Convierte datos de tarifa desde la BD (puede ser formato viejo o nuevo) a RangoPrecio[].
   140→   140→ * Formato viejo: { "1": 35000, "2": 30000, ... }
   141→   141→ * Formato nuevo: { "rangos": [{ minPersonas, maxPersonas, precio }] }
   142→   142→ */
   143→   143→function normalizarRangos(preciosDb: any): RangoPrecio[] {
   144→   144→  if (!preciosDb) return [];
   145→   145→  // Nuevo formato: tiene la propiedad "rangos" como array
   146→   146→  if (preciosDb.rangos && Array.isArray(preciosDb.rangos)) {
   147→   147→    return preciosDb.rangos.map((r: any) => ({
   148→   148→      minPersonas: Number(r.minPersonas) || 1,
   149→   149→      maxPersonas: r.maxPersonas != null ? Number(r.maxPersonas) : null,
   150→   150→      precio: Number(r.precio) || 0,
   151→   151→    }));
   152→   152→  }
   153→   153→  // Formato viejo: keys numéricos { "1": 35000, "2": 30000, ... }
   154→   154→  const keys = Object.keys(preciosDb).filter(k => !isNaN(Number(k)) && Number(k) >= 1);
   155→   155→  if (keys.length > 0) {
   156→   156→    return keys.map(k => ({
   157→   157→      minPersonas: Number(k),
   158→   158→      maxPersonas: Number(k),
   159→   159→      precio: Number(preciosDb[k]) || 0,
   160→   160→    }));
   161→   161→  }
   162→   162→  return [];
   163→   163→}

   165→   165→/**
   166→   166→ * Calcula las noches de cortesía según la modalidad.
   167→   167→ * Devuelve la cantidad de noches a descontar.
   168→   168→ */
   169→   169→function calcularNochesGratis(promociones: PromocionesTarifa, noches: number, checkin?: string): number {
   170→   170→  const nc = promociones.nochesCortesia;
   171→   171→  if (!nc?.activo || !nc.modalidad) return 0;
   172→   172→
   173→   173→  const mod = nc.modalidad;
   174→   174→  if (mod.tipo === 'cadaX') {
   175→   175→    const cada = mod.cada || 999;
   176→   176→    if (noches < cada) return 0;
   177→   177→    return Math.floor(noches / cada);
   178→   178→  }
   179→   179→  if (mod.tipo === 'aPartirDe') {
   180→   180→    if (noches < mod.minNoches) return 0;
   181→   181→    return mod.nochesGratis || 0;
   182→   182→  }
   183→   183→  if (mod.tipo === 'diaSemana' && checkin) {
   184→   184→    // Contar cuántas noches de la estadía caen en el día de la semana elegido
   185→   185→    const fechaInicio = new Date(checkin + 'T12:00:00');
   186→   186→    let count = 0;
   187→   187→    for (let i = 0; i < noches; i++) {
   188→   188→      const d = new Date(fechaInicio);
   189→   189→      d.setDate(d.getDate() + i);
   190→   190→      if (d.getDay() === mod.dia) count++;
   191→   191→    }
   192→   192→    return count;
   193→   193→  }
   194→   194→  return 0;
   195→   195→}

   197→   197→/**
   198→   198→ * Obtiene las promociones efectivas de una tarifa, migrando datos viejos (choferCortesia) si es necesario.
   199→   199→ */
   200→   200→function getPromocionesEfectivas(tarifa: TarifaPrecios): PromocionesTarifa {
   201→   201→  // Si tiene promociones nuevas, usar esas
   202→   202→  if (tarifa.promociones) return tarifa.promociones;
   203→   203→  // Migración: si tiene choferCortesia viejo, convertir a nuevo formato
   204→   204→  if (tarifa.choferCortesia) {
   205→   205→    return {
   206→   206→      acompananteSinCargo: {
   207→   207→        activo: true,
   208→   208→        etiqueta: 'Chofer de cortesía',
   209→   209→        habitacionAsignada: tarifa.habitacionChofer || undefined,
   210→   210→      },
   211→   211→    };
   212→   212→  }
   213→   213→  return {};
   214→   214→}

   220→   220→
   221→   221→function calcularTotalSegunTarifa(
   222→   222→  tarifas: Record<string, TarifaPrecios>,
   223→   223→  tipoTarifa: string,
   224→   224→  personas: number,
   225→   225→  noches: number,
   226→   226→  options?: CalcTarifaOptions
   227→   227→): number {
   228→   228→  // Buscar la tarifa (fallback a 'normal')
   229→   229→  const tarifa = tarifas[tipoTarifa] || tarifas['normal'];
   230→   230→  if (!tarifa || !tarifa.rangos || tarifa.rangos.length === 0) return 0;
   231→   231→
   232→   232→  const promociones = getPromocionesEfectivas(tarifa);
   233→   233→  const modo: ModoCobro = tarifa.modoCobro || 'porGrupo';
   234→   234→
   235→   235→  // ─── Noches de cortesía ───
   236→   236→  const nochesGratis = calcularNochesGratis(promociones, noches, options?.checkin);
   237→   237→  const nochesCobrables = Math.max(0, noches - nochesGratis);
   238→   238→
   239→   239→  // ─── Niños diferenciado ───
   240→   240→  const ninosDif = promociones.ninosDiferenciado;
   241→   241→  const cantNinos = (options?.ninos && ninosDif?.activo) ? options.ninos : 0;
   242→   242→  const adultos = Math.max(1, personas - cantNinos);
   243→   243→
   244→   244→  // ─── Acompañante sin cargo (no descuenta: el acompañante va a otra habitación) ───
   245→   245→  // Ya NO se resta 1 adulto. Todos los adultos de esta reserva pagan.
   246→   246→  // El acompañante se gestiona como una reserva separada con total=0.
   247→   247→
   248→   248→  // ─── Cálculo por modo ───
   249→   249→  if (modo === 'porCama') {
   250→   250→    // Modo porCama: el precio del rango es por persona por noche
   251→   251→    const rango = encontrarRango(tarifa.rangos, adultos);
   252→   252→    const precioCama = rango?.precio || tarifa.rangos[0]?.precio || 0;
   253→   253→    let total = nochesCobrables * adultos * precioCama;
   254→   254→    // Sumar niños
   255→   255→    if (cantNinos > 0 && ninosDif?.activo) {
   256→   256→      total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   257→   257→    }
   258→   258→    return total;
   259→   259→  }
   260→   260→
   261→   261→  if (modo === 'porHabitacion') {
   262→   262→    const precio = tarifa.rangos[0]?.precio || 0;
   263→   263→    let total = nochesCobrables * precio;
   264→   264→    // Sumar niños si aplica
   265→   265→    if (cantNinos > 0 && ninosDif?.activo) {
   266→   266→      total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   267→   267→    }
   268→   268→    return total;
   269→   269→  }
   270→   270→
   271→   271→  const rango = encontrarRango(tarifa.rangos, adultos);
   272→   272→  if (!rango) return 0;
   273→   273→
   274→   274→  if (modo === 'porPersona') {
   275→   275→    let total = nochesCobrables * adultos * rango.precio;
   276→   276→    // Sumar niños
   277→   277→    if (cantNinos > 0 && ninosDif?.activo) {
   278→   278→      total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   279→   279→    }
   280→   280→    return total;
   281→   281→  }
   282→   282→
   283→   283→  // porGrupo (default): el precio es el total del grupo
   284→   284→  let total = nochesCobrables * rango.precio;
   285→   285→  // Sumar niños
   286→   286→  if (cantNinos > 0 && ninosDif?.activo) {
   287→   287→    total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   288→   288→  }
   289→   289→  return total;
   290→   290→}
