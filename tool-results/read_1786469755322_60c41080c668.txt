    45→function mapDbReservaToStore(r: any, totalOverride?: number): Reserva {
    46→  // Map API estado to store estado
    47→  let estado: EstadoReserva = 'Confirmada';
    48→  if (r.estado === 'CheckIn_realizado') estado = 'Check-In realizado';
    49→  else if (r.estado === 'Checkout_realizado') estado = 'Check-Out realizado';
    50→  else if (r.estado === 'Cancelada') estado = 'Cancelada';
    51→
    52→  return {
    53→    id: r.id,
    54→    idCliente: r.clienteId || '',
    55→    huesped: r.huesped, dni: r.dni,
    56→    telefono: r.telefono || '', email: r.email || '',
    57→    domicilio: r.domicilio || '', habitacion: r.habitacion,
    58→    checkin: r.checkin?.split('T')[0] || r.checkin,
    59→    checkout: r.checkout?.split('T')[0] || r.checkout,
    60→    personas: r.personas, estadoPago: (r.estadoPago || 'Pendiente') as Reserva['estadoPago'],
    61→    notas: r.notas || '', estado,
    62→    tipoTarifa: r.tipoTarifa || undefined,
    63→    metodoPagoId: r.metodoPagoId || undefined,
    64→    cuotas: r.cuotas || undefined, recargoPorcentaje: r.recargoPorcentaje || undefined,
    65→    total: totalOverride !== undefined ? totalOverride : (r.total != null ? r.total / 100 : undefined),
    66→    ninos: r.ninos || undefined,
    67→    agencia: (r.agenciaNombre ? { nombre: r.agenciaNombre, convenio: r.agenciaConvenio, vendedor: undefined } : undefined) as any,
    68→    contactoEmergencia: (r.contactoEmergenciaNombre ? { nombre: r.contactoEmergenciaNombre, telefono: r.contactoEmergenciaTel || '' } : undefined) as any,

   100→    estado: 'abierta',
   101→    apertura: { montoInicial: turno.montoInicial / 100, empleado: turno.empleadoNombre, fecha: turno.fechaApertura },

   106→      movimientos: (t.movimientos || []).map((m: any) => ({ id: m.id, tipo: m.tipo as 'ingreso' | 'egreso', monto: m.monto / 100, descripcion: m.descripcion, metodo: m.metodo, empleado: m.empleadoNombre, fecha: m.fecha, gastoId: m.gastoId || null })),
   107→    })),
   108→  };
   109→}

   146→function nochesEntre(checkin: string, checkout: string): number {
   147→  const fechaIn = new Date(checkin + 'T12:00:00');
   148→  const fechaOut = new Date(checkout + 'T12:00:00');

   154→/** Busca el rango que corresponde a la cantidad de personas dada. */
   155→function encontrarRango(rangos: RangoPrecio[], personas: number): RangoPrecio | undefined {
   156→  // Buscar rango donde minPersonas <= personas <= maxPersonas
   157→  for (const r of rangos) {
   158→    if (personas >= r.minPersonas && (r.maxPersonas === null || personas <= r.maxPersonas)) {
   159→      return r;
   160→    }
   161→  }
   162→  // Fallback: usar el último rango (el que tiene maxPersonas null, o el de mayor cobertura)
   163→  return rangos.length > 0 ? rangos[rangos.length - 1] : undefined;
   164→}

   166→/**
   167→ * Convierte datos de tarifa desde la BD (puede ser formato viejo o nuevo) a RangoPrecio[].
   168→ * Formato viejo: { "1": 35000, "2": 30000, ... }
   169→ * Formato nuevo: { "rangos": [{ minPersonas, maxPersonas, precio }] }
   170→ */
   171→function normalizarRangos(preciosDb: any): RangoPrecio[] {
   172→  if (!preciosDb) return [];
   173→  // Nuevo formato: tiene la propiedad "rangos" como array
   174→  if (preciosDb.rangos && Array.isArray(preciosDb.rangos)) {
   175→    return preciosDb.rangos.map((r: any) => ({
   176→      minPersonas: Number(r.minPersonas) || 1,
   177→      maxPersonas: r.maxPersonas != null ? Number(r.maxPersonas) : null,
   178→      precio: Number(r.precio) || 0,
   179→    }));
   180→  }
   181→  // Formato viejo: keys numéricos { "1": 35000, "2": 30000, ... }
   182→  const keys = Object.keys(preciosDb).filter(k => !isNaN(Number(k)) && Number(k) >= 1);
   183→  if (keys.length > 0) {
   184→    return keys.map(k => ({
   185→      minPersonas: Number(k),
   186→      maxPersonas: Number(k),
   187→      precio: Number(preciosDb[k]) || 0,
   188→    }));
   189→  }
   190→  return [];
   191→}

   193→/**
   194→ * Calcula las noches de cortesía según la modalidad.
   195→ * Devuelve la cantidad de noches a descontar.
   196→ */
   197→function calcularNochesGratis(promociones: PromocionesTarifa, noches: number, checkin?: string): number {
   198→  const nc = promociones.nochesCortesia;
   199→  if (!nc?.activo || !nc.modalidad) return 0;
   200→
   201→  const mod = nc.modalidad;
   202→  if (mod.tipo === 'cadaX') {
   203→    const cada = mod.cada || 999;
   204→    if (noches < cada) return 0;
   205→    return Math.floor(noches / cada);
   206→  }
   207→  if (mod.tipo === 'aPartirDe') {
   208→    if (noches < mod.minNoches) return 0;
   209→    return mod.nochesGratis || 0;
   210→  }
   211→  if (mod.tipo === 'diaSemana' && checkin) {
   212→    // Contar cuántas noches de la estadía caen en el día de la semana elegido
   213→    const fechaInicio = new Date(checkin + 'T12:00:00');
   214→    let count = 0;
   215→    for (let i = 0; i < noches; i++) {
   216→      const d = new Date(fechaInicio);
   217→      d.setDate(d.getDate() + i);
   218→      if (d.getDay() === mod.dia) count++;
   219→    }
   220→    return count;
   221→  }
   222→  return 0;
   223→}

   225→/**
   226→ * Obtiene las promociones efectivas de una tarifa, migrando datos viejos (choferCortesia) si es necesario.
   227→ */
   228→function getPromocionesEfectivas(tarifa: TarifaPrecios): PromocionesTarifa {
   229→  // Si tiene promociones nuevas, usar esas
   230→  if (tarifa.promociones) return tarifa.promociones;
   231→  // Migración: si tiene choferCortesia viejo, convertir a nuevo formato
   232→  if (tarifa.choferCortesia) {
   233→    return {
   234→      acompananteSinCargo: {
   235→        activo: true,
   236→        etiqueta: 'Chofer de cortesía',
   237→        habitacionAsignada: tarifa.habitacionChofer || undefined,
   238→        cantidad: 1,
   239→      },
   240→    };
   241→  }
   242→  return {};
   243→}

   250→function calcularTotalSegunTarifa(
   251→  tarifas: Record<string, TarifaPrecios>,
   252→  tipoTarifa: string,
   253→  personas: number,
   254→  noches: number,
   255→  options?: CalcTarifaOptions
   256→): number {
   257→  // Buscar la tarifa (fallback a 'normal')
   258→  const tarifa = tarifas[tipoTarifa] || tarifas['normal'];
   259→  if (!tarifa || !tarifa.rangos || tarifa.rangos.length === 0) return 0;
   260→
   261→  const promociones = getPromocionesEfectivas(tarifa);
   262→  const modo: ModoCobro = tarifa.modoCobro || 'porGrupo';
   263→
   264→  // ─── Noches de cortesía ───
   265→  const nochesGratis = calcularNochesGratis(promociones, noches, options?.checkin);
   266→  const nochesCobrables = Math.max(0, noches - nochesGratis);
   267→
   268→  // ─── Niños diferenciado ───
   269→  const ninosDif = promociones.ninosDiferenciado;
   270→  const cantNinos = (options?.ninos && ninosDif?.activo) ? options.ninos : 0;
   271→  const adultos = Math.max(1, personas - cantNinos);
   272→
   273→  // ─── Acompañante sin cargo (no descuenta: el acompañante va a otra habitación) ───
   274→  // Ya NO se resta 1 adulto. Todos los adultos de esta reserva pagan.
   275→  // El acompañante se gestiona como una reserva separada con total=0.
   276→
   277→  // ─── Cálculo por modo ───
   278→  if (modo === 'porCama') {
   279→    // Modo porCama: el precio del rango es por persona por noche
   280→    const rango = encontrarRango(tarifa.rangos, adultos);
   281→    const precioCama = rango?.precio || tarifa.rangos[0]?.precio || 0;
   282→    let total = nochesCobrables * adultos * precioCama;
   283→    // Sumar niños
   284→    if (cantNinos > 0 && ninosDif?.activo) {
   285→      total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   286→    }
   287→    return total;
   288→  }
   289→
   290→  if (modo === 'porHabitacion') {
   291→    const precio = tarifa.rangos[0]?.precio || 0;
   292→    let total = nochesCobrables * precio;
   293→    // Sumar niños si aplica
   294→    if (cantNinos > 0 && ninosDif?.activo) {
   295→      total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   296→    }
   297→    return total;
   298→  }
   299→
   300→  const rango = encontrarRango(tarifa.rangos, adultos);
   301→  if (!rango) return 0;
   302→
   303→  if (modo === 'porPersona') {
   304→    let total = nochesCobrables * adultos * rango.precio;
   305→    // Sumar niños
   306→    if (cantNinos > 0 && ninosDif?.activo) {
   307→      total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   308→    }
   309→    return total;
   310→  }
   311→
   312→  // porGrupo (default): el precio es el total del grupo
   313→  let total = nochesCobrables * rango.precio;
   314→  // Sumar niños
   315→  if (cantNinos > 0 && ninosDif?.activo) {
   316→    total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   317→  }
   318→  return total;
   319→}

   811→          try {
   812→            const dbCliente = await api.clientes.create({ nombre: datos.huesped, dni: datos.dni, telefono: datos.telefono || '', email: datos.email });

   816→              set({ clientes: [...currentClientes, { id: dbCliente.id, nombre: dbCliente.nombre, dni: dbCliente.dni, telefono: dbCliente.telefono, email: dbCliente.email || '', preferencias: dbCliente.preferencias, historialEstadias: [], fechaCreacion: dbCliente.createdAt?.split('T')[0] || '' }] });
   817→            }
   818→          } catch (createErr: any) {
   819→            if (createErr?.status === 409) {
   820→              // Cliente ya existe — buscar por DNI en store local
   821→              const existing = get().clientes.find(c => c.dni === datos.dni);
   822→              clienteRealId = existing?.id;
   823→            }
   824→            // Si es otro error (no 409), no bloqueamos la reserva
