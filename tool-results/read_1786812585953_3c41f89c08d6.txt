    34→// ── DatePickerInline helper ──
    35→function DatePickerInline({
    36→  value,
    37→  onChange,
    38→  placeholder,
    39→  label,
    40→  disabled,
    41→}: {
    42→  value: string;
    43→  onChange: (val: string) => void;
    44→  placeholder: string;
    45→  label?: string;
    46→  disabled?: boolean;
    47→}) {
    48→  const [open, setOpen] = useState(false);
    49→  const selectedDate = value ? parseISO(value) : undefined;
    50→
    51→  const handleSelect = (date: Date | undefined) => {
    52→    if (date) {
    53→      const yyyy = date.getFullYear();
    54→      const mm = String(date.getMonth() + 1).padStart(2, '0');
    55→      const dd = String(date.getDate()).padStart(2, '0');
    56→      onChange(`${yyyy}-${mm}-${dd}`);
    57→    }
    58→    setOpen(false);
    59→  };
    60→
    61→  return (
    62→    <div className="grid gap-1.5">
    63→      {label && <Label>{label}</Label>}
    64→      <Popover open={open} onOpenChange={setOpen}>
    65→        <PopoverTrigger asChild>
    66→          <Button
    67→            variant="outline"
    68→            className={cn(
    69→              'w-full justify-start text-left font-normal h-9',
    70→              !value && 'text-muted-foreground'
    71→            )}
    72→            disabled={disabled}
    73→          >
    74→            {selectedDate
    75→              ? format(selectedDate, 'dd/MM/yyyy')
    76→              : placeholder}
    77→          </Button>
    78→        </PopoverTrigger>
    79→        <PopoverContent className="w-auto p-0" align="start">
    80→          <Calendar
    81→            mode="single"
    82→            selected={selectedDate}
    83→            onSelect={handleSelect}
    84→            autoFocus
    85→          />
    86→        </PopoverContent>
    87→      </Popover>
    88→    </div>
    89→  );
    90→}

    92→/** Get last checkout timestamp for a room (ms). */
    93→function getLastCheckoutMs(habitacion: string, reservas: Reserva[]): number {
    94→  const checkouts = reservas
    95→    .filter(r => r.habitacion === habitacion && (r.estado === 'Check-Out realizado' || r.estado === 'Check-In realizado'))
    96→    .map(r => {
    97→      const base = r.checkout ? new Date(r.checkout + 'T12:00:00') : null;
    98→      if (base && r.horaCheckout) {
    99→        const [hh, mm] = r.horaCheckout.split(':').map(n => parseInt(n, 10));
   100→        if (!isNaN(hh)) base.setHours(hh, isNaN(mm) ? 0 : mm, 0, 0);
   101→      }
   102→      return base ? base.getTime() : 0;
   103→    })
   104→    .filter(t => t > 0)
   105→    .sort((a, b) => b - a);
   106→  return checkouts[0] || 0;
   107→}

   109→/** Format time since a timestamp as "hace 30 min", "hace 2h", etc. */
   110→function formatTimeSince(ms: number): string {
   111→  const diff = Date.now() - ms;
   112→  if (diff < 60000) return 'hace un momento';
   113→  if (diff < 3600000) return `hace ${Math.floor(diff / 60000)} min`;
   114→  if (diff < 86400000) return `hace ${Math.floor(diff / 3600000)}h`;
   115→  return `hace ${Math.floor(diff / 86400000)}d`;
   116→}

   452→            <Button variant="ghost" size="sm">{showReportForm ? 'Cancelar' : 'Abrir formulario'}</Button>
   453→          </div>
   454→        </CardHeader>
   455→        {showReportForm && (
   456→          <CardContent className="space-y-3">
   457→            {repConfirm && reservasAfectadas > 0 && (
   458→              <div className="flex items-center gap-2 p-2.5 bg-amber-900/60 rounded-lg text-amber-300 text-sm">
   459→                <AlertTriangle className="w-4 h-4 shrink-0" />
   460→                <span>Esta habitación tiene <strong>{reservasAfectadas}</strong> reserva{reservasAfectadas !== 1 ? 's' : ''} activa{reservasAfectadas !== 1 ? 's' : ''} que serán canceladas.</span>
   461→                <Button size="sm" variant="ghost" className="h-6 w-6 p-0 ml-auto" onClick={() => setRepConfirm(false)}>
   462→                  <X className="w-3.5 h-3.5" />
   463→                </Button>
   464→              </div>
   465→            )}
   466→            <Select value={repHab} onValueChange={v => { setRepHab(v); setRepConfirm(false); }}>
   467→              <SelectTrigger><SelectValue placeholder="-- Elegir habitación --" /></SelectTrigger>
   468→              <SelectContent>
   469→                {habDisponibles.map(([num, h]) => (
   470→                  <SelectItem key={num} value={num}>{num} - {h.tipo}</SelectItem>
   471→                ))}
   472→              </SelectContent>
   473→            </Select>
   474→            <Textarea
   475→              placeholder="Describí el problema detectado..."
   476→              value={repDesc}
   477→              onChange={e => setRepDesc(e.target.value)}
   478→              rows={2}
   479→            />
   480→            <Button
   481→              onClick={handleReportar}
   482→              variant="destructive"
   483→              disabled={!repHab || !repDesc.trim() || reportando}
   484→            >
   485→              {reportando ? <Loader2 className="w-4 h-4 mr-1 animate-spin" /> : <Wrench className="w-4 h-4 mr-1" />}
   486→              {repConfirm ? 'Confirmar y reportar' : 'Reportar mantenimiento'}
   487→            </Button>
   488→          </CardContent>
   489→        )}
   490→      </Card>
   491→
   492→      {/* ── Historial de Mantenimiento ── */}
   493→      <Card>
   494→        <CardHeader className="bg-muted/50">
   495→          <CardTitle className="text-base flex items-center gap-2">
   496→            <Search className="w-4 h-4" /> Historial de Mantenimiento
   497→          </CardTitle>
   498→        </CardHeader>
   499→        <CardContent className="space-y-3">
   500→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-5 gap-2 items-end">
   501→            <DatePickerInline value={fDesde} onChange={v => { setFDesde(v); setPagina(1); }} placeholder="Desde" label="Desde" />
   502→            <DatePickerInline value={fHasta} onChange={v => { setFHasta(v); setPagina(1); }} placeholder="Hasta" label="Hasta" />
   503→            <div className="space-y-1">
   504→              <Label className="text-xs">Habitación</Label>
   505→              <Input placeholder="Ej: 101" value={fHab} onChange={e => { setFHab(e.target.value); setPagina(1); }} />
   506→            </div>
   507→            <div className="space-y-1">
   508→              <Label className="text-xs">Descripción</Label>
   509→              <Input placeholder="Buscar..." value={fDesc} onChange={e => { setFDesc(e.target.value); setPagina(1); }} />
   510→            </div>
   511→            <div className="flex gap-2 items-end">
   512→              <div className="space-y-1 flex-1">
   513→                <Label className="text-xs">Monto mín.</Label>
   514→                <Input type="number" placeholder="$0" value={fMonto} onChange={e => { setFMonto(e.target.value); setPagina(1); }} />
   515→              </div>
   516→              <Button variant="outline" size="icon" className="h-9 w-9" onClick={limpiarFiltros} title="Limpiar filtros">
   517→                {hasFiltros ? <X className="w-4 h-4" /> : <Search className="w-4 h-4" />}
   518→              </Button>
   519→            </div>
   520→          </div>
   521→
   522→          {listaPaginada.length === 0 ? (
   523→            <div className="text-center py-8">
   524→              <Wrench className="w-8 h-8 mx-auto mb-2 text-muted-foreground/40" />
   525→              <p className="text-sm text-muted-foreground">No hay reparaciones registradas.</p>
   526→            </div>
   527→          ) : (
   528→            <div className="overflow-x-auto -mx-6 px-6">
   529→              <Table>
   530→                <TableHeader>
   531→                  <TableRow>
   532→                    <TableHead>Fecha</TableHead>
   533→                    <TableHead>Hab.</TableHead>
   534→                    <TableHead>Problema</TableHead>
   535→                    <TableHead className="hidden md:table-cell">Reparación</TableHead>
   536→                    <TableHead className="text-right">Monto</TableHead>
   537→                    <TableHead className="hidden sm:table-cell">Empleado</TableHead>
   538→                  </TableRow>
   539→                </TableHeader>
   540→                <TableBody>
   541→                  {listaPaginada.map(item => (
   542→                    <TableRow key={item.id}>
   543→                      <TableCell className="text-xs whitespace-nowrap">{formatFechaHora(item.fecha)}</TableCell>
   544→                      <TableCell className="font-medium">{item.habitacion}</TableCell>
   545→                      <TableCell className="max-w-[200px] truncate">{item.problema}</TableCell>
   546→                      <TableCell className="hidden md:table-cell max-w-[200px] truncate">{item.reparacion}</TableCell>
   547→                      <TableCell className={cn('text-right font-medium', item.monto > 0 ? 'text-red-300' : 'text-muted-foreground')}>
   548→                        {formatMoney(item.monto)}
   549→                      </TableCell>
   550→                      <TableCell className="hidden sm:table-cell text-xs text-muted-foreground">{item.empleado}</TableCell>
   551→                    </TableRow>
   552→                  ))}
   553→                </TableBody>
   554→              </Table>
   555→            </div>
   556→          )}
   557→          <PaginationBar
   558→            page={paginaActual}
   559→            totalPages={totalPaginas}
   560→            onPageChange={setPagina}
   561→            totalItems={listaFiltrada.length}
   562→            pageSize={PAGE_SIZE}
   563→          />
   564→        </CardContent>
   565→      </Card>
   566→
   567→      {/* ── Modal: Resolver Mantenimiento ── */}
   568→      <Dialog open={!!modalResolver} onOpenChange={() => setModalResolver(null)}>
   569→        <DialogContent className="sm:max-w-md">
   570→          <DialogHeader>
   571→            <DialogTitle>Resolver Mantenimiento - Habitación {modalResolver}</DialogTitle>
   572→          </DialogHeader>
   573→          <div className="space-y-4 py-2">
   574→            {modalResolver && habitaciones[modalResolver] && (
   575→              <div className="bg-red-900/60 rounded-lg p-2.5">
   576→                <p className="text-sm text-red-300">
   577→                  <strong>Problema reportado:</strong> {habitaciones[modalResolver].problema || 'Sin descripción'}
   578→                </p>
   579→              </div>
   580→            )}
   581→            <div className="space-y-2">
   582→              <Label>Descripción de la reparación *</Label>
   583→              <Textarea
   584→                value={reparacion}
   585→                onChange={e => setReparacion(e.target.value)}
   586→                rows={3}
   587→                placeholder="Ej: Se reemplazó la placa controladora..."
   588→              />
   589→            </div>
   590→            <div className="space-y-2">
   591→              <Label>Monto de la reparación</Label>
   592→              <Input
   593→                type="number"
   594→                min="0"
   595→                step="0.01"
   596→                value={monto}
   597→                onChange={e => setMonto(e.target.value)}
   598→                placeholder="Dejar en 0 si no tuvo costo"
   599→              />
   600→              <p className="text-xs text-muted-foreground">Si no tuvo costo, dejá en $0.</p>
   601→            </div>
   602→            {parseFloat(monto) > 0 && (
   603→              <div className="space-y-2">
   604→                <Label className="text-sm font-medium">¿De dónde sale el pago?</Label>
   605→                <div className="grid grid-cols-2 gap-2">
   606→                  <button
   607→                    type="button"
   608→                    onClick={() => setSacarDeCaja(true)}
   609→                    className={cn(
   610→                      'flex flex-col items-center gap-1.5 p-3 rounded-lg border-2 transition-all',
   611→                      sacarDeCaja ? 'border-[#4ADE80] bg-emerald-900/60' : 'border-muted hover:border-muted-foreground/30'
   612→                    )}
   613→                  >
   614→                    <Wallet className={cn('w-5 h-5', sacarDeCaja ? 'text-emerald-300' : 'text-muted-foreground')} />
   615→                    <span className={cn('text-xs font-medium', sacarDeCaja ? 'text-emerald-300' : 'text-muted-foreground')}>De caja</span>
   616→                    <span className="text-[10px] text-muted-foreground">Sale de la caja del turno</span>
   617→                  </button>
   618→                  <button
   619→                    type="button"
   620→                    onClick={() => setSacarDeCaja(false)}
   621→                    className={cn(
   622→                      'flex flex-col items-center gap-1.5 p-3 rounded-lg border-2 transition-all',
   623→                      !sacarDeCaja ? 'border-sky-700/40 bg-sky-900/30' : 'border-muted hover:border-muted-foreground/30'
   624→                    )}
   625→                  >
   626→                    <Banknote className={cn('w-5 h-5', !sacarDeCaja ? 'text-sky-300' : 'text-muted-foreground')} />
   627→                    <span className={cn('text-xs font-medium', !sacarDeCaja ? 'text-sky-300' : 'text-muted-foreground')}>Pago aparte</span>
   628→                    <span className="text-[10px] text-muted-foreground">Gerente / dinero propio</span>
   629→                  </button>
   630→                </div>
   631→                {sacarDeCaja && (
   632→                  <p className="text-xs text-amber-300 flex items-center gap-1">
   633→                    <AlertTriangle className="w-3 h-3" />Se registrará un egreso en la caja activa
   634→                  </p>
   635→                )}
   636→              </div>
   637→            )}
   638→          </div>
   639→          <DialogFooter>
   640→            <DialogClose asChild><Button variant="secondary">Cancelar</Button></DialogClose>
   641→            <Button
   642→              onClick={handleResolver}
   643→              disabled={!reparacion.trim() || resolviendo}
   644→              className="bg-primary hover:bg-primary/80 text-white"
   645→            >
   646→              {resolviendo ? <Loader2 className="w-4 h-4 mr-1 animate-spin" /> : <Check className="w-4 h-4 mr-1" />}
   647→              Marcar como Resuelto
   648→            </Button>
   649→          </DialogFooter>
   650→        </DialogContent>
   651→      </Dialog>
   652→    </div>
   653→  );
   654→}
