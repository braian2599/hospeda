    13→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
    14→import {

    16→} from '@/components/ui/dialog';
    17→import {

    19→} from '@/components/ui/select';

    44→

   147→            <FileText className="w-4 h-4 mr-1" />Historial de pagos

   149→        </TabsList>

   300→                <div className="grid gap-1.5 flex-1 w-full sm:w-auto sm:min-w-[160px]">

   335→          {/* Payments table */}
