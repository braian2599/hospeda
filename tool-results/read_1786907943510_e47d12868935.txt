   148→   148→function formatoRango(r: RangoPrecio): string {
   149→   149→  if (r.maxPersonas === null) return `${r.minPersonas}+`;
   150→   150→  if (r.minPersonas === r.maxPersonas) return `${r.minPersonas}`;
   151→   151→  return `${r.minPersonas}-${r.maxPersonas}`;
   152→   152→}

   180→   180→// Count active promotions on a tariff
   181→   181→function countPromos(promos?: PromocionesTarifa | null): number {
   182→   182→  if (!promos) return 0;
   183→   183→  let n = 0;
   184→   184→  if (promos.acompananteSinCargo?.activo) n++;
   185→   185→  if (promos.ninosDiferenciado?.activo) n++;
   186→   186→  if (promos.nochesCortesia?.activo) n++;
   187→   187→  return n;
   188→   188→}

   190→   190→// Compute "Desde" price (min positive price, fallback to first range)
   191→   191→function precioDesde(rangos: RangoPrecio[]): number {
   192→   192→  const preciosPositivos = rangos.map(r => r.precio).filter(p => p > 0);
   193→   193→  return preciosPositivos.length > 0 ? Math.min(...preciosPositivos) : (rangos[0]?.precio || 0);
   194→   194→}

   196→   196→// Build a label describing a noches cortesía modalidad
   197→   197→function describeNochesCortesia(mod: ModalidadNochesCortesia): string {
   198→   198→  if (mod.tipo === 'cadaX') return `Cada ${mod.cada} noches, 1 gratis`;
   199→   199→  if (mod.tipo === 'aPartirDe') return `Desde ${mod.minNoches} noches, ${mod.nochesGratis} gratis`;
   200→   200→  const dias = ['Domingo', 'Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado'];
   201→   201→  return `${dias[mod.dia]} gratis`;
   202→   202→}

   204→   204→// Export a single tariff as CSV (client-side download)
   205→   205→function exportTariffCSV(tipo: string, t: TarifaPrecios) {
   206→   206→  const rows: (string | number)[][] = [];
   207→   207→  rows.push(['Campo', 'Valor']);
   208→   208→  rows.push(['Nombre', tipo]);
   209→   209→  rows.push(['Modo de cobro', t.modoCobro || 'porGrupo']);
   210→   210→  const rangos = t.rangos || [];
   211→   211→  rows.push(['Cantidad de rangos', rangos.length]);
   212→   212→  rangos.forEach((r, i) => {
   213→   213→    rows.push([`Rango ${i + 1} (${formatoRango(r)} pers.)`, r.precio]);
   214→   214→  });
   215→   215→  rows.push(['Acompañante sin cargo', t.promociones?.acompananteSinCargo?.activo ? 'Sí' : 'No']);
   216→   216→  if (t.promociones?.acompananteSinCargo?.activo) {
   217→   217→    rows.push(['  Etiqueta', t.promociones.acompananteSinCargo.etiqueta || '—']);
   218→   218→    rows.push(['  Cantidad', t.promociones.acompananteSinCargo.cantidad ?? 1]);
   219→   219→  }
   220→   220→  rows.push(['Niños diferenciado', t.promociones?.ninosDiferenciado?.activo ? 'Sí' : 'No']);
   221→   221→  if (t.promociones?.ninosDiferenciado?.activo) {
   222→   222→    rows.push(['  Precio niño/noche', t.promociones.ninosDiferenciado.precioNino || 0]);
   223→   223→  }
   224→   224→  rows.push(['Noches cortesía', t.promociones?.nochesCortesia?.activo ? 'Sí' : 'No']);
   225→   225→  if (t.promociones?.nochesCortesia?.activo) {
   226→   226→    rows.push(['  Modalidad', describeNochesCortesia(t.promociones.nochesCortesia.modalidad)]);
   227→   227→  }
   228→   228→  const campos = t.camposPersonalizados || [];
   229→   229→  rows.push(['Cantidad de campos personalizados', campos.length]);
   230→   230→  campos.forEach((c, i) => {
   231→   231→    rows.push([`  Campo ${i + 1}`, `${c.nombre} (${c.tipo}${c.requerido ? ', requerido' : ''})`]);
   232→   232→  });
   233→   233→
   234→   234→  const csv = rows.map(r => r.map(cell => {
   235→   235→    const s = String(cell);
   236→   236→    return s.includes(',') || s.includes('"') ? `"${s.replace(/"/g, '""')}"` : s;
   237→   237→  }).join(',')).join('\n');
   238→   238→
   239→   239→  const blob = new Blob([`\uFEFF${csv}`], { type: 'text/csv;charset=utf-8;' });
   240→   240→  const url = URL.createObjectURL(blob);
   241→   241→  const a = document.createElement('a');
   242→   242→  a.href = url;
   243→   243→  a.download = `tarifa_${tipo.toLowerCase().replace(/\s+/g, '_')}.csv`;
   244→   244→  document.body.appendChild(a);
   245→   245→  a.click();
   246→   246→  document.body.removeChild(a);
   247→   247→  URL.revokeObjectURL(url);
   248→   248→}

   252→   252→function WizardStepper({ current, onSelect }: { current: 1 | 2 | 3; onSelect: (s: 1 | 2 | 3) => void }) {
   253→   253→  const steps: { n: 1 | 2 | 3; label: string; icon: typeof Info }[] = [
   254→   254→    { n: 1, label: 'Información básica', icon: Info },
   255→   255→    { n: 2, label: 'Rangos de precios', icon: Tags },
   256→   256→    { n: 3, label: 'Promociones', icon: Sparkles },
   257→   257→  ];
   258→   258→  return (
   259→   259→    <div className="space-y-2">
   260→   260→      <div className="flex items-center gap-1">
   261→   261→        {steps.map((s, i) => {
   262→   262→          const Icon = s.icon;
   263→   263→          const isDone = current > s.n;
   264→   264→          const isActive = current === s.n;
   265→   265→          return (
   266→   266→            <div key={s.n} className="flex items-center flex-1 last:flex-none">
   267→   267→              <button
   268→   268→                type="button"
   269→   269→                onClick={() => onSelect(s.n)}
   270→   270→                className={`flex items-center gap-2 px-3 py-1.5 rounded-md text-xs font-semibold transition-all whitespace-nowrap ${
   271→   271→                  isActive
   272→   272→                    ? 'bg-primary text-white shadow-sm'
   273→   273→                    : isDone
   274→   274→                      ? 'bg-primary/10 text-primary hover:bg-primary/20'
   275→   275→                      : 'bg-muted text-muted-foreground hover:bg-muted/80'
   276→   276→                }`}
   277→   277→                aria-current={isActive ? 'step' : undefined}
   278→   278→              >
   279→   279→                <span className={`w-5 h-5 rounded-full flex items-center justify-center text-[10px] ${
   280→   280→                  isActive ? 'bg-white/20' : isDone ? 'bg-primary text-white' : 'bg-background'
   281→   281→                }`}>
   282→   282→                  {isDone ? <Check className="w-3 h-3" /> : s.n}
   283→   283→                </span>
   284→   284→                <span className="hidden sm:inline">{s.label}</span>
   285→   285→                <Icon className="w-3.5 h-3.5 sm:hidden" />
   286→   286→              </button>
   287→   287→              {i < steps.length - 1 && (
   288→   288→                <div className={`flex-1 h-0.5 mx-1 rounded ${isDone ? 'bg-primary' : 'bg-muted'}`} />
   289→   289→              )}
   290→   290→            </div>
   291→   291→          );
   292→   292→        })}
   293→   293→      </div>
   294→   294→      <Progress value={(current / 3) * 100} className="h-1 bg-muted" />
   295→   295→    </div>
   296→   296→  );
   297→   297→}
