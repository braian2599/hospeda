    85→    85→function RangoFila({ rango, index, onRemove, onUpdate, modoCobro, totalRangos }: {
    86→    86→  rango: RangoPrecio;
    87→    87→  index: number;
    88→    88→  onRemove: () => void;
    89→    89→  onUpdate: (r: RangoPrecio) => void;
    90→    90→  modoCobro: ModoCobro;
    91→    91→  totalRangos: number;
    92→    92→}) {
    93→    93→  const esInfinito = rango.maxPersonas === null;
    94→    94→  // Etiqueta legible del rango: "1 persona", "2 personas", "3+ personas", "2-4 personas"
    95→    95→  const rangoLabel = esInfinito
    96→    96→    ? `${rango.minPersonas}+ persona${rango.minPersonas > 1 ? 's' : ''}`
    97→    97→    : rango.minPersonas === rango.maxPersonas
    98→    98→      ? `${rango.minPersonas} persona${rango.minPersonas > 1 ? 's' : ''}`
    99→    99→      : `${rango.minPersonas} a ${rango.maxPersonas} personas`;
   100→   100→
   101→   101→  return (
   102→   102→    <div className="flex items-end gap-2">
   103→   103→      <div className="w-28 shrink-0 flex items-center h-8 text-sm font-medium text-foreground">
   104→   104→        {modoCobro === 'porHabitacion' ? 'Por noche' : modoCobro === 'porCama' ? 'Por cama/noche' : rangoLabel}
   105→   105→      </div>
   106→   106→      <div className="space-y-0.5 flex-1">
   107→   107→        <Label className="text-[10px] text-muted-foreground">
   108→   108→          {modoCobro === 'porPersona' || modoCobro === 'porCama' ? 'Precio c/u' : 'Precio'}
   109→   109→        </Label>
   110→   110→        <Input
   111→   111→          type="number" min={0} step="0.01"
   112→   112→          className="h-8 text-right"
   113→   113→          value={rango.precio}
   114→   114→          onChange={e => onUpdate({ ...rango, precio: parseFloat(e.target.value) || 0 })}
   115→   115→        />
   116→   116→      </div>
   117→   117→      <Button
   118→   118→        size="icon" variant="ghost"
   119→   119→        className="h-8 w-8 text-destructive shrink-0 mb-0.5"
   120→   120→        onClick={onRemove}

   124→   124→      </Button>
