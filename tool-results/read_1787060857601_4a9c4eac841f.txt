    45→function mapDbReservaToStore(r: any, totalOverride?: number): Reserva {
    46→  // Map API estado to store estado
    47→  let estado: EstadoReserva = 'Confirmada';
    48→  if (r.estado === 'CheckIn_realizado') estado = 'Check-In realizado';
    49→  else if (r.estado === 'Checkout_realizado') estado = 'Check-Out realizado';
    50→  else if (r.estado === 'Cancelada') estado = 'Cancelada';
    51→
    52→  return {
    53→    id: r.id,
    54→    idCliente: r.clienteId || '',
    55→    huesped: r.huesped, dni: r.dni,
    56→    telefono: r.telefono || '', email: r.email || '',
    57→    domicilio: r.domicilio || '', habitacion: r.habitacion,
    58→    checkin: r.checkin?.split('T')[0] || r.checkin,
    59→    checkout: r.checkout?.split('T')[0] || r.checkout,
    60→    personas: r.personas, estadoPago: (r.estadoPago || 'Pendiente') as Reserva['estadoPago'],
    61→    notas: r.notas || '', estado,
    62→    tipoTarifa: r.tipoTarifa || undefined,
    63→    metodoPagoId: r.metodoPagoId || undefined,
    64→    cuotas: r.cuotas || undefined, recargoPorcentaje: r.recargoPorcentaje || undefined,
    65→    total: totalOverride !== undefined ? totalOverride : (r.total != null ? r.total / 100 : undefined),
    66→    ninos: r.ninos || undefined,
    67→    agencia: r.agenciaNombre ? { nombre: r.agenciaNombre, convenio: r.agenciaConvenio, vendedor: undefined } : undefined,
    68→    contactoEmergencia: r.contactoEmergenciaNombre ? { nombre: r.contactoEmergenciaNombre, telefono: r.contactoEmergenciaTel || '' } : undefined,

    70→    llaveEntregada: (r as any).llaveEntregada || undefined,

   117→function mapDbCajaToStore(dbCaja: any): CajaState {
   118→  const turno = dbCaja.turnoActual;
   119→  if (!turno) {
   120→    const historial: TurnoCaja[] = (dbCaja.historial || []).map(mapDbTurnoToStore);
   121→    return { estado: 'cerrada', apertura: null, movimientos: [], historial };
   122→  }
   123→
   124→  const movimientos: MovimientoCaja[] = (turno.movimientos || []).map(mapDbMovimiento);
   125→
   126→  return {
   127→    estado: 'abierta',
   128→    apertura: { montoInicial: turno.montoInicial / 100, empleado: turno.empleadoNombre, fecha: turno.fechaApertura },
   129→    movimientos,
   130→    historial: (dbCaja.historial || []).map(mapDbTurnoToStore),
   131→  };
   132→}

   169→function nochesEntre(checkin: string, checkout: string): number {
   170→  const fechaIn = new Date(checkin + 'T12:00:00');
   171→  const fechaOut = new Date(checkout + 'T12:00:00');

   177→/** Busca el rango que corresponde a la cantidad de personas dada. */
   178→function encontrarRango(rangos: RangoPrecio[], personas: number): RangoPrecio | undefined {
   179→  // Buscar rango donde minPersonas <= personas <= maxPersonas
   180→  for (const r of rangos) {
   181→    if (personas >= r.minPersonas && (r.maxPersonas === null || personas <= r.maxPersonas)) {
   182→      return r;
   183→    }
   184→  }
   185→  // Fallback: usar el último rango (el que tiene maxPersonas null, o el de mayor cobertura)
   186→  return rangos.length > 0 ? rangos[rangos.length - 1] : undefined;
   187→}

   189→/**
   190→ * Convierte datos de tarifa desde la BD (puede ser formato viejo o nuevo) a RangoPrecio[].
   191→ * Formato viejo: { "1": 35000, "2": 30000, ... }
   192→ * Formato nuevo: { "rangos": [{ minPersonas, maxPersonas, precio }] }
   193→ */
   194→function normalizarRangos(preciosDb: any): RangoPrecio[] {
   195→  if (!preciosDb) return [];
   196→  // Nuevo formato: tiene la propiedad "rangos" como array
   197→  if (preciosDb.rangos && Array.isArray(preciosDb.rangos)) {
   198→    return preciosDb.rangos.map((r: any) => ({
   199→      minPersonas: Number(r.minPersonas) || 1,
   200→      maxPersonas: r.maxPersonas != null ? Number(r.maxPersonas) : null,
   201→      precio: Number(r.precio) || 0,
   202→    }));
   203→  }
   204→  // Formato viejo: keys numéricos { "1": 35000, "2": 30000, ... }
   205→  const keys = Object.keys(preciosDb).filter(k => !isNaN(Number(k)) && Number(k) >= 1);
   206→  if (keys.length > 0) {
   207→    return keys.map(k => ({
   208→      minPersonas: Number(k),
   209→      maxPersonas: Number(k),
   210→      precio: Number(preciosDb[k]) || 0,
   211→    }));
   212→  }
   213→  return [];
   214→}

   216→/**
   217→ * Calcula las noches de cortesía según la modalidad.
   218→ * Devuelve la cantidad de noches a descontar.
   219→ */
   220→function calcularNochesGratis(promociones: PromocionesTarifa, noches: number, checkin?: string): number {
   221→  const nc = promociones.nochesCortesia;
   222→  if (!nc?.activo || !nc.modalidad) return 0;
   223→
   224→  const mod = nc.modalidad;
   225→  if (mod.tipo === 'cadaX') {
   226→    const cada = mod.cada || 999;
   227→    if (noches < cada) return 0;
   228→    return Math.floor(noches / cada);
   229→  }
   230→  if (mod.tipo === 'aPartirDe') {
   231→    if (noches < mod.minNoches) return 0;
   232→    return mod.nochesGratis || 0;
   233→  }
   234→  if (mod.tipo === 'diaSemana' && checkin) {
   235→    // Contar cuántas noches de la estadía caen en el día de la semana elegido
   236→    const fechaInicio = new Date(checkin + 'T12:00:00');
   237→    let count = 0;
   238→    for (let i = 0; i < noches; i++) {
   239→      const d = new Date(fechaInicio);
   240→      d.setDate(d.getDate() + i);
   241→      if (d.getDay() === mod.dia) count++;
   242→    }
   243→    return count;
   244→  }
   245→  return 0;
   246→}

   248→/**
   249→ * Obtiene las promociones efectivas de una tarifa, migrando datos viejos (choferCortesia) si es necesario.
   250→ */
   251→function getPromocionesEfectivas(tarifa: TarifaPrecios): PromocionesTarifa {
   252→  // Si tiene promociones nuevas, usar esas
   253→  if (tarifa.promociones) return tarifa.promociones;
   254→  // Migración: si tiene choferCortesia viejo, convertir a nuevo formato
   255→  if (tarifa.choferCortesia) {
   256→    return {
   257→      acompananteSinCargo: {
   258→        activo: true,
   259→        etiqueta: 'Chofer de cortesía',
   260→        habitacionAsignada: tarifa.habitacionChofer || undefined,
   261→        cantidad: 1,
   262→      },
   263→    };
   264→  }
   265→  return {};
   266→}

   273→function calcularTotalSegunTarifa(
   274→  tarifas: Record<string, TarifaPrecios>,
   275→  tipoTarifa: string,
   276→  personas: number,
   277→  noches: number,
   278→  options?: CalcTarifaOptions
   279→): number {
   280→  // Buscar la tarifa (fallback a 'normal')
   281→  const tarifa = tarifas[tipoTarifa] || tarifas['normal'];
   282→  if (!tarifa || !tarifa.rangos || tarifa.rangos.length === 0) return 0;
   283→
   284→  const promociones = getPromocionesEfectivas(tarifa);
   285→  const modo: ModoCobro = tarifa.modoCobro || 'porGrupo';
   286→
   287→  // ─── Noches de cortesía ───
   288→  const nochesGratis = calcularNochesGratis(promociones, noches, options?.checkin);
   289→  const nochesCobrables = Math.max(0, noches - nochesGratis);
   290→
   291→  // ─── Niños diferenciado ───
   292→  const ninosDif = promociones.ninosDiferenciado;
   293→  const cantNinos = (options?.ninos && ninosDif?.activo) ? options.ninos : 0;
   294→  const adultos = Math.max(1, personas - cantNinos);
   295→
   296→  // ─── Acompañante sin cargo (no descuenta: el acompañante va a otra habitación) ───
   297→  // Ya NO se resta 1 adulto. Todos los adultos de esta reserva pagan.
   298→  // El acompañante se gestiona como una reserva separada con total=0.
   299→
   300→  // ─── Cálculo por modo ───
   301→  if (modo === 'porCama') {
   302→    // Modo porCama: el precio del rango es por persona por noche
   303→    const rango = encontrarRango(tarifa.rangos, adultos);
   304→    const precioCama = rango?.precio || tarifa.rangos[0]?.precio || 0;
   305→    let total = nochesCobrables * adultos * precioCama;
   306→    // Sumar niños
   307→    if (cantNinos > 0 && ninosDif?.activo) {
   308→      total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   309→    }
   310→    return total;
   311→  }
   312→
   313→  if (modo === 'porHabitacion') {
   314→    const precio = tarifa.rangos[0]?.precio || 0;
   315→    let total = nochesCobrables * precio;
   316→    // Sumar niños si aplica
   317→    if (cantNinos > 0 && ninosDif?.activo) {
   318→      total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   319→    }
   320→    return total;
   321→  }
   322→
   323→  const rango = encontrarRango(tarifa.rangos, adultos);
   324→  if (!rango) return 0;
   325→
   326→  // porGrupo (default): el precio es el total del grupo
   327→  let total = nochesCobrables * rango.precio;
   328→  // Sumar niños
   329→  if (cantNinos > 0 && ninosDif?.activo) {
   330→    total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   331→  }
   332→  return total;
   333→}

String((dbCliente as any).fechaNacimiento).split('T')[0] : undefined, nacionalidad: (dbCliente as any).nacionalidad || undefined, domicilio: (dbCliente as any).domicilio || datos.domicilio || undefined, preferencias: dbCliente.preferencias, historialEstadias: [], fechaCreacion: dbCliente.createdAt?.split('T')[0] || '' }] });
