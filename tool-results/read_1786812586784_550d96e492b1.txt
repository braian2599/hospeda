   238→// RoomTypeAnalytics — compact analytics section
   239→// ═══════════════════════════════════════════════════════════
   240→
   241→function RoomTypeAnalytics() {
   242→  const habitaciones = useHotelStore(s => s.habitaciones);
   243→  const reservas = useHotelStore(s => s.reservas);
   244→  const today = useMemo(() => todayLocal(), []);
   245→
   246→  const allRooms = Object.values(habitaciones);
   247→  const total = allRooms.length;
   248→
   249→  const typeData = useMemo(() => {
   250→    if (total === 0) return [];
   251→
   252→    const todayDate = safeDate(today).getTime();
   253→
   254→    // Group by type
   255→    const byType = new Map<string, Habitacion[]>();
   256→    allRooms.forEach(h => {
   257→      const t = h.tipo || 'Sin tipo';
   258→      if (!byType.has(t)) byType.set(t, []);
   259→      byType.get(t)!.push(h);
   260→    });
   261→
   262→    return Array.from(byType.entries())
   263→      .sort(([a], [b]) => a.localeCompare(b))
   264→      .map(([tipo, rooms]) => {
   265→        const count = rooms.length;
   266→        const occupied = rooms.filter(h => h.estado === 'Ocupada').length;
   267→        const occupancyRate = Math.round((occupied / count) * 100);
   268→
   269→        // Calculate revenue for occupied rooms
   270→        let revenue = 0;
   271→        rooms.forEach(h => {
   272→          if (h.estado === 'Ocupada' || h.estado === 'Reservada') {
   273→            const res = reservas.find(r =>
   274→              r.habitacion === h.numero &&
   275→              (r.estado === 'Check-In realizado' || r.estado === 'Confirmada') &&
   276→              safeDate(r.checkin).getTime() <= todayDate &&
   277→              safeDate(r.checkout).getTime() >= todayDate
   278→            );
   279→            if (res?.total) revenue += res.total;
   280→          }
   281→        });
   282→
   283→        const Icon = tipoIconMap[tipo] || Bed;
   284→
   285→        return { tipo, count, occupied, occupancyRate, revenue, Icon };
   286→      });
   287→  }, [allRooms, reservas, today, total]);
   288→
   289→  if (total === 0 || typeData.length === 0) return null;
   290→
   291→  const maxCount = Math.max(...typeData.map(d => d.count));
   292→
   293→  // Type color palette
   294→  const typeColors = ['#059669', '#D97706', '#0EA5E9', '#8B5CF6', '#EF4444'];
   295→
   296→  return (
   297→    <div className="rounded-xl border bg-card p-4 space-y-3">
   298→      <div className="flex items-center gap-2">
   299→        <div className="w-7 h-7 rounded-lg bg-primary/20 flex items-center justify-center">
   300→          <Bed className="w-3.5 h-3.5 text-primary" />
   301→        </div>
   302→        <span className="text-sm font-semibold text-foreground">Distribución por tipo</span>
   303→      </div>
   304→
   305→      {/* Horizontal distribution bar */}
   306→      <div className="h-2.5 rounded-full bg-muted overflow-hidden flex">
   307→        {typeData.map((d, i) => {
   308→          const pct = (d.count / total) * 100;
   309→          return (
   310→            <div
   311→              key={d.tipo}
   312→              className="transition-all duration-300"
   313→              style={{ width: `${pct}%`, backgroundColor: typeColors[i % typeColors.length] }}
   314→              title={`${d.tipo}: ${d.count} (${Math.round(pct)}%)`}
   315→            />
   316→          );
   317→        })}
   318→      </div>
   319→
   320→      {/* Type breakdown rows */}
   321→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-5 gap-2">
   322→        {typeData.map((d, i) => {
   323→          const color = typeColors[i % typeColors.length];
   324→          const barPct = (d.count / maxCount) * 100;
   325→          return (
   326→            <div
   327→              key={d.tipo}
   328→              className="rounded-lg border bg-muted/20 p-2.5 space-y-1.5"
   329→            >
   330→              <div className="flex items-center gap-2">
   331→                <d.Icon className="w-3.5 h-3.5 shrink-0" style={{ color }} />
   332→                <span className="text-xs font-semibold text-foreground truncate">{d.tipo}</span>
   333→                <span className="ml-auto text-xs font-bold text-foreground">{d.count}</span>
   334→              </div>
   335→              {/* Mini bar */}
   336→              <div className="h-1.5 rounded-full bg-muted overflow-hidden">
   337→                <div
   338→                  className="h-full rounded-full transition-all duration-300"
   339→                  style={{ width: `${barPct}%`, backgroundColor: color }}
   340→                />
   341→              </div>
   342→              <div className="flex items-center justify-between text-[10px] text-muted-foreground">
   343→                <span>Ocup. {d.occupancyRate}%</span>
   344→                {d.revenue > 0 && <span>${d.revenue.toLocaleString('es-AR')}</span>}
   345→              </div>
   346→            </div>
   347→          );
   348→        })}
   349→      </div>
   350→    </div>
   351→  );
   352→}

   355→// StatusChangePopover — Quick status change for a room
   356→// ═══════════════════════════════════════════════════════════
   357→
   358→function StatusChangePopover({
   359→  numero,
   360→  currentEstado,
   361→  onStatusChange,
   362→  children,
   363→}: {
   364→  numero: string;
   365→  currentEstado: EstadoHabitacion;
   366→  onStatusChange: (numero: string, newEstado: EstadoHabitacion) => void;
   367→  children: React.ReactNode;
   368→}) {
   369→  const [open, setOpen] = useState(false);
   370→  const [confirmDestructive, setConfirmDestructive] = useState<EstadoHabitacion | null>(null);
   371→
   372→  const handleSelect = (newEstado: EstadoHabitacion) => {
   373→    if (newEstado === currentEstado) {
   374→      setOpen(false);
   375→      return;
   376→    }
   377→    // Destructive states need confirmation
   378→    const isDestructive = newEstado === 'Fuera de servicio' || newEstado === 'Mantenimiento';
   379→    if (isDestructive && !confirmDestructive) {
   380→      setConfirmDestructive(newEstado);
   381→      return;
   382→    }
   383→    onStatusChange(numero, newEstado);
   384→    setConfirmDestructive(null);
   385→    setOpen(false);
   386→  };
   387→
   388→  return (
   389→    <Popover open={open} onOpenChange={(v) => { setOpen(v); setConfirmDestructive(null); }}>
   390→      <PopoverTrigger asChild>{children}</PopoverTrigger>
   391→      <PopoverContent className="w-56 p-2" align="start" side="bottom">
   392→        {confirmDestructive ? (
   393→          <div className="space-y-2 p-1">
   394→            <p className="text-xs text-muted-foreground">
   395→              ¿Cambiar a <strong className="text-foreground">{confirmDestructive}</strong>?
   396→              {confirmDestructive === 'Fuera de servicio' && ' La habitación no será asignable.'}
   397→              {confirmDestructive === 'Mantenimiento' && ' Se requiere reparación antes de habilitar.'}
   398→            </p>
   399→            <div className="flex gap-1.5">
   400→              <Button
   401→                size="sm"
   402→                variant="destructive"
   403→                className="h-7 text-xs flex-1"
   404→                onClick={() => handleSelect(confirmDestructive)}
   405→              >
   406→                Confirmar
   407→              </Button>
   408→              <Button
   409→                size="sm"
   410→                variant="ghost"
   411→                className="h-7 text-xs"
   412→                onClick={() => setConfirmDestructive(null)}
   413→              >
   414→                Cancelar
   415→              </Button>
   416→            </div>
   417→          </div>
   418→        ) : (
   419→          <div className="space-y-0.5">
   420→            <div className="text-[10px] uppercase tracking-wider text-muted-foreground font-semibold px-2 pt-1 pb-1.5">
   421→              Cambiar estado
   422→            </div>
   423→            {ALL_ESTADOS.map(estado => {
   424→              const vis = STATUS_VISUAL[estado];
   425→              const Icon = vis.icon;
   426→              const isCurrent = estado === currentEstado;
   427→              return (
   428→                <button
   429→                  key={estado}
   430→                  type="button"
   431→                  onClick={() => handleSelect(estado)}
   432→                  className={`
   433→                    w-full flex items-center gap-2 rounded-md px-2 py-1.5 text-xs
   434→                    transition-colors cursor-pointer
   435→                    ${isCurrent
   436→                      ? 'bg-accent/50 font-semibold text-foreground'
   437→                      : 'text-foreground/80 hover:bg-accent/40'
   438→                    }
   439→                  `}
   440→                >
   441→                  <Icon className="w-3.5 h-3.5 shrink-0" style={{ color: vis.color }} />
   442→                  <span className="truncate">{estado}</span>
   443→                  {isCurrent && (
   444→                    <span className="ml-auto text-[10px] text-muted-foreground">actual</span>
   445→                  )}
   446→                </button>
   447→              );
   448→            })}
   449→          </div>
   450→        )}
   451→      </PopoverContent>
   452→    </Popover>
   453→  );
   454→}

   599→function FloorGroup({

   604→  getHuespedActual,

   623→    rooms.forEach(h => { c[h.estado] = (c[h.estado] || 0) + 1; });

   625→  }, [rooms]);

   628→    <Collapsible open={isOpen} onOpenChange={setIsOpen}>
   629→      <div className="rounded-xl border bg-card overflow-hidden">
   630→        {/* Floor header */}

   632→          <button
   633→            type="button"

   637→            <div className="flex items-center justify-center w-8 h-8 rounded-lg bg-primary/20 text-primary shrink-0">

   651→            <div className="ml-auto flex items-center gap-1.5">

   654→                if (count === 0) return null;
   655→                const vis = STATUS_VISUAL[estado];

   657→                  <div key={estado} className="flex items-center gap-0.5" title={`${estado}: ${count}`}>
   658→                    <span
   659→                      className="w-2 h-2 rounded-sm"
   660→                      style={{ backgroundColor: vis.color }}

   662→                    <span className="text-[10px] text-muted-foreground">{count}</span>
   663→                  </div>
   664→                );

   667→            <div className="shrink-0 ml-1">

   669→                <ChevronDown className="w-4 h-4 text-muted-foreground" />

   671→                <ChevronRight className="w-4 h-4 text-muted-foreground" />

   673→            </div>

   675→        </CollapsibleTrigger>

   686→                  <EnhancedRoomCard
   687→                    key={hab.numero}
   688→                    hab={hab}
   689→                    huesped={huesped}
   690→                    onEdit={onEdit}
   691→                    onDelete={onDelete}
   692→                    onStatusChange={onStatusChange}
   693→                  />
   694→                );
   695→              })}
   696→            </div>
   697→          </div>
   698→        </CollapsibleContent>
   699→      </div>
   700→    </Collapsible>
   701→  );
   702→}

   708→export default function HabitacionesModule() {
   709→  // Granular selectors — avoids re-rendering on unrelated store changes
   710→  const habitaciones = useHotelStore(s => s.habitaciones);
   711→  const reservas = useHotelStore(s => s.reservas);

   716→  const [modal, setModal] = useState<'nueva' | 'editar' | 'eliminar' | null>(null);

   729→  const today = useMemo(() => todayLocal(), []);

   731→    () => Object.entries(habitaciones).sort(([a], [b]) => a.localeCompare(b)),
   732→    [habitaciones]
   733→  );
   734→
   735→  // Find the active guest for a room
   736→  const getHuespedActual = useCallback((num: string) => {
   737→    const todayDate = safeDate(today).getTime();
   738→    return reservas.find(r =>
   739→      r.habitacion === num &&
   740→      (r.estado === 'Check-In realizado' || r.estado === 'Confirmada') &&
   741→      safeDate(r.checkin).getTime() <= todayDate &&
   742→      safeDate(r.checkout).getTime() >= todayDate

   757→      // Group by floor

   760→        const floor = getRoomFloor(hab);
   761→        if (!map.has(floor)) map.set(floor, []);
   762→        map.get(floor)!.push(hab);
   763→      });

   765→        .sort(([a], [b]) => a - b)
   766→        .map(([floorNum, rooms]) => ({
   767→          key: `floor-${floorNum}`,
   768→          label: floorNum === 0 ? 'Sin piso' : `Piso ${floorNum}`,
   769→          floorNum,
   770→          rooms,
   771→        }));
   772→    }
   773→    // Group by type
   774→    const map = new Map<string, Habitacion[]>();
   775→    sorted.forEach(([, hab]) => {
   776→      const tipo = hab.tipo || 'Sin tipo';
   777→      if (!map.has(tipo)) map.set(tipo, []);
   778→      map.get(tipo)!.push(hab);
   779→    });
   780→    return Array.from(map.entries())
   781→      .sort(([a], [b]) => a.localeCompare(b))
   782→      .map(([tipo, rooms], idx) => ({
   783→        key: `type-${tipo}`,
   784→        label: tipo,
   785→        floorNum: idx + 1,
   786→        rooms,
   787→      }));
   788→  }, [sorted, isFloorPattern]);
   789→
   790→  // ── Helpers ──
   791→  const capacidadDesdeTipo = (tipo: TipoHabitacion) => {
   792→    const cap = CAPACIDAD_POR_TIPO[tipo];
   793→    return cap !== null ? String(cap) : '1';
   794→  };
   795→
   796→  const handleTipoChange = (tipo: TipoHabitacion) => {
   797→    setForm(prev => ({
   798→      ...prev,
   799→      tipo,
   800→      capacidad: CAPACIDAD_POR_TIPO[tipo] !== null ? String(CAPACIDAD_POR_TIPO[tipo]) : prev.capacidad,
   801→    }));
   802→  };
   803→
   804→  const getCapacidadFinal = (): number => {

   814→

   819→    setSel(num);

   822→      tipo,
   823→      capacidad: CAPACIDAD_POR_TIPO[tipo] !== null ? String(CAPACIDAD_POR_TIPO[tipo]) : String(h.capacidad),
   824→      camasMatrimoniales: String(h.camasMatrimoniales),
   825→      camasSimples: String(h.camasSimples),
   826→      piso: h.piso ? String(h.piso) : '',
   827→    });
   828→    setModal('editar');
   829→  };
   830→
   831→  const openDelete = (num: string) => { setSel(num); setModal('eliminar'); };
   832→
   833→  // ── Status change handler ──

   836→    if (ok) {
   837→      toast.success(`Hab. ${numero} → ${nuevoEstado}`);

   839→      toast.error('No se pudo cambiar el estado');

   900→                transition-all duration-200

   902→                  ? 'bg-card text-foreground shadow-sm'
   903→                  : 'text-muted-foreground hover:text-foreground hover:bg-accent/50'
