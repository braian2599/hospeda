    33→function mapDbReservaToStore(r: any, totalOverride?: number): Reserva {
    34→  // Map API estado to store estado
    35→  let estado: EstadoReserva = 'Confirmada';
    36→  if (r.estado === 'CheckIn_realizado') estado = 'Check-In realizado';
    37→  else if (r.estado === 'Checkout_realizado') estado = 'Check-Out realizado';
    38→  else if (r.estado === 'Cancelada') estado = 'Cancelada';
    39→
    40→  return {
    41→    id: r.id,
    42→    idCliente: r.clienteId || '',
    43→    huesped: r.huesped, dni: r.dni,
    44→    telefono: r.telefono || '', email: r.email || '',
    45→    domicilio: r.domicilio || '', habitacion: r.habitacion,
    46→    checkin: r.checkin?.split('T')[0] || r.checkin,
    47→    checkout: r.checkout?.split('T')[0] || r.checkout,
    48→    personas: r.personas, estadoPago: (r.estadoPago || 'Pendiente') as Reserva['estadoPago'],
    49→    notas: r.notas || '', estado,
    50→    tipoTarifa: r.tipoTarifa || undefined,
    51→    metodoPagoId: r.metodoPagoId || undefined,
    52→    cuotas: r.cuotas || undefined, recargoPorcentaje: r.recargoPorcentaje || undefined,
    53→    total: totalOverride !== undefined ? totalOverride : (r.total != null ? r.total / 100 : undefined),
    54→    ninos: r.ninos || undefined,
    55→    agencia: (r.agenciaNombre ? { nombre: r.agenciaNombre, convenio: r.agenciaConvenio, vendedor: undefined } : undefined) as any,
    56→    contactoEmergencia: (r.contactoEmergenciaNombre ? { nombre: r.contactoEmergenciaNombre, telefono: r.contactoEmergenciaTel || '' } : undefined) as any,

   134→function nochesEntre(checkin: string, checkout: string): number {

   142→/** Busca el rango que corresponde a la cantidad de personas dada. */
   143→function encontrarRango(rangos: RangoPrecio[], personas: number): RangoPrecio | undefined {
   144→  // Buscar rango donde minPersonas <= personas <= maxPersonas
   145→  for (const r of rangos) {
   146→    if (personas >= r.minPersonas && (r.maxPersonas === null || personas <= r.maxPersonas)) {
   147→      return r;
   148→    }
   149→  }
   150→  // Fallback: usar el último rango (el que tiene maxPersonas null, o el de mayor cobertura)
   151→  return rangos.length > 0 ? rangos[rangos.length - 1] : undefined;
   152→}

   154→/**
   155→ * Convierte datos de tarifa desde la BD (puede ser formato viejo o nuevo) a RangoPrecio[].
   156→ * Formato viejo: { "1": 35000, "2": 30000, ... }
   157→ * Formato nuevo: { "rangos": [{ minPersonas, maxPersonas, precio }] }
   158→ */
   159→function normalizarRangos(preciosDb: any): RangoPrecio[] {
   160→  if (!preciosDb) return [];
   161→  // Nuevo formato: tiene la propiedad "rangos" como array
   162→  if (preciosDb.rangos && Array.isArray(preciosDb.rangos)) {
   163→    return preciosDb.rangos.map((r: any) => ({
   164→      minPersonas: Number(r.minPersonas) || 1,
   165→      maxPersonas: r.maxPersonas != null ? Number(r.maxPersonas) : null,
   166→      precio: Number(r.precio) || 0,
   167→    }));
   168→  }
   169→  // Formato viejo: keys numéricos { "1": 35000, "2": 30000, ... }
   170→  const keys = Object.keys(preciosDb).filter(k => !isNaN(Number(k)) && Number(k) >= 1);
   171→  if (keys.length > 0) {
   172→    return keys.map(k => ({
   173→      minPersonas: Number(k),
   174→      maxPersonas: Number(k),
   175→      precio: Number(preciosDb[k]) || 0,
   176→    }));
   177→  }
   178→  return [];
   179→}

   181→/**
   182→ * Calcula las noches de cortesía según la modalidad.
   183→ * Devuelve la cantidad de noches a descontar.
   184→ */
   185→function calcularNochesGratis(promociones: PromocionesTarifa, noches: number, checkin?: string): number {
   186→  const nc = promociones.nochesCortesia;
   187→  if (!nc?.activo || !nc.modalidad) return 0;
   188→
   189→  const mod = nc.modalidad;
   190→  if (mod.tipo === 'cadaX') {
   191→    const cada = mod.cada || 999;
   192→    if (noches < cada) return 0;
   193→    return Math.floor(noches / cada);
   194→  }
   195→  if (mod.tipo === 'aPartirDe') {
   196→    if (noches < mod.minNoches) return 0;
   197→    return mod.nochesGratis || 0;
   198→  }
   199→  if (mod.tipo === 'diaSemana' && checkin) {
   200→    // Contar cuántas noches de la estadía caen en el día de la semana elegido
   201→    const fechaInicio = new Date(checkin + 'T12:00:00');
   202→    let count = 0;
   203→    for (let i = 0; i < noches; i++) {
   204→      const d = new Date(fechaInicio);
   205→      d.setDate(d.getDate() + i);
   206→      if (d.getDay() === mod.dia) count++;
   207→    }
   208→    return count;
   209→  }
   210→  return 0;
   211→}

   213→/**
   214→ * Obtiene las promociones efectivas de una tarifa, migrando datos viejos (choferCortesia) si es necesario.
   215→ */
   216→function getPromocionesEfectivas(tarifa: TarifaPrecios): PromocionesTarifa {
   217→  // Si tiene promociones nuevas, usar esas
   218→  if (tarifa.promociones) return tarifa.promociones;
   219→  // Migración: si tiene choferCortesia viejo, convertir a nuevo formato
   220→  if (tarifa.choferCortesia) {
   221→    return {
   222→      acompananteSinCargo: {
   223→        activo: true,
   224→        etiqueta: 'Chofer de cortesía',
   225→        habitacionAsignada: tarifa.habitacionChofer || undefined,
   226→        cantidad: 1,
   227→      },
   228→    };
   229→  }
   230→  return {};
   231→}

   238→function calcularTotalSegunTarifa(
   239→  tarifas: Record<string, TarifaPrecios>,
   240→  tipoTarifa: string,
   241→  personas: number,
   242→  noches: number,
   243→  options?: CalcTarifaOptions
   244→): number {
   245→  // Buscar la tarifa (fallback a 'normal')
   246→  const tarifa = tarifas[tipoTarifa] || tarifas['normal'];
   247→  if (!tarifa || !tarifa.rangos || tarifa.rangos.length === 0) return 0;
   248→
   249→  const promociones = getPromocionesEfectivas(tarifa);
   250→  const modo: ModoCobro = tarifa.modoCobro || 'porGrupo';
   251→
   252→  // ─── Noches de cortesía ───
   253→  const nochesGratis = calcularNochesGratis(promociones, noches, options?.checkin);
   254→  const nochesCobrables = Math.max(0, noches - nochesGratis);
   255→
   256→  // ─── Niños diferenciado ───
   257→  const ninosDif = promociones.ninosDiferenciado;
   258→  const cantNinos = (options?.ninos && ninosDif?.activo) ? options.ninos : 0;
   259→  const adultos = Math.max(1, personas - cantNinos);
   260→
   261→  // ─── Acompañante sin cargo (no descuenta: el acompañante va a otra habitación) ───
   262→  // Ya NO se resta 1 adulto. Todos los adultos de esta reserva pagan.
   263→  // El acompañante se gestiona como una reserva separada con total=0.
   264→
   265→  // ─── Cálculo por modo ───
   266→  if (modo === 'porCama') {
   267→    // Modo porCama: el precio del rango es por persona por noche
   268→    const rango = encontrarRango(tarifa.rangos, adultos);
   269→    const precioCama = rango?.precio || tarifa.rangos[0]?.precio || 0;
   270→    let total = nochesCobrables * adultos * precioCama;
   271→    // Sumar niños
   272→    if (cantNinos > 0 && ninosDif?.activo) {
   273→      total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   274→    }
   275→    return total;
   276→  }
   277→
   278→  if (modo === 'porHabitacion') {
   279→    const precio = tarifa.rangos[0]?.precio || 0;
   280→    let total = nochesCobrables * precio;
   281→    // Sumar niños si aplica
   282→    if (cantNinos > 0 && ninosDif?.activo) {
   283→      total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   284→    }
   285→    return total;
   286→  }
   287→
   288→  const rango = encontrarRango(tarifa.rangos, adultos);
   289→  if (!rango) return 0;
   290→
   291→  if (modo === 'porPersona') {
   292→    let total = nochesCobrables * adultos * rango.precio;
   293→    // Sumar niños
   294→    if (cantNinos > 0 && ninosDif?.activo) {
   295→      total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   296→    }
   297→    return total;
   298→  }
   299→
   300→  // porGrupo (default): el precio es el total del grupo
   301→  let total = nochesCobrables * rango.precio;
   302→  // Sumar niños
   303→  if (cantNinos > 0 && ninosDif?.activo) {
   304→    total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   305→  }
   306→  return total;
   307→}

   810→              clienteRealId = existing?.id;
   811→            }
   812→            // Si es otro error (no 409), no bloqueamos la reserva
   813→          }
   814→        } catch (clienteErr) {

   818→        const tempId = generarId();
