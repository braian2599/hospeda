    42→/** Get initials from a name string (up to 2 chars) */
    43→function getInitials(name: string): string {
    44→  if (!name) return '?';
    45→  const parts = name.trim().split(/\s+/);
    46→  if (parts.length >= 2) return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
    47→  return parts[0].slice(0, 2).toUpperCase();
    48→}

    50→/** Calculate days between a date string and today */
    51→function daysSince(dateStr: string): number {
    52→  if (!dateStr) return 0;
    53→  const then = new Date(dateStr.includes('T') ? dateStr : dateStr + 'T12:00:00');
    54→  const now = new Date();
    55→  return Math.floor((now.getTime() - then.getTime()) / (1000 * 60 * 60 * 24));
    56→}
    57→
    58→/** Format a relative time string from a date string */
    59→function relativeTime(dateStr: string): string {
    60→  if (!dateStr) return '';
    61→  const then = new Date(dateStr.includes('T') ? dateStr : dateStr + 'T12:00:00');
    62→  const now = new Date();
    63→  const diffMs = now.getTime() - then.getTime();
    64→  const diffMins = Math.floor(diffMs / (1000 * 60));
    65→  const diffHours = Math.floor(diffMs / (1000 * 60 * 60));
    66→  const diffDays = Math.floor(diffMs / (1000 * 60 * 60 * 24));
    67→  if (diffMins < 1) return 'ahora';
    68→  if (diffMins < 60) return `hace ${diffMins}m`;
    69→  if (diffHours < 24) return `hace ${diffHours}h`;
    70→  if (diffDays < 30) return `hace ${diffDays}d`;
    71→  if (diffDays < 365) return `hace ${Math.floor(diffDays / 30)}mes`;
    72→  return `hace ${Math.floor(diffDays / 365)}a`;
    73→}

    75→/** Get payment method icon component name based on method name */
    76→function getMetodoIcon(metodo: string): 'credit' | 'bank' | 'wallet' | 'cash' {
    77→  const lower = metodo.toLowerCase();
    78→  if (lower.includes('tarjeta') || lower.includes('crédito') || lower.includes('credito') || lower.includes('débito') || lower.includes('debito')) return 'credit';
    79→  if (lower.includes('transfer') || lower.includes('banco') || lower.includes('depósito') || lower.includes('deposito')) return 'bank';
    80→  if (lower.includes('mercadopago') || lower.includes('mp') || lower.includes('digital') || lower.includes('qr')) return 'wallet';
    81→  return 'cash';
    82→}

    84→/** Generate a receipt number from reserva ID */
    85→function receiptNumber(reservaId: string): string {
    86→  const num = reservaId.replace(/\D/g, '');
    87→  const suffix = num ? num.padStart(4, '0') : reservaId.slice(0, 4).toUpperCase().padEnd(4, '0');
    88→  const date = new Date();
    89→  const yy = String(date.getFullYear()).slice(2);
    90→  const mm = String(date.getMonth() + 1).padStart(2, '0');
    91→  return `RCP-${yy}${mm}-${suffix}`;
    92→}
