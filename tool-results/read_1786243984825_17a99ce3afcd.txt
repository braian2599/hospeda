    10→import { Badge } from '@/components/ui/badge';

    12→import { Textarea } from '@/components/ui/textarea';

    14→import {

    16→} from '@/components/ui/dialog';
    17→import {
    18→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    19→} from '@/components/ui/select';

    22→} from '@/components/ui/table';

    26→} from 'lucide-react';

    28→import { toast } from 'sonner';
    29→

    33→  return d.toLocaleDateString('es-AR');
    34→};
    35→

    37→  new Intl.NumberFormat('es-AR', { style: 'currency', currency: 'ARS', minimumFractionDigits: 0 }).format(n);

    44→

    48→    calcularTotalReserva, calcularTotalPagado, registrarPago, nochesEntre,
    49→  } = useHotelStore();
    50→
    51→  // Pending payments
    52→  const pendientes = reservas.filter(r => {
    53→    if (r.estado === 'Cancelada' || r.estado === 'Check-Out realizado') return false;
    54→    const total = calcularTotalReserva(r.id);
    55→    const pagado = calcularTotalPagado(r.id);
    56→    return pagado < total;
    57→  });

    59→  // History filters
    60→  const [histFiltroHuesped, setHistFiltroHuesped] = useState('');

    62→  const [histFiltroDesde, setHistFiltroDesde] = useState('');
    63→  const [histFiltroHasta, setHistFiltroHasta] = useState('');
    64→

   112→      toast.error('Ingresá un monto válido');

   153→      <Tabs defaultValue="pendientes">
   154→        <TabsList>
   155→          <TabsTrigger value="pendientes">
   156→            <CreditCard className="w-4 h-4 mr-1" />Cobros pendientes
   157→          </TabsTrigger>
   158→          <TabsTrigger value="historial">
   159→            <FileText className="w-4 h-4 mr-1" />Historial de pagos

   161→        </TabsList>

   163→        {/* =================== TAB: COBROS PENDIENTES =================== */}

   201→                          <Badge className={estadoPagoBadge[r.estadoPago] || ''}>{r.estadoPago}</Badge>
   202→                          {/* Money summary */}
   203→                          <div className="grid grid-cols-3 gap-2 text-center rounded-lg border p-2.5 bg-muted/30">

   222→                              onClick={() => openPagoDialog(r.id)}

   232→                              <FileText className="w-3.5 h-3.5" />
   233→                            </Button>

   235→                        </div>

   238→                  </div>

   240→              </div>

   242→              {/* ── Desktop: Table ── */}

   272→                            <TableCell className="font-medium">
   273→                              <div>{r.huesped}</div>
   274→                              <div className="text-xs text-muted-foreground">{r.dni}</div>

   277→                            <TableCell className="hidden md:table-cell">{formatFecha(r.checkin)}</TableCell>

   292→                                </Button>
   293→                              </div>
   294→                            </TableCell>
   295→                          </TableRow>
   296→                        );
   297→                      })
   298→                    )}
   299→                  </TableBody>
   300→                </Table>
   301→              </div>
   302→            </CardContent>

   306→        {/* =================== TAB: HISTORIAL =================== */}
   307→        <TabsContent value="historial" className="mt-4 space-y-4">
   308→          {/* Filter bar */}
   309→          <Card>
   310→            <CardContent className="p-4">
   311→              <div className="flex flex-wrap gap-3 items-end">
   312→                <div className="grid gap-1.5 flex-1 w-full sm:w-auto sm:min-w-[160px]">
   313→                  <Label className="text-xs text-muted-foreground">Huésped / DNI</Label>

   333→                  <Label className="text-xs text-muted-foreground">Desde</Label>

   337→                  <Label className="text-xs text-muted-foreground">Hasta</Label>

   341→                  <XCircle className="w-3.5 h-3.5 mr-1" />Limpiar

   346→
   347→          {/* Payments table */}
   348→          <Card>
   349→            <CardContent className="p-0">
   350→              {/* ── Mobile: Cards ── */}
   351→              <div className="sm:hidden">

   388→                            </div>

   432→                        const totalR = reserva ? calcularTotalReserva(reserva.id) : 0;
   433→                        const pagadoR = reserva ? calcularTotalPagado(reserva.id) : 0;

   435→                        return (
   436→                          <TableRow key={p.id}>
