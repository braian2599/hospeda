    48→/** Relative time since a date — "recién", "hace 5 min", "hace 2h 15min", "hace 3d" */
    49→function formatRelative(dateStr: string, now: number = Date.now()): string {
    50→  const t = new Date(dateStr).getTime();
    51→  if (isNaN(t)) return '';
    52→  const diff = Math.max(0, now - t);
    53→  const mins = Math.floor(diff / 60000);
    54→  if (mins < 1) return 'recién';
    55→  if (mins < 60) return `hace ${mins} min`;
    56→  const hours = Math.floor(mins / 60);
    57→  const restMins = mins % 60;
    58→  if (hours < 24) return restMins > 0 ? `hace ${hours}h ${restMins}min` : `hace ${hours}h`;
    59→  const days = Math.floor(hours / 24);
    60→  return `hace ${days}d`;
    61→}

    63→/** Time since caja was opened — "abierta hace 2h 15min" */
    64→function formatTimeSinceOpen(openedAt: string, now: number = Date.now()): string {
    65→  const t = new Date(openedAt).getTime();
    66→  if (isNaN(t)) return '';
    67→  const diff = Math.max(0, now - t);
    68→  const mins = Math.floor(diff / 60000);
    69→  if (mins < 1) return 'recién abierta';
    70→  if (mins < 60) return `abierta hace ${mins}min`;
    71→  const hours = Math.floor(mins / 60);
    72→  const restMins = mins % 60;
    73→  return restMins > 0 ? `abierta hace ${hours}h ${restMins}min` : `abierta hace ${hours}h`;
    74→}

   125→/**
   126→ * Suggest a category based on the description text.
   127→ * Returns null if no keyword matches.
   128→ */
   129→function suggestCategory(descripcion: string): MovementCategory | null {
   130→  const lower = descripcion.toLowerCase().trim();
   131→  if (!lower) return null;
   132→  for (const rule of CATEGORY_KEYWORDS) {
   133→    if (rule.keywords.some(kw => lower.includes(kw))) {
   134→      return rule.category;
   135→    }
   136→  }
   137→  return null;
   138→}
   139→
   140→/**
   141→ * Determine the category for a movement based on its fields:
   142→ * - ingresos → "Ingresos varios"
   143→ * - egresos with desc/gasto.tipo containing "mantenimiento" → "Mantenimiento"
   144→ * - egresos with desc containing "retiro" → "Retiros"
   145→ * - egresos with gastoId (linked gasto) → "Gastos"
   146→ * - otherwise → "Otros"
   147→ */
   148→function categorizeMovement(
   149→  mov: { tipo: 'ingreso' | 'egreso'; descripcion: string; gastoId?: string | null; categoria?: string },
   150→  gastoTipo?: string,
   151→): MovementCategory {
   152→  if (mov.tipo === 'ingreso') {
   153→    const suggested = suggestCategory(mov.descripcion);
   154→    return suggested === 'Ingresos varios' ? 'Ingresos varios' : 'Ingresos varios';
   155→  }
   156→  // egreso
   157→  const text = `${mov.descripcion} ${gastoTipo || ''} ${mov.categoria || ''}`.toLowerCase();
   158→  if (text.includes('retiro')) return 'Retiros';
   159→  if (text.includes('mantenim') || text.includes('reparac') || text.includes('arreglo')) return 'Mantenimiento';
   160→  if (mov.gastoId) return 'Gastos';
   161→  // Try suggestion from keywords
   162→  const suggested = suggestCategory(mov.descripcion);
   163→  if (suggested === 'Retiros' || suggested === 'Mantenimiento') return suggested;
   164→  if (suggested === 'Gastos') return 'Gastos';
   165→  return 'Otros';
   166→}
