    34→const formatHora = (f: string) => {

    39→/** Relative time since a date — "recién", "hace 5 min", "hace 2h 15min", "hace 3d" */
    40→function formatRelative(dateStr: string, now: number = Date.now()): string {
    41→  const t = new Date(dateStr).getTime();
    42→  if (isNaN(t)) return '';
    43→  const diff = Math.max(0, now - t);
    44→  const mins = Math.floor(diff / 60000);
    45→  if (mins < 1) return 'recién';
    46→  if (mins < 60) return `hace ${mins} min`;
    47→  const hours = Math.floor(mins / 60);
    48→  const restMins = mins % 60;
    49→  if (hours < 24) return restMins > 0 ? `hace ${hours}h ${restMins}min` : `hace ${hours}h`;
    50→  const days = Math.floor(hours / 24);
    51→  return `hace ${days}d`;
    52→}
    53→
    54→/** Time since caja was opened — "abierta hace 2h 15min" */
    55→function formatTimeSinceOpen(openedAt: string, now: number = Date.now()): string {
    56→  const t = new Date(openedAt).getTime();
    57→  if (isNaN(t)) return '';
    58→  const diff = Math.max(0, now - t);
    59→  const mins = Math.floor(diff / 60000);
    60→  if (mins < 1) return 'recién abierta';
    61→  if (mins < 60) return `abierta hace ${mins}min`;
    62→  const hours = Math.floor(mins / 60);
    63→  const restMins = mins % 60;
    64→  return restMins > 0 ? `abierta hace ${hours}h ${restMins}min` : `abierta hace ${hours}h`;
    65→}
