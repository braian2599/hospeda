   238→   238→// RoomTypeAnalytics — compact analytics section
   239→   239→// ═══════════════════════════════════════════════════════════
   240→   240→
   241→   241→function RoomTypeAnalytics() {
   242→   242→  const habitaciones = useHotelStore(s => s.habitaciones);
   243→   243→  const reservas = useHotelStore(s => s.reservas);
   244→   244→  const today = useMemo(() => todayLocal(), []);
   245→   245→
   246→   246→  const allRooms = Object.values(habitaciones);
   247→   247→  const total = allRooms.length;
   248→   248→
   249→   249→  const typeData = useMemo(() => {
   250→   250→    if (total === 0) return [];
   251→   251→
   252→   252→    const todayDate = safeDate(today).getTime();
   253→   253→
   254→   254→    // Group by type
   255→   255→    const byType = new Map<string, Habitacion[]>();
   256→   256→    allRooms.forEach(h => {
   257→   257→      const t = h.tipo || 'Sin tipo';
   258→   258→      if (!byType.has(t)) byType.set(t, []);
   259→   259→      byType.get(t)!.push(h);
   260→   260→    });
   261→   261→
   262→   262→    return Array.from(byType.entries())
   263→   263→      .sort(([a], [b]) => a.localeCompare(b))
   264→   264→      .map(([tipo, rooms]) => {
   265→   265→        const count = rooms.length;
   266→   266→        const occupied = rooms.filter(h => h.estado === 'Ocupada').length;
   267→   267→        const occupancyRate = Math.round((occupied / count) * 100);
   268→   268→
   269→   269→        // Calculate revenue for occupied rooms
   270→   270→        let revenue = 0;
   271→   271→        rooms.forEach(h => {
   272→   272→          if (h.estado === 'Ocupada' || h.estado === 'Reservada') {
   273→   273→            const res = reservas.find(r =>
   274→   274→              r.habitacion === h.numero &&
   275→   275→              (r.estado === 'Check-In realizado' || r.estado === 'Confirmada') &&
   276→   276→              safeDate(r.checkin).getTime() <= todayDate &&
   277→   277→              safeDate(r.checkout).getTime() >= todayDate
   278→   278→            );
   279→   279→            if (res?.total) revenue += res.total;
   280→   280→          }
   281→   281→        });
   282→   282→
   283→   283→        const Icon = tipoIconMap[tipo] || Bed;
   284→   284→
   285→   285→        return { tipo, count, occupied, occupancyRate, revenue, Icon };
   286→   286→      });
   287→   287→  }, [allRooms, reservas, today, total]);
   288→   288→
   289→   289→  if (total === 0 || typeData.length === 0) return null;
   290→   290→
   291→   291→  const maxCount = Math.max(...typeData.map(d => d.count));
   292→   292→
   293→   293→  // Type color palette
   294→   294→  const typeColors = ['#059669', '#D97706', '#0EA5E9', '#8B5CF6', '#EF4444'];
   295→   295→
   296→   296→  return (
   297→   297→    <div className="rounded-xl border bg-card p-4 space-y-3">
   298→   298→      <div className="flex items-center gap-2">
   299→   299→        <div className="w-7 h-7 rounded-lg bg-primary/20 flex items-center justify-center">
   300→   300→          <Bed className="w-3.5 h-3.5 text-primary" />
   301→   301→        </div>
   302→   302→        <span className="text-sm font-semibold text-foreground">Distribución por tipo</span>
   303→   303→      </div>
   304→   304→
   305→   305→      {/* Horizontal distribution bar */}
   306→   306→      <div className="h-2.5 rounded-full bg-muted overflow-hidden flex">
   307→   307→        {typeData.map((d, i) => {
   308→   308→          const pct = (d.count / total) * 100;
   309→   309→          return (
   310→   310→            <div
   311→   311→              key={d.tipo}
   312→   312→              className="transition-all duration-300"
   313→   313→              style={{ width: `${pct}%`, backgroundColor: typeColors[i % typeColors.length] }}
   314→   314→              title={`${d.tipo}: ${d.count} (${Math.round(pct)}%)`}
   315→   315→            />
   316→   316→          );
   317→   317→        })}
   318→   318→      </div>
   319→   319→
   320→   320→      {/* Type breakdown rows */}
   321→   321→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-5 gap-2">
   322→   322→        {typeData.map((d, i) => {
   323→   323→          const color = typeColors[i % typeColors.length];
   324→   324→          const barPct = (d.count / maxCount) * 100;
   325→   325→          return (
   326→   326→            <div
   327→   327→              key={d.tipo}
   328→   328→              className="rounded-lg border bg-muted/20 p-2.5 space-y-1.5"
   329→   329→            >
   330→   330→              <div className="flex items-center gap-2">
   331→   331→                <d.Icon className="w-3.5 h-3.5 shrink-0" style={{ color }} />
   332→   332→                <span className="text-xs font-semibold text-foreground truncate">{d.tipo}</span>
   333→   333→                <span className="ml-auto text-xs font-bold text-foreground">{d.count}</span>
   334→   334→              </div>
   335→   335→              {/* Mini bar */}
   336→   336→              <div className="h-1.5 rounded-full bg-muted overflow-hidden">
   337→   337→                <div
   338→   338→                  className="h-full rounded-full transition-all duration-300"
   339→   339→                  style={{ width: `${barPct}%`, backgroundColor: color }}
   340→   340→                />
   341→   341→              </div>
   342→   342→              <div className="flex items-center justify-between text-[10px] text-muted-foreground">
   343→   343→                <span>Ocup. {d.occupancyRate}%</span>
   344→   344→                {d.revenue > 0 && <span>${d.revenue.toLocaleString('es-AR')}</span>}
   345→   345→              </div>
   346→   346→            </div>
   347→   347→          );
   348→   348→        })}
   349→   349→      </div>
   350→   350→    </div>
   351→   351→  );
   352→   352→}

   355→   355→// StatusChangePopover — Quick status change for a room
   356→   356→// ═══════════════════════════════════════════════════════════
   357→   357→
   358→   358→function StatusChangePopover({
   359→   359→  numero,
   360→   360→  currentEstado,
   361→   361→  onStatusChange,
   362→   362→  children,
   363→   363→}: {
   364→   364→  numero: string;
   365→   365→  currentEstado: EstadoHabitacion;
   366→   366→  onStatusChange: (numero: string, newEstado: EstadoHabitacion) => void;
   367→   367→  children: React.ReactNode;
   368→   368→}) {
   369→   369→  const [open, setOpen] = useState(false);
   370→   370→  const [confirmDestructive, setConfirmDestructive] = useState<EstadoHabitacion | null>(null);
   371→   371→
   372→   372→  const handleSelect = (newEstado: EstadoHabitacion) => {
   373→   373→    if (newEstado === currentEstado) {
   374→   374→      setOpen(false);
   375→   375→      return;
   376→   376→    }
   377→   377→    // Destructive states need confirmation
   378→   378→    const isDestructive = newEstado === 'Fuera de servicio' || newEstado === 'Mantenimiento';
   379→   379→    if (isDestructive && !confirmDestructive) {
   380→   380→      setConfirmDestructive(newEstado);
   381→   381→      return;
   382→   382→    }
   383→   383→    onStatusChange(numero, newEstado);
   384→   384→    setConfirmDestructive(null);
   385→   385→    setOpen(false);
   386→   386→  };
   387→   387→
   388→   388→  return (
   389→   389→    <Popover open={open} onOpenChange={(v) => { setOpen(v); setConfirmDestructive(null); }}>
   390→   390→      <PopoverTrigger asChild>{children}</PopoverTrigger>
   391→   391→      <PopoverContent className="w-56 p-2" align="start" side="bottom">
   392→   392→        {confirmDestructive ? (
   393→   393→          <div className="space-y-2 p-1">
   394→   394→            <p className="text-xs text-muted-foreground">
   395→   395→              ¿Cambiar a <strong className="text-foreground">{confirmDestructive}</strong>?
   396→   396→              {confirmDestructive === 'Fuera de servicio' && ' La habitación no será asignable.'}
   397→   397→              {confirmDestructive === 'Mantenimiento' && ' Se requiere reparación antes de habilitar.'}
   398→   398→            </p>
   399→   399→            <div className="flex gap-1.5">
   400→   400→              <Button
   401→   401→                size="sm"
   402→   402→                variant="destructive"
   403→   403→                className="h-7 text-xs flex-1"
   404→   404→                onClick={() => handleSelect(confirmDestructive)}
   405→   405→              >
   406→   406→                Confirmar
   407→   407→              </Button>
   408→   408→              <Button
   409→   409→                size="sm"
   410→   410→                variant="ghost"
   411→   411→                className="h-7 text-xs"
   412→   412→                onClick={() => setConfirmDestructive(null)}
   413→   413→              >
   414→   414→                Cancelar
   415→   415→              </Button>
   416→   416→            </div>
   417→   417→          </div>
   418→   418→        ) : (
   419→   419→          <div className="space-y-0.5">
   420→   420→            <div className="text-[10px] uppercase tracking-wider text-muted-foreground font-semibold px-2 pt-1 pb-1.5">
   421→   421→              Cambiar estado
   422→   422→            </div>
   423→   423→            {ALL_ESTADOS.map(estado => {
   424→   424→              const vis = STATUS_VISUAL[estado];
   425→   425→              const Icon = vis.icon;
   426→   426→              const isCurrent = estado === currentEstado;
   427→   427→              return (
   428→   428→                <button
   429→   429→                  key={estado}
   430→   430→                  type="button"
   431→   431→                  onClick={() => handleSelect(estado)}
   432→   432→                  className={`
   433→   433→                    w-full flex items-center gap-2 rounded-md px-2 py-1.5 text-xs
