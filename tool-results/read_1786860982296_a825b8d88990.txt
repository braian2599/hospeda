    68→/** Relative time since a date — "recién", "hace 5 min", "hace 2h 15min", "hace 3d" */
    69→function formatRelative(dateStr: string, now: number = Date.now()): string {
    70→  const t = new Date(dateStr).getTime();
    71→  if (isNaN(t)) return '';
    72→  const diff = Math.max(0, now - t);
    73→  const mins = Math.floor(diff / 60000);
    74→  if (mins < 1) return 'recién';
    75→  if (mins < 60) return `hace ${mins} min`;
    76→  const hours = Math.floor(mins / 60);
    77→  const restMins = mins % 60;
    78→  if (hours < 24) return restMins > 0 ? `hace ${hours}h ${restMins}min` : `hace ${hours}h`;
    79→  const days = Math.floor(hours / 24);
    80→  return `hace ${days}d`;
    81→}

    83→/** Time since caja was opened — "abierta hace 2h 15min" */
    84→function formatTimeSinceOpen(openedAt: string, now: number = Date.now()): string {
    85→  const t = new Date(openedAt).getTime();
    86→  if (isNaN(t)) return '';
    87→  const diff = Math.max(0, now - t);
    88→  const mins = Math.floor(diff / 60000);
    89→  if (mins < 1) return 'recién abierta';
    90→  if (mins < 60) return `abierta hace ${mins}min`;
    91→  const hours = Math.floor(mins / 60);
    92→  const restMins = mins % 60;
    93→  return restMins > 0 ? `abierta hace ${hours}h ${restMins}min` : `abierta hace ${hours}h`;
    94→}

   145→/**
   146→ * Suggest a category based on the description text.
   147→ * Returns null if no keyword matches.
   148→ */
   149→function suggestCategory(descripcion: string): MovementCategory | null {
   150→  const lower = descripcion.toLowerCase().trim();
   151→  if (!lower) return null;
   152→  for (const rule of CATEGORY_KEYWORDS) {
   153→    if (rule.keywords.some(kw => lower.includes(kw))) {
   154→      return rule.category;
   155→    }
   156→  }
   157→  return null;
   158→}
   159→
   160→/**
   161→ * Determine the category for a movement based on its fields:
   162→ * - ingresos → "Ingresos varios"
   163→ * - egresos with desc/gasto.tipo containing "mantenimiento" → "Mantenimiento"
   164→ * - egresos with desc containing "retiro" → "Retiros"
   165→ * - egresos with gastoId (linked gasto) → "Gastos"
   166→ * - otherwise → "Otros"
   167→ */
   168→function categorizeMovement(
   169→  mov: { tipo: 'ingreso' | 'egreso'; descripcion: string; gastoId?: string | null; categoria?: string },
   170→  gastoTipo?: string,
   171→): MovementCategory {
   172→  if (mov.tipo === 'ingreso') {
   173→    const suggested = suggestCategory(mov.descripcion);
   174→    return suggested === 'Ingresos varios' ? 'Ingresos varios' : 'Ingresos varios';
   175→  }
   176→  // egreso
   177→  const text = `${mov.descripcion} ${gastoTipo || ''} ${mov.categoria || ''}`.toLowerCase();
   178→  if (text.includes('retiro')) return 'Retiros';
   179→  if (text.includes('mantenim') || text.includes('reparac') || text.includes('arreglo')) return 'Mantenimiento';
   180→  if (mov.gastoId) return 'Gastos';
   181→  // Try suggestion from keywords
   182→  const suggested = suggestCategory(mov.descripcion);
   183→  if (suggested === 'Retiros' || suggested === 'Mantenimiento') return suggested;
   184→  if (suggested === 'Gastos') return 'Gastos';
   185→  return 'Otros';
   186→}

   188→/** Get the appropriate icon for a payment method */
   189→function getMetodoIcon(metodo: string): ComponentType<{ className?: string }> {
   190→  const lower = metodo.toLowerCase();
   191→  if (lower.includes('efectivo')) return Banknote;
   192→  if (lower.includes('tarjeta') && lower.includes('crédito')) return CreditCard;
   193→  if (lower.includes('tarjeta') && lower.includes('débito')) return CreditCard;
   194→  if (lower.includes('tarjeta')) return CreditCard;
   195→  if (lower.includes('transferencia')) return ArrowRightLeft;
   196→  if (lower.includes('mercado')) return QrCode;
   197→  return Wallet;
   198→}
