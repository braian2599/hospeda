    53→export default function CheckInModule() {
    54→  // ── Granular Zustand selectors (no destructuring) ──
    55→  const reservas = useHotelStore(s => s.reservas);
    56→  const habitaciones = useHotelStore(s => s.habitaciones);
    57→  const realizarCheckIn = useHotelStore(s => s.realizarCheckIn);
    58→  const realizarCheckOut = useHotelStore(s => s.realizarCheckOut);
    59→  const calcularTotalReserva = useHotelStore(s => s.calcularTotalReserva);
    60→  const calcularTotalPagado = useHotelStore(s => s.calcularTotalPagado);
    61→  const nochesEntre = useHotelStore(s => s.nochesEntre);
    62→  const setModulo = useHotelStore(s => s.setModulo);
    63→
    64→  const pendientesCheckIn = useMemo(
    65→    () => reservas.filter(r => r.estado === 'Confirmada'),
    66→    [reservas]
    67→  );
    68→  const pendientesCheckOut = useMemo(
    69→    () => reservas.filter(r => r.estado === 'Check-In realizado'),
    70→    [reservas]
    71→  );
    72→
    73→  // ── Local loading state (brief, for skeleton display) ──
    74→  // The store has no `loading` flag; we simulate a brief fetch on mount so
    75→  // skeleton placeholders are visible to the user.
    76→  const [loading, setLoading] = useState(true);
    77→  useEffect(() => {
    78→    const t = setTimeout(() => setLoading(false), 400);
    79→    return () => clearTimeout(t);
    80→  }, []);
    81→
    82→  // Modal state
    83→  const [modalTipo, setModalTipo] = useState<'checkin' | 'checkout' | null>(null);
    84→  const [selReserva, setSelReserva] = useState<Reserva | null>(null);
    85→  const [dialogKey, setDialogKey] = useState(0);
    86→
    87→  // Check-in form
    88→  const [docVerificado, setDocVerificado] = useState(false);
    89→  const [firmaConformidad, setFirmaConformidad] = useState(false);
    90→  const [llave, setLlave] = useState('');
    91→  const [emergenciaNombre, setEmergenciaNombre] = useState('');
    92→  const [emergenciaTelefono, setEmergenciaTelefono] = useState('');
    93→  const [observaciones, setObservaciones] = useState('');
    94→  const [acompanantes, setAcompanantes] = useState<Acompanante[]>([
    95→    { nombre: '', dni: '', celular: '' },
    96→  ]);
    97→
    98→  // Menores form state
    99→  const [menoresForms, setMenoresForms] = useState<MenorForm[]>([]);
   100→  const [menoresErrors, setMenoresErrors] = useState<string[]>([]);
   101→
   102→  const openCheckIn = (r: Reserva) => {
   103→    setSelReserva(r);
   104→    setDocVerificado(false);
   105→    setFirmaConformidad(false);
   106→    setLlave('');
   107→    setEmergenciaNombre(r.contactoEmergencia?.nombre || '');
   108→    setEmergenciaTelefono(r.contactoEmergencia?.telefono || '');
   109→    setObservaciones(r.observacionesHuesped || '');
   110→    setAcompanantes(r.acompanantes?.length ? [...r.acompanantes] : [{ nombre: '', dni: '', celular: '' }]);
   111→
   112→    // Si ya tiene menores registrados (check-in previo), precargar
   113→    if (r.menores && r.menores.length > 0) {
   114→      setMenoresForms(r.menores.map(m => ({
   115→        nombre: m.nombre,
   116→        documento: m.documento,
   117→        edad: String(m.edad),
   118→        parentesco: m.parentesco,
   119→      })));
   120→    } else if ((r.ninos || 0) > 0) {
   121→      // Generar formularios vacíos según la cantidad de menores declarados
   122→      const forms: MenorForm[] = [];
   123→      for (let i = 0; i < r.ninos!; i++) {
   124→        forms.push(emptyMenorForm());
   125→      }
   126→      setMenoresForms(forms);
   127→    } else {
   128→      setMenoresForms([]);
   129→    }
   130→    setMenoresErrors([]);
   131→
   132→    setModalTipo('checkin');
   133→  };
   134→
   135→  const openCheckOut = (r: Reserva) => {
   136→    setSelReserva(r);
   137→    setModalTipo('checkout');
   138→  };
   139→
   140→  const closeModal = () => {
   141→    setModalTipo(null);
   142→    setSelReserva(null);
   143→    setDialogKey(prev => prev + 1);
   144→  };
   145→
   146→  const validateMenores = (): boolean => {
   147→    const errs: string[] = [];
   148→    for (let i = 0; i < menoresForms.length; i++) {
   149→      const m = menoresForms[i];
   150→      if (!m.nombre.trim()) errs.push(`Menor ${i + 1}: nombre obligatorio`);
   151→      if (!m.documento.trim()) errs.push(`Menor ${i + 1}: documento obligatorio`);
   152→      const edad = parseInt(m.edad);
   153→      if (isNaN(edad) || edad < 0 || edad > 17) errs.push(`Menor ${i + 1}: edad debe ser entre 0 y 17`);
   154→      if (!m.parentesco.trim()) errs.push(`Menor ${i + 1}: parentesco obligatorio`);
   155→    }
   156→    setMenoresErrors(errs);
   157→    return errs.length === 0;
   158→  };
   159→
   160→  const [checkLoading, setCheckLoading] = useState(false);
   161→
   162→  const handleConfirmCheckIn = async () => {
   163→    if (!selReserva || !llave.trim()) return;
   164→
   165→    // Validar menores si la reserva tiene ninos > 0
   166→    const cantNinos = selReserva.ninos || 0;
   167→    if (cantNinos > 0) {
   168→      if (menoresForms.length !== cantNinos) {
   169→        toast.error(`Se requieren datos de ${cantNinos} menor${cantNinos > 1 ? 'es' : ''}`);
   170→        return;
   171→      }
   172→      if (!validateMenores()) {
   173→        toast.error('Completá todos los datos obligatorios de los menores');
   174→        return;
   175→      }
   176→    }
   177→
   178→    const datos: Parameters<typeof realizarCheckIn>[1] = {};
   179→    if (docVerificado) datos.documentoVerificado = true;
   180→    if (firmaConformidad) datos.firmaConformidad = true;
   181→    datos.llaveEntregada = llave.trim();
   182→    if (emergenciaNombre.trim() || emergenciaTelefono.trim()) {
   183→      datos.contactoEmergencia = { nombre: emergenciaNombre.trim(), telefono: emergenciaTelefono.trim() };
   184→    }
   185→    if (observaciones.trim()) datos.observacionesHuesped = observaciones.trim();
   186→    const validAcomp = acompanantes.filter(a => a.nombre.trim());
   187→    if (validAcomp.length > 0) datos.acompanantes = validAcomp;
   188→
   189→    // Pasar menores al store
   190→    if (cantNinos > 0 && menoresForms.length > 0) {
   191→      datos.menores = menoresForms.map(m => ({
   192→        nombre: m.nombre.trim(),
   193→        documento: m.documento.trim(),
   194→        edad: parseInt(m.edad) || 0,
   195→        parentesco: m.parentesco.trim(),
   196→      }));
   197→    }
   198→
   199→    try {
   200→      setCheckLoading(true);
   201→      const resultado = await realizarCheckIn(selReserva.id, datos);
   202→      if (resultado) {
   203→        closeModal();
   204→        const desc = `${selReserva.huesped} - Hab. ${selReserva.habitacion}${cantNinos > 0 ? ` (${cantNinos} menor${cantNinos > 1 ? 'es' : ''} registrados)` : ''}`;
   205→        notifySuccess('Check-in realizado', desc);
   206→      } else {
   207→        toast.error('No se pudo realizar el check-in', { description: 'La reserva ya no está en estado Confirmada, faltan datos de menores, o hubo un error de conexión.' });
   208→      }
   209→    } catch (err) {
   210→      console.error('Error en check-in:', err);
   211→      toast.error('Error al realizar check-in');
   212→    } finally {
   213→      setCheckLoading(false);
   214→    }
   215→  };
   216→
   217→  const handleConfirmCheckOut = async () => {
   218→    if (!selReserva) return;
   219→    try {
   220→      setCheckLoading(true);
   221→      const resultado = await realizarCheckOut(selReserva.id);
   222→      if (resultado) {
   223→        closeModal();
   224→        notifySuccess('Check-out realizado', `${selReserva.huesped} - Hab. ${selReserva.habitacion}`);
   225→      } else {
   226→        toast.error('No se pudo realizar el check-out', { description: 'La reserva ya no está en estado Check-In realizado o hubo un error de conexión.' });
   227→      }
   228→    } catch (err) {
   229→      console.error('Error en check-out:', err);
   230→      toast.error('Error al realizar check-out');
   231→    } finally {
   232→      setCheckLoading(false);
   233→    }
   234→  };
   235→
   236→  const addAcompanante = () => {
   237→    setAcompanantes(prev => [...prev, { nombre: '', dni: '', celular: '' }]);
   238→  };
   239→
   240→  const removeAcompanante = (idx: number) => {
   241→    setAcompanantes(prev => prev.filter((_, i) => i !== idx));
   242→  };
   243→
   244→  const updateAcompanante = (idx: number, field: keyof Acompanante, value: string) => {
   245→    setAcompanantes(prev => prev.map((a, i) => (i === idx ? { ...a, [field]: value } : a)));
   246→  };
   247→
   248→  const updateMenor = (idx: number, field: keyof MenorForm, value: string) => {
   249→    setMenoresForms(prev => prev.map((m, i) => (i === idx ? { ...m, [field]: value } : m)));
   250→  };
   251→
   252→  // Determinar si la reserva seleccionada requiere datos de menores
   253→  const requiereMenores = selReserva && (selReserva.ninos || 0) > 0;
   254→  const menoresCompletos = requiereMenores && menoresForms.length === (selReserva?.ninos || 0)
   255→    && menoresForms.every(m => m.nombre.trim() && m.documento.trim() && m.edad.trim() && m.parentesco.trim());
   256→
   257→  const puedeConfirmarCheckIn = llave.trim() && (!requiereMenores || menoresCompletos);
   258→
   259→  // ── Quick action: navigate to another module ──
   260→  const goTo = (m: ModuloId) => setModulo(m);
   261→
   262→  return (
   263→    <div className="space-y-6">
   264→      <ModuleHeader icon={LogIn} title="Check-In / Check-Out" subtitle="Gestioná ingresos y egresos de huéspedes" />
   265→
   266→      {/* ═══════════ TODAY'S ACTIVITY SUMMARY ═══════════ */}
   267→      {loading ? (
   268→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   269→          <StatCardSkeleton />
   270→          <StatCardSkeleton />
   271→          <StatCardSkeleton />
   272→        </div>
   273→      ) : (
   274→        <TodayActivitySummary />
   275→      )}
   276→
   277→      {/* ═══════════ QUICK ACTIONS ═══════════ */}
   278→      <QuickActions onNavigate={goTo} />
   279→
   280→      {/* ═══════════ CHECK-IN / CHECK-OUT CARDS ═══════════ */}
   281→      {loading ? (
   282→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   283→          <Card>
   284→            <CardHeader className="pb-3">
   285→              <Skeleton className="h-6 w-48" />
   286→            </CardHeader>
   287→            <CardContent className="space-y-3">
   288→              <ListItemSkeleton />
   289→              <ListItemSkeleton />
   290→              <ListItemSkeleton />
   291→            </CardContent>
   292→          </Card>
   293→          <Card>
   294→            <CardHeader className="pb-3">
   295→              <Skeleton className="h-6 w-48" />
   296→            </CardHeader>
   297→            <CardContent className="space-y-3">
   298→              <ListItemSkeleton />
   299→              <ListItemSkeleton />
   300→            </CardContent>
   301→          </Card>
   302→        </div>
   303→      ) : (
   304→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   305→          {/* Pendientes Check-In */}
   306→          <Card className="border-[#BBF7D0]/60 bg-gradient-to-br from-[#ECFDF5]/30 to-white wave-border-hover">
   307→            <CardHeader className="pb-3">
   308→              <CardTitle className="text-lg flex items-center gap-2">
   309→                <LogIn className="w-5 h-5 text-[#059669]" />
   310→                Check-Ins Pendientes
   311→                {pendientesCheckIn.length > 0 && <PulsingDot color="bg-emerald-500" />}
   312→                <Badge
   313→                  key={`cin-${pendientesCheckIn.length}`}
   314→                  variant="secondary"
   315→                  className="ml-auto count-pop bg-[#059669]/15 text-[#047857] border-[#BBF7D0] hover:bg-[#059669]/20"
   316→                >
   317→                  {pendientesCheckIn.length}
   318→                </Badge>
   319→              </CardTitle>
   320→            </CardHeader>
   321→            <CardContent>
   322→              {pendientesCheckIn.length === 0 ? (
   323→                <CelebratoryEmptyState />
   324→              ) : (
   325→                <div className="space-y-3 max-h-[28rem] overflow-y-auto pr-1">
   326→                  {pendientesCheckIn.map(r => (
   327→                    <div key={r.id} className="border rounded-lg p-3 flex items-center justify-between gap-3 hover:bg-[#ECFDF5]/40 transition-colors duration-200 group">
   328→                      <div className="min-w-0 flex-1">
   329→                        <div className="flex items-center gap-2 flex-wrap">
   330→                          <span className="font-semibold text-sm group-hover:text-[#0F2B28] transition-colors">{r.huesped}</span>
   331→                          <Badge className={`font-semibold shadow-sm ${estadoPagoBadge[r.estadoPago] || ''}`}>{r.estadoPago}</Badge>
   332→                          {(r.ninos || 0) > 0 && (
   333→                            <Badge variant="outline" className="bg-[#F5F3FF] text-[#5B21B6] border-[#DDD6FE]">
   334→                              <Baby className="w-3 h-3 mr-1" />{r.ninos} menor{(r.ninos || 0) > 1 ? 'es' : ''}
   335→                            </Badge>
   336→                          )}
   337→                        </div>
   338→                        <div className="text-xs text-muted-foreground mt-1 flex flex-wrap gap-x-3 gap-y-0.5">
   339→                          <span className="flex items-center gap-1"><BedDouble className="w-3 h-3" /> Hab. {r.habitacion}</span>
   340→                          <span>{formatFecha(r.checkin)} → {formatFecha(r.checkout)}</span>
   341→                          <span>{nochesEntre(r.checkin, r.checkout)} noche{s(nochesEntre(r.checkin, r.checkout))}</span>
   342→                          <span>{r.personas} adulto{s(r.personas)}{(r.ninos || 0) > 0 ? ` + ${r.ninos} niño${(r.ninos || 0) > 1 ? 's' : ''}` : ''}</span>
   343→                        </div>
   344→                      </div>
   345→                      <Button size="sm" className="bg-[#059669] hover:bg-[#047857] text-white shadow-sm" onClick={() => openCheckIn(r)}>
   346→                        <KeyRound className="w-3.5 h-3.5 mr-1" />Check-In
   347→                      </Button>
   348→                    </div>
   349→                  ))}
   350→                </div>
   351→              )}
   352→            </CardContent>
   353→          </Card>
   354→
   355→          {/* Pendientes Check-Out */}
   356→          <Card className="border-[#FED7AA]/60 bg-gradient-to-br from-[#FFF7ED]/30 to-white wave-border-hover-amber">
   357→            <CardHeader className="pb-3">
   358→              <CardTitle className="text-lg flex items-center gap-2">
   359→                <LogOut className="w-5 h-5 text-[#EA580C]" />
   360→                Check-Outs Pendientes
   361→                {pendientesCheckOut.length > 0 && <PulsingDot color="bg-orange-500" />}
   362→                <Badge
   363→                  key={`cout-${pendientesCheckOut.length}`}
   364→                  variant="secondary"
   365→                  className="ml-auto count-pop bg-[#EA580C]/15 text-[#9A3412] border-[#FED7AA] hover:bg-[#EA580C]/20"
   366→                >
   367→                  {pendientesCheckOut.length}
   368→                </Badge>
   369→              </CardTitle>
   370→            </CardHeader>
   371→            <CardContent>
   372→              {pendientesCheckOut.length === 0 ? (
   373→                <CelebratoryEmptyState />
   374→              ) : (
   375→                <div className="space-y-3 max-h-[28rem] overflow-y-auto pr-1">
   376→                  {pendientesCheckOut.map(r => (
   377→                    <div key={r.id} className="border rounded-lg p-3 flex items-center justify-between gap-3 hover:bg-[#FFF7ED]/40 transition-colors duration-200 group">
   378→                      <div className="min-w-0 flex-1">
   379→                        <div className="flex items-center gap-2 flex-wrap">
   380→                          <span className="font-semibold text-sm group-hover:text-[#0F2B28] transition-colors">{r.huesped}</span>
   381→                          <Badge className={`font-semibold shadow-sm ${estadoPagoBadge[r.estadoPago] || ''}`}>{r.estadoPago}</Badge>
   382→                          {r.menores && r.menores.length > 0 && (
   383→                            <Badge variant="outline" className="bg-[#F5F3FF] text-[#5B21B6] border-[#DDD6FE]">
   384→                              <Baby className="w-3 h-3 mr-1" />{r.menores.length} menor{r.menores.length > 1 ? 'es' : ''}
   385→                            </Badge>
   386→                          )}
   387→                        </div>
   388→                        <div className="text-xs text-muted-foreground mt-1 flex flex-wrap gap-x-3 gap-y-0.5">
   389→                          <span className="flex items-center gap-1"><BedDouble className="w-3 h-3" /> Hab. {r.habitacion}</span>
   390→                          <span>{formatFecha(r.checkin)} → {formatFecha(r.checkout)}</span>
   391→                          <span>{nochesEntre(r.checkin, r.checkout)} noche{s(nochesEntre(r.checkin, r.checkout))}</span>
   392→                        </div>
   393→                      </div>
   394→                      <Button size="sm" variant="outline" onClick={() => openCheckOut(r)}>
   395→                        <LogOut className="w-3.5 h-3.5 mr-1" />Check-Out
   396→                      </Button>
   397→                    </div>
   398→                  ))}
   399→                </div>
   400→              )}
   401→            </CardContent>
   402→          </Card>
   403→        </div>
   404→      )}
   405→
   406→      {/* =================== MODAL CHECK-IN =================== */}
   407→      <Dialog key={`checkin-${dialogKey}`} open={modalTipo === 'checkin'} onOpenChange={closeModal}>
   408→        <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   409→          {selReserva && (
   410→            <>
   411→              <DialogHeader>
   412→                <DialogTitle className="flex items-center gap-2">
   413→                  <KeyRound className="w-5 h-5" />
   414→                  Check-In — {selReserva.huesped}
   415→                </DialogTitle>
   416→              </DialogHeader>
   417→
   418→              {/* Estado de cuenta */}
   419→              <CheckInAccountStatus reserva={selReserva} />
   420→
   421→              <Separator />
   422→
   423→              {/* ═══════════ DATOS DE MENORES (OBLIGATORIO si ninos > 0) ═══════════ */}
   424→              {requiereMenores && (
   425→                <div className="space-y-4">
   426→                  <div className="flex items-center gap-2">
   427→                    <Baby className="w-5 h-5 text-[#5B21B6]" />
   428→                    <h4 className="font-semibold text-sm">
   429→                      Datos de menores ({selReserva.ninos}) — <span className="text-[#EF4444]">Obligatorio</span>
   430→                    </h4>
   431→                  </div>
   432→                  <p className="text-xs text-muted-foreground">
   433→                    La reserva registra {selReserva.ninos} menor{selReserva.ninos! > 1 ? 'es' : ''}.
   434→                    Debes completar los datos de cada uno para poder realizar el check-in.
   435→                  </p>
   436→
   437→                  {menoresErrors.length > 0 && (
   438→                    <div className="rounded-lg border border-[#FECACA] bg-[#FEE2E2] p-3 space-y-1">
   439→                      {menoresErrors.map((err, i) => (
   440→                        <p key={i} className="text-xs text-[#991B1B]">{err}</p>
   441→                      ))}
   442→                    </div>
   443→                  )}
   444→
   445→                  <div className="space-y-4">
   446→                    {menoresForms.map((menor, idx) => (
   447→                      <div key={idx} className="rounded-lg border p-4 space-y-3 bg-muted/30">
   448→                        <div className="flex items-center justify-between">
   449→                          <span className="text-sm font-semibold flex items-center gap-1.5">
   450→                            <Baby className="w-4 h-4 text-[#7C3AED]" />
   451→                            Menor {idx + 1}

   759→/**
   760→ * TodayActivitySummary
   761→ *
   762→ * Banner con 3 stat cards mostrando la actividad de hoy:
   763→ *  - Check-ins completados hoy
   764→ *  - Check-outs completados hoy
   765→ *  - Estadías activas
   766→ *
   767→ * Usa selectores Zustand granulares (sin destructuring) y `todayLocal` / `daysAgo`
   768→ * de `@/lib/format` para evitar drift UTC.
   769→ */
   770→function TodayActivitySummary() {
   771→  // Granular selectors (no destructuring) — each subscribes to its own slice.
   772→  const reservas = useHotelStore(s => s.reservas);
   773→  const habitaciones = useHotelStore(s => s.habitaciones);
   774→
   775→  const hoyStr = todayLocal();
   776→  const ayerStr = daysAgo(1);
   777→
   778→  // ── Check-ins completados hoy (estado === 'Check-In realizado' AND checkin === hoy) ──
   779→  const checkinsHoy = reservas.filter(
   780→    r => r.estado === 'Check-In realizado' && r.checkin === hoyStr
   781→  ).length;
   782→  const checkinsAyer = reservas.filter(
   783→    r => r.estado === 'Check-In realizado' && r.checkin === ayerStr
   784→  ).length;
   785→
   786→  // ── Check-outs completados hoy (estado === 'Check-Out realizado' AND checkout === hoy) ──
   787→  const checkoutsHoy = reservas.filter(
   788→    r => r.estado === 'Check-Out realizado' && r.checkout === hoyStr
   789→  ).length;
   790→  const checkoutsAyer = reservas.filter(
   791→    r => r.estado === 'Check-Out realizado' && r.checkout === ayerStr
   792→  ).length;
   793→
   794→  // ── Estadías activas (snapshot: estado === 'Check-In realizado') ──
   795→  const estadiasActivas = reservas.filter(
   796→    r => r.estado === 'Check-In realizado'
   797→  ).length;
   798→  // Snapshot aproximado de "estadías activas antes de los check-outs de hoy"
   799→  // = activas actuales + las que hicieron check-out hoy.
   800→  const estadiasPrevias = reservas.filter(
   801→    r => r.estado === 'Check-In realizado'
   802→      || (r.estado === 'Check-Out realizado' && r.checkout === hoyStr)
   803→  ).length;
   804→
   805→  // Ocupación (% de habitaciones ocupadas) — usa `habitaciones` selector.
   806→  const totalHabitaciones = Object.keys(habitaciones).length;
   807→  const ocupadas = Object.values(habitaciones).filter(h => h.estado === 'Ocupada').length;
   808→  const ocupacionPct = totalHabitaciones === 0
   809→    ? null
   810→    : Math.round((ocupadas / totalHabitaciones) * 100);
   811→
   812→  const stats: Array<{
   813→    key: string;
   814→    label: string;
   815→    value: number;
   816→    icon: LucideIcon;
   817→    color: string;
   818→    gradient: string;
   819→    trend?: number;
   820→    sublabel?: string;
   821→  }> = [
   822→    {
   823→      key: 'checkins',
   824→      label: 'Check-ins completados hoy',
   825→      value: checkinsHoy,
   826→      icon: LogIn,
   827→      color: '#059669',
   828→      gradient: 'from-[#059669]/20',
   829→      trend: checkinsHoy === 0 && checkinsAyer === 0 ? undefined : checkinsHoy - checkinsAyer,
   830→    },
   831→    {
   832→      key: 'checkouts',
   833→      label: 'Check-outs completados hoy',
   834→      value: checkoutsHoy,
   835→      icon: LogOut,
   836→      color: '#EA580C',
   837→      gradient: 'from-[#EA580C]/20',
   838→      trend: checkoutsHoy === 0 && checkoutsAyer === 0 ? undefined : checkoutsHoy - checkoutsAyer,
   839→    },
   840→    {
   841→      key: 'estadias',
   842→      label: 'Estadías activas',
   843→      value: estadiasActivas,
   844→      icon: Bed,
   845→      color: '#0F2B28',
   846→      gradient: 'from-[#0F2B28]/20',
   847→      trend: estadiasActivas === 0 && estadiasPrevias === 0 ? undefined : estadiasActivas - estadiasPrevias,
   848→      sublabel: ocupacionPct !== null ? `${ocupacionPct}% ocup.` : undefined,
   849→    },
   850→  ];
   851→
   852→  return (
   853→    <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   854→      {stats.map((s, i) => (
   855→        <StatCard
   856→          key={s.key}
   857→          label={s.label}
   858→          value={s.value}
   859→          icon={s.icon}
   860→          color={s.color}
   861→          gradient={s.gradient}
   862→          trend={s.trend}
   863→          sublabel={s.sublabel}
