   100→function DateRangePickerInline({
   101→  checkin,
   102→  checkout,
   103→  onChangeCheckin,
   104→  onChangeCheckout,
   105→  label,
   106→  disabled,
   107→}: {
   108→  checkin: string;
   109→  checkout: string;
   110→  onChangeCheckin: (val: string) => void;
   111→  onChangeCheckout: (val: string) => void;
   112→  label?: string;
   113→  disabled?: boolean;
   114→}) {
   115→  const [open, setOpen] = useState(false);
   116→  const [localRange, setLocalRange] = useState<{ from?: Date; to?: Date }>({});
   117→  const syncedRef = useRef(false);
   118→
   119→  // Al abrir el popover, sincronizar el estado local con el form.
   120→  // Al cerrar, resetear el flag para la próxima vez.
   121→  useEffect(() => {
   122→    if (open && !syncedRef.current) {
   123→      syncedRef.current = true;
   124→      const f = checkin ? parseISO(checkin) : undefined;
   125→      const t = checkout ? parseISO(checkout) : undefined;
   126→      setLocalRange(f ? { from: f, to: t } : {});
   127→    }
   128→    if (!open) syncedRef.current = false;
   129→  }, [open, checkin, checkout]);
   130→
   131→  const handleSelect = (range: { from?: Date; to?: Date } | undefined) => {
   132→    if (!range?.from) {
   133→      setLocalRange({});
   134→      return;
   135→    }
   136→    setLocalRange(range);
   137→  };
   138→
   139→  const fmt = (d: Date) => {
   140→    const yyyy = d.getFullYear();
   141→    const mm = String(d.getMonth() + 1).padStart(2, '0');
   142→    const dd = String(d.getDate()).padStart(2, '0');
   143→    return `${yyyy}-${mm}-${dd}`;
   144→  };
   145→
   146→  const handleConfirm = () => {
   147→    if (!localRange.from) return;
   148→    onChangeCheckin(fmt(localRange.from));
   149→    onChangeCheckout(localRange.to ? fmt(localRange.to) : '');
   150→    setOpen(false);
   151→  };
   152→
   153→  const from = checkin ? parseISO(checkin) : undefined;
   154→  const to = checkout ? parseISO(checkout) : undefined;
   155→
   156→  const displayText = from && to
   157→    ? `${format(from, 'dd/MM/yyyy')} → ${format(to, 'dd/MM/yyyy')}`
   158→    : from
   159→      ? `${format(from, 'dd/MM/yyyy')} → ...`
   160→      : 'Seleccionar fechas';
   161→
   162→  return (
   163→    <div className="grid gap-1.5">
   164→      {label && <Label>{label} *</Label>}
   165→      <Button
   166→        variant="outline"
   167→        className={cn(
   168→          'w-full justify-start text-left font-normal h-9',
   169→          !checkin && 'text-muted-foreground'
   170→        )}
   171→        disabled={disabled}
   172→        onClick={() => setOpen(true)}
   173→      >
   174→        <CalendarDays className="w-4 h-4 mr-2 text-muted-foreground" />
   175→        {displayText}
   176→      </Button>
   177→      {open
   178→        ? createPortal(
   179→            <div className="fixed inset-0 z-[200] flex items-center justify-center bg-black/40" onClick={() => setOpen(false)}>
   180→              <div
   181→                className="bg-white rounded-xl shadow-2xl border border-[#E2E8F0] pointer-events-auto"
   182→                onClick={e => e.stopPropagation()}
   183→                onMouseDown={e => e.stopPropagation()}
   184→              >
   185→                <Calendar
   186→                  mode="range"
   187→                  selected={localRange.from ? { from: localRange.from, to: localRange.to } as { from: Date; to: Date } : undefined}
   188→                  onSelect={handleSelect}
   189→                  numberOfMonths={2}
   190→                  autoFocus
   191→                  classNames={{
   192→                    range_start: 'rounded-l-md bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   193→                    range_end: 'rounded-r-md bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   194→                    range_middle: 'rounded-none bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   195→                    day: 'relative w-full h-full p-0 text-center [&:first-child[data-selected=true]_button]:rounded-l-md [&:last-child[data-selected=true]_button]:rounded-r-md group/day aspect-square select-none',
   196→                  }}
   197→                />
   198→                <div className="flex justify-end border-t border-[#E2E8F0] p-3">
   199→                  <Button
   200→                    size="sm"
   201→                    onClick={handleConfirm}
   202→                    disabled={!localRange.from}
   203→                    className="bg-[#059669] hover:bg-[#047857] text-white text-xs h-8"
   204→                  >
   205→                    Confirmar fechas
   206→                  </Button>
   207→                </div>
   208→              </div>
   209→            </div>,
   210→            document.body
   211→          )
   212→        : null}
   213→    </div>
   214→  );
   215→}

   245→// ==================== DESGLOSE PRECIO COMPONENT ====================
   246→function DesglosePrecio({ form, computed, formatMoney, s }: {
   247→  form: { tipoTarifa: string; reservaMultiple: boolean; habitacion: string; habitacion2?: string; personas: string; personas2?: string };
   248→  computed: { noches: number; subtotal: number; subtotal2: number; desglose: any };
   249→  formatMoney: (n: number) => string;
   250→  s: (n: number) => string;
   251→}) {
   252→  const d = computed.desglose;
   253→
   254→  const adultoLinea = (d: any) => {
   255→    if (!d || d.totalAdultos === 0) return null;
   256→    const modo = d.modoCobro;
   257→    if (modo === 'porCama') {
   258→      return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''} × ${formatMoney(d.precioRango)}/cama/noche`;
   259→    }
   260→    if (modo === 'porHabitacion') return 'Habitación';
   261→    if (modo === 'porPersona') {
   262→      return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''} × ${formatMoney(d.precioRango)}/noche`;
   263→    }
   264→    // porGrupo
   265→    return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''}`;
   266→  };
   267→
   268→  const renderLines = (sub: number, dLocal: any) => {
   269→    if (!dLocal) return (
   270→      <div className="space-y-1">
   271→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Subtotal</span><span className="font-semibold text-[#1E293B]">{formatMoney(sub)}</span></div>
   272→      </div>
   273→    );
   274→
   275→    const lines: { key: string; label: string; value: string; dim?: boolean }[] = [];
   276→
   277→    const aLabel = adultoLinea(dLocal);
   278→    if (aLabel) {
   279→      lines.push({ key: 'adultos', label: aLabel, value: formatMoney(dLocal.totalAdultos) });
   280→    }
   281→
   282→    if (dLocal.ninosCount > 0 && dLocal.totalNinos > 0) {
   283→      lines.push({
   284→        key: 'ninos',
   285→        label: `${dLocal.ninosCount} niño${dLocal.ninosCount > 1 ? 's' : ''} × ${formatMoney(dLocal.precioNino)}/noche`,
   286→        value: formatMoney(dLocal.totalNinos),
   287→      });
   288→    }
   289→
   290→    if (dLocal.nochesGratis > 0) {
   291→      const totalBase = dLocal.totalAdultos + dLocal.totalNinos;
   292→      const ahorro = totalBase > 0 && dLocal.nochesCobrables > 0
   293→        ? Math.round(totalBase / dLocal.nochesCobrables * dLocal.nochesGratis)
   294→        : 0;
   295→      lines.push({
   296→        key: 'cortesia',
   297→        label: `${dLocal.nochesGratis} noche${dLocal.nochesGratis > 1 ? 's' : ''} de cortesía`,
   298→        value: `- ${formatMoney(ahorro)}`,
   299→        dim: true,
   300→      });
   301→    }
   302→
   303→    return (
   304→      <div className="space-y-1">
   305→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Tarifa</span><span className="font-semibold text-[#1E293B] capitalize">{form.tipoTarifa}</span></div>
   306→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Noches</span><span className="font-semibold text-[#1E293B]">{computed.noches}{dLocal?.nochesGratis > 0 ? ` (${dLocal.nochesCobrables} cobrables)` : ''}</span></div>
   307→        {lines.map(l => (
   308→          <div key={l.key} className="flex justify-between items-center py-1 text-[13px]">
   309→            <span className={l.dim ? 'text-[#94A3B8] line-through' : 'text-[#64748B]'}>{l.label}</span>
   310→            <span className={l.dim ? 'font-medium text-[#94A3B8]' : 'font-semibold text-[#1E293B]'}>{l.value}</span>
   311→          </div>
   312→        ))}
   313→        <div className="flex justify-between items-center py-1.5 mt-1 border-t border-[#E2E8F0] text-[13px]">
   314→          <span className="font-medium text-[#475569]">Subtotal</span>
   315→          <span className="font-bold text-[#1E293B]">{formatMoney(sub)}</span>
   316→        </div>
   317→      </div>
   318→    );
   319→  };
   320→
   321→  return (
   322→    <>
   323→      {form.reservaMultiple ? (
   324→        <>
   325→          <p className="text-[11px] font-semibold text-[#64748B] uppercase tracking-wider">Habitación {form.habitacion}</p>
   326→          {renderLines(computed.subtotal, d)}
   327→          {computed.subtotal2 > 0 && (
   328→            <>
   329→              <p className="text-[11px] font-semibold text-[#64748B] uppercase tracking-wider pt-2">Habitación {form.habitacion2}</p>
   330→              {renderLines(computed.subtotal2, null)}
   331→            </>
   332→          )}
   333→        </>
   334→      ) : (
   335→        renderLines(computed.subtotal, d)
   336→      )}
   337→    </>
   338→  );
   339→}
