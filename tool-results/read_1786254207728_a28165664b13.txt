    32→const formatHora = (f: string) => {
    33→  const d = new Date(f);

    37→/** Relative time since a date — "recién", "hace 5 min", "hace 2h 15min", "hace 3d" */
    38→function formatRelative(dateStr: string, now: number = Date.now()): string {
    39→  const t = new Date(dateStr).getTime();
    40→  if (isNaN(t)) return '';
    41→  const diff = Math.max(0, now - t);
    42→  const mins = Math.floor(diff / 60000);
    43→  if (mins < 1) return 'recién';
    44→  if (mins < 60) return `hace ${mins} min`;
    45→  const hours = Math.floor(mins / 60);
    46→  const restMins = mins % 60;
    47→  if (hours < 24) return restMins > 0 ? `hace ${hours}h ${restMins}min` : `hace ${hours}h`;
    48→  const days = Math.floor(hours / 24);
    49→  return `hace ${days}d`;
    50→}
    51→
    52→/** Time since caja was opened — "abierta hace 2h 15min" */
    53→function formatTimeSinceOpen(openedAt: string, now: number = Date.now()): string {
    54→  const t = new Date(openedAt).getTime();
    55→  if (isNaN(t)) return '';
    56→  const diff = Math.max(0, now - t);
    57→  const mins = Math.floor(diff / 60000);
    58→  if (mins < 1) return 'recién abierta';
    59→  if (mins < 60) return `abierta hace ${mins}min`;
    60→  const hours = Math.floor(mins / 60);
    61→  const restMins = mins % 60;
    62→  return restMins > 0 ? `abierta hace ${hours}h ${restMins}min` : `abierta hace ${hours}h`;
    63→}
