    83→    83→function RangoFila({ rango, index, onRemove, onUpdate, modoCobro, totalRangos }: {
    84→    84→  rango: RangoPrecio;
    85→    85→  index: number;

    96→    96→      ? `${rango.minPersonas} persona${rango.minPersonas > 1 ? 's' : ''}`

   110→   110→          className="h-8 text-right"
   111→   111→          value={rango.precio}
