     1→'use client';
     2→
     3→import { useState, useMemo, useCallback } from 'react';
     4→import { useHotelStore } from '@/lib/store';
     5→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
     6→import { Button } from '@/components/ui/button';
     7→import { Input } from '@/components/ui/input';
     8→import { Label } from '@/components/ui/label';
     9→import { Badge } from '@/components/ui/badge';
    10→import {
    11→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter, DialogClose,
    12→} from '@/components/ui/dialog';
    13→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table';
    14→import {
    15→  Wallet, Lock, Unlock, Plus, Minus, Loader2, Pencil, Trash2, AlertTriangle, Tag,
    16→} from 'lucide-react';
    17→import ModuleHeader from '@/components/layout/ModuleHeader';
    18→import { toast } from 'sonner';
    19→import { DialogTrigger } from '@/components/ui/dialog';
    20→import { BILLETES } from '@/lib/types';
    21→
    22→const formatFechaHora = (f: string) => {
    23→  const d = new Date(f);
    24→  return d.toLocaleDateString('es-AR') + ' ' + d.toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit' });
    25→};
    26→
    27→const formatHora = (f: string) => {
    28→  const d = new Date(f);
    29→  return d.toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit' });
    30→};
    31→
    32→const formatMoney = (n: number) => '$' + n.toLocaleString('es-AR', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    33→
    34→const METODOS = ['Efectivo', 'Transferencia', 'Tarjeta de Credito', 'Tarjeta de Debito', 'Mercado Pago', 'Otro'];
    35→
    36→export default function CajaModule() {
    37→  const {
    38→    caja, abrirCaja, registrarMovimientoCaja, cerrarCaja, saldoActualCaja,
    39→    editarMovimientoCaja, eliminarMovimientoCaja, usuarioActual,
    40→    metodosPago, categoriasGastos,
    41→  } = useHotelStore();
    42→
    43→  // Loading states
    44→  const [loadingAbrir, setLoadingAbrir] = useState(false);
    45→  const [loadingMovimiento, setLoadingMovimiento] = useState(false);
    46→  const [loadingCerrar, setLoadingCerrar] = useState(false);
    47→
    48→  // Form states
    49→  const [montoInicial, setMontoInicial] = useState('');
    50→  const [showApertura, setShowApertura] = useState(false);
    51→  const [showMovForm, setShowMovForm] = useState<'ingreso' | 'egreso' | null>(null);
    52→  const [movMonto, setMovMonto] = useState('');
    53→  const [movDesc, setMovDesc] = useState('');
    54→  const [movMetodo, setMovMetodo] = useState('Efectivo');
    55→  const [movCategoria, setMovCategoria] = useState('');
    56→
    57→  // Close dialog
    58→  const [showCierre, setShowCierre] = useState(false);
    59→  const [billetes, setBilletes] = useState<Record<number, number>>(() => Object.fromEntries(BILLETES.map(b => [b, 0])));
    60→
    61→  // Edit dialog
    62→  const [editingMov, setEditingMov] = useState<{ id: string; monto: number; descripcion: string } | null>(null);
    63→  const [editMonto, setEditMonto] = useState('');
    64→  const [editDesc, setEditDesc] = useState('');
    65→  const [loadingEdit, setLoadingEdit] = useState(false);
    66→  const [loadingDelete, setLoadingDelete] = useState(false);
    67→
    68→  const saldo = saldoActualCaja();
    69→  const movimientos = caja.movimientos || [];
    70→
    71→  const isAdminOrOwner = usuarioActual?.rol === 'owner' || usuarioActual?.rol === 'admin';
    72→
    73→  // Summary calculations
    74→  const resumenOtros = useMemo(() => {
    75→    const res: Record<string, { ingresos: number; egresos: number }> = {};
    76→    movimientos.forEach(m => {
    77→      if (m.metodo !== 'Efectivo') {
    78→        if (!res[m.metodo]) res[m.metodo] = { ingresos: 0, egresos: 0 };
    79→        res[m.metodo][m.tipo === 'ingreso' ? 'ingresos' : 'egresos'] += m.monto;
    80→      }
    81→    });
    82→    return res;
    83→  }, [movimientos]);
    84→
    85→  const totalIngresosPorMetodo = useMemo(() => {
    86→    const res: Record<string, number> = {};
    87→    movimientos.forEach(m => {
    88→      if (m.tipo === 'ingreso') res[m.metodo] = (res[m.metodo] || 0) + m.monto;
    89→    });
    90→    return res;
    91→  }, [movimientos]);
    92→
    93→  const totalOtros = Object.values(resumenOtros).reduce((s, v) => s + v.ingresos - v.egresos, 0);
    94→  const totalEfectivo = useMemo(() => BILLETES.reduce((s, b) => s + b * (billetes[b] || 0), 0), [billetes]);
    95→  const totalTurno = totalEfectivo + totalOtros;
    96→
    97→  // Handlers
    98→  const handleAbrir = async () => {
    99→    const m = parseFloat(montoInicial);
   100→    if (isNaN(m) || m < 0) return toast.error('Monto invalido');
   101→    setLoadingAbrir(true);
   102→    try {
   103→      const ok = await abrirCaja(m);
   104→      if (ok) {
   105→        toast.success('Caja abierta', { description: 'Turno iniciado' });
   106→        setShowApertura(false);
   107→        setMontoInicial('');
   108→      } else {
   109→        toast.error('Error al abrir la caja');
   110→      }
   111→    } catch (e) {
   112→      toast.error('Error al abrir la caja');
   113→    }
   114→    setLoadingAbrir(false);
   115→  };
   116→
   117→  const handleMovimiento = async () => {
   118→    if (!showMovForm) return;
   119→    const m = parseFloat(movMonto);
   120→    if (isNaN(m) || m <= 0) return toast.error('Monto invalido');
   121→    if (showMovForm === 'egreso' && !movCategoria) return toast.error('Selecciona una categoria de gasto');
   122→    const desc = movDesc.trim() || (showMovForm === 'ingreso' ? 'Ingreso manual' : `Egreso: ${movCategoria}`);
   123→    setLoadingMovimiento(true);
   124→    try {
   125→      const ok = await registrarMovimientoCaja(showMovForm, m, desc, movMetodo, showMovForm === 'egreso' ? movCategoria : undefined);
   126→      if (ok) {
   127→        toast.success(`${showMovForm === 'ingreso' ? 'Ingreso' : 'Egreso'} registrado`, { description: formatMoney(m) });
   128→        setShowMovForm(null);
   129→        setMovMonto('');
   130→        setMovDesc('');
   131→        setMovMetodo('Efectivo');
   132→        setMovCategoria('');
   133→      } else {
   134→        toast.error(`Error al registrar ${showMovForm}`);
   135→      }
   136→    } catch (e) {
   137→      toast.error(`Error al registrar ${showMovForm}`);
   138→    }
   139→    setLoadingMovimiento(false);
   140→  };
   141→
   142→  const handleCerrar = async () => {
   143→    setLoadingCerrar(true);
   144→    try {
   145→      const cierre = await cerrarCaja(billetes, totalOtros);
   146→      if (cierre) {
   147→        toast.success('Caja cerrada', { description: 'Turno finalizado' });
   148→        setShowCierre(false);
   149→        setBilletes(Object.fromEntries(BILLETES.map(b => [b, 0])));
   150→      } else {
   151→        toast.error('Error al cerrar la caja');
   152→      }
   153→    } catch (e) {
   154→      toast.error('Error al cerrar la caja');
   155→    }
   156→    setLoadingCerrar(false);
   157→  };
   158→
   159→  const handleEditOpen = (mov: typeof movimientos[0]) => {
   160→    setEditingMov({ id: mov.id, monto: mov.monto, descripcion: mov.descripcion });
   161→    setEditMonto(String(mov.monto));
   162→    setEditDesc(mov.descripcion);
   163→  };
   164→
   165→  const handleEditSave = async () => {
   166→    if (!editingMov) return;
   167→    const m = parseFloat(editMonto);
   168→    if (isNaN(m) || m <= 0) return toast.error('Monto invalido');
   169→    if (!editDesc.trim()) return toast.error('Descripcion requerida');
   170→    setLoadingEdit(true);
   171→    try {
   172→      const ok = await editarMovimientoCaja(editingMov.id, { monto: m, descripcion: editDesc.trim() });
   173→      if (ok) {
   174→        toast.success('Movimiento actualizado');
   175→        setEditingMov(null);
   176→      } else {
   177→        toast.error('Error al editar');
   178→      }
   179→    } catch (e) {
   180→      toast.error('Error al editar movimiento');
   181→    }
   182→    setLoadingEdit(false);
   183→  };
   184→
   185→  const handleDelete = async (movId: string) => {
   186→    if (!confirm('Eliminar este movimiento?')) return;
   187→    setLoadingDelete(true);
   188→    try {
   189→      const ok = await eliminarMovimientoCaja(movId);
   190→      if (ok) {
   191→        toast.success('Movimiento eliminado');
   192→      } else {
   193→        toast.error('Error al eliminar');
   194→      }
   195→    } catch (e) {
   196→      toast.error('Error al eliminar movimiento');
   197→    }
   198→    setLoadingDelete(false);
   199→  };
   200→
   201→  // Saldo esperado (efectivo only)
   202→  const saldoEsperadoEfectivo = useMemo(() => {
   203→    if (!caja.apertura) return 0;
   204→    let s = caja.apertura.montoInicial;
   205→    movimientos.forEach(m => {
   206→      if (m.metodo === 'Efectivo') s += m.tipo === 'ingreso' ? m.monto : -m.monto;
   207→    });
   208→    return s;
   209→  }, [caja.apertura, movimientos]);
   210→
   211→  const diferencia = totalEfectivo - saldoEsperadoEfectivo;
   212→
   213→  const cancelarForm = () => { setShowMovForm(null); setMovMonto(''); setMovDesc(''); setMovMetodo('Efectivo'); setMovCategoria(''); };
   214→
   215→  return (
   216→    <div className="space-y-6">
   217→      <ModuleHeader icon={Wallet} title="Caja" subtitle="Controla los movimientos de dinero del dia" />
   218→
   219→      {caja.estado === 'cerrada' ? (
   220→        /* ═══════ CAJA CERRADA ═══════ */
   221→        <Card>
   222→          <CardContent className="text-center py-10 space-y-4">
   223→            <Lock className="w-12 h-12 mx-auto text-destructive" />
   224→            <div>
   225→              <h3 className="text-lg font-semibold">Caja cerrada</h3>
   226→              <p className="text-sm text-muted-foreground">Inicie un nuevo turno para comenzar a operar.</p>
   227→            </div>
   228→            {!showApertura ? (
   229→              <Button size="lg" onClick={() => setShowApertura(true)}><Unlock className="w-4 h-4 mr-1" />Abrir caja</Button>
   230→            ) : (
   231→              <div className="max-w-xs mx-auto space-y-2">
   232→                <Label className="text-sm text-muted-foreground">Monto inicial en efectivo</Label>
   233→                <Input type="number" placeholder="0.00" step="0.01" min="0" value={montoInicial} onChange={e => setMontoInicial(e.target.value)} autoFocus />
   234→                <div className="flex gap-2">
   235→                  <Button onClick={handleAbrir} className="flex-1" disabled={loadingAbrir}>
   236→                    {loadingAbrir ? <Loader2 className="w-4 h-4 animate-spin mr-1" /> : null}Confirmar apertura
   237→                  </Button>
   238→                  <Button variant="secondary" onClick={() => setShowApertura(false)} disabled={loadingAbrir}>Cancelar</Button>
   239→                </div>
   240→              </div>
   241→            )}
   242→            {caja.historial && caja.historial.length > 0 && (
   243→              <p className="text-xs text-muted-foreground">Ultimo cierre: {formatFechaHora(caja.historial[caja.historial.length - 1].cierre.fecha)}</p>
   244→            )}
   245→          </CardContent>
   246→        </Card>
   247→      ) : (
   248→        /* ═══════ CAJA ABIERTA ═══════ */
   249→        <div className="space-y-4">
   250→          {/* ── Mobile: compact status bar ── */}
   251→          <Card className="lg:hidden">
   252→            <CardContent className="p-3 space-y-3">
   253→              <div className="flex items-center justify-between">
   254→                <h3 className="font-semibold flex items-center gap-2 text-sm"><Unlock className="w-4 h-4 text-[#166534]" /> Caja abierta</h3>
   255→                <Dialog open={showCierre} onOpenChange={setShowCierre}>
   256→                  <DialogTrigger asChild>
   257→                    <Button variant="destructive" size="sm" className="h-7 text-xs"><Lock className="w-3.5 h-3.5 mr-1" />Cerrar</Button>
   258→                  </DialogTrigger>
   259→                  <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
   260→                    <ClosingDialogContent
   261→                      saldoEsperado={saldoEsperadoEfectivo}
   262→                      totalEfectivo={totalEfectivo}
   263→                      diferencia={diferencia}
   264→                      totalOtros={totalOtros}
   265→                      resumenOtros={resumenOtros}
   266→                      totalIngresosPorMetodo={totalIngresosPorMetodo}
   267→                      billetes={billetes}
   268→                      setBilletes={setBilletes}
   269→                      loadingCerrar={loadingCerrar}
   270→                      handleCerrar={handleCerrar}
   271→                      setShowCierre={setShowCierre}
   272→                    />
   273→                  </DialogContent>
   274→                </Dialog>
   275→              </div>
   276→              {/* Saldo + info */}
   277→              <div className="grid grid-cols-3 gap-2 text-center rounded-lg border p-2.5 bg-muted/30">
   278→                <div>
   279→                  <p className="text-[10px] text-muted-foreground leading-tight">Saldo</p>
   280→                  <p className="text-sm font-bold">{formatMoney(saldo)}</p>
   281→                </div>
   282→                <div>
   283→                  <p className="text-[10px] text-muted-foreground leading-tight">Movimientos</p>
   284→                  <p className="text-sm font-semibold">{movimientos.length}</p>
   285→                </div>
   286→                <div>
   287→                  <p className="text-[10px] text-muted-foreground leading-tight">Apertura</p>
   288→                  <p className="text-sm font-medium">{caja.apertura ? formatHora(caja.apertura.fecha) : '--'}</p>
   289→                </div>
   290→              </div>
   291→              {/* Action buttons */}
   292→              <div className="grid grid-cols-2 gap-2">
   293→                <Button variant="outline" className="h-9 text-sm" onClick={() => setShowMovForm('ingreso')}><Plus className="w-4 h-4 mr-1" />Ingreso</Button>
   294→                <Button variant="outline" className="h-9 text-sm" onClick={() => setShowMovForm('egreso')}><Minus className="w-4 h-4 mr-1" />Egreso</Button>
   295→              </div>
   296→              {/* Movement form inline */}
   297→              {showMovForm && (
   298→                <MovFormInline
   299→                  tipo={showMovForm}
   300→                  movMonto={movMonto}
   301→                  setMovMonto={setMovMonto}
   302→                  movDesc={movDesc}
   303→                  setMovDesc={setMovDesc}
   304→                  movMetodo={movMetodo}
   305→                  setMovMetodo={setMovMetodo}
   306→                  movCategoria={movCategoria}
   307→                  setMovCategoria={setMovCategoria}
   308→                  loading={loadingMovimiento}
   309→                  onGuardar={handleMovimiento}
   310→                  onCancelar={cancelarForm}
   311→                  metodosPago={metodosPago}
   312→                  categoriasGastos={categoriasGastos}
   313→                />
   314→              )}
   315→            </CardContent>
   316→          </Card>
   317→
   318→          {/* ── Mobile: movement cards ── */}
   319→          <Card className="lg:hidden">
   320→            <CardHeader className="pb-3"><CardTitle className="text-base">Movimientos del turno</CardTitle></CardHeader>
   321→            <CardContent className="p-0">
   322→              {movimientos.length === 0 ? (
   323→                <div className="text-center py-10 text-muted-foreground">Sin movimientos.</div>
   324→              ) : (
   325→                <div className="divide-y">
   326→                  {[...movimientos].reverse().map((m) => (
   327→                    <div key={m.id} className="p-3.5 space-y-1.5">
   328→                      <div className="flex items-start justify-between gap-2">
   329→                        <div className="flex items-center gap-2">
   330→                          {m.tipo === 'ingreso' ? (
   331→                            <Badge className="bg-[#DCFCE7] text-[#166534]"><Plus className="w-3 h-3 mr-0.5" />Ingreso</Badge>
   332→                          ) : (
   333→                            <Badge className="bg-[#FEE2E2] text-[#991B1B]"><Minus className="w-3 h-3 mr-0.5" />Egreso</Badge>
   334→                          )}
   335→                          <Badge variant="secondary">{m.metodo}</Badge>
   336→                        </div>
   337→                        <p className={`text-sm font-bold shrink-0 ${m.tipo === 'ingreso' ? 'text-[#166534]' : 'text-[#991B1B]'}`}>
   338→                          {m.tipo === 'ingreso' ? '+' : '-'}{formatMoney(m.monto)}
   339→                        </p>
   340→                      </div>
   341→                      {m.descripcion && (
   342→                        <p className="text-xs text-muted-foreground leading-relaxed">{m.descripcion}</p>
   343→                      )}
   344→                      <div className="flex items-center justify-between">
   345→                        <p className="text-[10px] text-muted-foreground">{formatFechaHora(m.fecha)}</p>
   346→                        {isAdminOrOwner && (
   347→                          <div className="flex gap-1">
   348→                            <Button size="icon" variant="ghost" className="h-6 w-6" onClick={() => handleEditOpen(m)}><Pencil className="w-3 h-3" /></Button>
   349→                            <Button size="icon" variant="ghost" className="h-6 w-6 text-destructive" onClick={() => handleDelete(m.id)} disabled={loadingDelete}>
   350→                              {loadingDelete ? <Loader2 className="w-3 h-3 animate-spin" /> : <Trash2 className="w-3 h-3" />}
   351→                            </Button>
   352→                          </div>
   353→                        )}
   354→                      </div>
   355→                    </div>
   356→                  ))}
   357→                </div>
   358→              )}
   359→            </CardContent>
   360→          </Card>
   361→
   362→          {/* ═══════ DESKTOP ═══════ */}
   363→          <div className="hidden lg:grid lg:grid-cols-3 gap-4">
   364→            <div className="lg:col-span-2 space-y-4">
   365→              {/* Status bar */}
   366→              <Card>
   367→                <CardContent className="flex items-center justify-between py-3">
   368→                  <h3 className="font-semibold flex items-center gap-2"><Unlock className="w-5 h-5 text-[#166534]" /> Caja abierta</h3>
   369→                  <Dialog open={showCierre} onOpenChange={setShowCierre}>
   370→                    <DialogTrigger asChild>
   371→                      <Button variant="destructive" disabled={loadingCerrar}><Lock className="w-4 h-4 mr-1" />Cerrar caja</Button>
   372→                    </DialogTrigger>
   373→                    <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
   374→                      <ClosingDialogContent
   375→                        saldoEsperado={saldoEsperadoEfectivo}
   376→                        totalEfectivo={totalEfectivo}
   377→                        diferencia={diferencia}
   378→                        totalOtros={totalOtros}
   379→                        resumenOtros={resumenOtros}
   380→                        totalIngresosPorMetodo={totalIngresosPorMetodo}
   381→                        billetes={billetes}
   382→                        setBilletes={setBilletes}
   383→                        loadingCerrar={loadingCerrar}
   384→                        handleCerrar={handleCerrar}
   385→                        setShowCierre={setShowCierre}
   386→                      />
   387→                    </DialogContent>
   388→                  </Dialog>
   389→                </CardContent>
   390→              </Card>
   391→
   392→              {/* Movements table */}
   393→              <Card>
   394→                <CardHeader className="pb-3"><CardTitle className="text-base">Movimientos del turno</CardTitle></CardHeader>
   395→                <CardContent>
   396→                  {movimientos.length === 0 ? <p className="text-sm text-muted-foreground">Sin movimientos.</p> : (
   397→                    <Table>
   398→                      <TableHeader>
   399→                        <TableRow>
   400→                          <TableHead>Hora</TableHead>
   401→                          <TableHead>Tipo</TableHead>
   402→                          <TableHead>Monto</TableHead>
   403→                          <TableHead>Metodo</TableHead>
   404→                          <TableHead>Descripcion</TableHead>
   405→                          {isAdminOrOwner && <TableHead className="w-[80px]">Acciones</TableHead>}
   406→                        </TableRow>
   407→                      </TableHeader>
   408→                      <TableBody>
   409→                        {movimientos.map((m) => (
   410→                          <TableRow key={m.id}>
   411→                            <TableCell className="text-xs">{formatFechaHora(m.fecha)}</TableCell>
   412→                            <TableCell>
   413→                              {m.tipo === 'ingreso' ? (
   414→                                <Badge className="bg-[#DCFCE7] text-[#166534]"><Plus className="w-3 h-3 mr-1" />Ingreso</Badge>
   415→                              ) : (
   416→                                <Badge className="bg-[#FEE2E2] text-[#991B1B]"><Minus className="w-3 h-3 mr-1" />Egreso</Badge>
   417→                              )}
   418→                            </TableCell>
   419→                            <TableCell className={`font-medium ${m.tipo === 'ingreso' ? 'text-[#166534]' : 'text-[#991B1B]'}`}>
   420→                              {m.tipo === 'ingreso' ? '+' : '-'}{formatMoney(m.monto)}
   421→                            </TableCell>
   422→                            <TableCell>{m.metodo}</TableCell>
   423→                            <TableCell className="text-sm text-muted-foreground">{m.descripcion}</TableCell>
   424→                            {isAdminOrOwner && (
   425→                              <TableCell>
   426→                                <div className="flex gap-1">
   427→                                  <Button size="icon" variant="ghost" className="h-7 w-7" onClick={() => handleEditOpen(m)}><Pencil className="w-3.5 h-3.5" /></Button>
   428→                                  <Button size="icon" variant="ghost" className="h-7 w-7 text-destructive" onClick={() => handleDelete(m.id)} disabled={loadingDelete}>
   429→                                    {loadingDelete ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Trash2 className="w-3.5 h-3.5" />}
   430→                                  </Button>
   431→                                </div>
   432→                              </TableCell>
   433→                            )}
   434→                          </TableRow>
   435→                        ))}
   436→                      </TableBody>
   437→                    </Table>
   438→                  )}
   439→                </CardContent>
   440→              </Card>
   441→            </div>
   442→
   443→            {/* Info Panel */}
   444→            <div className="space-y-4">
   445→              <Card>
   446→                <CardHeader className="bg-[#F1F5F9] pb-3"><CardTitle className="text-sm">Informacion</CardTitle></CardHeader>
   447→                <CardContent className="space-y-2 text-sm">
   448→                  {caja.apertura && (
   449→                    <>

   527→  resumenOtros: Record<string, { ingresos: number; egresos: number }>;
   528→  totalIngresosPorMetodo: Record<string, number>;
   529→  billetes: Record<number, number>; setBilletes: (b: Record<number, number>) => void;
   530→  loadingCerrar: boolean; handleCerrar: () => void; setShowCierre: (v: boolean) => void;

   533→  return (
   534→    <>

'bg-[#DCFCE7]' : 'bg-[#FEE2E2]'}`}><p className="text-[10px] text-muted-foreground">Diferencia</p><p className={`font-bold text-sm ${diferencia === 0 ?    544→        <div className="flex items-center gap-2 p-2 bg-[#FEF3C7] rounded-lg text-[#92400E] text-sm"><AlertTriangle className="w-4 h-4 shrink-0" /><span>Hay una diferencia de {fmt(Math.abs(diferencia))} en efectivo</span></div>

   596→
   597→function MovFormInline({
   598→  tipo, movMonto, setMovMonto, movDesc, setMovDesc,
   599→  movMetodo, setMovMetodo, movCategoria, setMovCategoria,
   600→  loading, onGuardar, onCancelar, metodosPago, categoriasGastos,
   601→}: {
   602→  tipo: 'ingreso' | 'egreso';
   603→  movMonto: string; setMovMonto: (v: string) => void;
   604→  movDesc: string; setMovDesc: (v: string) => void;
   605→  movMetodo: string; setMovMetodo: (v: string) => void;
   606→  movCategoria: string; setMovCategoria: (v: string) => void;
   607→  loading: boolean; onGuardar: () => void; onCancelar: () => void;
   608→  metodosPago: { id: string; nombre: string }[];
   609→  categoriasGastos: string[];
   610→}) {
   611→  const methods = metodosPago.length > 0 ? metodosPago.map(m => m.nombre) : METODOS;
   612→  return (
   613→    <div className="border rounded-lg p-3 space-y-2">
   614→      <p className="font-medium text-sm">
   615→        {tipo === 'ingreso' ? (
   616→          <span className="flex items-center gap-1"><Plus className="w-3.5 h-3.5 text-[#166534]" />Registrar ingreso</span>
   617→        ) : (
   618→          <span className="flex items-center gap-1"><Minus className="w-3.5 h-3.5 text-[#991B1B]" />Registrar egreso</span>
   619→        )}
   620→      </p>
   621→      <div className="grid grid-cols-2 gap-2">
   622→        <Input type="number" placeholder="Monto" step="0.01" min="0.01" value={movMonto} onChange={e => setMovMonto(e.target.value)} />
   623→        <select
   624→          className="flex h-9 w-full rounded-md border border-input bg-background px-3 py-1 text-sm shadow-sm transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring"
   625→          value={movMetodo}
   626→          onChange={e => setMovMetodo(e.target.value)}
   627→        >
   628→          {methods.map(m => (<option key={m} value={m}>{m}</option>))}
   629→        </select>
   630→      </div>
   631→      {/* Categoria de gasto - solo para egresos */}
   632→      {tipo === 'egreso' && (
   633→        <div className="relative">
   634→          <Tag className="absolute left-2.5 top-1/2 -translate-y-1/2 w-3.5 h-3.5 text-muted-foreground pointer-events-none" />
   635→          <select
   636→            className="flex h-9 w-full rounded-md border border-input bg-background pl-8 pr-3 py-1 text-sm shadow-sm transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring appearance-none"
   637→            value={movCategoria}
   638→            onChange={e => setMovCategoria(e.target.value)}
   639→          >
   640→            <option value="">Categoria de gasto...</option>
   641→            {categoriasGastos.map(c => (<option key={c} value={c}>{c}</option>))}
   642→          </select>
   643→        </div>
   644→      )}
   645→      <Input placeholder="Descripcion (opcional)" value={movDesc} onChange={e => setMovDesc(e.target.value)} />

   648→          {loading ? <Loader2 className="w-4 h-4 animate-spin mr-1" /> : null}Guardar
   649→        </Button>
   650→        <Button variant="secondary" onClick={onCancelar} disabled={loading} className="flex-1">Cancelar</Button>
   651→      </div>
   652→    </div>
   653→  );
