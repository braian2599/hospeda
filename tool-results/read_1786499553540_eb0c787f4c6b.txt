   153→function WeatherIcon({ code }: { code: number }) {
   154→  if (code === 0) return <Sun className="w-5 h-5 text-[#F59E0B]" />;
   155→  if (code >= 1 && code <= 3) return <CloudSun className="w-5 h-5 text-[#94A3B8]" />;
   156→  if (code >= 45 && code <= 48) return <CloudFog className="w-5 h-5 text-[#94A3B8]" />;
   157→  if (code >= 51 && code <= 55) return <CloudDrizzle className="w-5 h-5 text-[#3B82F6]" />;
   158→  if (code >= 56 && code <= 57) return <CloudDrizzle className="w-5 h-5 text-[#3B82F6]" />;
   159→  if (code >= 61 && code <= 67) return <CloudRain className="w-5 h-5 text-[#3B82F6]" />;
   160→  if (code >= 71 && code <= 77) return <CloudSnow className="w-5 h-5 text-[#3B82F6]" />;
   161→  if (code >= 80 && code <= 82) return <CloudLightning className="w-5 h-5 text-[#92400E]" />;
   162→  if (code >= 95) return <CloudLightning className="w-5 h-5 text-[#5B21B6]" />;
   163→  return <Thermometer className="w-5 h-5 text-[#94A3B8]" />;
   164→}

   353→// ==================== CALENDARIO GANTT (con navegación y popover mejorado) ====================
   354→
   355→interface GanttReserva {
   356→  tipo: string;
   357→  checkin: string;
   358→  checkout: string;
   359→  huesped: string;
   360→  horaCheckin: Date;
   361→  horaCheckout: Date;
   362→  tarifa?: string;
   363→  monto?: number;
   364→  estadoPago?: string;
   365→  ninos?: number;
   366→}
   367→
   368→function CalendarioGantt({ habitaciones, reservas, fechaInicioBase }: {
   369→  habitaciones: Record<string, { tipo: string; estado: string; problema?: string }>;
   370→  reservas: { habitacion: string; estado: string; checkin: string; checkout: string; huesped: string; horaCheckin?: string; horaCheckout?: string; tipoTarifa?: string; total?: number; estadoPago?: string; ninos?: number }[];
   371→  fechaInicioBase: Date;
   372→}) {
   373→  const [offset, setOffset] = useState(0);
   374→  const [ganttDays, setGanttDays] = useState(14);
   375→  const [mostrarHistorial, setMostrarHistorial] = useState(false);
   376→  const [popoverData, setPopoverData] = useState<PopoverData | null>(null);
   377→  const [popoverPos, setPopoverPos] = useState({ top: 0, left: 0 });
   378→
   379→  const fechaInicio = useMemo(() => {
   380→    const d = new Date(fechaInicioBase);
   381→    d.setDate(d.getDate() + offset * 7);
   382→    return d;
   383→  }, [fechaInicioBase, offset]);
   384→
   385→  const columnas = useMemo(() => {
   386→    const cols: string[] = [];
   387→    for (let i = 0; i < ganttDays; i++) {
   388→      const f = new Date(fechaInicio);
   389→      f.setDate(f.getDate() + i);
   390→      cols.push(toLocalDateStr(f));
   391→    }
   392→    return cols;
   393→  }, [fechaInicio, ganttDays]);
   394→
   395→  const colIdx = useMemo(() => {
   396→    const idx: Record<string, number> = {};
   397→    columnas.forEach((c, i) => idx[c] = i);
   398→    return idx;
   399→  }, [columnas]);
   400→
   401→  const hoyStr = todayLocal();
   402→
   403→  const handleBarClick = useCallback((e: React.MouseEvent, res: GanttReserva, num: string) => {
   404→    e.stopPropagation();
   405→    const rect = (e.currentTarget as HTMLElement).getBoundingClientRect();
   406→    let left = rect.left;
   407→    if (left + 280 > window.innerWidth) left = window.innerWidth - 290;
   408→    setPopoverPos({ top: rect.bottom + 6, left: Math.max(left, 10) });
   409→    setPopoverData({
   410→      estado: res.tipo,
   411→      habitacion: num,
   412→      huesped: res.huesped,
   413→      checkin: res.checkin,
   414→      checkout: res.checkout,
   415→      problema: res.tipo === 'Mantenimiento' ? res.huesped : undefined,
   416→      tarifa: res.tarifa,
   417→      monto: res.monto,
   418→      estadoPago: res.estadoPago,
   419→      ninos: res.ninos,
   420→    });
   421→  }, []);
   422→
   423→  const rows = useMemo(() => {
   424→    const habNumbers = Object.keys(habitaciones).sort((a, b) => parseInt(a) - parseInt(b));
   425→    const result: React.ReactNode[] = [];
   426→
   427→    habNumbers.forEach((num, rowIndex) => {
   428→      const hab = habitaciones[num];
   429→      const reservasHab: GanttReserva[] = [];
   430→
   431→      reservas.forEach(r => {
   432→        if (r.habitacion !== num || r.estado === 'Cancelada') return;
   433→        if (r.checkin > columnas[ganttDays - 1] || r.checkout < columnas[0]) return;
   434→
   435→        // Usar tanto el formato del store como el de la BD para comparar
   436→        const esCheckout = r.estado === 'Checkout_realizado' || r.estado === 'Check-Out realizado';
   437→        const esCheckin = r.estado === 'CheckIn_realizado' || r.estado === 'Check-In realizado';
   438→
   439→        const esHistorica = esCheckout && !mostrarHistorial;
   440→        if (esHistorica) return;
   441→
   442→        const horaCheckin = r.horaCheckin ? new Date(r.horaCheckin) : new Date(r.checkin + 'T14:00:00');
   443→        const horaCheckout = r.horaCheckout ? new Date(r.horaCheckout) : new Date(r.checkout + 'T09:00:00');
   444→
   445→        let estado: string;
   446→        if (esCheckout) {
   447→          estado = 'Finalizada';
   448→        } else if (esCheckin) {
   449→          estado = 'Ocupada';
   450→        } else {
   451→          estado = 'Reservada';
   452→        }
   453→        reservasHab.push({
   454→          tipo: estado, checkin: r.checkin, checkout: r.checkout, huesped: r.huesped,
   455→          horaCheckin, horaCheckout, tarifa: r.tipoTarifa, monto: r.total, estadoPago: r.estadoPago, ninos: r.ninos,
   456→        });
   457→      });

   657→function getBarColorClass(tipo: string): string {
   658→  const map: Record<string, string> = {

   660→    Ocupada: 'bg-[#4ADE80] shadow-[0_2px_6px_rgba(74,222,128,0.35)]',

   662→    Limpieza: 'bg-[#F59E0B] shadow-[0_2px_6px_rgba(245,158,11,0.30)]',
   663→    Mantenimiento: 'bg-[#64748B] shadow-[0_2px_6px_rgba(100,116,139,0.25)]',

   668→function calcularBarra(res: GanttReserva, colIdx: Record<string, number>, columnas: string[], DIAS: number) {
   669→  const COL_PCT = 100 / DIAS;
   670→  const MITAD_COL_PCT = COL_PCT / 2;
   671→  let startCol = Math.max(colIdx[res.checkin] ?? 0, 0);
   672→  let endCol = Math.min(colIdx[res.checkout] !== undefined ? colIdx[res.checkout] : DIAS - 1, DIAS - 1);
   673→
   674→  const esCheckinTarde = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckin && res.horaCheckin.getHours() >= 12;
   675→  const esCheckoutManana = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckout && res.horaCheckout.getHours() < 12;
   676→
   677→  let leftPct: number, widthPct: number;
   678→
   679→  if (esCheckinTarde && startCol >= 0 && !(res.checkin < columnas[0])) {
   680→    leftPct = startCol * COL_PCT + MITAD_COL_PCT;
   681→  } else {
   682→    leftPct = startCol * COL_PCT;
   683→  }
   684→
   685→  if (esCheckoutManana && endCol >= 0 && !(res.checkout > columnas[DIAS - 1])) {
   686→    widthPct = ((endCol - startCol) * COL_PCT) + (esCheckinTarde ? MITAD_COL_PCT : COL_PCT) - (esCheckoutManana ? MITAD_COL_PCT : 0);
   687→  } else {
   688→    widthPct = (endCol - startCol + 1) * COL_PCT;
   689→  }
   690→
   691→  widthPct = Math.max(widthPct, MITAD_COL_PCT);
   692→  return { leftPct, widthPct };
   693→}

   697→function RoomHeatmap({ habitaciones, reservas }: {
   698→  habitaciones: Record<string, { tipo: string; estado: string; capacidad: number; problema?: string }>;
   699→  reservas: { habitacion: string; estado: string; huesped: string; checkin: string; checkout: string }[];
   700→}) {
   701→  const hoyStr = todayLocal();
   702→
   703→  const habInfo = useMemo(() => {
   704→    const map: Record<string, { huesped: string; estado: string }> = {};
   705→    reservas.forEach(r => {
   706→      if ((r.estado === 'Check-In realizado' || r.estado === 'Confirmada') && r.checkin <= hoyStr && r.checkout >= hoyStr) {
   707→        map[r.habitacion] = { huesped: r.huesped, estado: r.estado === 'Check-In realizado' ? 'Ocupada' : 'Reservada' };
   708→      }
   709→    });
   710→    return map;
   711→  }, [reservas, hoyStr]);
   712→
   713→  const colors: Record<string, string> = {
   714→    Disponible: 'bg-[#DCFCE7] text-[#166534]',
   715→    Ocupada: 'bg-[#FEE2E2] text-[#991B1B]',
   716→    Limpieza: 'bg-[#FEF3C7] text-[#92400E]',

   718→    Reservada: 'bg-[#E0E7FF] text-[#3730A3]',

   726→      <CardHeader className="pb-3">
   727→        <CardTitle className="text-base">Estado de habitaciones</CardTitle>

   735→              : hab.estado === 'Mantenimiento'
   736→                ? `${num} - ${hab.problema || 'Mantenimiento'}`
   737→                : `${num} - ${hab.estado} · Cap. ${hab.capacidad} · ${hab.tipo}`;
   738→            return (

   740→                <div className={`rounded-lg border-2 p-2 text-center text-xs font-semibold transition-all duration-200 hover:-translate-y-0.5 hover:scale-[1.04] hover:shadow-lg cursor-default ${colors[hab.estado] || colors.Disponible}`}>

   755→          </span>
   756→        </div>
   757→      </CardContent>
   758→    </Card>
