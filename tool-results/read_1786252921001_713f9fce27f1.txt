    48→function KPIAnimated({ icon: Icon, label, value, sub, color, bgGradient, trend, numericValue, suffix }: {
    49→  icon: ComponentType<{ className?: string }>;
    50→  label: string;
    51→  value: string;
    52→  sub?: string;
    53→  color: string;
    54→  bgGradient?: string;
    55→  trend?: { value: number; label: string };
    56→  numericValue?: number;
    57→  suffix?: string;
    58→}) {
    59→  const trendUp = trend && trend.value > 0;
    60→  const trendDown = trend && trend.value < 0;
    61→  const trendIcon = trendUp ? '\u2191' : trendDown ? '\u2193' : '';
    62→  const trendColor = trendUp ? 'text-[#4ADE80]' : trendDown ? 'text-[#EF4444]' : 'text-muted-foreground';
    63→
    64→  return (
    65→    <Card className={`overflow-hidden group hover:shadow-lg hover:-translate-y-0.5 transition-all duration-200 !py-0 !gap-0 border-0 ${bgGradient || 'bg-gradient-to-br from-slate-50 to-slate-100'}`}>
    66→      <CardContent className="p-4">
    67→        <div className="flex items-start justify-between">
    68→          <div className="min-w-0 flex-1">
    69→            <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-semibold">{label}</p>
    70→            <p className="text-2xl font-extrabold mt-1 text-[#0F172A]">
    71→              {numericValue !== undefined ? (
    72→                <><AnimatedNumber value={numericValue} duration={600} format={suffix === '%' ? (n: number) => `${Math.round(n)}%` : (n: number) => String(Math.round(n))} className="text-2xl font-extrabold text-[#0F172A]" /></>
    73→              ) : value}
    74→            </p>
    75→            {sub && <p className="text-xs text-muted-foreground mt-0.5">{sub}</p>}
    76→          </div>
    77→          <div className="p-2.5 rounded-xl bg-white/80 shadow-sm group-hover:scale-110 group-hover:shadow-md transition-all duration-200">
    78→            <Icon className={`w-5 h-5 ${color}`} />
    79→          </div>
    80→        </div>
    81→        {trend && (
    82→          <div className={`flex items-center gap-1 mt-2 text-xs font-medium ${trendColor}`}>
    83→            <span>{trendIcon}</span>
    84→            <span>{Math.abs(trend.value)}%</span>

    95→function WeatherIcon({ code }: { code: number }) {
    96→  if (code === 0) return <Sun className="w-5 h-5 text-[#F59E0B]" />;
    97→  if (code >= 1 && code <= 3) return <CloudSun className="w-5 h-5 text-[#94A3B8]" />;
    98→  if (code >= 45 && code <= 48) return <CloudFog className="w-5 h-5 text-[#94A3B8]" />;
    99→  if (code >= 51 && code <= 55) return <CloudDrizzle className="w-5 h-5 text-[#3B82F6]" />;
   100→  if (code >= 56 && code <= 57) return <CloudDrizzle className="w-5 h-5 text-[#3B82F6]" />;
   101→  if (code >= 61 && code <= 67) return <CloudRain className="w-5 h-5 text-[#3B82F6]" />;
   102→  if (code >= 71 && code <= 77) return <CloudSnow className="w-5 h-5 text-[#3B82F6]" />;
   103→  if (code >= 80 && code <= 82) return <CloudLightning className="w-5 h-5 text-[#92400E]" />;
   104→  if (code >= 95) return <CloudLightning className="w-5 h-5 text-[#5B21B6]" />;
   105→  return <Thermometer className="w-5 h-5 text-[#94A3B8]" />;
   106→}

   297→function GraficoIngresosEgresos({ pagos, gastos }: { pagos: { fecha: string; monto: number }[]; gastos: { fecha: string; monto: number }[] }) {

   301→  const datos = useMemo(() => {
   302→    const hoy = new Date();

   304→    for (let i = 6; i >= 0; i--) {
   305→      const d = new Date(hoy);
   306→      d.setDate(d.getDate() - i);

   372→// ==================== TENDENCIA DE OCUPACIÓN (14 días) ====================
   373→
   374→interface OccupancyDayData {
   375→  fecha: string;
   376→  ocupadas: number;
   377→  porcentaje: number;
   378→  label: string;
   379→}
   380→
   381→function OccupancyTrendChart() {
   382→  // Granular Zustand selectors (no destructuring) — only re-renders when these slices change
   383→  const habitaciones = useHotelStore(s => s.habitaciones);
   384→  const reservas = useHotelStore(s => s.reservas);
   385→
   386→  const [mounted, setMounted] = useState(false);
   387→  useEffect(() => {
   388→    const t = setTimeout(() => setMounted(true), 60);
   389→    return () => clearTimeout(t);
   390→  }, []);
   391→
   392→  const totalHabitaciones = Object.keys(habitaciones).length;
   393→
   394→  const datos = useMemo<OccupancyDayData[]>(() => {
   395→    const hoy = new Date();
   396→    const out: OccupancyDayData[] = [];
   397→    for (let i = 13; i >= 0; i--) {
   398→      const d = new Date(hoy);
   399→      d.setDate(d.getDate() - i);
   400→      const diaStr = toLocalDateStr(d);
   401→      // Count unique rooms occupied on this day (checkin <= day < checkout, excluding cancelled)
   402→      const ocupadasSet = new Set<string>();
   403→      reservas.forEach(r => {
   404→        if (r.estado === 'Cancelada') return;
   405→        if (r.checkin <= diaStr && r.checkout > diaStr) {
   406→          ocupadasSet.add(r.habitacion);
   407→        }
   408→      });
   409→      const ocupadas = ocupadasSet.size;
   410→      const porcentaje = totalHabitaciones > 0 ? Math.round((ocupadas / totalHabitaciones) * 100) : 0;
   411→      const weekday = NOMBRES_DIAS[d.getDay()];
   412→      out.push({
   413→        fecha: diaStr,
   414→        ocupadas,
   415→        porcentaje,
   416→        label: `${weekday} ${d.getDate()}`,
   417→      });
   418→    }
   419→    return out;
   420→  }, [reservas, totalHabitaciones]);
   421→
   422→  const hoyData = datos[datos.length - 1];
   423→  const ayerData = datos[datos.length - 2];
   424→  const ultimos7 = datos.slice(-7);
   425→  const promedio7d = ultimos7.length > 0 ? ultimos7.reduce((s, d) => s + d.porcentaje, 0) / ultimos7.length : 0;
   426→
   427→  const diffVsAyer = hoyData && ayerData ? hoyData.porcentaje - ayerData.porcentaje : 0;
   428→  const trendUp = diffVsAyer > 0;
   429→  const trendDown = diffVsAyer < 0;
   430→  const trendIcon = trendUp ? '\u2191' : trendDown ? '\u2193' : '\u2192';
   431→  const trendColor = trendUp ? 'text-[#166534]' : trendDown ? 'text-[#991B1B]' : 'text-muted-foreground';
   432→
   433→  const renderTooltip = ({ active, payload }: TooltipProps<number, string>) => {
   434→    if (!active || !payload || !payload.length) return null;
   435→    const d = payload[0]?.payload as OccupancyDayData | undefined;
   436→    if (!d) return null;
   437→    return (
   438→      <div className="bg-[#0F2B28] text-white px-3 py-2 rounded-lg shadow-xl text-xs border border-[#059669]/30">
   439→        <p className="font-semibold text-[#34d399]">{formatearFecha(d.fecha)}</p>
   440→        <p className="mt-0.5">Ocupación: <span className="font-bold">{d.porcentaje}%</span></p>
   441→        <p className="text-white/70">{d.ocupadas} de {totalHabitaciones} hab.</p>
   442→      </div>
   443→    );
   444→  };
   445→
   446→  return (
   447→    <Card className={cn('transition-opacity duration-500', mounted ? 'opacity-100' : 'opacity-0')}>

   822→function getBarColorClass(tipo: string): string {
   823→  const map: Record<string, string> = {
   824→    Reservada: 'bg-[#3B82F6] shadow-[0_2px_6px_rgba(59,130,246,0.35)]',
   825→    Ocupada: 'bg-[#4ADE80] shadow-[0_2px_6px_rgba(74,222,128,0.35)]',
   826→    Finalizada: 'bg-[#94A3B8] opacity-50 border border-dashed border-[#CBD5E1]',
   827→    Limpieza: 'bg-[#F59E0B] shadow-[0_2px_6px_rgba(245,158,11,0.30)]',
   828→    Mantenimiento: 'bg-[#64748B] shadow-[0_2px_6px_rgba(100,116,139,0.25)]',
   829→  };
   830→  return map[tipo] || map.Reservada;
   831→}
   832→
   833→function calcularBarra(res: GanttReserva, colIdx: Record<string, number>, columnas: string[], DIAS: number) {
   834→  const COL_PCT = 100 / DIAS;
   835→  const MITAD_COL_PCT = COL_PCT / 2;
   836→  let startCol = Math.max(colIdx[res.checkin] ?? 0, 0);
   837→  let endCol = Math.min(colIdx[res.checkout] !== undefined ? colIdx[res.checkout] : DIAS - 1, DIAS - 1);
   838→
   839→  const esCheckinTarde = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckin && res.horaCheckin.getHours() >= 12;
   840→  const esCheckoutManana = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckout && res.horaCheckout.getHours() < 12;
   841→
   842→  let leftPct: number, widthPct: number;
   843→
   844→  if (esCheckinTarde && startCol >= 0 && !(res.checkin < columnas[0])) {
   845→    leftPct = startCol * COL_PCT + MITAD_COL_PCT;
   846→  } else {
   847→    leftPct = startCol * COL_PCT;
   848→  }
   849→
   850→  if (esCheckoutManana && endCol >= 0 && !(res.checkout > columnas[DIAS - 1])) {
   851→    widthPct = ((endCol - startCol) * COL_PCT) + (esCheckinTarde ? MITAD_COL_PCT : COL_PCT) - (esCheckoutManana ? MITAD_COL_PCT : 0);
   852→  } else {
   853→    widthPct = (endCol - startCol + 1) * COL_PCT;
   854→  }
   855→
   856→  widthPct = Math.max(widthPct, MITAD_COL_PCT);
   857→  return { leftPct, widthPct };
   858→}
