   356→// RoomTypeAnalytics — compact analytics section
   357→// ═══════════════════════════════════════════════════════════
   358→
   359→function RoomTypeAnalytics() {
   360→  const habitaciones = useHotelStore(s => s.habitaciones);
   361→  const reservas = useHotelStore(s => s.reservas);
   362→  const today = useMemo(() => todayLocal(), []);
   363→
   364→  const allRooms = Object.values(habitaciones);
   365→  const total = allRooms.length;
   366→
   367→  const typeData = useMemo(() => {
   368→    if (total === 0) return [];
   369→
   370→    const todayDate = safeDate(today).getTime();
   371→
   372→    // Group by type
   373→    const byType = new Map<string, Habitacion[]>();
   374→    allRooms.forEach(h => {
   375→      const t = h.tipo || 'Sin tipo';
   376→      if (!byType.has(t)) byType.set(t, []);
   377→      byType.get(t)!.push(h);
   378→    });
   379→
   380→    return Array.from(byType.entries())
   381→      .sort(([a], [b]) => a.localeCompare(b))
   382→      .map(([tipo, rooms]) => {
   383→        const count = rooms.length;
   384→        const occupied = rooms.filter(h => h.estado === 'Ocupada').length;
   385→        const occupancyRate = Math.round((occupied / count) * 100);
   386→
   387→        // Calculate revenue for occupied rooms
   388→        let revenue = 0;
   389→        rooms.forEach(h => {
   390→          if (h.estado === 'Ocupada' || h.estado === 'Reservada') {
   391→            const res = reservas.find(r =>
   392→              r.habitacion === h.numero &&
   393→              (r.estado === 'Check-In realizado' || r.estado === 'Confirmada') &&
   394→              safeDate(r.checkin).getTime() <= todayDate &&
   395→              safeDate(r.checkout).getTime() >= todayDate
   396→            );
   397→            if (res?.total) revenue += res.total;
   398→          }
   399→        });
   400→
   401→        const Icon = tipoIconMap[tipo] || Bed;
   402→
   403→        return { tipo, count, occupied, occupancyRate, revenue, Icon };
   404→      });
   405→  }, [allRooms, reservas, today, total]);
   406→
   407→  if (total === 0 || typeData.length === 0) return null;
   408→
   409→  const maxCount = Math.max(...typeData.map(d => d.count));
   410→
   411→  // Type color palette
   412→  const typeColors = ['#059669', '#D97706', '#0EA5E9', '#8B5CF6', '#EF4444'];
   413→
   414→  return (
   415→    <div className="rounded-xl border bg-card p-4 space-y-3">
   416→      <div className="flex items-center gap-2">
   417→        <div className="w-7 h-7 rounded-lg bg-[#0F2B28]/10 flex items-center justify-center">
   418→          <Bed className="w-3.5 h-3.5 text-[#0F2B28]" />
   419→        </div>
   420→        <span className="text-sm font-semibold text-foreground">Distribución por tipo</span>
   421→      </div>
   422→
   423→      {/* Horizontal distribution bar */}
   424→      <div className="h-2.5 rounded-full bg-muted overflow-hidden flex">
   425→        {typeData.map((d, i) => {
   426→          const pct = (d.count / total) * 100;
   427→          return (
   428→            <div
   429→              key={d.tipo}
   430→              className="transition-all duration-300"
   431→              style={{ width: `${pct}%`, backgroundColor: typeColors[i % typeColors.length] }}
   432→              title={`${d.tipo}: ${d.count} (${Math.round(pct)}%)`}
   433→            />
   434→          );
   435→        })}
   436→      </div>
   437→
   438→      {/* Type breakdown rows */}
   439→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-5 gap-2">
   440→        {typeData.map((d, i) => {
   441→          const color = typeColors[i % typeColors.length];
   442→          const barPct = (d.count / maxCount) * 100;
   443→          return (
   444→            <div
   445→              key={d.tipo}
   446→              className="rounded-lg border bg-muted/20 p-2.5 space-y-1.5"
   447→            >
   448→              <div className="flex items-center gap-2">
   449→                <d.Icon className="w-3.5 h-3.5 shrink-0" style={{ color }} />
   450→                <span className="text-xs font-semibold text-foreground truncate">{d.tipo}</span>
   451→                <span className="ml-auto text-xs font-bold text-foreground">{d.count}</span>
   452→              </div>
   453→              {/* Mini bar */}
   454→              <div className="h-1.5 rounded-full bg-muted overflow-hidden">
   455→                <div
   456→                  className="h-full rounded-full transition-all duration-300"
   457→                  style={{ width: `${barPct}%`, backgroundColor: color }}
   458→                />
   459→              </div>
   460→              <div className="flex items-center justify-between text-[10px] text-muted-foreground">
   461→                <span>Ocup. {d.occupancyRate}%</span>
   462→                {d.revenue > 0 && <span>${d.revenue.toLocaleString('es-AR')}</span>}
   463→              </div>
   464→            </div>
   465→          );
   466→        })}
   467→      </div>
   468→    </div>
   469→  );
   470→}

   472→// ═══════════════════════════════════════════════════════════
   473→// StatusChangePopover — Quick status change for a room
   474→// ═══════════════════════════════════════════════════════════
   475→
   476→function StatusChangePopover({
   477→  numero,
   478→  currentEstado,
   479→  onStatusChange,
   480→  children,
   481→}: {
   482→  numero: string;
   483→  currentEstado: EstadoHabitacion;
   484→  onStatusChange: (numero: string, newEstado: EstadoHabitacion) => void;
   485→  children: React.ReactNode;
   486→}) {
   487→  const [open, setOpen] = useState(false);
   488→  const [confirmDestructive, setConfirmDestructive] = useState<EstadoHabitacion | null>(null);
   489→
   490→  const handleSelect = (newEstado: EstadoHabitacion) => {
   491→    if (newEstado === currentEstado) {
   492→      setOpen(false);
   493→      return;
   494→    }
   495→    // Destructive states need confirmation
   496→    const isDestructive = newEstado === 'Fuera de servicio' || newEstado === 'Mantenimiento';
   497→    if (isDestructive && !confirmDestructive) {
   498→      setConfirmDestructive(newEstado);
   499→      return;
   500→    }
   501→    onStatusChange(numero, newEstado);
   502→    setConfirmDestructive(null);
   503→    setOpen(false);
   504→  };
   505→
   506→  return (
   507→    <Popover open={open} onOpenChange={(v) => { setOpen(v); setConfirmDestructive(null); }}>
   508→      <PopoverTrigger asChild>{children}</PopoverTrigger>
   509→      <PopoverContent className="w-56 p-2" align="start" side="bottom">
   510→        {confirmDestructive ? (
   511→          <div className="space-y-2 p-1">
   512→            <p className="text-xs text-muted-foreground">
   513→              ¿Cambiar a <strong className="text-foreground">{confirmDestructive}</strong>?
   514→              {confirmDestructive === 'Fuera de servicio' && ' La habitación no será asignable.'}
   515→              {confirmDestructive === 'Mantenimiento' && ' Se requiere reparación antes de habilitar.'}
   516→            </p>

   739→function FloorGroup({
   740→  floorLabel,
   741→  floorNum,
   742→  rooms,
   743→  isFloorPattern,
   744→  getHuespedActual,
   745→  onEdit,

   756→  onStatusChange: (num: string, estado: EstadoHabitacion) => void;
   757→}) {

   759→

   767→  return (

   769→      <div className="rounded-xl border bg-card overflow-hidden">
   770→        {/* Floor header */}

   799→                      className="w-2 h-2 rounded-sm"

   804→                );
   805→              })}
   806→            </div>
   807→            <div className="shrink-0 ml-1">
   808→              {isOpen ? (
   809→                <ChevronDown className="w-4 h-4 text-muted-foreground" />
   810→              ) : (
   811→                <ChevronRight className="w-4 h-4 text-muted-foreground" />

   813→            </div>
   814→          </button>
   815→        </CollapsibleTrigger>
   816→
   817→        {/* Room grid */}
   818→        <CollapsibleContent>
   819→          <div className="p-3 pt-0">
   820→            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6 gap-3 card-grid-stagger">

   823→                  ? getHuespedActual(hab.numero)
   824→                  : null;

   826→                  <EnhancedRoomCard
   827→                    key={hab.numero}
   828→                    hab={hab}
   829→                    huesped={huesped}

   831→                    onDelete={onDelete}
   832→                    onStatusChange={onStatusChange}
   833→                  />
   834→                );
   835→              })}
   836→            </div>

   838→        </CollapsibleContent>
