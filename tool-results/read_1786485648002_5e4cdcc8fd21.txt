   185→   185→// RoomTypeAnalytics — compact analytics section
   186→   186→// ═══════════════════════════════════════════════════════════
   187→   187→
   188→   188→function RoomTypeAnalytics() {
   189→   189→  const habitaciones = useHotelStore(s => s.habitaciones);
   190→   190→  const reservas = useHotelStore(s => s.reservas);
   191→   191→  const today = useMemo(() => todayLocal(), []);
   192→   192→
   193→   193→  const allRooms = Object.values(habitaciones);
   194→   194→  const total = allRooms.length;
   195→   195→
   196→   196→  const typeData = useMemo(() => {
   197→   197→    if (total === 0) return [];
   198→   198→
   199→   199→    const todayDate = safeDate(today).getTime();
   200→   200→
   201→   201→    // Group by type
   202→   202→    const byType = new Map<string, Habitacion[]>();
   203→   203→    allRooms.forEach(h => {
   204→   204→      const t = h.tipo || 'Sin tipo';
   205→   205→      if (!byType.has(t)) byType.set(t, []);
   206→   206→      byType.get(t)!.push(h);
   207→   207→    });
   208→   208→
   209→   209→    return Array.from(byType.entries())
   210→   210→      .sort(([a], [b]) => a.localeCompare(b))
   211→   211→      .map(([tipo, rooms]) => {
   212→   212→        const count = rooms.length;
   213→   213→        const occupied = rooms.filter(h => h.estado === 'Ocupada').length;
   214→   214→        const occupancyRate = Math.round((occupied / count) * 100);
   215→   215→
   216→   216→        // Calculate revenue for occupied rooms
   217→   217→        let revenue = 0;
   218→   218→        rooms.forEach(h => {
   219→   219→          if (h.estado === 'Ocupada' || h.estado === 'Reservada') {
   220→   220→            const res = reservas.find(r =>
   221→   221→              r.habitacion === h.numero &&
   222→   222→              (r.estado === 'Check-In realizado' || r.estado === 'Confirmada') &&
   223→   223→              safeDate(r.checkin).getTime() <= todayDate &&
   224→   224→              safeDate(r.checkout).getTime() >= todayDate
   225→   225→            );
   226→   226→            if (res?.total) revenue += res.total;
   227→   227→          }
   228→   228→        });
   229→   229→
   230→   230→        const Icon = tipoIconMap[tipo] || Bed;
   231→   231→
   232→   232→        return { tipo, count, occupied, occupancyRate, revenue, Icon };
   233→   233→      });
   234→   234→  }, [allRooms, reservas, today, total]);
   235→   235→
   236→   236→  if (total === 0 || typeData.length === 0) return null;
   237→   237→
   238→   238→  const maxCount = Math.max(...typeData.map(d => d.count));
   239→   239→
   240→   240→  // Type color palette
   241→   241→  const typeColors = ['#059669', '#D97706', '#0EA5E9', '#8B5CF6', '#EF4444'];
   242→   242→
   243→   243→  return (
   244→   244→    <div className="rounded-xl border bg-card p-4 space-y-3">
   245→   245→      <div className="flex items-center gap-2">
   246→   246→        <div className="w-7 h-7 rounded-lg bg-[#0F2B28]/10 flex items-center justify-center">
   247→   247→          <Bed className="w-3.5 h-3.5 text-[#0F2B28]" />
   248→   248→        </div>
   249→   249→        <span className="text-sm font-semibold text-foreground">Distribución por tipo</span>
   250→   250→      </div>
   251→   251→
   252→   252→      {/* Horizontal distribution bar */}
   253→   253→      <div className="h-2.5 rounded-full bg-muted overflow-hidden flex">
   254→   254→        {typeData.map((d, i) => {
   255→   255→          const pct = (d.count / total) * 100;
   256→   256→          return (
   257→   257→            <div
   258→   258→              key={d.tipo}
   259→   259→              className="transition-all duration-300"
   260→   260→              style={{ width: `${pct}%`, backgroundColor: typeColors[i % typeColors.length] }}
   261→   261→              title={`${d.tipo}: ${d.count} (${Math.round(pct)}%)`}
   262→   262→            />
   263→   263→          );
   264→   264→        })}
   265→   265→      </div>
   266→   266→
   267→   267→      {/* Type breakdown rows */}
   268→   268→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-5 gap-2">
   269→   269→        {typeData.map((d, i) => {
   270→   270→          const color = typeColors[i % typeColors.length];
   271→   271→          const barPct = (d.count / maxCount) * 100;
   272→   272→          return (
   273→   273→            <div
   274→   274→              key={d.tipo}
   275→   275→              className="rounded-lg border bg-muted/20 p-2.5 space-y-1.5"
   276→   276→            >
   277→   277→              <div className="flex items-center gap-2">
   278→   278→                <d.Icon className="w-3.5 h-3.5 shrink-0" style={{ color }} />
   279→   279→                <span className="text-xs font-semibold text-foreground truncate">{d.tipo}</span>
   280→   280→                <span className="ml-auto text-xs font-bold text-foreground">{d.count}</span>
   281→   281→              </div>
   282→   282→              {/* Mini bar */}
   283→   283→              <div className="h-1.5 rounded-full bg-muted overflow-hidden">
   284→   284→                <div
   285→   285→                  className="h-full rounded-full transition-all duration-300"
   286→   286→                  style={{ width: `${barPct}%`, backgroundColor: color }}
   287→   287→                />
   288→   288→              </div>
   289→   289→              <div className="flex items-center justify-between text-[10px] text-muted-foreground">
   290→   290→                <span>Ocup. {d.occupancyRate}%</span>
   291→   291→                {d.revenue > 0 && <span>${d.revenue.toLocaleString('es-AR')}</span>}
   292→   292→              </div>
   293→   293→            </div>
   294→   294→          );
   295→   295→        })}
   296→   296→      </div>
   297→   297→    </div>
   298→   298→  );
   299→   299→}

   302→   302→// StatusChangePopover — Quick status change for a room
   303→   303→// ═══════════════════════════════════════════════════════════

   305→   305→function StatusChangePopover({
   306→   306→  numero,
   307→   307→  currentEstado,
   308→   308→  onStatusChange,
   309→   309→  children,
   310→   310→}: {
   311→   311→  numero: string;
   312→   312→  currentEstado: EstadoHabitacion;
   313→   313→  onStatusChange: (numero: string, newEstado: EstadoHabitacion) => void;
   314→   314→  children: React.ReactNode;
   315→   315→}) {
   316→   316→  const [open, setOpen] = useState(false);
   317→   317→  const [confirmDestructive, setConfirmDestructive] = useState<EstadoHabitacion | null>(null);
   318→   318→
   319→   319→  const handleSelect = (newEstado: EstadoHabitacion) => {
   320→   320→    if (newEstado === currentEstado) {
   321→   321→      setOpen(false);
   322→   322→      return;
   323→   323→    }
   324→   324→    // Destructive states need confirmation
   325→   325→    const isDestructive = newEstado === 'Fuera de servicio' || newEstado === 'Mantenimiento';
   326→   326→    if (isDestructive && !confirmDestructive) {
   327→   327→      setConfirmDestructive(newEstado);
   328→   328→      return;
   329→   329→    }
   330→   330→    onStatusChange(numero, newEstado);
   331→   331→    setConfirmDestructive(null);
   332→   332→    setOpen(false);
   333→   333→  };
   334→   334→
   335→   335→  return (
   336→   336→    <Popover open={open} onOpenChange={(v) => { setOpen(v); setConfirmDestructive(null); }}>
   337→   337→      <PopoverTrigger asChild>{children}</PopoverTrigger>
   338→   338→      <PopoverContent className="w-56 p-2" align="start" side="bottom">
   339→   339→        {confirmDestructive ? (
   340→   340→          <div className="space-y-2 p-1">
   341→   341→            <p className="text-xs text-muted-foreground">
   342→   342→              ¿Cambiar a <strong className="text-foreground">{confirmDestructive}</strong>?
   343→   343→              {confirmDestructive === 'Fuera de servicio' && ' La habitación no será asignable.'}
   344→   344→              {confirmDestructive === 'Mantenimiento' && ' Se requiere reparación antes de habilitar.'}
   345→   345→            </p>
   346→   346→            <div className="flex gap-1.5">
   347→   347→              <Button
   348→   348→                size="sm"
   349→   349→                variant="destructive"
   350→   350→                className="h-7 text-xs flex-1"
   351→   351→                onClick={() => handleSelect(confirmDestructive)}
   352→   352→              >
   353→   353→                Confirmar
   354→   354→              </Button>
   355→   355→              <Button
   356→   356→                size="sm"
   357→   357→                variant="ghost"
   358→   358→                className="h-7 text-xs"
   359→   359→                onClick={() => setConfirmDestructive(null)}
   360→   360→              >
   361→   361→                Cancelar
   362→   362→              </Button>
   363→   363→            </div>
   364→   364→          </div>
   365→   365→        ) : (
   366→   366→          <div className="space-y-0.5">
   367→   367→            <div className="text-[10px] uppercase tracking-wider text-muted-foreground font-semibold px-2 pt-1 pb-1.5">
   368→   368→              Cambiar estado
   369→   369→            </div>
   370→   370→            {ALL_ESTADOS.map(estado => {
   371→   371→              const vis = STATUS_VISUAL[estado];
   372→   372→              const Icon = vis.icon;
   373→   373→              const isCurrent = estado === currentEstado;
   374→   374→              return (
   375→   375→                <button
   376→   376→                  key={estado}
   377→   377→                  type="button"
   378→   378→                  onClick={() => handleSelect(estado)}
   379→   379→                  className={`
   380→   380→                    w-full flex items-center gap-2 rounded-md px-2 py-1.5 text-xs
   381→   381→                    transition-colors cursor-pointer
   382→   382→                    ${isCurrent
   383→   383→                      ? 'bg-accent/50 font-semibold text-foreground'
   384→   384→                      : 'text-foreground/80 hover:bg-accent/40'
   385→   385→                    }
   386→   386→                  `}
   387→   387→                >
   388→   388→                  <Icon className="w-3.5 h-3.5 shrink-0" style={{ color: vis.color }} />
   389→   389→                  <span className="truncate">{estado}</span>
   390→   390→                  {isCurrent && (
   391→   391→                    <span className="ml-auto text-[10px] text-muted-foreground">actual</span>
   392→   392→                  )}
   393→   393→                </button>
   394→   394→              );
   395→   395→            })}
   396→   396→          </div>
   397→   397→        )}
   398→   398→      </PopoverContent>
   399→   399→    </Popover>
   400→   400→  );
   401→   401→}
