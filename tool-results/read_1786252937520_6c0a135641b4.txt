    19→/* ─── Intersection Observer hook ─── */
    20→function useInView(threshold = 0.12) {
    21→  const ref = useRef<HTMLDivElement>(null);
    22→  const [inView, setInView] = useState(false);
    23→  useEffect(() => {
    24→    const el = ref.current;
    25→    if (!el) return;
    26→    const obs = new IntersectionObserver(
    27→      ([e]) => { if (e.isIntersecting) { setInView(true); obs.disconnect(); } },
    28→      { threshold }
    29→    );
    30→    obs.observe(el);
    31→    return () => obs.disconnect();
    32→  }, [threshold]);
    33→  return { ref, inView };
    34→}

    41→/* ─── Scroll progress indicator (top gradient bar) ─── */
    42→function ScrollProgress() {
    43→  const [progress, setProgress] = useState(0);
    44→  useEffect(() => {
    45→    const onScroll = () => {
    46→      const scrollTop = window.scrollY;
    47→      const docHeight = document.documentElement.scrollHeight - window.innerHeight;
    48→      setProgress(docHeight > 0 ? Math.min(100, (scrollTop / docHeight) * 100) : 0);
    49→    };
    50→    onScroll();
    51→    window.addEventListener('scroll', onScroll, { passive: true });
    52→    return () => window.removeEventListener('scroll', onScroll);
    53→  }, []);
    54→  return (
    55→    <div
    56→      className="scroll-progress"
    57→      style={{ width: `${progress}%`, opacity: progress > 0.5 ? 1 : 0 }}
    58→      aria-hidden="true"
    59→    />
    60→  );
    61→}

   258→/* ─── Fade-in wrapper ─── */
   259→function FadeIn({ children, className = '', delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   260→  const { ref, inView } = useInView();
   261→  return (
   262→    <div
   263→      ref={ref}
   264→      className={`transition-all duration-700 ease-out ${className}`}
   265→      style={{
   266→        opacity: inView ? 1 : 0,
   267→        transform: inView ? 'translateY(0)' : 'translateY(32px)',
   268→        transitionDelay: `${delay}ms`,
   269→      }}
   270→    >
   271→      {children}
   272→    </div>
   273→  );
   274→}
