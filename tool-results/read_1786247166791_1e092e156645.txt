    88→function WeatherIcon({ code }: { code: number }) {
    89→  if (code === 0) return <Sun className="w-5 h-5 text-[#F59E0B]" />;
    90→  if (code >= 1 && code <= 3) return <CloudSun className="w-5 h-5 text-[#94A3B8]" />;
    91→  if (code >= 45 && code <= 48) return <CloudFog className="w-5 h-5 text-[#94A3B8]" />;
    92→  if (code >= 51 && code <= 55) return <CloudDrizzle className="w-5 h-5 text-[#3B82F6]" />;
    93→  if (code >= 56 && code <= 57) return <CloudDrizzle className="w-5 h-5 text-[#3B82F6]" />;
    94→  if (code >= 61 && code <= 67) return <CloudRain className="w-5 h-5 text-[#3B82F6]" />;
    95→  if (code >= 71 && code <= 77) return <CloudSnow className="w-5 h-5 text-[#3B82F6]" />;
    96→  if (code >= 80 && code <= 82) return <CloudLightning className="w-5 h-5 text-[#92400E]" />;
    97→  if (code >= 95) return <CloudLightning className="w-5 h-5 text-[#5B21B6]" />;
    98→  return <Thermometer className="w-5 h-5 text-[#94A3B8]" />;

   290→function GraficoIngresosEgresos({ pagos, gastos }: { pagos: { fecha: string; monto: number }[]; gastos: { fecha: string; monto: number }[] }) {
   291→  const [mounted, setMounted] = useState(false);
   292→  useEffect(() => { const t = setTimeout(() => setMounted(true), 50); return () => clearTimeout(t); }, []);
   293→
   294→  const datos = useMemo(() => {
   295→    const hoy = new Date();
   296→    const dias: string[] = [];
   297→    for (let i = 6; i >= 0; i--) {
   298→      const d = new Date(hoy);
   299→      d.setDate(d.getDate() - i);
   300→      dias.push(toLocalDateStr(d));
   301→    }
   302→    return dias.map(dia => {
   303→      const ing = pagos.filter(p => p.fecha === dia).reduce((s, p) => s + p.monto, 0);
   304→      const egr = gastos.filter(g => g.fecha === dia).reduce((s, g) => s + g.monto, 0);
   305→      const d = new Date(dia + 'T12:00:00');
   306→      return { dia: d.toLocaleDateString('es-AR', { weekday: 'short', day: 'numeric' }).replace(/^\w/, c => c.toUpperCase()), ing, egr, fullDate: formatearFecha(dia) };
   307→    });
   308→  }, [pagos, gastos]);
   309→
   310→  const maxVal = Math.max(...datos.map(d => Math.max(d.ing, d.egr)), 1);
   311→
   312→  return (
   313→    <Card>
   314→      <CardHeader className="pb-3">
   315→        <CardTitle className="text-base flex items-center gap-2">
   316→          <BarChart3 className="w-4 h-4 text-[#3B82F6]" />
   317→          Ingresos vs Egresos (7 días)
   318→        </CardTitle>
   319→      </CardHeader>
   320→      <CardContent>
   321→        <div className="flex items-end gap-2" style={{ height: 160 }}>
   322→          {datos.map((d, i) => (
   323→            <div key={i} className="flex-1 flex flex-col items-center h-full">
   324→              <Tooltip text={`Ingresos: ${formatMoney(d.ing)}`}>
   325→                <div
   326→                  className="w-5 rounded-t-md transition-all duration-700 ease-out hover:opacity-80 cursor-default"
   327→                  style={{
   328→                    height: mounted ? `${(d.ing / maxVal) * 100}%` : '0%',
   329→                    background: 'linear-gradient(to top, #059669, #34d399)',
   330→                    boxShadow: '0 2px 8px rgba(5,150,105,0.3)',
   331→                    transitionDelay: `${i * 80}ms`,
   332→                  }}
   333→                />
   334→              </Tooltip>
   335→              <Tooltip text={`Egresos: ${formatMoney(d.egr)}`}>
   336→                <div
   337→                  className="w-5 rounded-t-md transition-all duration-700 ease-out hover:opacity-80 cursor-default mt-0.5"
   338→                  style={{
   339→                    height: mounted ? `${(d.egr / maxVal) * 100}%` : '0%',
   340→                    background: 'linear-gradient(to top, #dc2626, #fca5a5)',
   341→                    boxShadow: '0 2px 8px rgba(220,38,38,0.3)',

   443→      reservas.forEach(r => {
   444→        if (r.habitacion !== num || r.estado === 'Cancelada') return;

   670→  const map: Record<string, string> = {

   673→    Finalizada: 'bg-[#94A3B8] opacity-50 border border-dashed border-[#CBD5E1]',
   674→    Limpieza: 'bg-[#F59E0B] shadow-[0_2px_6px_rgba(245,158,11,0.30)]',
   675→    Mantenimiento: 'bg-[#64748B] shadow-[0_2px_6px_rgba(100,116,139,0.25)]',

   680→function calcularBarra(res: GanttReserva, colIdx: Record<string, number>, columnas: string[], DIAS: number) {
   681→  const COL_PCT = 100 / DIAS;
   682→  const MITAD_COL_PCT = COL_PCT / 2;
   683→  let startCol = Math.max(colIdx[res.checkin] ?? 0, 0);
   684→  let endCol = Math.min(colIdx[res.checkout] !== undefined ? colIdx[res.checkout] : DIAS - 1, DIAS - 1);
   685→
   686→  const esCheckinTarde = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckin && res.horaCheckin.getHours() >= 12;
   687→  const esCheckoutManana = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckout && res.horaCheckout.getHours() < 12;
   688→
   689→  let leftPct: number, widthPct: number;
   690→
   691→  if (esCheckinTarde && startCol >= 0 && !(res.checkin < columnas[0])) {
   692→    leftPct = startCol * COL_PCT + MITAD_COL_PCT;
   693→  } else {
   694→    leftPct = startCol * COL_PCT;
   695→  }
   696→
   697→  if (esCheckoutManana && endCol >= 0 && !(res.checkout > columnas[DIAS - 1])) {
   698→    widthPct = ((endCol - startCol) * COL_PCT) + (esCheckinTarde ? MITAD_COL_PCT : COL_PCT) - (esCheckoutManana ? MITAD_COL_PCT : 0);
   699→  } else {
   700→    widthPct = (endCol - startCol + 1) * COL_PCT;
   701→  }
   702→
   703→  widthPct = Math.max(widthPct, MITAD_COL_PCT);
   704→  return { leftPct, widthPct };
   705→}

   767→          </span>
