    77→interface KpiProps {

    88→function KpiCard({ label, value, icon, colorFamily, tooltip, subtext, trend, onClick }: KpiProps) {
    89→  const c = KPI_COLORS[colorFamily] || KPI_COLORS.emerald;
    90→  const card = (
    91→    <div
    92→      className={`relative overflow-hidden rounded-xl border-l-[3px] ${c.borderL} ${c.bg} ${c.darkBg} p-4 shadow-sm hover:shadow-md hover:-translate-y-0.5 transition-all duration-200 card-interactive ${onClick ? 'cursor-pointer group' : ''}`}
    93→      onClick={onClick}
    94→    >
    95→      <div className="flex items-start justify-between">
    96→        <div className="space-y-1">
    97→          <p className={`text-xs font-medium ${c.label}`}>{label}</p>
    98→          <p className={`text-xl font-bold ${c.value} truncate`}>{value}</p>
    99→          {trend && (
   100→            <div className="flex items-center gap-1 mt-1">
   101→              {trend.value > 0 ? (
   102→                <ArrowUpRight className="w-3 h-3 text-[#059669]" />
   103→              ) : trend.value < 0 ? (
   104→                <ArrowDownRight className="w-3 h-3 text-[#EF4444]" />
   105→              ) : (
   106→                <Minus className="w-3 h-3 text-muted-foreground" />
   107→              )}
   108→              <span className={`text-[11px] font-medium ${trend.value > 0 ? 'text-[#059669]' : trend.value < 0 ? 'text-[#EF4444]' : 'text-muted-foreground'}`}>
   109→                {Math.abs(trend.value)}%
   110→              </span>
   111→              <span className="text-[11px] text-muted-foreground hidden sm:inline">{trend.label}</span>
   112→            </div>
   113→          )}
   114→          {subtext && !trend && <p className={`text-[10px] ${c.sub} mt-1 truncate`}>{subtext}</p>}
   115→        </div>
   116→        <div className={`w-10 h-10 rounded-full ${c.iconBg} ${c.iconColor} flex items-center justify-center shrink-0`}>
   117→          {icon}
   118→        </div>
   119→      </div>
   120→      {onClick && (
   121→        <div className="absolute inset-x-0 bottom-0 h-0.5 bg-primary opacity-0 group-hover:opacity-100 transition-opacity" />
   122→      )}
   123→    </div>
   124→  );
   125→
   126→  if (tooltip) {
   127→    return (
   128→      <TooltipProvider delayDuration={300}>
   129→        <Tooltip>
   130→          <TooltipTrigger asChild>{card}</TooltipTrigger>
   131→          <TooltipContent side="bottom" className="text-xs max-w-[220px]">{tooltip}</TooltipContent>
   132→        </Tooltip>
   133→      </TooltipProvider>
   134→    );
   135→  }
   136→  return card;
   137→}
