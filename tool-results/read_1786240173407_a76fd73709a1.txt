    39→// ==================== ANIMATED NUMBER HOOK ====================
    40→
    41→function useCountUp(target: number, duration = 800, enabled = true) {
    42→  const [value, setValue] = useState(0);
    43→  const prevTarget = useRef(0);
    44→
    45→  useEffect(() => {
    46→    if (!enabled) { setValue(target); return; }
    47→    const startValue = prevTarget.current;
    48→    const diff = target - startValue;
    49→    if (Math.abs(diff) < 1) { setValue(target); prevTarget.current = target; return; }
    50→    const startTime = performance.now();
    51→    let raf: number;
    52→    const animate = (now: number) => {
    53→      const elapsed = now - startTime;
    54→      const progress = Math.min(elapsed / duration, 1);
    55→      const eased = 1 - Math.pow(1 - progress, 3); // ease-out cubic
    56→      setValue(Math.round(startValue + diff * eased));
    57→      if (progress < 1) { raf = requestAnimationFrame(animate); }
    58→      else { prevTarget.current = target; }
    59→    };
    60→    raf = requestAnimationFrame(animate);
    61→    return () => cancelAnimationFrame(raf);
    62→  }, [target, duration, enabled]);
    63→
    64→  return value;
    65→}

    69→function KPIAnimated({ icon: Icon, label, value, sub, color, bgGradient, trend, numeric = false }: {
    70→  icon: React.ComponentType<{ className?: string }>;
    71→  label: string;
    72→  value: string;
    73→  sub?: string;
    74→  color: string;
    75→  bgGradient?: string;
    76→  trend?: { value: number; label: string };
    77→  numeric?: boolean;
    78→}) {
    79→  const numVal = numeric ? parseInt(value.replace(/[^0-9]/g, '')) || 0 : 0;
    80→  const animated = useCountUp(numVal, 600, numeric && numVal > 0);
    81→  const displayValue = numeric ? value.replace(/\d+/, String(animated)) : value;
    82→
    83→  const trendUp = trend && trend.value > 0;
    84→  const trendDown = trend && trend.value < 0;
    85→  const trendIcon = trendUp ? '\u2191' : trendDown ? '\u2193' : '';
    86→  const trendColor = trendUp ? 'text-[#4ADE80]' : trendDown ? 'text-[#EF4444]' : 'text-muted-foreground';
    87→
    88→  return (
    89→    <Card className={`overflow-hidden group hover:shadow-md transition-shadow duration-200 !py-0 !gap-0 ${bgGradient || 'bg-gradient-to-br from-slate-50 to-slate-100'}`}>
    90→      <CardContent className="p-4">
    91→        <div className="flex items-start justify-between">
    92→          <div className="min-w-0 flex-1">
    93→            <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-semibold">{label}</p>
    94→            <p className="text-2xl font-extrabold mt-1 text-[#0F172A]">{displayValue}</p>
    95→            {sub && <p className="text-xs text-muted-foreground mt-0.5">{sub}</p>}

    98→            <Icon className={`w-5 h-5 ${color}`} />
    99→          </div>
   100→        </div>
   101→        {trend && (
   102→          <div className={`flex items-center gap-1 mt-2 text-xs font-medium ${trendColor}`}>
   103→            <span>{trendIcon}</span>
   104→            <span>{Math.abs(trend.value)}%</span>

   115→function WeatherIcon({ code }: { code: number }) {
   116→  if (code === 0) return <Sun className="w-5 h-5 text-[#F59E0B]" />;
   117→  if (code >= 1 && code <= 3) return <CloudSun className="w-5 h-5 text-[#94A3B8]" />;
   118→  if (code >= 45 && code <= 48) return <CloudFog className="w-5 h-5 text-[#94A3B8]" />;
   119→  if (code >= 51 && code <= 55) return <CloudDrizzle className="w-5 h-5 text-[#3B82F6]" />;
   120→  if (code >= 56 && code <= 57) return <CloudDrizzle className="w-5 h-5 text-[#3B82F6]" />;
   121→  if (code >= 61 && code <= 67) return <CloudRain className="w-5 h-5 text-[#3B82F6]" />;
   122→  if (code >= 71 && code <= 77) return <CloudSnow className="w-5 h-5 text-[#3B82F6]" />;
   123→  if (code >= 80 && code <= 82) return <CloudLightning className="w-5 h-5 text-[#92400E]" />;
   124→  if (code >= 95) return <CloudLightning className="w-5 h-5 text-[#5B21B6]" />;
   125→  return <Thermometer className="w-5 h-5 text-[#94A3B8]" />;

   316→function GraficoIngresosEgresos({ pagos, gastos }: { pagos: { fecha: string; monto: number }[]; gastos: { fecha: string; monto: number }[] }) {
   317→  const [mounted, setMounted] = useState(false);
   318→  useEffect(() => { const t = setTimeout(() => setMounted(true), 50); return () => clearTimeout(t); }, []);
   319→
   320→  const datos = useMemo(() => {
   321→    const hoy = new Date();
   322→    const dias: string[] = [];
   323→    for (let i = 6; i >= 0; i--) {
   324→      const d = new Date(hoy);
   325→      d.setDate(d.getDate() - i);
   326→      dias.push(format(d, 'yyyy-MM-dd'));
   327→    }
   328→    return dias.map(dia => {
   329→      const ing = pagos.filter(p => p.fecha === dia).reduce((s, p) => s + p.monto, 0);
   330→      const egr = gastos.filter(g => g.fecha === dia).reduce((s, g) => s + g.monto, 0);
   331→      const d = new Date(dia + 'T12:00:00');
   332→      return { dia: d.toLocaleDateString('es-AR', { weekday: 'short', day: 'numeric' }).replace(/^\w/, c => c.toUpperCase()), ing, egr, fullDate: formatearFecha(dia) };
   333→    });
   334→  }, [pagos, gastos]);
   335→
   336→  const maxVal = Math.max(...datos.map(d => Math.max(d.ing, d.egr)), 1);

   338→  return (

   677→function getBarColorClass(tipo: string): string {
   678→  const map: Record<string, string> = {
   679→    Reservada: 'bg-[#3B82F6] shadow-[0_2px_6px_rgba(59,130,246,0.35)]',
   680→    Ocupada: 'bg-[#4ADE80] shadow-[0_2px_6px_rgba(74,222,128,0.35)]',
   681→    Finalizada: 'bg-[#94A3B8] opacity-50 border border-dashed border-[#CBD5E1]',
   682→    Limpieza: 'bg-[#F59E0B] shadow-[0_2px_6px_rgba(245,158,11,0.30)]',
   683→    Mantenimiento: 'bg-[#64748B] shadow-[0_2px_6px_rgba(100,116,139,0.25)]',
   684→  };
   685→  return map[tipo] || map.Reservada;
   686→}

   688→function calcularBarra(res: GanttReserva, colIdx: Record<string, number>, columnas: string[], DIAS: number) {
   689→  let startCol = Math.max(colIdx[res.checkin] ?? 0, 0);
   690→  let endCol = Math.min(colIdx[res.checkout] !== undefined ? colIdx[res.checkout] : DIAS - 1, DIAS - 1);
   691→
   692→  const esCheckinTarde = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckin && res.horaCheckin.getHours() >= 12;
   693→  const esCheckoutManana = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckout && res.horaCheckout.getHours() < 12;
   694→
   695→  let leftPct: number, widthPct: number;
   696→
   697→  if (esCheckinTarde && startCol >= 0 && !(res.checkin < columnas[0])) {
   698→    leftPct = startCol * COL_PCT + MITAD_COL_PCT;
   699→  } else {
   700→    leftPct = startCol * COL_PCT;
   701→  }
   702→
   703→  if (esCheckoutManana && endCol >= 0 && !(res.checkout > columnas[DIAS - 1])) {
   704→    widthPct = ((endCol - startCol) * COL_PCT) + (esCheckinTarde ? MITAD_COL_PCT : COL_PCT) - (esCheckoutManana ? MITAD_COL_PCT : 0);
   705→  } else {
   706→    widthPct = (endCol - startCol + 1) * COL_PCT;
   707→  }
   708→
   709→  return { leftPct, widthPct };
   710→}

   714→function RoomHeatmap({ habitaciones, reservas }: {
   715→  habitaciones: Record<string, { tipo: string; estado: string; capacidad: number; problema?: string }>;

   772→          </span>
   773→        </div>
