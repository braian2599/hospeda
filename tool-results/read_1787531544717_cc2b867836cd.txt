    38→/** Smooth scroll to a section by id. */
    39→function scrollTo(id: string) {
    40→  document.getElementById(id)?.scrollIntoView({ behavior: 'smooth', block: 'start' });
    41→}
    42→
    43→/** Intersection Observer hook for fade-in animations. */
    44→function useInView<T extends HTMLElement = HTMLDivElement>(threshold = 0.12) {
    45→  const ref = useRef<T>(null);
    46→  const [inView, setInView] = useState(false);
    47→  useEffect(() => {
    48→    const el = ref.current;
    49→    if (!el) return;
    50→    const obs = new IntersectionObserver(
    51→      ([e]) => {
    52→        if (e.isIntersecting) {
    53→          setInView(true);
    54→          obs.disconnect();
    55→        }
    56→      },
    57→      { threshold }
    58→    );
    59→    obs.observe(el);
    60→    return () => obs.disconnect();
    61→  }, [threshold]);
    62→  return { ref, inView };
    63→}

    65→/** Fade-in wrapper that animates children on scroll. */
    66→function FadeIn({
    67→  children,
    68→  delay = 0,
    69→  className = '',
    70→}: {
    71→  children: ReactNode;
    72→  delay?: number;
    73→  className?: string;
    74→}) {
    75→  const { ref, inView } = useInView();
    76→  return (
    77→    <div
    78→      ref={ref}
    79→      className={className}
    80→      style={{
    81→        opacity: inView ? 1 : 0,
    82→        transform: inView ? 'translateY(0)' : 'translateY(24px)',
    83→        transition: `opacity 0.6s ease ${delay}ms, transform 0.6s ease ${delay}ms`,
    84→      }}
    85→    >
    86→      {children}
    87→    </div>
    88→  );
    89→}

    91→/** Top scroll progress bar. */
    92→function ScrollProgress() {
    93→  const [progress, setProgress] = useState(0);
    94→  useEffect(() => {
    95→    const onScroll = () => {
    96→      const scrollTop = window.scrollY;
    97→      const docHeight = document.documentElement.scrollHeight - window.innerHeight;
    98→      setProgress(docHeight > 0 ? (scrollTop / docHeight) * 100 : 0);
    99→    };
   100→    onScroll();
   101→    window.addEventListener('scroll', onScroll, { passive: true });
   102→    return () => window.removeEventListener('scroll', onScroll);
   103→  }, []);
   104→  return (
   105→    <div
   106→      className="fixed top-0 left-0 z-[60] h-[3px] w-full bg-primary transition-[width] duration-150"
   107→      style={{ width: `${progress}%` }}
   108→      aria-hidden="true"
   109→    />
   110→  );
   111→}
   112→
   113→/** Back to top floating button. */
   114→function BackToTop() {
   115→  const [visible, setVisible] = useState(false);
   116→  useEffect(() => {
   117→    const onScroll = () => setVisible(window.scrollY > 600);
   118→    window.addEventListener('scroll', onScroll, { passive: true });
   119→    return () => window.removeEventListener('scroll', onScroll);
   120→  }, []);
   121→  return (
   122→    <button
   123→      type="button"
   124→      onClick={() => window.scrollTo({ top: 0, behavior: 'smooth' })}
   125→      aria-label="Volver arriba"
   126→      className={`fixed bottom-6 right-6 z-50 h-11 w-11 rounded-full bg-primary text-primary-foreground shadow-lg transition-all duration-300 hover:bg-primary/90 hover:-translate-y-1 ${
   127→        visible ? 'opacity-100 translate-y-0' : 'opacity-0 pointer-events-none translate-y-4'
   128→      }`}
   129→    >
   130→      <ArrowRight className="h-5 w-5 mx-auto -rotate-90" />
   131→    </button>
   132→  );
   133→}
