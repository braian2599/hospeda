    53→// ==================== KPI CARD ====================
    54→
    55→interface KpiProps {
    56→  label: string;
    57→  value: string | number;
    58→  icon: React.ReactNode;
    59→  color: string;
    60→  bgColor: string;
    61→  tooltip?: string;
    62→  subtext?: string;
    63→  trend?: { value: number; label: string };
    64→  onClick?: () => void;
    65→}
    66→
    67→function KpiCard({ label, value, icon, color, bgColor, tooltip, subtext, trend, onClick }: KpiProps) {
    68→  const card = (
    69→    <Card
    70→      className={`relative overflow-hidden transition-all duration-200 hover:shadow-md ${onClick ? 'cursor-pointer group' : ''}`}
    71→      onClick={onClick}
    72→    >
    73→      <CardContent className="p-3 sm:p-4">
    74→        <div className="flex items-start justify-between gap-2 sm:gap-3">
    75→          <div className="flex-1 min-w-0 text-center">
    76→            <p className="text-[10px] sm:text-[11px] font-medium uppercase tracking-wider text-muted-foreground mb-1 sm:mb-1.5">{label}</p>
    77→            <p className={`text-lg sm:text-2xl font-bold tracking-tight ${color} truncate`}>{value}</p>
    78→            {trend && (
    79→              <div className="flex items-center justify-center gap-1 mt-1 sm:mt-1.5">
    80→                {trend.value > 0 ? (
    81→                  <ArrowUpRight className="w-3 h-3 sm:w-3.5 sm:h-3.5 text-[#059669]" />
    82→                ) : trend.value < 0 ? (
    83→                  <ArrowDownRight className="w-3 h-3 sm:w-3.5 sm:h-3.5 text-[#EF4444]" />
    84→                ) : (
    85→                  <Minus className="w-3 h-3 sm:w-3.5 sm:h-3.5 text-muted-foreground" />
    86→                )}
    87→                <span className={`text-[11px] sm:text-xs font-medium ${trend.value > 0 ? 'text-[#059669]' : trend.value < 0 ? 'text-[#EF4444]' : 'text-muted-foreground'}`}>
    88→                  {Math.abs(trend.value)}%
    89→                </span>
    90→                <span className="text-[11px] sm:text-xs text-muted-foreground hidden sm:inline">{trend.label}</span>
    91→              </div>
    92→            )}
    93→            {subtext && !trend && <p className="text-[11px] sm:text-xs text-muted-foreground mt-1 truncate hidden sm:block">{subtext}</p>}
    94→          </div>
    95→          <div className={`shrink-0 w-8 h-8 sm:w-10 sm:h-10 rounded-lg sm:rounded-xl flex items-center justify-center ${bgColor}`}>
    96→            <div className="sm:scale-100 scale-90">{icon}</div>
    97→          </div>
    98→        </div>
    99→      </CardContent>
   100→      {onClick && (
   101→        <div className="absolute inset-x-0 bottom-0 h-0.5 bg-primary opacity-0 group-hover:opacity-100 transition-opacity" />
   102→      )}
   103→    </Card>
   104→  );
   105→
   106→  if (tooltip) {
   107→    return (
   108→      <TooltipProvider delayDuration={300}>
   109→        <Tooltip>
   110→          <TooltipTrigger asChild>{card}</TooltipTrigger>
   111→          <TooltipContent side="bottom" className="text-xs max-w-[220px]">{tooltip}</TooltipContent>
   112→        </Tooltip>
   113→      </TooltipProvider>
   114→    );
   115→  }
   116→  return card;
   117→}

   131→function ProgressKpi({ label, value, max, color = 'bg-primary', suffix = '%', description }: { label: string; value: number; max: number; color?: string; suffix?: string; description?: string }) {
   132→  const pct = max > 0 ? Math.round((value / max) * 100) : 0;
   133→  return (
   134→    <Card>
   135→      <CardContent className="p-3 sm:p-4">
   136→        <div className="flex items-center justify-between mb-1.5 sm:mb-2">
   137→          <p className="text-[10px] sm:text-[11px] font-medium uppercase tracking-wider text-muted-foreground">{label}</p>
   138→          <span className="text-base sm:text-lg font-bold">{value}{suffix}</span>
   139→        </div>
   140→        <Progress value={pct} className="h-2 sm:h-2.5" />
   141→        <p className="text-[11px] sm:text-xs text-muted-foreground mt-1 sm:mt-1.5 text-center">{description || `${pct}%`}</p>
   142→      </CardContent>
   143→    </Card>
   144→  );
   145→}
