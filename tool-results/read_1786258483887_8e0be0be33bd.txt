    85→function RangoFila({ rango, index, onRemove, onUpdate, modoCobro, totalRangos }: {
    86→  rango: RangoPrecio;
    87→  index: number;
    88→  onRemove: () => void;
    89→  onUpdate: (r: RangoPrecio) => void;
    90→  modoCobro: ModoCobro;
    91→  totalRangos: number;
    92→}) {
    93→  const esInfinito = rango.maxPersonas === null;
    94→  // Etiqueta legible del rango: "1 persona", "2 personas", "3+ personas", "2-4 personas"
    95→  const rangoLabel = esInfinito
    96→    ? `${rango.minPersonas}+ persona${rango.minPersonas > 1 ? 's' : ''}`
    97→    : rango.minPersonas === rango.maxPersonas
    98→      ? `${rango.minPersonas} persona${rango.minPersonas > 1 ? 's' : ''}`
    99→      : `${rango.minPersonas} a ${rango.maxPersonas} personas`;
   100→
   101→  return (
   102→    <div className="flex items-end gap-2">
   103→      <div className="w-28 shrink-0 flex items-center h-8 text-sm font-medium text-foreground">
   104→        {modoCobro === 'porHabitacion' ? 'Por noche' : modoCobro === 'porCama' ? 'Por cama/noche' : rangoLabel}
   105→      </div>
   106→      <div className="space-y-0.5 flex-1">
   107→        <Label className="text-[10px] text-muted-foreground">
   108→          {modoCobro === 'porPersona' || modoCobro === 'porCama' ? 'Precio c/u' : 'Precio'}
   109→        </Label>
   110→        <Input
   111→          type="number" min={0} step="0.01"
   112→          className="h-8 text-right"
   113→          value={rango.precio}
   114→          onChange={e => onUpdate({ ...rango, precio: parseFloat(e.target.value) || 0 })}
   115→        />
   116→      </div>
   117→      <Button
   118→        size="icon" variant="ghost"
   119→        className="h-8 w-8 text-destructive shrink-0 mb-0.5"
   120→        onClick={onRemove}
   121→        disabled={modoCobro === 'porHabitacion' || modoCobro === 'porCama' || totalRangos <= 1}
   122→      >
   123→        <Trash2 className="w-3.5 h-3.5" />
   124→      </Button>
   125→    </div>
   126→  );
   127→}

   138→function formatoRango(r: RangoPrecio): string {
   139→ if (r.maxPersonas === null) return `${r.minPersonas}+`;
   140→  if (r.minPersonas === r.maxPersonas) return `${r.minPersonas}`;
   141→  return `${r.minPersonas}-${r.maxPersonas}`;
   142→}
   143→
   144→function modoLabel(m: ModoCobro): string {
   145→  return MODO_OPTIONS.find(o => o.value === m)?.label || m;
   146→}
   147→
   148→function modoBadgeColor(m: ModoCobro): string {
   149→  switch (m) {
   150→    case 'porPersona': return 'bg-[#F5F3FF] text-[#6D28D9] border-0';
   151→    case 'porHabitacion': return 'bg-[#FEF3C7] text-[#92400E] border-0';
   152→    case 'porCama': return 'bg-[#DCFCE7] text-[#166534] border-0';
   153→    default: return 'bg-[#D1FAE5] text-[#065F46] border-0';
   154→  }
   155→}

   157→// Subtle gradient tint per modo for rate cards (decorative, soft tints)
   158→function modoGradient(m: ModoCobro): string {
   159→  switch (m) {
   160→    case 'porPersona': return 'from-[#FFFBEB] to-white';
   161→    case 'porHabitacion': return 'from-[#FEF2F2] to-white';
   162→    case 'porCama': return 'from-[#F5F3FF] to-white';
   163→    default: return 'from-[#F0FDF4] to-white';
   164→  }
   165→}

   167→// Colored circle for rate card icon
   168→function modoIconCircle(m: ModoCobro): string {
   169→  switch (m) {
   170→    case 'porPersona': return 'bg-[#FEF3C7] text-[#92400E]';
   171→    case 'porHabitacion': return 'bg-[#FEE2E2] text-[#991B1B]';
   172→    case 'porCama': return 'bg-[#EDE9FE] text-[#5B21B6]';
   173→    default: return 'bg-[#DCFCE7] text-[#166534]';

   177→function crearRangoDefault(modo: ModoCobro, minPersonas: number = 1): RangoPrecio {
   178→  if (modo === 'porHabitacion') {
   179→    return { minPersonas: 1, maxPersonas: null, precio: 0 };
