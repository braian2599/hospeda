    18→/* ─── Intersection Observer hook ─── */
    19→function useInView(threshold = 0.12) {
    20→  const ref = useRef<HTMLDivElement>(null);
    21→  const [inView, setInView] = useState(false);
    22→  useEffect(() => {
    23→    const el = ref.current;
    24→    if (!el) return;
    25→    const obs = new IntersectionObserver(
    26→      ([e]) => { if (e.isIntersecting) { setInView(true); obs.disconnect(); } },
    27→      { threshold }
    28→    );
    29→    obs.observe(el);
    30→    return () => obs.disconnect();
    31→  }, [threshold]);
    32→  return { ref, inView };
    33→}

    40→/* ─── Scroll progress indicator (top gradient bar) ─── */
    41→function ScrollProgress() {
    42→  const [progress, setProgress] = useState(0);
    43→  useEffect(() => {
    44→    const onScroll = () => {
    45→      const scrollTop = window.scrollY;
    46→      const docHeight = document.documentElement.scrollHeight - window.innerHeight;
    47→      setProgress(docHeight > 0 ? Math.min(100, (scrollTop / docHeight) * 100) : 0);
    48→    };
    49→    onScroll();
    50→    window.addEventListener('scroll', onScroll, { passive: true });
    51→    return () => window.removeEventListener('scroll', onScroll);
    52→  }, []);
    53→  return (
    54→    <div
    55→      className="scroll-progress"
    56→      style={{ width: `${progress}%`, opacity: progress > 0.5 ? 1 : 0 }}
    57→      aria-hidden="true"
    58→    />
    59→  );
    60→}

   170→/* ─── Fade-in wrapper ─── */
   171→function FadeIn({ children, className = '', delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   172→  const { ref, inView } = useInView();
   173→  return (
   174→    <div
   175→      ref={ref}
   176→      className={`transition-all duration-700 ease-out ${className}`}
   177→      style={{
   178→        opacity: inView ? 1 : 0,
   179→        transform: inView ? 'translateY(0)' : 'translateY(32px)',
   180→        transitionDelay: `${delay}ms`,
   181→      }}
   182→    >
   183→      {children}
   184→    </div>
   185→  );
   186→}

   188→/* ─── Typewriter text effect ─── */
   189→function TypewriterText({ text, className = '' }: { text: string; className?: string }) {
   190→  const [displayed, setDisplayed] = useState('');
   191→  const [done, setDone] = useState(false);
   192→
   193→  useEffect(() => {
   194→    let i = 0;
   195→    const interval = setInterval(() => {
   196→      i++;
   197→      if (i <= text.length) {
   198→        setDisplayed(text.slice(0, i));
   199→      } else {
   200→        setDone(true);
   201→        clearInterval(interval);
   202→      }
   203→    }, 65);
   204→    return () => clearInterval(interval);
   205→  }, [text]);
   206→
   207→  return (
   208→    <span className={className}>
   209→      {displayed}
   210→      {!done && <span className="typewriter-cursor" aria-hidden="true" />}
   211→    </span>
   212→  );
   213→}

   215→/* ─── Back to top button ─── */
   216→function BackToTop() {
   217→  const [visible, setVisible] = useState(false);
   218→
   219→  useEffect(() => {
   220→    const onScroll = () => {
   221→      const scrollTop = window.scrollY;
   222→      const docHeight = document.documentElement.scrollHeight - window.innerHeight;
   223→      setVisible(docHeight > 0 && scrollTop / docHeight > 0.5);
   224→    };
   225→    onScroll();
   226→    window.addEventListener('scroll', onScroll, { passive: true });
   227→    return () => window.removeEventListener('scroll', onScroll);
   228→  }, []);
   229→
   230→  if (!visible) return null;
   231→
   232→  return (
   233→    <button
   234→      className="back-to-top-btn"
   235→      onClick={() => window.scrollTo({ top: 0, behavior: 'smooth' })}
   236→      aria-label="Volver arriba"
   237→    >
   238→      <ArrowUp className="w-4 h-4" />
   239→    </button>
   240→  );
   241→}
