    84→/** Argentine CUIT/CUIL verification digit (dígito verificador). */
    85→function calcularDigitoVerificadorCuit(cuit: string): number | null {
    86→  const digits = cuit.replace(/\D/g, '');
    87→  if (digits.length < 10) return null;
    88→  const first10 = digits.slice(0, 10).split('').map(Number);
    89→  if (first10.some(n => Number.isNaN(n))) return null;
    90→  const mult = [5, 4, 3, 2, 7, 6, 5, 4, 3, 2];
    91→  let sum = 0;
    92→  for (let i = 0; i < 10; i++) sum += first10[i] * mult[i];
    93→  const rest = sum % 11;
    94→  const digit = 11 - rest;
    95→  if (digit === 11) return 0;
    96→  if (digit === 10) return null;
    97→  return digit;
    98→}

   100→type StrengthLevel = 'weak' | 'medium' | 'strong';
   101→
   102→function getPasswordStrength(pw: string): { level: StrengthLevel; label: string; pct: number; color: string; textColor: string } {
   103→  if (!pw) return { level: 'weak', label: '—', pct: 0, color: 'bg-muted', textColor: 'text-muted-foreground' };
   104→  const len = pw.length;
   105→  const hasLetters = /[a-zA-Z]/.test(pw);
   106→  const hasNumbers = /[0-9]/.test(pw);
   107→  const hasSymbols = /[^a-zA-Z0-9]/.test(pw);
   108→  const hasMixed = /[a-z]/.test(pw) && /[A-Z]/.test(pw);
   109→
   110→  if (len >= 10 && hasNumbers && (hasSymbols || hasMixed)) {
   111→    return { level: 'strong', label: 'Fuerte', pct: 100, color: 'bg-[#059669]', textColor: 'text-[#059669]' };
   112→  }
   113→  if (len >= 6 && hasLetters && hasNumbers) {
   114→    return { level: 'medium', label: 'Media', pct: 60, color: 'bg-[#F59E0B]', textColor: 'text-[#F59E0B]' };
   115→  }
   116→  return { level: 'weak', label: 'Débil', pct: 25, color: 'bg-red-500', textColor: 'text-red-500' };
   117→}

   119→function formatBytes(bytes: number): string {
   120→  if (bytes < 1024) return `${bytes} B`;
   121→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   122→  return `${(bytes / 1024 / 1024).toFixed(1)} MB`;
   123→}
