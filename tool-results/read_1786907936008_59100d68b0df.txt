   148→function formatoRango(r: RangoPrecio): string {
   149→  if (r.maxPersonas === null) return `${r.minPersonas}+`;
   150→  if (r.minPersonas === r.maxPersonas) return `${r.minPersonas}`;
   151→  return `${r.minPersonas}-${r.maxPersonas}`;
   152→}

   180→// Count active promotions on a tariff
   181→function countPromos(promos?: PromocionesTarifa | null): number {
   182→  if (!promos) return 0;
   183→  let n = 0;
   184→  if (promos.acompananteSinCargo?.activo) n++;
   185→  if (promos.ninosDiferenciado?.activo) n++;
   186→  if (promos.nochesCortesia?.activo) n++;
   187→  return n;
   188→}

   190→// Compute "Desde" price (min positive price, fallback to first range)
   191→function precioDesde(rangos: RangoPrecio[]): number {
   192→  const preciosPositivos = rangos.map(r => r.precio).filter(p => p > 0);
   193→  return preciosPositivos.length > 0 ? Math.min(...preciosPositivos) : (rangos[0]?.precio || 0);
   194→}
   195→
   196→// Build a label describing a noches cortesía modalidad
   197→function describeNochesCortesia(mod: ModalidadNochesCortesia): string {
   198→  if (mod.tipo === 'cadaX') return `Cada ${mod.cada} noches, 1 gratis`;
   199→  if (mod.tipo === 'aPartirDe') return `Desde ${mod.minNoches} noches, ${mod.nochesGratis} gratis`;
   200→  const dias = ['Domingo', 'Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado'];
   201→  return `${dias[mod.dia]} gratis`;
   202→}

   204→// Export a single tariff as CSV (client-side download)
   205→function exportTariffCSV(tipo: string, t: TarifaPrecios) {
   206→  const rows: (string | number)[][] = [];
   207→  rows.push(['Campo', 'Valor']);
   208→  rows.push(['Nombre', tipo]);
   209→  rows.push(['Modo de cobro', t.modoCobro || 'porGrupo']);
   210→  const rangos = t.rangos || [];
   211→  rows.push(['Cantidad de rangos', rangos.length]);
   212→  rangos.forEach((r, i) => {
   213→    rows.push([`Rango ${i + 1} (${formatoRango(r)} pers.)`, r.precio]);
   214→  });
   215→  rows.push(['Acompañante sin cargo', t.promociones?.acompananteSinCargo?.activo ? 'Sí' : 'No']);
   216→  if (t.promociones?.acompananteSinCargo?.activo) {
   217→    rows.push(['  Etiqueta', t.promociones.acompananteSinCargo.etiqueta || '—']);
   218→    rows.push(['  Cantidad', t.promociones.acompananteSinCargo.cantidad ?? 1]);
   219→  }
   220→  rows.push(['Niños diferenciado', t.promociones?.ninosDiferenciado?.activo ? 'Sí' : 'No']);
   221→  if (t.promociones?.ninosDiferenciado?.activo) {
   222→    rows.push(['  Precio niño/noche', t.promociones.ninosDiferenciado.precioNino || 0]);
   223→  }
   224→  rows.push(['Noches cortesía', t.promociones?.nochesCortesia?.activo ? 'Sí' : 'No']);
   225→  if (t.promociones?.nochesCortesia?.activo) {
   226→    rows.push(['  Modalidad', describeNochesCortesia(t.promociones.nochesCortesia.modalidad)]);
   227→  }
   228→  const campos = t.camposPersonalizados || [];
   229→  rows.push(['Cantidad de campos personalizados', campos.length]);
   230→  campos.forEach((c, i) => {
   231→    rows.push([`  Campo ${i + 1}`, `${c.nombre} (${c.tipo}${c.requerido ? ', requerido' : ''})`]);
   232→  });
   233→
   234→  const csv = rows.map(r => r.map(cell => {
   235→    const s = String(cell);
   236→    return s.includes(',') || s.includes('"') ? `"${s.replace(/"/g, '""')}"` : s;
   237→  }).join(',')).join('\n');
   238→
   239→  const blob = new Blob([`\uFEFF${csv}`], { type: 'text/csv;charset=utf-8;' });
   240→  const url = URL.createObjectURL(blob);
   241→  const a = document.createElement('a');
   242→  a.href = url;
   243→  a.download = `tarifa_${tipo.toLowerCase().replace(/\s+/g, '_')}.csv`;
   244→  document.body.appendChild(a);
   245→  a.click();
   246→  document.body.removeChild(a);
   247→  URL.revokeObjectURL(url);
   248→}

   252→function WizardStepper({ current, onSelect }: { current: 1 | 2 | 3; onSelect: (s: 1 | 2 | 3) => void }) {
   253→  const steps: { n: 1 | 2 | 3; label: string; icon: typeof Info }[] = [
   254→    { n: 1, label: 'Información básica', icon: Info },
   255→    { n: 2, label: 'Rangos de precios', icon: Tags },
   256→    { n: 3, label: 'Promociones', icon: Sparkles },
   257→  ];
   258→  return (
   259→    <div className="space-y-2">
   260→      <div className="flex items-center gap-1">
   261→        {steps.map((s, i) => {
   262→          const Icon = s.icon;
   263→          const isDone = current > s.n;
   264→          const isActive = current === s.n;
   265→          return (
   266→            <div key={s.n} className="flex items-center flex-1 last:flex-none">
   267→              <button
   268→                type="button"
   269→                onClick={() => onSelect(s.n)}
   270→                className={`flex items-center gap-2 px-3 py-1.5 rounded-md text-xs font-semibold transition-all whitespace-nowrap ${
   271→                  isActive
   272→                    ? 'bg-primary text-white shadow-sm'
   273→                    : isDone
   274→                      ? 'bg-primary/10 text-primary hover:bg-primary/20'
   275→                      : 'bg-muted text-muted-foreground hover:bg-muted/80'
   276→                }`}
   277→                aria-current={isActive ? 'step' : undefined}
   278→              >
   279→                <span className={`w-5 h-5 rounded-full flex items-center justify-center text-[10px] ${
   280→                  isActive ? 'bg-white/20' : isDone ? 'bg-primary text-white' : 'bg-background'
   281→                }`}>
   282→                  {isDone ? <Check className="w-3 h-3" /> : s.n}
   283→                </span>
   284→                <span className="hidden sm:inline">{s.label}</span>
   285→                <Icon className="w-3.5 h-3.5 sm:hidden" />
   286→              </button>
   287→              {i < steps.length - 1 && (
   288→                <div className={`flex-1 h-0.5 mx-1 rounded ${isDone ? 'bg-primary' : 'bg-muted'}`} />
   289→              )}
   290→            </div>
   291→          );
   292→        })}
   293→      </div>
   294→      <Progress value={(current / 3) * 100} className="h-1 bg-muted" />
   295→    </div>
   296→  );
   297→}

   402→function ComparisonModal({ tariffs, onClose }: {
   403→  tariffs: { tipo: string; t: TarifaPrecios }[];
   404→  onClose: () => void;
   405→}) {
   406→  // Compute base price per person for each tariff (min positive price)
   407→  const basePrices = tariffs.map(({ t }) => {
   408→    const rangos = t.rangos || [];
   409→    const pos = rangos.map(r => r.precio).filter(p => p > 0);
   410→    return pos.length > 0 ? Math.min(...pos) : (rangos[0]?.precio || 0);
   411→  });
   412→
   413→  // Determine which rows have differing values for highlighting
   414→  const modos = tariffs.map(({ t }) => t.modoCobro || 'porGrupo');
   415→  const modoDiffers = new Set(modos).size > 1;
   416→  const priceDiffers = new Set(basePrices).size > 1;
   417→  const rangoCounts = tariffs.map(({ t }) => (t.rangos || []).length);
   418→  const rangoDiffers = new Set(rangoCounts).size > 1;
   419→  const promoCounts = tariffs.map(({ t }) => countPromos(t.promociones));
   420→  const promoDiffers = new Set(promoCounts).size > 1;
   421→  const campoCounts = tariffs.map(({ t }) => (t.camposPersonalizados || []).length);
   422→  const campoDiffers = new Set(campoCounts).size > 1;
   423→
   424→  return (
   425→    <DialogContent className="sm:max-w-4xl max-h-[90vh] overflow-y-auto">
   426→      <DialogHeader>
   427→        <DialogTitle className="flex items-center gap-2 text-xl">
   428→          <GitCompareArrows className="w-5 h-5 text-primary" />
   429→          Comparación de tarifas
   430→        </DialogTitle>
   431→        <p className="text-sm text-muted-foreground mt-1">
   432→          Comparando {tariffs.length} tarifas. Las filas con diferencias están resaltadas.
   433→        </p>
   434→      </DialogHeader>
   435→
   436→      <div className="overflow-x-auto rounded-lg border border-border">
   437→        <Table>
