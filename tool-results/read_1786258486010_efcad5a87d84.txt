    44→    44→function UsageBar({ label, current, max, icon: Icon }: { label: string; current: number; max: number; icon: React.ComponentType<{ className?: string }> }) {
    45→    45→  const pct = max === 0 ? 0 : Math.min(100, Math.round((current / max) * 100));
    46→    46→  const isUnlimited = max === 0;
    47→    47→  return (
    48→    48→    <div className="space-y-1.5">
    49→    49→      <div className="flex items-center justify-between text-sm">
    50→    50→        <div className="flex items-center gap-2">
    51→    51→          <Icon className="w-3.5 h-3.5 text-muted-foreground" />
    52→    52→          <span>{label}</span>
    53→    53→        </div>
    54→    54→        <span className="text-muted-foreground">
    55→    55→          {current} / {isUnlimited ? 'Ilimitado' : max}
    56→    56→        </span>
    57→    57→      </div>
    58→    58→      {isUnlimited ? (
    59→    59→        <div className="h-2 rounded-full bg-muted" />
    60→    60→      ) : (
    61→    61→        <Progress value={pct} className="h-2" />

    63→    63→    </div>
    64→    64→  );
    65→    65→}

   790→   790→  const handleExport = async (tipo: string) => {
   791→   791→    setExporting(tipo);
   792→   792→    try {
   793→   793→      const endpoints: Record<string, string> = {
   794→   794→        habitaciones: '/api/habitaciones',
   795→   795→        clientes: '/api/clientes',
   796→   796→        reservas: '/api/reservas',
   797→   797→      };
   798→   798→      const url = endpoints[tipo];
   799→   799→      if (!url) return;
   800→   800→
   801→   801→      const res = await fetch(url);
   802→   802→      const data = await res.json();
   803→   803→      const items = Array.isArray(data) ? data : data.data || data.habitaciones || data.clientes || data.reservas || [];
   804→   804→
   805→   805→      if (!items.length) { toast.info('No hay datos para exportar'); setExporting(null); return; }
   806→   806→
   807→   807→      // Build CSV
   808→   808→      const headers = Object.keys(items[0]).filter(k => typeof items[0][k] !== 'object');
   809→   809→      const csv = [
   810→   810→        headers.join(','),
   811→   811→        ...items.map((row: any) => headers.map(h => {
   812→   812→          let val = row[h];
   813→   813→          if (val === null || val === undefined) val = '';
   814→   814→          if (typeof val === 'string' && (val.includes(',') || val.includes('"'))) val = `"${val.replace(/"/g, '""')}"`;
   815→   815→          return val;
   816→   816→        }).join(','))
   817→   817→      ].join('\n');
   818→   818→
   819→   819→      const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
   820→   820→      const link = document.createElement('a');
   821→   821→      link.href = URL.createObjectURL(blob);
   822→   822→      link.download = `hospeda-${tipo}-${new Date().toLocaleDateString('en-CA')}.csv`;
   823→   823→      link.click();
   824→   824→      URL.revokeObjectURL(link.href);
   825→   825→      toast.success(`${tipo} exportados correctamente`);
   826→   826→    } catch {
   827→   827→      toast.error('Error al exportar');
   828→   828→    }
   829→   829→    setExporting(null);
   830→   830→  };
   831→   831→
   832→   832→  const exports = [
   833→   833→    { id: 'habitaciones', label: 'Habitaciones', desc: 'Listado completo de habitaciones y sus estados', icon: Hotel },
   834→   834→    { id: 'clientes', label: 'Clientes', desc: 'Base de huéspedes con datos de contacto', icon: Mail },
   835→   835→    { id: 'reservas', label: 'Reservas', desc: 'Historial de reservas con estados y pagos', icon: CreditCard },
   836→   836→  ];
   837→   837→
   838→   838→  return (
   839→   839→    <Card>
   840→   840→      <CardHeader>
   841→   841→        <CardTitle className="text-lg">Exportar Datos</CardTitle>
   842→   842→        <CardDescription>Descargá tus datos en formato CSV para usar en otras herramientas</CardDescription>
   843→   843→      </CardHeader>
   844→   844→      <CardContent className="space-y-3">
   845→   845→        {exports.map(exp => {
   846→   846→          const Icon = exp.icon;
   847→   847→          return (
   848→   848→            <div key={exp.id} className="flex items-center justify-between p-4 rounded-lg border hover:bg-muted/30 transition-colors">
   849→   849→              <div className="flex items-center gap-3">
   850→   850→                <div className="w-9 h-9 rounded-lg bg-muted flex items-center justify-center">
   851→   851→                  <Icon className="w-4 h-4 text-muted-foreground" />
   852→   852→                </div>
   853→   853→                <div>
   854→   854→                  <p className="text-sm font-medium">{exp.label}</p>
   855→   855→                  <p className="text-xs text-muted-foreground">{exp.desc}</p>
   856→   856→                </div>
   857→   857→              </div>
   858→   858→              <Button
   859→   859→                variant="outline"
   860→   860→                size="sm"
   861→   861→                onClick={() => handleExport(exp.id)}
   862→   862→                disabled={exporting === exp.id}
   863→   863→              >
   864→   864→                {exporting === exp.id ? <Loader2 className="w-4 h-4 animate-spin" /> : <Download className="w-4 h-4 mr-1.5" />}
   865→   865→                CSV
   866→   866→              </Button>
   867→   867→            </div>
   868→   868→          );
   869→   869→        })}
   870→   870→      </CardContent>
   871→   871→    </Card>
   872→   872→  );
