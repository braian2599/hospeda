   293→   293→function GraficoIngresosEgresos({ pagos, gastos }: { pagos: { fecha: string; monto: number }[]; gastos: { fecha: string; monto: number }[] }) {
   294→   294→  const [mounted, setMounted] = useState(false);
   295→   295→  useEffect(() => { const t = setTimeout(() => setMounted(true), 50); return () => clearTimeout(t); }, []);
   296→   296→
   297→   297→  const datos = useMemo(() => {
   298→   298→    const hoy = new Date();
   299→   299→    const dias: string[] = [];
   300→   300→    for (let i = 6; i >= 0; i--) {
   301→   301→      const d = new Date(hoy);
   302→   302→      d.setDate(d.getDate() - i);
   303→   303→      dias.push(toLocalDateStr(d));
   304→   304→    }
   305→   305→    return dias.map(dia => {
   306→   306→      const ing = pagos.filter(p => p.fecha === dia).reduce((s, p) => s + p.monto, 0);
   307→   307→      const egr = gastos.filter(g => g.fecha === dia).reduce((s, g) => s + g.monto, 0);
   308→   308→      const d = new Date(dia + 'T12:00:00');
   309→   309→      return { dia: d.toLocaleDateString('es-AR', { weekday: 'short', day: 'numeric' }).replace(/^\w/, c => c.toUpperCase()), ing, egr, fullDate: formatearFecha(dia) };
   310→   310→    });
   311→   311→  }, [pagos, gastos]);
   312→   312→
   313→   313→  const maxVal = Math.max(...datos.map(d => Math.max(d.ing, d.egr)), 1);

   315→   315→  return (

   329→   329→                  className="w-5 rounded-t-md transition-all duration-700 ease-out hover:opacity-80 cursor-default"
   330→   330→                  style={{
   331→   331→                    height: mounted ? `${(d.ing / maxVal) * 100}%` : '0%',
   332→   332→                    background: 'linear-gradient(to top, #059669, #34d399)',
   333→   333→                    boxShadow: '0 2px 8px rgba(5,150,105,0.3)',
   334→   334→                    transitionDelay: `${i * 80}ms`,
   335→   335→                  }}

   665→   665→function calcularBarra(res: GanttReserva, colIdx: Record<string, number>, columnas: string[], DIAS: number) {
   666→   666→  let startCol = Math.max(colIdx[res.checkin] ?? 0, 0);
   667→   667→  let endCol = Math.min(colIdx[res.checkout] !== undefined ? colIdx[res.checkout] : DIAS - 1, DIAS - 1);
   668→   668→
   669→   669→  const esCheckinTarde = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckin && res.horaCheckin.getHours() >= 12;
   670→   670→  const esCheckoutManana = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckout && res.horaCheckout.getHours() < 12;
   671→   671→
   672→   672→  let leftPct: number, widthPct: number;
   673→   673→
   674→   674→  if (esCheckinTarde && startCol >= 0 && !(res.checkin < columnas[0])) {
   675→   675→    leftPct = startCol * COL_PCT + MITAD_COL_PCT;
   676→   676→  } else {
   677→   677→    leftPct = startCol * COL_PCT;
   678→   678→  }
   679→   679→
   680→   680→  if (esCheckoutManana && endCol >= 0 && !(res.checkout > columnas[DIAS - 1])) {
   681→   681→    widthPct = ((endCol - startCol) * COL_PCT) + (esCheckinTarde ? MITAD_COL_PCT : COL_PCT) - (esCheckoutManana ? MITAD_COL_PCT : 0);
   682→   682→  } else {
   683→   683→    widthPct = (endCol - startCol + 1) * COL_PCT;
   684→   684→  }
   685→   685→
   686→   686→  widthPct = Math.max(widthPct, MITAD_COL_PCT);
   687→   687→  return { leftPct, widthPct };
   688→   688→}
