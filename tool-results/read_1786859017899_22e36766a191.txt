    21→    21→/* ─── Intersection Observer hook ─── */
    22→    22→function useInView(threshold = 0.12) {
    23→    23→  const ref = useRef<HTMLDivElement>(null);
    24→    24→  const [inView, setInView] = useState(false);
    25→    25→  useEffect(() => {
    26→    26→    const el = ref.current;
    27→    27→    if (!el) return;
    28→    28→    const obs = new IntersectionObserver(
    29→    29→      ([e]) => { if (e.isIntersecting) { setInView(true); obs.disconnect(); } },
    30→    30→      { threshold }
    31→    31→    );
    32→    32→    obs.observe(el);
    33→    33→    return () => obs.disconnect();
    34→    34→  }, [threshold]);
    35→    35→  return { ref, inView };
    36→    36→}

    44→    44→function ScrollProgress() {
    45→    45→  const [progress, setProgress] = useState(0);
    46→    46→  useEffect(() => {
    47→    47→    const onScroll = () => {
    48→    48→      const scrollTop = window.scrollY;
    49→    49→      const docHeight = document.documentElement.scrollHeight - window.innerHeight;
    50→    50→      setProgress(docHeight > 0 ? Math.min(100, (scrollTop / docHeight) * 100) : 0);
    51→    51→    };
    52→    52→    onScroll();
    53→    53→    window.addEventListener('scroll', onScroll, { passive: true });
    54→    54→    return () => window.removeEventListener('scroll', onScroll);
    55→    55→  }, []);
    56→    56→  return (
    57→    57→    <div
    58→    58→      className="scroll-progress"
    59→    59→      style={{ width: `${progress}%`, opacity: progress > 0.5 ? 1 : 0 }}
    60→    60→      aria-hidden="true"
    61→    61→    />
    62→    62→  );
    63→    63→}

   264→   264→/* ─── Fade-in wrapper ─── */
   265→   265→function FadeIn({ children, className = '', delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   266→   266→  const { ref, inView } = useInView();
   267→   267→  return (
   268→   268→    <div
   269→   269→      ref={ref}
   270→   270→      className={`transition-all duration-700 ease-out ${className}`}
   271→   271→      style={{
   272→   272→        opacity: inView ? 1 : 0,
   273→   273→        transform: inView ? 'translateY(0)' : 'translateY(32px)',
   274→   274→        transitionDelay: `${delay}ms`,
   275→   275→      }}
   276→   276→    >
   277→   277→      {children}
   278→   278→    </div>
   279→   279→  );
   280→   280→}

   282→   282→/* ─── Typewriter text effect ─── */
   283→   283→function TypewriterText({ text, className = '' }: { text: string; className?: string }) {
   284→   284→  const [displayed, setDisplayed] = useState('');
   285→   285→  const [done, setDone] = useState(false);
   286→   286→
   287→   287→  useEffect(() => {
   288→   288→    let i = 0;
   289→   289→    const interval = setInterval(() => {
   290→   290→      i++;
   291→   291→      if (i <= text.length) {
   292→   292→        setDisplayed(text.slice(0, i));
   293→   293→      } else {
   294→   294→        setDone(true);
   295→   295→        clearInterval(interval);
   296→   296→      }
   297→   297→    }, 65);
   298→   298→    return () => clearInterval(interval);
   299→   299→  }, [text]);
   300→   300→
   301→   301→  return (
   302→   302→    <span className={className}>
   303→   303→      {displayed}
   304→   304→      {!done && <span className="typewriter-cursor" aria-hidden="true" />}
   305→   305→    </span>
   306→   306→  );
   307→   307→}

   309→   309→/* ─── Back to top button ─── */
   310→   310→function BackToTop() {
   311→   311→  const [visible, setVisible] = useState(false);
   312→   312→
   313→   313→  useEffect(() => {
   314→   314→    const onScroll = () => {
   315→   315→      const scrollTop = window.scrollY;
   316→   316→      const docHeight = document.documentElement.scrollHeight - window.innerHeight;
   317→   317→      setVisible(docHeight > 0 && scrollTop / docHeight > 0.5);
   318→   318→    };
   319→   319→    onScroll();
   320→   320→    window.addEventListener('scroll', onScroll, { passive: true });
   321→   321→    return () => window.removeEventListener('scroll', onScroll);
   322→   322→  }, []);
   323→   323→
   324→   324→  if (!visible) return null;
   325→   325→
   326→   326→  return (
   327→   327→    <button
   328→   328→      className="back-to-top-btn"
   329→   329→      onClick={() => window.scrollTo({ top: 0, behavior: 'smooth' })}
   330→   330→      aria-label="Volver arriba"
   331→   331→    >
   332→   332→      <ArrowUp className="w-4 h-4" />
   333→   333→    </button>
