    91→function WeatherIcon({ code }: { code: number }) {
    92→  if (code === 0) return <Sun className="w-5 h-5 text-[#F59E0B]" />;
    93→  if (code >= 1 && code <= 3) return <CloudSun className="w-5 h-5 text-[#94A3B8]" />;
    94→  if (code >= 45 && code <= 48) return <CloudFog className="w-5 h-5 text-[#94A3B8]" />;
    95→  if (code >= 51 && code <= 55) return <CloudDrizzle className="w-5 h-5 text-[#3B82F6]" />;
    96→  if (code >= 56 && code <= 57) return <CloudDrizzle className="w-5 h-5 text-[#3B82F6]" />;
    97→  if (code >= 61 && code <= 67) return <CloudRain className="w-5 h-5 text-[#3B82F6]" />;
    98→  if (code >= 71 && code <= 77) return <CloudSnow className="w-5 h-5 text-[#3B82F6]" />;
    99→  if (code >= 80 && code <= 82) return <CloudLightning className="w-5 h-5 text-[#92400E]" />;
   100→  if (code >= 95) return <CloudLightning className="w-5 h-5 text-[#5B21B6]" />;
   101→  return <Thermometer className="w-5 h-5 text-[#94A3B8]" />;
   102→}

   293→function GraficoIngresosEgresos({ pagos, gastos }: { pagos: { fecha: string; monto: number }[]; gastos: { fecha: string; monto: number }[] }) {
   294→  const [mounted, setMounted] = useState(false);
   295→  useEffect(() => { const t = setTimeout(() => setMounted(true), 50); return () => clearTimeout(t); }, []);
   296→
   297→  const datos = useMemo(() => {
   298→    const hoy = new Date();
   299→    const dias: string[] = [];
   300→    for (let i = 6; i >= 0; i--) {
   301→      const d = new Date(hoy);
   302→      d.setDate(d.getDate() - i);
   303→      dias.push(toLocalDateStr(d));
   304→    }
   305→    return dias.map(dia => {
   306→      const ing = pagos.filter(p => p.fecha === dia).reduce((s, p) => s + p.monto, 0);
   307→      const egr = gastos.filter(g => g.fecha === dia).reduce((s, g) => s + g.monto, 0);
   308→      const d = new Date(dia + 'T12:00:00');
   309→      return { dia: d.toLocaleDateString('es-AR', { weekday: 'short', day: 'numeric' }).replace(/^\w/, c => c.toUpperCase()), ing, egr, fullDate: formatearFecha(dia) };
   310→    });
   311→  }, [pagos, gastos]);
   312→
   313→  const maxVal = Math.max(...datos.map(d => Math.max(d.ing, d.egr)), 1);

   315→  return (
   316→    <Card>
   317→      <CardHeader className="pb-3">
   318→        <CardTitle className="text-base flex items-center gap-2">
   319→          <BarChart3 className="w-4 h-4 text-[#3B82F6]" />

   322→      </CardHeader>

   324→        <div className="flex items-end gap-2" style={{ height: 160 }}>

   326→            <div key={i} className="flex-1 flex flex-col items-center h-full">

   331→                    height: mounted ? `${(d.ing / maxVal) * 100}%` : '0%',
   332→                    background: 'linear-gradient(to top, #059669, #34d399)',
   333→                    boxShadow: '0 2px 8px rgba(5,150,105,0.3)',

   424→      estado: res.tipo,
   425→      habitacion: num,

   427→      checkin: res.checkin,

   439→    const result: React.ReactNode[] = [];

   441→    habNumbers.forEach((num, rowIndex) => {
   442→      const hab = habitaciones[num];
   443→      const reservasHab: GanttReserva[] = [];

   445→      reservas.forEach(r => {
   446→        if (r.habitacion !== num || r.estado === 'Cancelada') return;

   665→function calcularBarra(res: GanttReserva, colIdx: Record<string, number>, columnas: string[], DIAS: number) {
   666→  let startCol = Math.max(colIdx[res.checkin] ?? 0, 0);
   667→  let endCol = Math.min(colIdx[res.checkout] !== undefined ? colIdx[res.checkout] : DIAS - 1, DIAS - 1);
   668→
   669→  const esCheckinTarde = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckin && res.horaCheckin.getHours() >= 12;
   670→  const esCheckoutManana = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckout && res.horaCheckout.getHours() < 12;
   671→
   672→  let leftPct: number, widthPct: number;
   673→
   674→  if (esCheckinTarde && startCol >= 0 && !(res.checkin < columnas[0])) {
   675→    leftPct = startCol * COL_PCT + MITAD_COL_PCT;
   676→  } else {
   677→    leftPct = startCol * COL_PCT;
   678→  }
   679→
   680→  if (esCheckoutManana && endCol >= 0 && !(res.checkout > columnas[DIAS - 1])) {
   681→    widthPct = ((endCol - startCol) * COL_PCT) + (esCheckinTarde ? MITAD_COL_PCT : COL_PCT) - (esCheckoutManana ? MITAD_COL_PCT : 0);
   682→  } else {
   683→    widthPct = (endCol - startCol + 1) * COL_PCT;
   684→  }
   685→
   686→  widthPct = Math.max(widthPct, MITAD_COL_PCT);
   687→  return { leftPct, widthPct };
   688→}

   692→function RoomHeatmap({ habitaciones, reservas }: {
   693→  habitaciones: Record<string, { tipo: string; estado: string; capacidad: number; problema?: string }>;
   694→  reservas: { habitacion: string; estado: string; huesped: string; checkin: string; checkout: string }[];
   695→}) {
   696→  const hoyStr = todayLocal();
   697→
   698→  const habInfo = useMemo(() => {
   699→    const map: Record<string, { huesped: string; estado: string }> = {};
   700→    reservas.forEach(r => {
   701→      if ((r.estado === 'Check-In realizado' || r.estado === 'Confirmada') && r.checkin <= hoyStr && r.checkout >= hoyStr) {
   702→        map[r.habitacion] = { huesped: r.huesped, estado: r.estado === 'Check-In realizado' ? 'Ocupada' : 'Reservada' };
   703→      }
   704→    });

   709→    Disponible: 'bg-[#DCFCE7] text-[#166534]',
   710→    Ocupada: 'bg-[#FEE2E2] text-[#991B1B]',
   711→    Limpieza: 'bg-[#FEF3C7] text-[#92400E]',

   713→    Reservada: 'bg-[#E0E7FF] text-[#3730A3]',

   750→          </span>
