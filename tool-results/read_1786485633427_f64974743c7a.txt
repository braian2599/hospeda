   185→// RoomTypeAnalytics — compact analytics section
   186→// ═══════════════════════════════════════════════════════════
   187→
   188→function RoomTypeAnalytics() {
   189→  const habitaciones = useHotelStore(s => s.habitaciones);
   190→  const reservas = useHotelStore(s => s.reservas);
   191→  const today = useMemo(() => todayLocal(), []);
   192→
   193→  const allRooms = Object.values(habitaciones);
   194→  const total = allRooms.length;
   195→
   196→  const typeData = useMemo(() => {
   197→    if (total === 0) return [];
   198→
   199→    const todayDate = safeDate(today).getTime();
   200→
   201→    // Group by type
   202→    const byType = new Map<string, Habitacion[]>();
   203→    allRooms.forEach(h => {
   204→      const t = h.tipo || 'Sin tipo';
   205→      if (!byType.has(t)) byType.set(t, []);
   206→      byType.get(t)!.push(h);
   207→    });
   208→
   209→    return Array.from(byType.entries())
   210→      .sort(([a], [b]) => a.localeCompare(b))
   211→      .map(([tipo, rooms]) => {
   212→        const count = rooms.length;
   213→        const occupied = rooms.filter(h => h.estado === 'Ocupada').length;
   214→        const occupancyRate = Math.round((occupied / count) * 100);
   215→
   216→        // Calculate revenue for occupied rooms
   217→        let revenue = 0;
   218→        rooms.forEach(h => {
   219→          if (h.estado === 'Ocupada' || h.estado === 'Reservada') {
   220→            const res = reservas.find(r =>
   221→              r.habitacion === h.numero &&
   222→              (r.estado === 'Check-In realizado' || r.estado === 'Confirmada') &&
   223→              safeDate(r.checkin).getTime() <= todayDate &&
   224→              safeDate(r.checkout).getTime() >= todayDate
   225→            );
   226→            if (res?.total) revenue += res.total;
   227→          }
   228→        });
   229→
   230→        const Icon = tipoIconMap[tipo] || Bed;
   231→
   232→        return { tipo, count, occupied, occupancyRate, revenue, Icon };
   233→      });
   234→  }, [allRooms, reservas, today, total]);
   235→
   236→  if (total === 0 || typeData.length === 0) return null;
   237→
   238→  const maxCount = Math.max(...typeData.map(d => d.count));
   239→
   240→  // Type color palette
   241→  const typeColors = ['#059669', '#D97706', '#0EA5E9', '#8B5CF6', '#EF4444'];
   242→
   243→  return (
   244→    <div className="rounded-xl border bg-card p-4 space-y-3">
   245→      <div className="flex items-center gap-2">
   246→        <div className="w-7 h-7 rounded-lg bg-[#0F2B28]/10 flex items-center justify-center">
   247→          <Bed className="w-3.5 h-3.5 text-[#0F2B28]" />
   248→        </div>
   249→        <span className="text-sm font-semibold text-foreground">Distribución por tipo</span>
   250→      </div>
   251→
   252→      {/* Horizontal distribution bar */}
   253→      <div className="h-2.5 rounded-full bg-muted overflow-hidden flex">
   254→        {typeData.map((d, i) => {
   255→          const pct = (d.count / total) * 100;
   256→          return (
   257→            <div
   258→              key={d.tipo}
   259→              className="transition-all duration-300"
   260→              style={{ width: `${pct}%`, backgroundColor: typeColors[i % typeColors.length] }}
   261→              title={`${d.tipo}: ${d.count} (${Math.round(pct)}%)`}
   262→            />
   263→          );
   264→        })}
   265→      </div>
   266→
   267→      {/* Type breakdown rows */}
   268→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-5 gap-2">
   269→        {typeData.map((d, i) => {
   270→          const color = typeColors[i % typeColors.length];
   271→          const barPct = (d.count / maxCount) * 100;
   272→          return (
   273→            <div
   274→              key={d.tipo}
   275→              className="rounded-lg border bg-muted/20 p-2.5 space-y-1.5"
   276→            >
   277→              <div className="flex items-center gap-2">
   278→                <d.Icon className="w-3.5 h-3.5 shrink-0" style={{ color }} />
   279→                <span className="text-xs font-semibold text-foreground truncate">{d.tipo}</span>
   280→                <span className="ml-auto text-xs font-bold text-foreground">{d.count}</span>
   281→              </div>
   282→              {/* Mini bar */}
   283→              <div className="h-1.5 rounded-full bg-muted overflow-hidden">
   284→                <div
   285→                  className="h-full rounded-full transition-all duration-300"
   286→                  style={{ width: `${barPct}%`, backgroundColor: color }}
   287→                />
   288→              </div>
   289→              <div className="flex items-center justify-between text-[10px] text-muted-foreground">
   290→                <span>Ocup. {d.occupancyRate}%</span>
   291→                {d.revenue > 0 && <span>${d.revenue.toLocaleString('es-AR')}</span>}
   292→              </div>
   293→            </div>
   294→          );
   295→        })}
   296→      </div>
   297→    </div>
   298→  );
   299→}

   302→// StatusChangePopover — Quick status change for a room
   303→// ═══════════════════════════════════════════════════════════
   304→
   305→function StatusChangePopover({
   306→  numero,
   307→  currentEstado,
   308→  onStatusChange,
   309→  children,
   310→}: {
   311→  numero: string;
   312→  currentEstado: EstadoHabitacion;
   313→  onStatusChange: (numero: string, newEstado: EstadoHabitacion) => void;
   314→  children: React.ReactNode;
   315→}) {
   316→  const [open, setOpen] = useState(false);
   317→  const [confirmDestructive, setConfirmDestructive] = useState<EstadoHabitacion | null>(null);
   318→
   319→  const handleSelect = (newEstado: EstadoHabitacion) => {
   320→    if (newEstado === currentEstado) {
   321→      setOpen(false);
   322→      return;
   323→    }
   324→    // Destructive states need confirmation
   325→    const isDestructive = newEstado === 'Fuera de servicio' || newEstado === 'Mantenimiento';
   326→    if (isDestructive && !confirmDestructive) {
   327→      setConfirmDestructive(newEstado);
   328→      return;
   329→    }
   330→    onStatusChange(numero, newEstado);
   331→    setConfirmDestructive(null);
   332→    setOpen(false);
   333→  };
   334→
   335→  return (
   336→    <Popover open={open} onOpenChange={(v) => { setOpen(v); setConfirmDestructive(null); }}>
   337→      <PopoverTrigger asChild>{children}</PopoverTrigger>
   338→      <PopoverContent className="w-56 p-2" align="start" side="bottom">
   339→        {confirmDestructive ? (
   340→          <div className="space-y-2 p-1">
   341→            <p className="text-xs text-muted-foreground">
   342→              ¿Cambiar a <strong className="text-foreground">{confirmDestructive}</strong>?
   343→              {confirmDestructive === 'Fuera de servicio' && ' La habitación no será asignable.'}
   344→              {confirmDestructive === 'Mantenimiento' && ' Se requiere reparación antes de habilitar.'}
   345→            </p>
   346→            <div className="flex gap-1.5">
   347→              <Button
   348→                size="sm"
   349→                variant="destructive"
   350→                className="h-7 text-xs flex-1"
   351→                onClick={() => handleSelect(confirmDestructive)}
   352→              >
   353→                Confirmar
   354→              </Button>
   355→              <Button
   356→                size="sm"
   357→                variant="ghost"
   358→                className="h-7 text-xs"
   359→                onClick={() => setConfirmDestructive(null)}
   360→              >
   361→                Cancelar
   362→              </Button>
   363→            </div>
   364→          </div>
   365→        ) : (
   366→          <div className="space-y-0.5">
   367→            <div className="text-[10px] uppercase tracking-wider text-muted-foreground font-semibold px-2 pt-1 pb-1.5">
   368→              Cambiar estado
   369→            </div>
   370→            {ALL_ESTADOS.map(estado => {
   371→              const vis = STATUS_VISUAL[estado];
   372→              const Icon = vis.icon;
   373→              const isCurrent = estado === currentEstado;
   374→              return (
   375→                <button
   376→                  key={estado}
   377→                  type="button"
   378→                  onClick={() => handleSelect(estado)}
   379→                  className={`
   380→                    w-full flex items-center gap-2 rounded-md px-2 py-1.5 text-xs
   381→                    transition-colors cursor-pointer
   382→                    ${isCurrent
   383→                      ? 'bg-accent/50 font-semibold text-foreground'
   384→                      : 'text-foreground/80 hover:bg-accent/40'
   385→                    }
   386→                  `}
   387→                >
   388→                  <Icon className="w-3.5 h-3.5 shrink-0" style={{ color: vis.color }} />
   389→                  <span className="truncate">{estado}</span>
   390→                  {isCurrent && (
   391→                    <span className="ml-auto text-[10px] text-muted-foreground">actual</span>
   392→                  )}
   393→                </button>
   394→              );
   395→            })}
   396→          </div>
   397→        )}
   398→      </PopoverContent>
   399→    </Popover>
   400→  );
   401→}

   568→function FloorGroup({
   569→  floorLabel,
   570→  floorNum,
   571→  rooms,
   572→  isFloorPattern,
   573→  getHuespedActual,
   574→  onEdit,
   575→  onDelete,
   576→  onStatusChange,
   577→}: {
   578→  floorLabel: string;
   579→  floorNum: number;
   580→  rooms: Habitacion[];
   581→  isFloorPattern: boolean;
   582→  getHuespedActual: (num: string) => any;
   583→  onEdit: (num: string) => void;
   584→  onDelete: (num: string) => void;
   585→  onStatusChange: (num: string, estado: EstadoHabitacion) => void;
   586→}) {
   587→  const [isOpen, setIsOpen] = useState(true);
   588→
   589→  // Count statuses in this floor
   590→  const statusCounts = useMemo(() => {
   591→    const c: Partial<Record<EstadoHabitacion, number>> = {};
   592→    rooms.forEach(h => { c[h.estado] = (c[h.estado] || 0) + 1; });
   593→    return c;
   594→  }, [rooms]);
   595→
   596→  return (
   597→    <Collapsible open={isOpen} onOpenChange={setIsOpen}>
   598→      <div className="rounded-xl border bg-card overflow-hidden">
   599→        {/* Floor header */}
   600→        <CollapsibleTrigger asChild>
   601→          <button
   602→            type="button"
   603→            className="w-full flex items-center gap-2.5 p-3 hover:bg-accent/30 transition-colors cursor-pointer"
   604→            aria-label={`${isOpen ? 'Colapsar' : 'Expandir'} ${floorLabel}`}
   605→          >
   606→            <div className="flex items-center justify-center w-8 h-8 rounded-lg bg-[#0F2B28]/10 text-[#0F2B28] shrink-0">
   607→              {isFloorPattern ? (
   608→                <span className="text-sm font-bold">{floorNum}</span>
   609→              ) : (
   610→                <Bed className="w-4 h-4" />
   611→              )}
   612→            </div>
   613→            <div className="text-left min-w-0">
   614→              <span className="text-sm font-semibold text-foreground">{floorLabel}</span>
   615→              <span className="text-xs text-muted-foreground ml-2">
   616→                {rooms.length} {rooms.length === 1 ? 'habitación' : 'habitaciones'}
   617→              </span>
   618→            </div>
   619→            {/* Mini status indicators */}
   620→            <div className="ml-auto flex items-center gap-1.5">
   621→              {ALL_ESTADOS.map(estado => {
   622→                const count = statusCounts[estado] || 0;
   623→                if (count === 0) return null;
   624→                const vis = STATUS_VISUAL[estado];
   625→                return (
   626→                  <div key={estado} className="flex items-center gap-0.5" title={`${estado}: ${count}`}>
   627→                    <span
   628→                      className="w-2 h-2 rounded-sm"
   629→                      style={{ backgroundColor: vis.color }}
   630→                    />
   631→                    <span className="text-[10px] text-muted-foreground">{count}</span>
   632→                  </div>
   633→                );
   634→              })}
   635→            </div>
   636→            <div className="shrink-0 ml-1">
   637→              {isOpen ? (
   638→                <ChevronDown className="w-4 h-4 text-muted-foreground" />
   639→              ) : (
   640→                <ChevronRight className="w-4 h-4 text-muted-foreground" />
   641→              )}
   642→            </div>
   643→          </button>
   644→        </CollapsibleTrigger>
   645→
   646→        {/* Room grid */}
   647→        <CollapsibleContent>
   648→          <div className="p-3 pt-0">
   649→            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6 gap-3 card-grid-stagger">
   650→              {rooms.map(hab => {
   651→                const huesped = (hab.estado === 'Ocupada' || hab.estado === 'Reservada')
   652→                  ? getHuespedActual(hab.numero)
   653→                  : null;
   654→                return (
   655→                  <EnhancedRoomCard
   656→                    key={hab.numero}
   657→                    hab={hab}
   658→                    huesped={huesped}
   659→                    onEdit={onEdit}
   660→                    onDelete={onDelete}
   661→                    onStatusChange={onStatusChange}
   662→                  />
   663→                );
   664→              })}
   665→            </div>
   666→          </div>
   667→        </CollapsibleContent>
   668→      </div>
   669→    </Collapsible>
   670→  );
   671→}

   674→// HabitacionesModule — Main component
   675→// ═══════════════════════════════════════════════════════════

   677→export default function HabitacionesModule() {
   678→  // Granular selectors — avoids re-rendering on unrelated store changes
   679→  const habitaciones = useHotelStore(s => s.habitaciones);
   680→  const reservas = useHotelStore(s => s.reservas);
   681→  const agregarHabitacion = useHotelStore(s => s.agregarHabitacion);
   682→  const editarHabitacion = useHotelStore(s => s.editarHabitacion);
   683→  const eliminarHabitacion = useHotelStore(s => s.eliminarHabitacion);
   684→  const cambiarEstadoHabitacion = useHotelStore(s => s.cambiarEstadoHabitacion);
   685→  const [modal, setModal] = useState<'nueva' | 'editar' | 'eliminar' | null>(null);
   686→  const [sel, setSel] = useState<string>('');
   687→  const [viewMode, setViewMode] = useState<'lista' | 'mapa'>('lista');
   688→  const [form, setForm] = useState({
   689→    numero: '',
   690→    tipo: 'Doble' as TipoHabitacion,
   691→    capacidad: '2',
   692→    camasMatrimoniales: '0',
   693→    camasSimples: '0',
   694→  });
   695→
   696→  const esCompartida = form.tipo === 'Compartida';
   697→  const today = useMemo(() => todayLocal(), []);
   698→  const sorted = useMemo(
   699→    () => Object.entries(habitaciones).sort(([a], [b]) => a.localeCompare(b)),
   700→    [habitaciones]
   701→  );
   702→
   703→  // Find the active guest for a room
   704→  const getHuespedActual = useCallback((num: string) => {
   705→    const todayDate = safeDate(today).getTime();
   706→    return reservas.find(r =>
   707→      r.habitacion === num &&
   708→      (r.estado === 'Check-In realizado' || r.estado === 'Confirmada') &&
   709→      safeDate(r.checkin).getTime() <= todayDate &&
   710→      safeDate(r.checkout).getTime() >= todayDate
   711→    );
   712→  }, [reservas, today]);
   713→
   714→  // Detect if room numbers follow a floor pattern (e.g., 1xx = floor 1)
   715→  const isFloorPattern = useMemo(() => {
   716→    if (sorted.length === 0) return false;
   717→    const floorSet = new Set<number>();
   718→    sorted.forEach(([num]) => floorSet.add(extractFloor(num)));
   719→    return floorSet.size > 1; // Multiple floors detected → use floor pattern
   720→  }, [sorted]);
   721→
   722→  // Group rooms by floor or type
   723→  const groupedRooms = useMemo(() => {
   724→    if (isFloorPattern) {
   725→      // Group by floor
   726→      const map = new Map<number, Habitacion[]>();
   727→      sorted.forEach(([, hab]) => {
   728→        const floor = extractFloor(hab.numero);
   729→        if (!map.has(floor)) map.set(floor, []);
   730→        map.get(floor)!.push(hab);
   731→      });
   732→      return Array.from(map.entries())
   733→        .sort(([a], [b]) => a - b)
   734→        .map(([floorNum, rooms]) => ({
   735→          key: `floor-${floorNum}`,
   736→          label: floorNum === 0 ? 'Sin piso' : `Piso ${floorNum}`,
   737→          floorNum,
   738→          rooms,
   739→        }));
   740→    }
   741→    // Group by type
   742→    const map = new Map<string, Habitacion[]>();
   743→    sorted.forEach(([, hab]) => {
   744→      const tipo = hab.tipo || 'Sin tipo';
   745→      if (!map.has(tipo)) map.set(tipo, []);
   746→      map.get(tipo)!.push(hab);
   747→    });
   748→    return Array.from(map.entries())
   749→      .sort(([a], [b]) => a.localeCompare(b))
   750→      .map(([tipo, rooms], idx) => ({
   751→        key: `type-${tipo}`,
   752→        label: tipo,
   753→        floorNum: idx + 1,
   754→        rooms,
   755→      }));
   756→  }, [sorted, isFloorPattern]);
   757→
   758→  // ── Helpers ──
   759→  const capacidadDesdeTipo = (tipo: TipoHabitacion) => {
   760→    const cap = CAPACIDAD_POR_TIPO[tipo];
   761→    return cap !== null ? String(cap) : '1';
   762→  };
   763→
   764→  const handleTipoChange = (tipo: TipoHabitacion) => {
   765→    setForm(prev => ({
   766→      ...prev,
   767→      tipo,
   768→      capacidad: CAPACIDAD_POR_TIPO[tipo] !== null ? String(CAPACIDAD_POR_TIPO[tipo]) : prev.capacidad,
   769→    }));
   770→  };
   771→
   772→  const getCapacidadFinal = (): number => {
   773→    if (esCompartida) return parseInt(form.capacidad) || 1;

   781→  };
   782→
   783→  const openEdit = (num: string) => {
   784→    const h = habitaciones[num];
   785→    if (!h) return;
   786→    const tipo = h.tipo as TipoHabitacion;
   787→    setSel(num);
   788→    setForm({
   789→      numero: num,
   790→      tipo,
   791→      capacidad: CAPACIDAD_POR_TIPO[tipo] !== null ? String(CAPACIDAD_POR_TIPO[tipo]) : String(h.capacidad),
   792→      camasMatrimoniales: String(h.camasMatrimoniales),
   793→      camasSimples: String(h.camasSimples),
   794→    });
   795→    setModal('editar');
   796→  };
   797→
   798→  const openDelete = (num: string) => { setSel(num); setModal('eliminar'); };
   799→
   800→  // ── Status change handler ──
   801→  const handleStatusChange = async (numero: string, nuevoEstado: EstadoHabitacion) => {
   802→    const ok = await cambiarEstadoHabitacion(numero, nuevoEstado);
   803→    if (ok) {
   804→      toast.success(`Hab. ${numero} → ${nuevoEstado}`);
   805→    } else {
   806→      toast.error('No se pudo cambiar el estado');
   807→    }
   808→  };
   809→

   816→    if (modal === 'nueva') {
   817→      ok = await agregarHabitacion(form.numero.trim(), form.tipo, capacidad, camasM, camasS);
   818→    } else if (modal === 'editar') {
   819→      ok = await editarHabitacion(sel, form.numero.trim(), form.tipo, capacidad, camasM, camasS);

   821→    if (ok) {
   822→      toast.success('Habitación guardada');

   825→      toast.error('No se pudo guardar');

   832→      toast.success('Habitación eliminada');

   835→      toast.error('No se pudo eliminar');
   836→    }

   838→
   839→  return (

   841→      <ModuleHeader icon={Bed} title="Habitaciones" subtitle="Gestioná las habitaciones de tu hotel">
   842→        <div className="flex items-center gap-2">
   843→          {/* ── View toggle ── */}

   850→                transition-all duration-200

   852→                  ? 'bg-card text-foreground shadow-sm'
   853→                  : 'text-muted-foreground hover:text-foreground hover:bg-accent/50'

   866→                transition-all duration-200

   873→            >
   874→              <LayoutGrid className="w-3.5 h-3.5" />
