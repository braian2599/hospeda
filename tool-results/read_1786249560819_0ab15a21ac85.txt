    17→/* ─── Intersection Observer hook ─── */
    18→function useInView(threshold = 0.12) {
    19→  const ref = useRef<HTMLDivElement>(null);
    20→  const [inView, setInView] = useState(false);
    21→  useEffect(() => {
    22→    const el = ref.current;
    23→    if (!el) return;
    24→    const obs = new IntersectionObserver(
    25→      ([e]) => { if (e.isIntersecting) { setInView(true); obs.disconnect(); } },
    26→      { threshold }
    27→    );
    28→    obs.observe(el);
    29→    return () => obs.disconnect();
    30→  }, [threshold]);
    31→  return { ref, inView };
    32→}

   169→/* ─── Fade-in wrapper ─── */
   170→function FadeIn({ children, className = '', delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   171→  const { ref, inView } = useInView();
   172→  return (
   173→    <div
   174→      ref={ref}
   175→      className={`transition-all duration-700 ease-out ${className}`}
   176→      style={{
   177→        opacity: inView ? 1 : 0,
   178→        transform: inView ? 'translateY(0)' : 'translateY(32px)',
   179→        transitionDelay: `${delay}ms`,
   180→      }}
   181→    >
   182→      {children}
   183→    </div>
   184→  );
   185→}
