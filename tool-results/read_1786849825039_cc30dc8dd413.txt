    78→interface KpiProps {
    79→  label: string;
    80→  value: string | number;
    81→  icon: React.ReactNode;
    82→  colorFamily: string;
    83→  tooltip?: string;
    84→  subtext?: string;
    85→  trend?: { value: number; label: string };

    88→
    89→function KpiCard({ label, value, icon, colorFamily, tooltip, subtext, trend, onClick }: KpiProps) {
    90→  const c = KPI_COLORS[colorFamily] || KPI_COLORS.primary;
    91→  const card = (
    92→    <div
    93→      className={`relative overflow-hidden rounded-xl border-l-[3px] ${c.borderL} ${c.bg} ${c.darkBg} p-4 shadow-sm hover:shadow-md hover:-translate-y-0.5 transition-all duration-200 card-interactive ${onClick ? 'cursor-pointer group' : ''}`}
    94→      onClick={onClick}
    95→    >
    96→      <div className="flex items-start justify-between">
    97→        <div className="space-y-1">
    98→          <p className={`text-xs font-medium ${c.label}`}>{label}</p>
    99→          <p className={`text-xl font-bold ${c.value} truncate`}>{value}</p>
   100→          {trend && (
   101→            <div className="flex items-center gap-1 mt-1">
   102→              {trend.value > 0 ? (
   103→                <ArrowUpRight className="w-3 h-3 text-primary" />
   104→              ) : trend.value < 0 ? (
   105→                <ArrowDownRight className="w-3 h-3 text-[#EF4444]" />
   106→              ) : (
   107→                <Minus className="w-3 h-3 text-muted-foreground" />
   108→              )}
   109→              <span className={`text-[11px] font-medium ${trend.value > 0 ? 'text-primary' : trend.value < 0 ? 'text-[#EF4444]' : 'text-muted-foreground'}`}>
   110→                {Math.abs(trend.value)}%
   111→              </span>
   112→              <span className="text-[11px] text-muted-foreground hidden sm:inline">{trend.label}</span>
   113→            </div>
   114→          )}
   115→          {subtext && !trend && <p className={`text-[10px] ${c.sub} mt-1 truncate`}>{subtext}</p>}
   116→        </div>
   117→        <div className={`w-10 h-10 rounded-full ${c.iconBg} ${c.iconColor} flex items-center justify-center shrink-0`}>
   118→          {icon}
   119→        </div>
   120→      </div>
   121→      {onClick && (
   122→        <div className="absolute inset-x-0 bottom-0 h-0.5 bg-primary opacity-0 group-hover:opacity-100 transition-opacity" />
   123→      )}
   124→    </div>
   125→  );
   126→
   127→  if (tooltip) {
   128→    return (
   129→      <TooltipProvider delayDuration={300}>
   130→        <Tooltip>
   131→          <TooltipTrigger asChild>{card}</TooltipTrigger>
   132→          <TooltipContent side="bottom" className="text-xs max-w-[220px]">{tooltip}</TooltipContent>
   133→        </Tooltip>
   134→      </TooltipProvider>
   135→    );
   136→  }
   137→  return card;
   138→}
