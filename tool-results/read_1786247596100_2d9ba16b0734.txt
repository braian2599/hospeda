    33→function escapeCSV(val: string | number | undefined): string {
    34→  const str = val == null ? '' : String(val);
    35→  if (str.includes(',') || str.includes('"') || str.includes('\n')) {
    36→    return `"${str.replace(/"/g, '""')}"`;
    37→  }
    38→  return str;
    39→}
    40→
    41→function downloadCSV(filename: string, headers: string[], rows: (string | number | undefined)[][]) {
    42→  const lines = [
    43→    headers.map(escapeCSV).join(','),
    44→    ...rows.map(r => r.map(escapeCSV).join(',')),
    45→  ];
    46→  const blob = new Blob([lines.join('\n')], { type: 'text/csv;charset=utf-8;' });
    47→  const url = URL.createObjectURL(blob);
    48→  const a = document.createElement('a');
    49→  a.href = url;
    50→  a.download = filename;
    51→  a.click();
    52→  URL.revokeObjectURL(url);
    53→}

    80→interface KpiProps {
    81→  label: string;
    82→  value: string | number;
    83→  icon: React.ReactNode;
    84→  color: string;
    85→  bgColor: string;
    86→  tooltip?: string;

    88→  trend?: { value: number; label: string };
    89→  onClick?: () => void;
    90→}
    91→
    92→function KpiCard({ label, value, icon, color, bgColor, tooltip, subtext, trend, onClick }: KpiProps) {
    93→  const card = (
    94→    <Card
    95→      className={`relative overflow-hidden transition-all duration-200 hover:shadow-md ${onClick ? 'cursor-pointer group' : ''}`}
    96→      onClick={onClick}
    97→    >
    98→      <CardContent className="p-3 sm:p-4">
    99→        <div className="flex items-start justify-between gap-2 sm:gap-3">
   100→          <div className="flex-1 min-w-0 text-center">
   101→            <p className="text-[10px] sm:text-[11px] font-medium uppercase tracking-wider text-muted-foreground mb-1 sm:mb-1.5">{label}</p>
   102→            <p className={`text-lg sm:text-2xl font-bold tracking-tight ${color} truncate`}>{value}</p>
   103→            {trend && (
   104→              <div className="flex items-center justify-center gap-1 mt-1 sm:mt-1.5">
   105→                {trend.value > 0 ? (
   106→                  <ArrowUpRight className="w-3 h-3 sm:w-3.5 sm:h-3.5 text-[#059669]" />
   107→                ) : trend.value < 0 ? (
   108→                  <ArrowDownRight className="w-3 h-3 sm:w-3.5 sm:h-3.5 text-[#EF4444]" />
   109→                ) : (
   110→                  <Minus className="w-3 h-3 sm:w-3.5 sm:h-3.5 text-muted-foreground" />
   111→                )}
   112→                <span className={`text-[11px] sm:text-xs font-medium ${trend.value > 0 ? 'text-[#059669]' : trend.value < 0 ? 'text-[#EF4444]' : 'text-muted-foreground'}`}>
   113→                  {Math.abs(trend.value)}%
   114→                </span>
   115→                <span className="text-[11px] sm:text-xs text-muted-foreground hidden sm:inline">{trend.label}</span>
   116→              </div>
   117→            )}
   118→            {subtext && !trend && <p className="text-[11px] sm:text-xs text-muted-foreground mt-1 truncate hidden sm:block">{subtext}</p>}
   119→          </div>
   120→          <div className={`shrink-0 w-8 h-8 sm:w-10 sm:h-10 rounded-lg sm:rounded-xl flex items-center justify-center ${bgColor}`}>
   121→            <div className="sm:scale-100 scale-90">{icon}</div>
   122→          </div>
   123→        </div>
   124→      </CardContent>
   125→      {onClick && (
   126→        <div className="absolute inset-x-0 bottom-0 h-0.5 bg-primary opacity-0 group-hover:opacity-100 transition-opacity" />
   127→      )}
   128→    </Card>
   129→  );
   130→
   131→  if (tooltip) {
   132→    return (
   133→      <TooltipProvider delayDuration={300}>
   134→        <Tooltip>
   135→          <TooltipTrigger asChild>{card}</TooltipTrigger>
   136→          <TooltipContent side="bottom" className="text-xs max-w-[220px]">{tooltip}</TooltipContent>
   137→        </Tooltip>
   138→      </TooltipProvider>
   139→    );
   140→  }
   141→  return card;
   142→}

   156→function ProgressKpi({ label, value, max, color = 'bg-primary', suffix = '%', description }: { label: string; value: number; max: number; color?: string; suffix?: string; description?: string }) {
   157→  const pct = max > 0 ? Math.round((value / max) * 100) : 0;
   158→  return (
   159→    <Card>
   160→      <CardContent className="p-3 sm:p-4">
   161→        <div className="flex items-center justify-between mb-1.5 sm:mb-2">
   162→          <p className="text-[10px] sm:text-[11px] font-medium uppercase tracking-wider text-muted-foreground">{label}</p>
   163→          <span className="text-base sm:text-lg font-bold">{value}{suffix}</span>
   164→        </div>
   165→        <Progress value={pct} className="h-2 sm:h-2.5" />
   166→        <p className="text-[11px] sm:text-xs text-muted-foreground mt-1 sm:mt-1.5 text-center">{description || `${pct}%`}</p>
   167→      </CardContent>
   168→    </Card>
   169→  );
   170→}
