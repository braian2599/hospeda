   100→   100→function DateRangePickerInline({
   101→   101→  checkin,
   102→   102→  checkout,
   103→   103→  onChangeCheckin,
   104→   104→  onChangeCheckout,
   105→   105→  label,
   106→   106→  disabled,
   107→   107→}: {
   108→   108→  checkin: string;
   109→   109→  checkout: string;
   110→   110→  onChangeCheckin: (val: string) => void;
   111→   111→  onChangeCheckout: (val: string) => void;
   112→   112→  label?: string;
   113→   113→  disabled?: boolean;
   114→   114→}) {
   115→   115→  const [open, setOpen] = useState(false);
   116→   116→  const [localRange, setLocalRange] = useState<{ from?: Date; to?: Date }>({});
   117→   117→  const syncedRef = useRef(false);
   118→   118→
   119→   119→  // Al abrir el popover, sincronizar el estado local con el form.
   120→   120→  // Al cerrar, resetear el flag para la próxima vez.
   121→   121→  useEffect(() => {
   122→   122→    if (open && !syncedRef.current) {
   123→   123→      syncedRef.current = true;
   124→   124→      const f = checkin ? parseISO(checkin) : undefined;
   125→   125→      const t = checkout ? parseISO(checkout) : undefined;
   126→   126→      setLocalRange(f ? { from: f, to: t } : {});
   127→   127→    }
   128→   128→    if (!open) syncedRef.current = false;
   129→   129→  }, [open, checkin, checkout]);
   130→   130→
   131→   131→  const handleSelect = (range: { from?: Date; to?: Date } | undefined) => {
   132→   132→    if (!range?.from) {
   133→   133→      setLocalRange({});
   134→   134→      return;
   135→   135→    }
   136→   136→    setLocalRange(range);
   137→   137→  };
   138→   138→
   139→   139→  const fmt = (d: Date) => {
   140→   140→    const yyyy = d.getFullYear();
   141→   141→    const mm = String(d.getMonth() + 1).padStart(2, '0');
   142→   142→    const dd = String(d.getDate()).padStart(2, '0');
   143→   143→    return `${yyyy}-${mm}-${dd}`;
   144→   144→  };
   145→   145→
   146→   146→  const handleConfirm = () => {
   147→   147→    if (!localRange.from) return;
   148→   148→    onChangeCheckin(fmt(localRange.from));
   149→   149→    onChangeCheckout(localRange.to ? fmt(localRange.to) : '');
   150→   150→    setOpen(false);
   151→   151→  };
   152→   152→
   153→   153→  const from = checkin ? parseISO(checkin) : undefined;
   154→   154→  const to = checkout ? parseISO(checkout) : undefined;
   155→   155→
   156→   156→  const displayText = from && to
   157→   157→    ? `${format(from, 'dd/MM/yyyy')} → ${format(to, 'dd/MM/yyyy')}`
   158→   158→    : from
   159→   159→      ? `${format(from, 'dd/MM/yyyy')} → ...`
   160→   160→      : 'Seleccionar fechas';
   161→   161→
   162→   162→  return (
   163→   163→    <div className="grid gap-1.5">
   164→   164→      {label && <Label>{label} *</Label>}
   165→   165→      <Button
   166→   166→        variant="outline"
   167→   167→        className={cn(
   168→   168→          'w-full justify-start text-left font-normal h-9',
   169→   169→          !checkin && 'text-muted-foreground'
   170→   170→        )}
   171→   171→        disabled={disabled}
   172→   172→        onClick={() => setOpen(true)}
   173→   173→      >
   174→   174→        <CalendarDays className="w-4 h-4 mr-2 text-muted-foreground" />
   175→   175→        {displayText}
   176→   176→      </Button>
   177→   177→      {open
   178→   178→        ? createPortal(
   179→   179→            <div className="fixed inset-0 z-[200] flex items-center justify-center bg-black/40" onClick={() => setOpen(false)}>
   180→   180→              <div
   181→   181→                className="bg-white rounded-xl shadow-2xl border border-[#E2E8F0] pointer-events-auto"
   182→   182→                onClick={e => e.stopPropagation()}
   183→   183→                onMouseDown={e => e.stopPropagation()}
   184→   184→              >
   185→   185→                <Calendar
   186→   186→                  mode="range"
   187→   187→                  selected={localRange.from ? { from: localRange.from, to: localRange.to } as { from: Date; to: Date } : undefined}
   188→   188→                  onSelect={handleSelect}
   189→   189→                  numberOfMonths={2}
   190→   190→                  autoFocus
   191→   191→                  classNames={{
   192→   192→                    range_start: 'rounded-l-md bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   193→   193→                    range_end: 'rounded-r-md bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   194→   194→                    range_middle: 'rounded-none bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   195→   195→                    day: 'relative w-full h-full p-0 text-center [&:first-child[data-selected=true]_button]:rounded-l-md [&:last-child[data-selected=true]_button]:rounded-r-md group/day aspect-square select-none',
   196→   196→                  }}
   197→   197→                />
   198→   198→                <div className="flex justify-end border-t border-[#E2E8F0] p-3">
   199→   199→                  <Button
   200→   200→                    size="sm"
   201→   201→                    onClick={handleConfirm}
   202→   202→                    disabled={!localRange.from}
   203→   203→                    className="bg-[#059669] hover:bg-[#047857] text-white text-xs h-8"
   204→   204→                  >
   205→   205→                    Confirmar fechas
   206→   206→                  </Button>
   207→   207→                </div>
   208→   208→              </div>
   209→   209→            </div>,
   210→   210→            document.body
   211→   211→          )
   212→   212→        : null}
   213→   213→    </div>
   214→   214→  );
   215→   215→}

   246→   246→function DesglosePrecio({ form, computed, formatMoney, s }: {
   247→   247→  form: { tipoTarifa: string; reservaMultiple: boolean; habitacion: string; habitacion2?: string; personas: string; personas2?: string };
   248→   248→  computed: { noches: number; subtotal: number; subtotal2: number; desglose: any };
   249→   249→  formatMoney: (n: number) => string;
   250→   250→  s: (n: number) => string;
   251→   251→}) {
   252→   252→  const d = computed.desglose;
   253→   253→
   254→   254→  const adultoLinea = (d: any) => {
   255→   255→    if (!d || d.totalAdultos === 0) return null;
   256→   256→    const modo = d.modoCobro;
   257→   257→    if (modo === 'porCama') {
   258→   258→      return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''} × ${formatMoney(d.precioRango)}/cama/noche`;
   259→   259→    }
   260→   260→    if (modo === 'porHabitacion') return 'Habitación';
   261→   261→    if (modo === 'porPersona') {
   262→   262→      return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''} × ${formatMoney(d.precioRango)}/noche`;
   263→   263→    }
   264→   264→    // porGrupo
   265→   265→    return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''}`;
   266→   266→  };
   267→   267→
   268→   268→  const renderLines = (sub: number, dLocal: any) => {
   269→   269→    if (!dLocal) return (
   270→   270→      <div className="space-y-1">
   271→   271→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Subtotal</span><span className="font-semibold text-[#1E293B]">{formatMoney(sub)}</span></div>
   272→   272→      </div>
   273→   273→    );
   274→   274→
   275→   275→    const lines: { key: string; label: string; value: string; dim?: boolean }[] = [];
   276→   276→
   277→   277→    const aLabel = adultoLinea(dLocal);
   278→   278→    if (aLabel) {
   279→   279→      lines.push({ key: 'adultos', label: aLabel, value: formatMoney(dLocal.totalAdultos) });
   280→   280→    }
   281→   281→
   282→   282→    if (dLocal.ninosCount > 0 && dLocal.totalNinos > 0) {
   283→   283→      lines.push({
   284→   284→        key: 'ninos',
   285→   285→        label: `${dLocal.ninosCount} niño${dLocal.ninosCount > 1 ? 's' : ''} × ${formatMoney(dLocal.precioNino)}/noche`,
   286→   286→        value: formatMoney(dLocal.totalNinos),
   287→   287→      });
   288→   288→    }
   289→   289→
   290→   290→    if (dLocal.nochesGratis > 0) {
   291→   291→      const totalBase = dLocal.totalAdultos + dLocal.totalNinos;
   292→   292→      const ahorro = totalBase > 0 && dLocal.nochesCobrables > 0
   293→   293→        ? Math.round(totalBase / dLocal.nochesCobrables * dLocal.nochesGratis)
   294→   294→        : 0;
   295→   295→      lines.push({
   296→   296→        key: 'cortesia',
   297→   297→        label: `${dLocal.nochesGratis} noche${dLocal.nochesGratis > 1 ? 's' : ''} de cortesía`,
   298→   298→        value: `- ${formatMoney(ahorro)}`,
   299→   299→        dim: true,
   300→   300→      });
   301→   301→    }
   302→   302→
   303→   303→    return (
   304→   304→      <div className="space-y-1">
   305→   305→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Tarifa</span><span className="font-semibold text-[#1E293B] capitalize">{form.tipoTarifa}</span></div>
   306→   306→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Noches</span><span className="font-semibold text-[#1E293B]">{computed.noches}{dLocal?.nochesGratis > 0 ? ` (${dLocal.nochesCobrables} cobrables)` : ''}</span></div>
   307→   307→        {lines.map(l => (
   308→   308→          <div key={l.key} className="flex justify-between items-center py-1 text-[13px]">
   309→   309→            <span className={l.dim ? 'text-[#94A3B8] line-through' : 'text-[#64748B]'}>{l.label}</span>
   310→   310→            <span className={l.dim ? 'font-medium text-[#94A3B8]' : 'font-semibold text-[#1E293B]'}>{l.value}</span>
   311→   311→          </div>
   312→   312→        ))}
   313→   313→        <div className="flex justify-between items-center py-1.5 mt-1 border-t border-[#E2E8F0] text-[13px]">
   314→   314→          <span className="font-medium text-[#475569]">Subtotal</span>
   315→   315→          <span className="font-bold text-[#1E293B]">{formatMoney(sub)}</span>
   316→   316→        </div>
   317→   317→      </div>
   318→   318→    );
   319→   319→  };

   321→   321→  return (

   335→   335→        renderLines(computed.subtotal, d)
