     1→'use client';
     2→
     3→import { useState } from 'react';
     4→import { useHotelStore } from '@/lib/store';
     5→import type { Reserva, Acompanante, Menor } from '@/lib/types';
     6→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
     7→import { Button } from '@/components/ui/button';
     8→import { Input } from '@/components/ui/input';
     9→import { Label } from '@/components/ui/label';
    10→import { Badge } from '@/components/ui/badge';
    11→import { Checkbox } from '@/components/ui/checkbox';
    12→import { Textarea } from '@/components/ui/textarea';
    13→import { Separator } from '@/components/ui/separator';
    14→import {
    15→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter, DialogClose,
    16→} from '@/components/ui/dialog';
    17→import {
    18→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    19→} from '@/components/ui/select';
    20→import {
    21→  LogIn, LogOut, KeyRound, UserPlus, Trash2, Users, AlertCircle, CreditCard, BedDouble, Baby,
    22→} from 'lucide-react';
    23→import ModuleHeader from '@/components/layout/ModuleHeader';
    24→import { toast } from 'sonner';
    25→import { notifySuccess } from '@/lib/notify';
    26→
    27→const formatFecha = (f: string) => {
    28→  if (!f) return '—';
    29→  const d = new Date(f + 'T12:00:00');
    30→  return d.toLocaleDateString('es-AR');
    31→};
    32→
    33→const formatMoney = (n: number) =>
    34→  new Intl.NumberFormat('es-AR', { style: 'currency', currency: 'ARS', minimumFractionDigits: 0 }).format(n);
    35→
    36→const estadoPagoBadge: Record<string, string> = {
    37→  Pendiente: 'bg-[#FEF3C7] text-[#92400E] border-[#FDE68A]',
    38→  Parcial: 'bg-[#FFEDD5] text-[#9A3412] border-[#FED7AA]',
    39→  Pagado: 'bg-[#DCFCE7] text-[#166534] border-[#BBF7D0]',
    40→};
    41→
    42→const PARENTESCO_OPTIONS = [
    43→  'Hijo/a', 'Hijo/astra', 'Nieto/a', 'Sobrino/a', 'Primo/a', 'Otro',
    44→];
    45→
    46→interface MenorForm {
    47→  nombre: string;
    48→  documento: string;
    49→  edad: string;
    50→  parentesco: string;
    51→}
    52→
    53→function emptyMenorForm(): MenorForm {
    54→  return { nombre: '', documento: '', edad: '', parentesco: 'Hijo/a' };
    55→}
    56→
    57→export default function CheckInModule() {
    58→  const {
    59→    reservas, habitaciones, realizarCheckIn, realizarCheckOut,
    60→    calcularTotalReserva, calcularTotalPagado, nochesEntre,
    61→  } = useHotelStore();
    62→
    63→  const pendientesCheckIn = reservas.filter(r => r.estado === 'Confirmada');
    64→  const pendientesCheckOut = reservas.filter(r => r.estado === 'Check-In realizado');
    65→
    66→  // Modal state
    67→  const [modalTipo, setModalTipo] = useState<'checkin' | 'checkout' | null>(null);
    68→  const [selReserva, setSelReserva] = useState<Reserva | null>(null);
    69→  const [dialogKey, setDialogKey] = useState(0);
    70→
    71→  // Check-in form
    72→  const [docVerificado, setDocVerificado] = useState(false);
    73→  const [firmaConformidad, setFirmaConformidad] = useState(false);
    74→  const [llave, setLlave] = useState('');
    75→  const [emergenciaNombre, setEmergenciaNombre] = useState('');
    76→  const [emergenciaTelefono, setEmergenciaTelefono] = useState('');
    77→  const [observaciones, setObservaciones] = useState('');
    78→  const [acompanantes, setAcompanantes] = useState<Acompanante[]>([
    79→    { nombre: '', dni: '', celular: '' },
    80→  ]);
    81→
    82→  // Menores form state
    83→  const [menoresForms, setMenoresForms] = useState<MenorForm[]>([]);
    84→  const [menoresErrors, setMenoresErrors] = useState<string[]>([]);
    85→
    86→  const openCheckIn = (r: Reserva) => {
    87→    setSelReserva(r);
    88→    setDocVerificado(false);
    89→    setFirmaConformidad(false);
    90→    setLlave('');
    91→    setEmergenciaNombre(r.contactoEmergencia?.nombre || '');
    92→    setEmergenciaTelefono(r.contactoEmergencia?.telefono || '');
    93→    setObservaciones(r.observacionesHuesped || '');
    94→    setAcompanantes(r.acompanantes?.length ? [...r.acompanantes] : [{ nombre: '', dni: '', celular: '' }]);
    95→
    96→    // Si ya tiene menores registrados (check-in previo), precargar
    97→    if (r.menores && r.menores.length > 0) {
    98→      setMenoresForms(r.menores.map(m => ({
    99→        nombre: m.nombre,
   100→        documento: m.documento,
   101→        edad: String(m.edad),
   102→        parentesco: m.parentesco,
   103→      })));
   104→    } else if ((r.ninos || 0) > 0) {
   105→      // Generar formularios vacíos según la cantidad de menores declarados
   106→      const forms: MenorForm[] = [];
   107→      for (let i = 0; i < r.ninos!; i++) {
   108→        forms.push(emptyMenorForm());
   109→      }
   110→      setMenoresForms(forms);
   111→    } else {
   112→      setMenoresForms([]);
   113→    }
   114→    setMenoresErrors([]);
   115→
   116→    setModalTipo('checkin');
   117→  };
   118→
   119→  const openCheckOut = (r: Reserva) => {
   120→    setSelReserva(r);
   121→    setModalTipo('checkout');
   122→  };
   123→
   124→  const closeModal = () => {
   125→    setModalTipo(null);
   126→    setSelReserva(null);
   127→    setDialogKey(prev => prev + 1);
   128→  };
   129→
   130→  const validateMenores = (): boolean => {
   131→    const errs: string[] = [];
   132→    for (let i = 0; i < menoresForms.length; i++) {
   133→      const m = menoresForms[i];
   134→      if (!m.nombre.trim()) errs.push(`Menor ${i + 1}: nombre obligatorio`);
   135→      if (!m.documento.trim()) errs.push(`Menor ${i + 1}: documento obligatorio`);
   136→      const edad = parseInt(m.edad);
   137→      if (isNaN(edad) || edad < 0 || edad > 17) errs.push(`Menor ${i + 1}: edad debe ser entre 0 y 17`);
   138→      if (!m.parentesco.trim()) errs.push(`Menor ${i + 1}: parentesco obligatorio`);
   139→    }
   140→    setMenoresErrors(errs);
   141→    return errs.length === 0;
   142→  };
   143→
   144→  const [checkLoading, setCheckLoading] = useState(false);
   145→
   146→  const handleConfirmCheckIn = async () => {
   147→    if (!selReserva || !llave.trim()) return;
   148→
   149→    // Validar menores si la reserva tiene ninos > 0
   150→    const cantNinos = selReserva.ninos || 0;
   151→    if (cantNinos > 0) {
   152→      if (menoresForms.length !== cantNinos) {
   153→        toast.error(`Se requieren datos de ${cantNinos} menor${cantNinos > 1 ? 'es' : ''}`);
   154→        return;
   155→      }
   156→      if (!validateMenores()) {
   157→        toast.error('Completá todos los datos obligatorios de los menores');
   158→        return;
   159→      }
   160→    }
   161→
   162→    const datos: Parameters<typeof realizarCheckIn>[1] = {};
   163→    if (docVerificado) datos.documentoVerificado = true;
   164→    if (firmaConformidad) datos.firmaConformidad = true;
   165→    datos.llaveEntregada = llave.trim();
   166→    if (emergenciaNombre.trim() || emergenciaTelefono.trim()) {
   167→      datos.contactoEmergencia = { nombre: emergenciaNombre.trim(), telefono: emergenciaTelefono.trim() };
   168→    }
   169→    if (observaciones.trim()) datos.observacionesHuesped = observaciones.trim();
   170→    const validAcomp = acompanantes.filter(a => a.nombre.trim());
   171→    if (validAcomp.length > 0) datos.acompanantes = validAcomp;
   172→
   173→    // Pasar menores al store
   174→    if (cantNinos > 0 && menoresForms.length > 0) {
   175→      datos.menores = menoresForms.map(m => ({
   176→        nombre: m.nombre.trim(),
   177→        documento: m.documento.trim(),
   178→        edad: parseInt(m.edad) || 0,
   179→        parentesco: m.parentesco.trim(),
   180→      }));
   181→    }
   182→
   183→    try {
   184→      setCheckLoading(true);
   185→      const resultado = await realizarCheckIn(selReserva.id, datos);
   186→      if (resultado) {
   187→        closeModal();
   188→        const desc = `${selReserva.huesped} - Hab. ${selReserva.habitacion}${cantNinos > 0 ? ` (${cantNinos} menor${cantNinos > 1 ? 'es' : ''} registrados)` : ''}`;
   189→        notifySuccess('Check-in realizado', desc);
   190→      } else {
   191→        toast.error('No se pudo realizar el check-in', { description: 'La reserva ya no está en estado Confirmada, faltan datos de menores, o hubo un error de conexión.' });
   192→      }
   193→    } catch (err) {
   194→      console.error('Error en check-in:', err);
   195→      toast.error('Error al realizar check-in');
   196→    } finally {
   197→      setCheckLoading(false);
   198→    }
   199→  };
   200→
   201→  const handleConfirmCheckOut = async () => {
   202→    if (!selReserva) return;
   203→    try {
   204→      setCheckLoading(true);
   205→      const resultado = await realizarCheckOut(selReserva.id);
   206→      if (resultado) {
   207→        closeModal();
   208→        notifySuccess('Check-out realizado', `${selReserva.huesped} - Hab. ${selReserva.habitacion}`);
   209→      } else {
   210→        toast.error('No se pudo realizar el check-out', { description: 'La reserva ya no está en estado Check-In realizado o hubo un error de conexión.' });
   211→      }
   212→    } catch (err) {
   213→      console.error('Error en check-out:', err);
   214→      toast.error('Error al realizar check-out');
   215→    } finally {
   216→      setCheckLoading(false);
   217→    }
   218→  };
   219→
   220→  const addAcompanante = () => {
   221→    setAcompanantes(prev => [...prev, { nombre: '', dni: '', celular: '' }]);
   222→  };
   223→
   224→  const removeAcompanante = (idx: number) => {
   225→    setAcompanantes(prev => prev.filter((_, i) => i !== idx));
   226→  };
   227→
   228→  const updateAcompanante = (idx: number, field: keyof Acompanante, value: string) => {
   229→    setAcompanantes(prev => prev.map((a, i) => (i === idx ? { ...a, [field]: value } : a)));
   230→  };
   231→
   232→  const updateMenor = (idx: number, field: keyof MenorForm, value: string) => {
   233→    setMenoresForms(prev => prev.map((m, i) => (i === idx ? { ...m, [field]: value } : m)));
   234→  };
   235→
   236→  // Determinar si la reserva seleccionada requiere datos de menores
   237→  const requiereMenores = selReserva && (selReserva.ninos || 0) > 0;
   238→  const menoresCompletos = requiereMenores && menoresForms.length === (selReserva?.ninos || 0)
   239→    && menoresForms.every(m => m.nombre.trim() && m.documento.trim() && m.edad.trim() && m.parentesco.trim());
   240→
   241→  const puedeConfirmarCheckIn = llave.trim() && (!requiereMenores || menoresCompletos);
   242→
   243→  return (
   244→    <div className="space-y-6">
   245→      <ModuleHeader icon={LogIn} title="Check-In / Check-Out" subtitle="Gestioná ingresos y egresos de huéspedes" />
   246→
   247→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   248→        {/* Pendientes Check-In */}
   249→        <Card className="border-[#BBF7D0]/60 bg-gradient-to-br from-[#ECFDF5]/30 to-white">
   250→          <CardHeader className="pb-3">
   251→            <CardTitle className="text-lg flex items-center gap-2">
   252→              <LogIn className="w-5 h-5 text-[#059669]" />
   253→              Check-Ins Pendientes
   254→              <Badge variant="secondary" className="ml-auto">{pendientesCheckIn.length}</Badge>
   255→            </CardTitle>
   256→          </CardHeader>
   257→          <CardContent>
   258→            {pendientesCheckIn.length === 0 ? (
   259→              <p className="text-sm text-muted-foreground text-center py-6">No hay check-ins pendientes.</p>
   260→            ) : (
   261→              <div className="space-y-3 max-h-[28rem] overflow-y-auto pr-1">
   262→                {pendientesCheckIn.map(r => (
   263→                  <div key={r.id} className="border rounded-lg p-3 flex items-center justify-between gap-3 hover:bg-[#ECFDF5]/40 transition-colors duration-200 group">
   264→                    <div className="min-w-0 flex-1">
   265→                      <div className="flex items-center gap-2 flex-wrap">
   266→                        <span className="font-semibold text-sm group-hover:text-[#0F2B28] transition-colors">{r.huesped}</span>
   267→                        <Badge className={`font-semibold shadow-sm ${estadoPagoBadge[r.estadoPago] || ''}`}>{r.estadoPago}</Badge>
   268→                        {(r.ninos || 0) > 0 && (
   269→                          <Badge variant="outline" className="bg-[#F5F3FF] text-[#5B21B6] border-[#DDD6FE]">
   270→                            <Baby className="w-3 h-3 mr-1" />{r.ninos} menor{(r.ninos || 0) > 1 ? 'es' : ''}
   271→                          </Badge>
   272→                        )}
   273→                      </div>
   274→                      <div className="text-xs text-muted-foreground mt-1 flex flex-wrap gap-x-3 gap-y-0.5">
   275→                        <span className="flex items-center gap-1"><BedDouble className="w-3 h-3" /> Hab. {r.habitacion}</span>
   276→                        <span>{formatFecha(r.checkin)} → {formatFecha(r.checkout)}</span>
   277→                        <span>{nochesEntre(r.checkin, r.checkout)} noche{s(nochesEntre(r.checkin, r.checkout))}</span>
   278→                        <span>{r.personas} adulto{s(r.personas)}{(r.ninos || 0) > 0 ? ` + ${r.ninos} niño${(r.ninos || 0) > 1 ? 's' : ''}` : ''}</span>
   279→                      </div>
   280→                    </div>
   281→                    <Button size="sm" className="bg-[#059669] hover:bg-[#047857] text-white shadow-sm" onClick={() => openCheckIn(r)}>
   282→                      <KeyRound className="w-3.5 h-3.5 mr-1" />Check-In
   283→                    </Button>
   284→                  </div>
   285→                ))}
   286→              </div>
   287→            )}
   288→          </CardContent>
   289→        </Card>
   290→
   291→        {/* Pendientes Check-Out */}
   292→        <Card className="border-[#FED7AA]/60 bg-gradient-to-br from-[#FFF7ED]/30 to-white">
   293→          <CardHeader className="pb-3">
   294→            <CardTitle className="text-lg flex items-center gap-2">
   295→              <LogOut className="w-5 h-5 text-[#EA580C]" />
   296→              Check-Outs Pendientes
   297→              <Badge variant="secondary" className="ml-auto">{pendientesCheckOut.length}</Badge>
   298→            </CardTitle>
   299→          </CardHeader>
   300→          <CardContent>
   301→            {pendientesCheckOut.length === 0 ? (
   302→              <p className="text-sm text-muted-foreground text-center py-6">No hay check-outs pendientes.</p>
   303→            ) : (
   304→              <div className="space-y-3 max-h-[28rem] overflow-y-auto pr-1">
   305→                {pendientesCheckOut.map(r => (
   306→                  <div key={r.id} className="border rounded-lg p-3 flex items-center justify-between gap-3 hover:bg-[#FFF7ED]/40 transition-colors duration-200 group">
   307→                    <div className="min-w-0 flex-1">
   308→                      <div className="flex items-center gap-2 flex-wrap">
   309→                        <span className="font-semibold text-sm group-hover:text-[#0F2B28] transition-colors">{r.huesped}</span>
   310→                        <Badge className={`font-semibold shadow-sm ${estadoPagoBadge[r.estadoPago] || ''}`}>{r.estadoPago}</Badge>
   311→                        {r.menores && r.menores.length > 0 && (
   312→                          <Badge variant="outline" className="bg-[#F5F3FF] text-[#5B21B6] border-[#DDD6FE]">
   313→                            <Baby className="w-3 h-3 mr-1" />{r.menores.length} menor{r.menores.length > 1 ? 'es' : ''}
   314→                          </Badge>
   315→                        )}
   316→                      </div>
   317→                      <div className="text-xs text-muted-foreground mt-1 flex flex-wrap gap-x-3 gap-y-0.5">
   318→                        <span className="flex items-center gap-1"><BedDouble className="w-3 h-3" /> Hab. {r.habitacion}</span>
   319→                        <span>{formatFecha(r.checkin)} → {formatFecha(r.checkout)}</span>
   320→                        <span>{nochesEntre(r.checkin, r.checkout)} noche{s(nochesEntre(r.checkin, r.checkout))}</span>
   321→                      </div>
   322→                    </div>
   323→                    <Button size="sm" variant="outline" onClick={() => openCheckOut(r)}>
   324→                      <LogOut className="w-3.5 h-3.5 mr-1" />Check-Out
   325→                    </Button>
   326→                  </div>
   327→                ))}
   328→              </div>
   329→            )}
   330→          </CardContent>
   331→        </Card>
   332→      </div>
   333→
   334→      {/* =================== MODAL CHECK-IN =================== */}
   335→      <Dialog key={`checkin-${dialogKey}`} open={modalTipo === 'checkin'} onOpenChange={closeModal}>
   336→        <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   337→          {selReserva && (
   338→            <>
   339→              <DialogHeader>
   340→                <DialogTitle className="flex items-center gap-2">
   341→                  <KeyRound className="w-5 h-5" />
   342→                  Check-In — {selReserva.huesped}
   343→                </DialogTitle>
   344→              </DialogHeader>
   345→
   346→              {/* Estado de cuenta */}
   347→              <CheckInAccountStatus reserva={selReserva} />
   348→
   349→              <Separator />
   350→
   351→              {/* ═══════════ DATOS DE MENORES (OBLIGATORIO si ninos > 0) ═══════════ */}
   352→              {requiereMenores && (
   353→                <div className="space-y-4">
   354→                  <div className="flex items-center gap-2">
   355→                    <Baby className="w-5 h-5 text-[#5B21B6]" />
   356→                    <h4 className="font-semibold text-sm">
   357→                      Datos de menores ({selReserva.ninos}) — <span className="text-[#EF4444]">Obligatorio</span>
   358→                    </h4>
   359→                  </div>
   360→                  <p className="text-xs text-muted-foreground">
   361→                    La reserva registra {selReserva.ninos} menor{selReserva.ninos! > 1 ? 'es' : ''}.
   362→                    Debes completar los datos de cada uno para poder realizar el check-in.
   363→                  </p>
   364→
   365→                  {menoresErrors.length > 0 && (
   366→                    <div className="rounded-lg border border-[#FECACA] bg-[#FEE2E2] p-3 space-y-1">
   367→                      {menoresErrors.map((err, i) => (
   368→                        <p key={i} className="text-xs text-[#991B1B]">{err}</p>
   369→                      ))}
   370→                    </div>
   371→                  )}
   372→
   373→                  <div className="space-y-4">
   374→                    {menoresForms.map((menor, idx) => (
   375→                      <div key={idx} className="rounded-lg border p-4 space-y-3 bg-muted/30">
   376→                        <div className="flex items-center justify-between">
   377→                          <span className="text-sm font-semibold flex items-center gap-1.5">
   378→                            <Baby className="w-4 h-4 text-[#7C3AED]" />
   379→                            Menor {idx + 1}
   380→                          </span>
   381→                        </div>
   382→                        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   383→                          <div className="grid gap-2">
   384→                            <Label htmlFor={`menor-nombre-${idx}`}>Nombre completo *</Label>
   385→                            <Input
   386→                              id={`menor-nombre-${idx}`}
   387→                              value={menor.nombre}
   388→                              onChange={e => updateMenor(idx, 'nombre', e.target.value)}
   389→                              placeholder="Nombre del menor"
   390→                            />
   391→                          </div>
   392→                          <div className="grid gap-2">
   393→                            <Label htmlFor={`menor-doc-${idx}`}>Documento / DNI *</Label>
   394→                            <Input
   395→                              id={`menor-doc-${idx}`}
   396→                              value={menor.documento}
   397→                              onChange={e => updateMenor(idx, 'documento', e.target.value)}
   398→                              placeholder="Documento del menor"
   399→                            />
   400→                          </div>
   401→                          <div className="grid gap-2">
   402→                            <Label htmlFor={`menor-edad-${idx}`}>Edad (años) *</Label>
   403→                            <Input
   404→                              id={`menor-edad-${idx}`}
   405→                              type="number"
   406→                              min="0"
   407→                              max="17"
   408→                              value={menor.edad}
   409→                              onChange={e => updateMenor(idx, 'edad', e.target.value)}
   410→                              placeholder="Ej: 8"
   411→                            />
   412→                          </div>
   413→                          <div className="grid gap-2">
   414→                            <Label htmlFor={`menor-parentesco-${idx}`}>Parentesco *</Label>
   415→                            <Select
   416→                              value={menor.parentesco}
   417→                              onValueChange={v => updateMenor(idx, 'parentesco', v)}
   418→                            >
   419→                              <SelectTrigger id={`menor-parentesco-${idx}`}>
   420→                                <SelectValue placeholder="Seleccionar" />
   421→                              </SelectTrigger>
   422→                              <SelectContent>
   423→                                {PARENTESCO_OPTIONS.map(p => (
   424→                                  <SelectItem key={p} value={p}>{p}</SelectItem>
   425→                                ))}
   426→                              </SelectContent>
   427→                            </Select>
   428→                          </div>
   429→                        </div>
   430→                      </div>
   431→                    ))}
   432→                  </div>
   433→
   434→                  <Separator />
   435→                </div>
   436→              )}
   437→
   438→              {/* Requerimientos */}
   439→              <div className="space-y-4">
   440→                <h4 className="font-semibold text-sm flex items-center gap-2">
   441→                  <AlertCircle className="w-4 h-4 text-[#F59E0B]" />
   442→                  Requisitos
   443→                </h4>
   444→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   445→                  <div className="flex items-start gap-3">
   446→                    <Checkbox
   447→                      id="docVerificado"
   448→                      checked={docVerificado}
   449→                      onCheckedChange={(v) => setDocVerificado(v === true)}
   450→                      className="mt-0.5"
   451→                    />
   452→                    <Label htmlFor="docVerificado" className="text-sm cursor-pointer">
   453→                      Documento verificado
   454→                    </Label>
