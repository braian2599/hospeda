   185→/** Legacy Priority (computed from checkout time) */
   186→type LegacyPriority = 'high' | 'medium' | 'low';
   187→
   188→function getLastCheckoutMs(habitacion: string, reservas: Reserva[]): number {
   189→  const checkouts = reservas
   190→    .filter(r => r.habitacion === habitacion && (r.estado === 'Check-Out realizado' || r.estado === 'Check-In realizado'))
   191→    .map(r => {
   192→      const base = r.checkout ? new Date(r.checkout + 'T12:00:00') : null;
   193→      if (base && r.horaCheckout) {
   194→        const [hh, mm] = r.horaCheckout.split(':').map(n => parseInt(n, 10));
   195→        if (!isNaN(hh)) base.setHours(hh, isNaN(mm) ? 0 : mm, 0, 0);
   196→      }
   197→      return base ? base.getTime() : 0;
   198→    })
   199→    .filter(t => t > 0)
   200→    .sort((a, b) => b - a);
   201→  return checkouts[0] || 0;
   202→}
   203→
   204→function computeLegacyPriority(habitacion: string, reservas: Reserva[], now: number): LegacyPriority {
   205→  const lastCheckout = getLastCheckoutMs(habitacion, reservas);
   206→  if (lastCheckout === 0) return 'low';
   207→  const diffHours = (now - lastCheckout) / 3_600_000;
   208→  if (diffHours >= 2) return 'high';
   209→  if (diffHours >= 1) return 'medium';
   210→  return 'low';
   211→}

   213→/** Format a duration in ms as "12m 30s" / "1h 05m" / "45s". */
   214→function formatDuration(ms: number): string {
   215→  if (ms < 0) ms = 0;
   216→  const totalSec = Math.floor(ms / 1000);
   217→  const h = Math.floor(totalSec / 3600);
   218→  const m = Math.floor((totalSec % 3600) / 60);
   219→  const s = totalSec % 60;
   220→  if (h > 0) return `${h}h ${String(m).padStart(2, '0')}m`;
   221→  if (m > 0) return `${m}m ${String(s).padStart(2, '0')}s`;
   222→  return `${s}s`;
   223→}

   225→/** Format time since a timestamp as "hace 30 min", "hace 2h", etc. */
   226→function formatTimeSince(ms: number): string {
   227→  const diff = Date.now() - ms;
   228→  if (diff < 60000) return 'hace un momento';
   229→  if (diff < 3600000) return `hace ${Math.floor(diff / 60000)} min`;
   230→  if (diff < 86400000) return `hace ${Math.floor(diff / 3600000)}h`;
   231→  return `hace ${Math.floor(diff / 86400000)}d`;
   232→}

   234→/** Get initials from a full name (max 2 chars). */
   235→function getInitials(name: string): string {
   236→  const parts = name.trim().split(/\s+/).filter(Boolean);
   237→  if (parts.length === 0) return '?';
   238→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   239→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   240→}
