   105→function DateRangePickerInline({
   106→  checkin,
   107→  checkout,
   108→  onChangeCheckin,
   109→  onChangeCheckout,
   110→  label,
   111→  disabled,
   112→}: {
   113→  checkin: string;
   114→  checkout: string;
   115→  onChangeCheckin: (val: string) => void;
   116→  onChangeCheckout: (val: string) => void;
   117→  label?: string;
   118→  disabled?: boolean;
   119→}) {
   120→  const [open, setOpen] = useState(false);
   121→  const [localRange, setLocalRange] = useState<{ from?: Date; to?: Date }>({});
   122→  const syncedRef = useRef(false);
   123→
   124→  // Sync local state with the form when the popover opens (event-driven, not effect-driven).
   125→  const handleOpenChange = (next: boolean) => {
   126→    setOpen(next);
   127→    if (next && !syncedRef.current) {
   128→      syncedRef.current = true;
   129→      const f = checkin ? new Date(checkin + 'T12:00:00') : undefined;
   130→      const t = checkout ? new Date(checkout + 'T12:00:00') : undefined;
   131→      setLocalRange(f ? { from: f, to: t } : {});
   132→    }
   133→    if (!next) syncedRef.current = false;
   134→  };
   135→
   136→  const handleSelect = (range: { from?: Date; to?: Date } | undefined) => {
   137→    if (!range?.from) {
   138→      setLocalRange({});
   139→      return;
   140→    }
   141→    setLocalRange(range);
   142→  };
   143→
   144→  const fmt = (d: Date) => {
   145→    const yyyy = d.getFullYear();
   146→    const mm = String(d.getMonth() + 1).padStart(2, '0');
   147→    const dd = String(d.getDate()).padStart(2, '0');
   148→    return `${yyyy}-${mm}-${dd}`;
   149→  };
   150→
   151→  const handleConfirm = () => {
   152→    if (!localRange.from) return;
   153→    onChangeCheckin(fmt(localRange.from));
   154→    onChangeCheckout(localRange.to ? fmt(localRange.to) : '');
   155→    handleOpenChange(false);
   156→  };
   157→
   158→  const from = checkin ? new Date(checkin + 'T12:00:00') : undefined;
   159→  const to = checkout ? new Date(checkout + 'T12:00:00') : undefined;
   160→
   161→  const displayText = from && to
   162→    ? `${format(from, 'dd/MM/yyyy')} → ${format(to, 'dd/MM/yyyy')}`
   163→    : from
   164→      ? `${format(from, 'dd/MM/yyyy')} → ...`
   165→      : 'Seleccionar fechas';
   166→
   167→  return (
   168→    <div className="grid gap-1.5">
   169→      {label && <Label>{label} *</Label>}
   170→      <Button
   171→        variant="outline"
   172→        className={cn(
   173→          'w-full justify-start text-left font-normal h-9',
   174→          !checkin && 'text-muted-foreground'
   175→        )}
   176→        disabled={disabled}
   177→        onClick={() => handleOpenChange(true)}
   178→      >
   179→        <CalendarDays className="w-4 h-4 mr-2 text-muted-foreground" />
   180→        {displayText}
   181→      </Button>
   182→      {open
   183→        ? createPortal(
   184→            <div className="fixed inset-0 z-[200] flex items-center justify-center bg-black/40" onClick={() => handleOpenChange(false)}>
   185→              <div
   186→                className="bg-white rounded-xl shadow-2xl border border-[#E2E8F0] pointer-events-auto"
   187→                onClick={e => e.stopPropagation()}
   188→                onMouseDown={e => e.stopPropagation()}
   189→              >
   190→                <Calendar
   191→                  mode="range"
   192→                  selected={localRange.from ? { from: localRange.from, to: localRange.to } as { from: Date; to: Date } : undefined}
   193→                  onSelect={handleSelect}
   194→                  numberOfMonths={2}
   195→                  autoFocus
   196→                  classNames={{
   197→                    range_start: 'rounded-l-md bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   198→                    range_end: 'rounded-r-md bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   199→                    range_middle: 'rounded-none bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   200→                    day: 'relative w-full h-full p-0 text-center [&:first-child[data-selected=true]_button]:rounded-l-md [&:last-child[data-selected=true]_button]:rounded-r-md group/day aspect-square select-none',
   201→                  }}
   202→                />
   203→                <div className="flex justify-end border-t border-[#E2E8F0] p-3">
   204→                  <Button
   205→                    size="sm"
   206→                    onClick={handleConfirm}
   207→                    disabled={!localRange.from}
   208→                    className="bg-[#059669] hover:bg-[#047857] text-white text-xs h-8"
   209→                  >
   210→                    Confirmar fechas
   211→                  </Button>
   212→                </div>
   213→              </div>
   214→            </div>,
   215→            document.body
   216→          )
   217→        : null}
   218→    </div>
   219→  );
   220→}

   243→// ==================== DESGLOSE PRECIO COMPONENT ====================
   244→function DesglosePrecio({ form, computed, formatMoney, s }: {
   245→  form: { tipoTarifa: string; reservaMultiple: boolean; habitacion: string; habitacion2?: string; personas: string; personas2?: string };
   246→  computed: { noches: number; subtotal: number; subtotal2: number; desglose: any };
   247→  formatMoney: (n: number) => string;
   248→  s: (n: number) => string;
   249→}) {
   250→  const d = computed.desglose;
   251→
   252→  const adultoLinea = (d: any) => {
   253→    if (!d || d.totalAdultos === 0) return null;
   254→    const modo = d.modoCobro;
   255→    if (modo === 'porCama') {
   256→      return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''} × ${formatMoney(d.precioRango)}/cama/noche`;
   257→    }
   258→    if (modo === 'porHabitacion') return 'Habitación';
   259→    if (modo === 'porPersona') {
   260→      return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''} × ${formatMoney(d.precioRango)}/noche`;
   261→    }
   262→    // porGrupo
   263→    return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''}`;
   264→  };
   265→
   266→  const renderLines = (sub: number, dLocal: any) => {
   267→    if (!dLocal) return (
   268→      <div className="space-y-1">
   269→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Subtotal</span><span className="font-semibold text-[#1E293B]">{formatMoney(sub)}</span></div>
   270→      </div>
   271→    );
   272→
   273→    const lines: { key: string; label: string; value: string; dim?: boolean }[] = [];
   274→
   275→    const aLabel = adultoLinea(dLocal);
   276→    if (aLabel) {
   277→      lines.push({ key: 'adultos', label: aLabel, value: formatMoney(dLocal.totalAdultos) });
   278→    }
   279→
   280→    if (dLocal.ninosCount > 0 && dLocal.totalNinos > 0) {
   281→      lines.push({
   282→        key: 'ninos',
   283→        label: `${dLocal.ninosCount} niño${dLocal.ninosCount > 1 ? 's' : ''} × ${formatMoney(dLocal.precioNino)}/noche`,
   284→        value: formatMoney(dLocal.totalNinos),
   285→      });
   286→    }
   287→
   288→    if (dLocal.nochesGratis > 0) {
   289→      const totalBase = dLocal.totalAdultos + dLocal.totalNinos;
   290→      const ahorro = totalBase > 0 && dLocal.nochesCobrables > 0
   291→        ? Math.round(totalBase / dLocal.nochesCobrables * dLocal.nochesGratis)
   292→        : 0;
   293→      lines.push({
   294→        key: 'cortesia',
   295→        label: `${dLocal.nochesGratis} noche${dLocal.nochesGratis > 1 ? 's' : ''} de cortesía`,
   296→        value: `- ${formatMoney(ahorro)}`,
   297→        dim: true,
   298→      });
   299→    }
   300→
   301→    return (
   302→      <div className="space-y-1">
   303→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Tarifa</span><span className="font-semibold text-[#1E293B] capitalize">{form.tipoTarifa}</span></div>
   304→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Noches</span><span className="font-semibold text-[#1E293B]">{computed.noches}{dLocal?.nochesGratis > 0 ? ` (${dLocal.nochesCobrables} cobrables)` : ''}</span></div>
   305→        {lines.map(l => (
   306→          <div key={l.key} className="flex justify-between items-center py-1 text-[13px]">
   307→            <span className={l.dim ? 'text-[#94A3B8] line-through' : 'text-[#64748B]'}>{l.label}</span>
   308→            <span className={l.dim ? 'font-medium text-[#94A3B8]' : 'font-semibold text-[#1E293B]'}>{l.value}</span>
   309→          </div>
   310→        ))}
   311→        <div className="flex justify-between items-center py-1.5 mt-1 border-t border-[#E2E8F0] text-[13px]">
   312→          <span className="font-medium text-[#475569]">Subtotal</span>
   313→          <span className="font-bold text-[#1E293B]">{formatMoney(sub)}</span>
   314→        </div>
   315→      </div>
   316→    );
   317→  };
   318→
   319→  return (
   320→    <>
   321→      {form.reservaMultiple ? (
   322→        <>
   323→          <p className="text-[11px] font-semibold text-[#64748B] uppercase tracking-wider">Habitación {form.habitacion}</p>
   324→          {renderLines(computed.subtotal, d)}
   325→          {computed.subtotal2 > 0 && (
   326→            <>
   327→              <p className="text-[11px] font-semibold text-[#64748B] uppercase tracking-wider pt-2">Habitación {form.habitacion2}</p>
   328→              {renderLines(computed.subtotal2, null)}
   329→            </>
   330→          )}
   331→        </>
   332→      ) : (
   333→        renderLines(computed.subtotal, d)
   334→      )}
   335→    </>
   336→  );
   337→}
