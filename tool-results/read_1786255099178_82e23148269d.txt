    20→/* ─── Intersection Observer hook ─── */
    21→function useInView(threshold = 0.12) {
    22→  const ref = useRef<HTMLDivElement>(null);
    23→  const [inView, setInView] = useState(false);
    24→  useEffect(() => {
    25→    const el = ref.current;
    26→    if (!el) return;
    27→    const obs = new IntersectionObserver(
    28→      ([e]) => { if (e.isIntersecting) { setInView(true); obs.disconnect(); } },
    29→      { threshold }
    30→    );
    31→    obs.observe(el);
    32→    return () => obs.disconnect();
    33→  }, [threshold]);
    34→  return { ref, inView };
    35→}

    42→/* ─── Scroll progress indicator (top gradient bar) ─── */
    43→function ScrollProgress() {
    44→  const [progress, setProgress] = useState(0);
    45→  useEffect(() => {
    46→    const onScroll = () => {
    47→      const scrollTop = window.scrollY;
    48→      const docHeight = document.documentElement.scrollHeight - window.innerHeight;
    49→      setProgress(docHeight > 0 ? Math.min(100, (scrollTop / docHeight) * 100) : 0);
    50→    };
    51→    onScroll();
    52→    window.addEventListener('scroll', onScroll, { passive: true });
    53→    return () => window.removeEventListener('scroll', onScroll);
    54→  }, []);
    55→  return (
    56→    <div
    57→      className="scroll-progress"
    58→      style={{ width: `${progress}%`, opacity: progress > 0.5 ? 1 : 0 }}
    59→      aria-hidden="true"
    60→    />
    61→  );
    62→}

   259→/* ─── Fade-in wrapper ─── */
   260→function FadeIn({ children, className = '', delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   261→  const { ref, inView } = useInView();
   262→  return (
   263→    <div
   264→      ref={ref}
   265→      className={`transition-all duration-700 ease-out ${className}`}
   266→      style={{
   267→        opacity: inView ? 1 : 0,
   268→        transform: inView ? 'translateY(0)' : 'translateY(32px)',
   269→        transitionDelay: `${delay}ms`,
   270→      }}
   271→    >
   272→      {children}
   273→    </div>
   274→  );
   275→}

   277→/* ─── Typewriter text effect ─── */
   278→function TypewriterText({ text, className = '' }: { text: string; className?: string }) {
   279→  const [displayed, setDisplayed] = useState('');
   280→  const [done, setDone] = useState(false);
   281→
   282→  useEffect(() => {
   283→    let i = 0;
   284→    const interval = setInterval(() => {
   285→      i++;
   286→      if (i <= text.length) {
   287→        setDisplayed(text.slice(0, i));
   288→      } else {
   289→        setDone(true);
   290→        clearInterval(interval);
   291→      }
   292→    }, 65);
   293→    return () => clearInterval(interval);
   294→  }, [text]);
   295→
   296→  return (
   297→    <span className={className}>
   298→      {displayed}
   299→      {!done && <span className="typewriter-cursor" aria-hidden="true" />}
   300→    </span>
   301→  );
   302→}

   304→/* ─── Back to top button ─── */
   305→function BackToTop() {
   306→  const [visible, setVisible] = useState(false);
   307→
   308→  useEffect(() => {
   309→    const onScroll = () => {
   310→      const scrollTop = window.scrollY;
   311→      const docHeight = document.documentElement.scrollHeight - window.innerHeight;
   312→      setVisible(docHeight > 0 && scrollTop / docHeight > 0.5);
   313→    };
   314→    onScroll();
   315→    window.addEventListener('scroll', onScroll, { passive: true });
   316→    return () => window.removeEventListener('scroll', onScroll);
   317→  }, []);
   318→
   319→  if (!visible) return null;
   320→
   321→  return (
   322→    <button
   323→      className="back-to-top-btn"
   324→      onClick={() => window.scrollTo({ top: 0, behavior: 'smooth' })}
   325→      aria-label="Volver arriba"
   326→    >
   327→      <ArrowUp className="w-4 h-4" />
   328→    </button>
   329→  );
   330→}
