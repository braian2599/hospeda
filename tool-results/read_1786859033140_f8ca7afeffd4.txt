    91→    91→function subEstadoBadge(estado: string) {
    92→    92→  const map: Record<string, string> = {
    93→    93→    trial: 'bg-amber-100/30 text-amber-700 border-amber-300',
    94→    94→    activa: 'bg-emerald-100/30 text-emerald-700 border-primary',
    95→    95→    vencida: 'bg-red-100/30 text-red-700 border-red-300',
    96→    96→    cancelada: 'bg-gray-100/30 text-gray-600 border-gray-300',
    97→    97→    suspensa: 'bg-orange-100/30 text-orange-700 border-orange-300',
    98→    98→  };
    99→    99→  return (
   100→   100→    <Badge variant="outline" className={map[estado] || ''}>
   101→   101→      {estado.charAt(0).toUpperCase() + estado.slice(1)}
   102→   102→    </Badge>
   103→   103→  );
   104→   104→}

   123→   123→
   124→   124→  // Dialog states

   301→   301→
