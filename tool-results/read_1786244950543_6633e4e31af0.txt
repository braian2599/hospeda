   133→   133→/** Busca el rango que corresponde a la cantidad de personas dada. */
   134→   134→function encontrarRango(rangos: RangoPrecio[], personas: number): RangoPrecio | undefined {
   135→   135→  // Buscar rango donde minPersonas <= personas <= maxPersonas
   136→   136→  for (const r of rangos) {
   137→   137→    if (personas >= r.minPersonas && (r.maxPersonas === null || personas <= r.maxPersonas)) {
   138→   138→      return r;
   139→   139→    }
   140→   140→  }
   141→   141→  // Fallback: usar el último rango (el que tiene maxPersonas null, o el de mayor cobertura)
   142→   142→  return rangos.length > 0 ? rangos[rangos.length - 1] : undefined;
   143→   143→}
   144→   144→
   145→   145→/**
   146→   146→ * Convierte datos de tarifa desde la BD (puede ser formato viejo o nuevo) a RangoPrecio[].
   147→   147→ * Formato viejo: { "1": 35000, "2": 30000, ... }
   148→   148→ * Formato nuevo: { "rangos": [{ minPersonas, maxPersonas, precio }] }
   149→   149→ */
   150→   150→function normalizarRangos(preciosDb: any): RangoPrecio[] {
   151→   151→  if (!preciosDb) return [];
   152→   152→  // Nuevo formato: tiene la propiedad "rangos" como array
   153→   153→  if (preciosDb.rangos && Array.isArray(preciosDb.rangos)) {
   154→   154→    return preciosDb.rangos.map((r: any) => ({
   155→   155→      minPersonas: Number(r.minPersonas) || 1,
   156→   156→      maxPersonas: r.maxPersonas != null ? Number(r.maxPersonas) : null,
   157→   157→      precio: Number(r.precio) || 0,
   158→   158→    }));
   159→   159→  }
   160→   160→  // Formato viejo: keys numéricos { "1": 35000, "2": 30000, ... }
   161→   161→  const keys = Object.keys(preciosDb).filter(k => !isNaN(Number(k)) && Number(k) >= 1);
   162→   162→  if (keys.length > 0) {
   163→   163→    return keys.map(k => ({
   164→   164→      minPersonas: Number(k),
   165→   165→      maxPersonas: Number(k),
   166→   166→      precio: Number(preciosDb[k]) || 0,
   167→   167→    }));
   168→   168→  }
   169→   169→  return [];
   170→   170→}

   172→   172→/**
   173→   173→ * Calcula las noches de cortesía según la modalidad.
   174→   174→ * Devuelve la cantidad de noches a descontar.
   175→   175→ */
   176→   176→function calcularNochesGratis(promociones: PromocionesTarifa, noches: number, checkin?: string): number {
   177→   177→  const nc = promociones.nochesCortesia;
   178→   178→  if (!nc?.activo || !nc.modalidad) return 0;
   179→   179→
   180→   180→  const mod = nc.modalidad;
   181→   181→  if (mod.tipo === 'cadaX') {
   182→   182→    const cada = mod.cada || 999;
   183→   183→    if (noches < cada) return 0;
   184→   184→    return Math.floor(noches / cada);
   185→   185→  }
   186→   186→  if (mod.tipo === 'aPartirDe') {
   187→   187→    if (noches < mod.minNoches) return 0;
   188→   188→    return mod.nochesGratis || 0;
   189→   189→  }
   190→   190→  if (mod.tipo === 'diaSemana' && checkin) {
   191→   191→    // Contar cuántas noches de la estadía caen en el día de la semana elegido
   192→   192→    const fechaInicio = new Date(checkin + 'T12:00:00');
   193→   193→    let count = 0;
   194→   194→    for (let i = 0; i < noches; i++) {
   195→   195→      const d = new Date(fechaInicio);
   196→   196→      d.setDate(d.getDate() + i);
   197→   197→      if (d.getDay() === mod.dia) count++;
   198→   198→    }
   199→   199→    return count;
   200→   200→  }
   201→   201→  return 0;
   202→   202→}

   204→   204→/**
   205→   205→ * Obtiene las promociones efectivas de una tarifa, migrando datos viejos (choferCortesia) si es necesario.
   206→   206→ */
   207→   207→function getPromocionesEfectivas(tarifa: TarifaPrecios): PromocionesTarifa {
   208→   208→  // Si tiene promociones nuevas, usar esas
   209→   209→  if (tarifa.promociones) return tarifa.promociones;
   210→   210→  // Migración: si tiene choferCortesia viejo, convertir a nuevo formato
   211→   211→  if (tarifa.choferCortesia) {
   212→   212→    return {
   213→   213→      acompananteSinCargo: {
   214→   214→        activo: true,
   215→   215→        etiqueta: 'Chofer de cortesía',
   216→   216→        habitacionAsignada: tarifa.habitacionChofer || undefined,
   217→   217→      },
   218→   218→    };
   219→   219→  }
   220→   220→  return {};
   221→   221→}

   227→   227→
   228→   228→function calcularTotalSegunTarifa(
   229→   229→  tarifas: Record<string, TarifaPrecios>,
   230→   230→  tipoTarifa: string,
   231→   231→  personas: number,
   232→   232→  noches: number,
   233→   233→  options?: CalcTarifaOptions
   234→   234→): number {
   235→   235→  // Buscar la tarifa (fallback a 'normal')
   236→   236→  const tarifa = tarifas[tipoTarifa] || tarifas['normal'];
   237→   237→  if (!tarifa || !tarifa.rangos || tarifa.rangos.length === 0) return 0;
   238→   238→
   239→   239→  const promociones = getPromocionesEfectivas(tarifa);
   240→   240→  const modo: ModoCobro = tarifa.modoCobro || 'porGrupo';
   241→   241→
   242→   242→  // ─── Noches de cortesía ───
   243→   243→  const nochesGratis = calcularNochesGratis(promociones, noches, options?.checkin);
   244→   244→  const nochesCobrables = Math.max(0, noches - nochesGratis);
   245→   245→
   246→   246→  // ─── Niños diferenciado ───
   247→   247→  const ninosDif = promociones.ninosDiferenciado;
   248→   248→  const cantNinos = (options?.ninos && ninosDif?.activo) ? options.ninos : 0;
   249→   249→  const adultos = Math.max(1, personas - cantNinos);
   250→   250→
   251→   251→  // ─── Acompañante sin cargo (no descuenta: el acompañante va a otra habitación) ───
   252→   252→  // Ya NO se resta 1 adulto. Todos los adultos de esta reserva pagan.
   253→   253→  // El acompañante se gestiona como una reserva separada con total=0.
   254→   254→
   255→   255→  // ─── Cálculo por modo ───
   256→   256→  if (modo === 'porCama') {
   257→   257→    // Modo porCama: el precio del rango es por persona por noche
   258→   258→    const rango = encontrarRango(tarifa.rangos, adultos);
   259→   259→    const precioCama = rango?.precio || tarifa.rangos[0]?.precio || 0;
   260→   260→    let total = nochesCobrables * adultos * precioCama;
   261→   261→    // Sumar niños
   262→   262→    if (cantNinos > 0 && ninosDif?.activo) {
   263→   263→      total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   264→   264→    }
   265→   265→    return total;
   266→   266→  }
   267→   267→
   268→   268→  if (modo === 'porHabitacion') {
   269→   269→    const precio = tarifa.rangos[0]?.precio || 0;
   270→   270→    let total = nochesCobrables * precio;
   271→   271→    // Sumar niños si aplica
   272→   272→    if (cantNinos > 0 && ninosDif?.activo) {
   273→   273→      total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   274→   274→    }
   275→   275→    return total;
   276→   276→  }
   277→   277→
   278→   278→  const rango = encontrarRango(tarifa.rangos, adultos);
   279→   279→  if (!rango) return 0;
   280→   280→
   281→   281→  if (modo === 'porPersona') {
   282→   282→    let total = nochesCobrables * adultos * rango.precio;
   283→   283→    // Sumar niños
   284→   284→    if (cantNinos > 0 && ninosDif?.activo) {
   285→   285→      total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   286→   286→    }
   287→   287→    return total;
   288→   288→  }
   289→   289→
   290→   290→  // porGrupo (default): el precio es el total del grupo
   291→   291→  let total = nochesCobrables * rango.precio;
   292→   292→  // Sumar niños
   293→   293→  if (cantNinos > 0 && ninosDif?.activo) {
   294→   294→    total += cantNinos * (ninosDif.precioNino || 0) * nochesCobrables;
   295→   295→  }
   296→   296→  return total;
   297→   297→}
