    31→
    32→// formatFecha and formatMoney imported from @/lib/format
    33→
    34→const estadoPagoBadge: Record<string, string> = {
    35→  Pendiente: 'bg-[#FEF3C7] text-[#92400E] border-[#FDE68A]',

    37→  Pagado: 'bg-[#DCFCE7] text-[#166534] border-[#BBF7D0]',
    38→};
    39→
    40→export default function FacturacionModule() {
    41→  const reservas = useHotelStore(s => s.reservas);
    42→  const pagos = useHotelStore(s => s.pagos);
    43→  const metodosPago = useHotelStore(s => s.metodosPago);
    44→  const habitaciones = useHotelStore(s => s.habitaciones);
    45→  const caja = useHotelStore(s => s.caja);
    46→  const calcularTotalReserva = useHotelStore(s => s.calcularTotalReserva);
    47→  const calcularTotalPagado = useHotelStore(s => s.calcularTotalPagado);
    48→  const registrarPago = useHotelStore(s => s.registrarPago);
    49→  const nochesEntre = useHotelStore(s => s.nochesEntre);
    50→
    51→  // Pending payments
    52→  const pendientes = reservas.filter(r => {

    54→    const total = calcularTotalReserva(r.id);
    55→    const pagado = calcularTotalPagado(r.id);

    57→  });
    58→
    59→  // History filters

    61→  const [histFiltroMetodo, setHistFiltroMetodo] = useState('todos');
    62→  const [histFiltroDesde, setHistFiltroDesde] = useState('');
    63→  const [histFiltroHasta, setHistFiltroHasta] = useState('');
    64→
    65→  // Pagination
    66→  const [pendPage, setPendPage] = useState(1);
    67→  const [histPage, setHistPage] = useState(1);
    68→  const PAGE_SIZE = 15;
    69→
    70→  // Payment dialog
    71→  const [savingPago, setSavingPago] = useState(false);
    72→  const [pagoDialogOpen, setPagoDialogOpen] = useState(false);
    73→  const [pagoReservaId, setPagoReservaId] = useState<string | null>(null);
    74→  const [pagoMonto, setPagoMonto] = useState('');
    75→  const [pagoMetodo, setPagoMetodo] = useState('');
    76→  const [pagoNota, setPagoNota] = useState('');

    78→  // Receipt dialog
    79→  const [reciboDialogOpen, setReciboDialogOpen] = useState(false);
    80→  const [reciboReservaId, setReciboReservaId] = useState<string | null>(null);
    81→
    82→  // Resolve method filter ID to name for comparison
    83→  const filtroMetodoNombre = histFiltroMetodo !== 'todos'
    84→    ? metodosPago.find(m => m.id === histFiltroMetodo)?.nombre
    85→    : 'todos';
    86→

    88→  const filteredPagos = pagos

   108→  const pagedPagos = filteredPagos.slice((histPage - 1) * PAGE_SIZE, histPage * PAGE_SIZE);

   125→      toast.error('Ingresá un monto válido');
   126→      return;

   128→    if (!pagoReservaId || !pagoMetodo) return;

   130→      toast.error('Caja cerrada', { description: 'Debés abrir la caja antes de registrar un cobro.' });

   149→    setPagoReservaId(null);
   150→    setSavingPago(false);

   156→    setReciboDialogOpen(true);
   157→  };
   158→
   159→  const pagoReserva = reservas.find(r => r.id === pagoReservaId);
   160→  const reciboReserva = reservas.find(r => r.id === reciboReservaId);
   161→
   162→  return (
   163→    <div className="space-y-6">
   164→      <ModuleHeader icon={Receipt} title="Facturación" subtitle="Comprobantes y pagos de tus reservas" />
   165→
   166→      <Tabs defaultValue="pendientes">
   167→        <TabsList>
   168→          <TabsTrigger value="pendientes">
   169→            <CreditCard className="w-4 h-4 mr-1" />Cobros pendientes
   170→          </TabsTrigger>
   171→          <TabsTrigger value="historial">
   172→            <FileText className="w-4 h-4 mr-1" />Historial de pagos
   173→          </TabsTrigger>
   174→        </TabsList>
   175→
   176→        {/* =================== TAB: COBROS PENDIENTES =================== */}
   177→        <TabsContent value="pendientes" className="mt-4">
   178→          <Card>
   179→            <CardHeader className="pb-3">
   180→              <CardTitle className="text-base">
   181→                Reservas con saldo pendiente
   182→                <Badge variant="secondary" className="ml-2">{pendientes.length}</Badge>
   183→              </CardTitle>
   184→            </CardHeader>
   185→            <CardContent className="p-0">

   189→                  <div className="text-center py-10 text-muted-foreground">No hay cobros pendientes.</div>

   208→                            <CalendarDays className="w-3.5 h-3.5 shrink-0" />
   209→                            <span>{formatFecha(r.checkin)}</span>
   210→                            <span>→</span>
   211→                            <span>{formatFecha(r.checkout)}</span>
   212→                          </div>
   213→                          {/* Payment status badge */}
   214→                          <Badge className={estadoPagoBadge[r.estadoPago] || ''}>{r.estadoPago}</Badge>
   215→                          {/* Money summary */}
   216→                          <div className="grid grid-cols-3 gap-2 text-center rounded-lg border p-2.5 bg-muted/30">
   217→                            <div>
   218→                              <p className="text-[10px] text-muted-foreground leading-tight">Total</p>
   219→                              <p className="text-sm font-semibold">{formatMoney(total)}</p>
   220→                            </div>
   221→                            <div>
   222→                              <p className="text-[10px] text-muted-foreground leading-tight">Pagado</p>

   235→                              onClick={() => openPagoDialog(r.id)}

   250→                    })}
   251→                  </div>
   252→                )}
   253→              </div>
   254→
   255→              {/* ── Desktop: Table ── */}
   256→              <div className="hidden sm:block overflow-x-auto">
   257→                <Table>
   258→                  <TableHeader>
   259→                    <TableRow>
   260→                      <TableHead>Huésped</TableHead>
   261→                      <TableHead>Hab.</TableHead>
   262→                      <TableHead className="hidden md:table-cell">Check-in</TableHead>
   263→                      <TableHead className="hidden md:table-cell">Check-out</TableHead>
   264→                      <TableHead className="text-right">Total</TableHead>
   265→                      <TableHead className="text-right">Pagado</TableHead>
   266→                      <TableHead className="text-right">Saldo</TableHead>
   267→                      <TableHead>Pago</TableHead>
   268→                      <TableHead className="text-right">Acciones</TableHead>
   269→                    </TableRow>
   270→                  </TableHeader>
   271→                  <TableBody>
   272→                    {pendientes.length === 0 ? (
   273→                      <TableRow>
   274→                        <TableCell colSpan={9} className="text-center py-8 text-muted-foreground">
   275→                          No hay cobros pendientes.
   276→                        </TableCell>
   277→                      </TableRow>
   278→                    ) : (

   280→                        const total = calcularTotalReserva(r.id);

   302→                                </Button>
   303→                                <Button size="icon" variant="ghost" className="h-8 w-8" onClick={() => openRecibo(r.id)}>
   304→                                  <FileText className="w-3.5 h-3.5" />
   305→                                </Button>
   306→                              </div>
   307→                            </TableCell>
   308→                          </TableRow>
   309→                        );
   310→                      })
   311→                    )}
   312→                  </TableBody>
   313→                </Table>
   314→              </div>
   315→              <PaginationBar page={pendPage} totalPages={pendTotalPages} onPageChange={setPendPage} totalItems={pendientes.length} pageSize={PAGE_SIZE} />
   316→            </CardContent>
   317→          </Card>
   318→        </TabsContent>
   319→
   320→        {/* =================== TAB: HISTORIAL =================== */}
   321→        <TabsContent value="historial" className="mt-4 space-y-4">
   322→          {/* Filter bar */}
   323→          <Card>
   324→            <CardContent className="p-4">
   325→              <div className="flex flex-wrap gap-3 items-end">
   326→                <div className="grid gap-1.5 flex-1 w-full sm:w-auto sm:min-w-[160px]">
   327→                  <Label className="text-xs text-muted-foreground">Huésped / DNI</Label>

   329→                    placeholder="Buscar..."
   330→                    value={histFiltroHuesped}
   331→                    onChange={e => { setHistFiltroHuesped(e.target.value); setHistPage(1); }}
   332→                  />
   333→                </div>
   334→                <div className="grid gap-1.5">
   335→                  <Label className="text-xs text-muted-foreground">Método</Label>

   338→                    <SelectContent>
   339→                      <SelectItem value="todos">Todos</SelectItem>

   341→                        <SelectItem key={m.id} value={m.id}>{m.nombre}</SelectItem>
   342→                      ))}
   343→                    </SelectContent>
   344→                  </Select>
   345→                </div>
   346→                <div className="grid gap-1.5">
   347→                  <Label className="text-xs text-muted-foreground">Desde</Label>
   348→                  <Input type="date" value={histFiltroDesde} onChange={e => { setHistFiltroDesde(e.target.value); setHistPage(1); }} className="w-full sm:w-auto sm:min-w-[140px]" />
   349→                </div>
   350→                <div className="grid gap-1.5">
   351→                  <Label className="text-xs text-muted-foreground">Hasta</Label>
   352→                  <Input type="date" value={histFiltroHasta} onChange={e => { setHistFiltroHasta(e.target.value); setHistPage(1); }} className="w-full sm:w-auto sm:min-w-[140px]" />
   353→                </div>
   354→                <Button variant="outline" size="sm" onClick={() => { setHistFiltroHuesped(''); setHistFiltroMetodo('todos'); setHistFiltroDesde(''); setHistFiltroHasta(''); setHistPage(1); }}>
   355→                  <XCircle className="w-3.5 h-3.5 mr-1" />Limpiar
   356→                </Button>
   357→              </div>
   358→            </CardContent>
   359→          </Card>
   360→
   361→          {/* Payments table */}
   362→          <Card>
   363→            <CardContent className="p-0">
   364→              {/* ── Mobile: Cards ── */}
   365→              <div className="sm:hidden">
   366→                {filteredPagos.length === 0 ? (

   380→                            <div className="flex items-center gap-1.5 text-xs text-muted-foreground">
   381→                              <CalendarDays className="w-3.5 h-3.5 shrink-0" />
   382→                              <span>{formatFecha(p.fecha)}</span>
   383→                            </div>
   384→                            <p className="text-base font-bold text-[#166534] shrink-0">{formatMoney(p.monto)}</p>
   385→                          </div>

   387→                          <div className="flex items-center gap-2">
   388→                            <p className="font-semibold text-sm truncate flex-1">{reserva?.huesped || `Reserva #${p.idReserva}`}</p>

   402→                            </div>

   421→                <Table>

   442→                    ) : (
