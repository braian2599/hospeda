   767→   767→ * Usa selectores Zustand granulares (sin destructuring) y `todayLocal` / `daysAgo`

   769→   769→ */
   770→   770→function TodayActivitySummary() {
   771→   771→  // Granular selectors (no destructuring) — each subscribes to its own slice.
   772→   772→  const reservas = useHotelStore(s => s.reservas);
   773→   773→  const habitaciones = useHotelStore(s => s.habitaciones);
   774→   774→
   775→   775→  const hoyStr = todayLocal();

   777→   777→

   780→   780→    r => r.estado === 'Check-In realizado' && r.checkin === hoyStr
   781→   781→  ).length;

   783→   783→    r => r.estado === 'Check-In realizado' && r.checkin === ayerStr
   784→   784→  ).length;

   794→   794→  // ── Estadías activas (snapshot: estado === 'Check-In realizado') ──
   795→   795→  const estadiasActivas = reservas.filter(
   796→   796→    r => r.estado === 'Check-In realizado'
   797→   797→  ).length;
   798→   798→  // Snapshot aproximado de "estadías activas antes de los check-outs de hoy"
   799→   799→  // = activas actuales + las que hicieron check-out hoy.
   800→   800→  const estadiasPrevias = reservas.filter(
   801→   801→    r => r.estado === 'Check-In realizado'
   802→   802→      || (r.estado === 'Check-Out realizado' && r.checkout === hoyStr)
   803→   803→  ).length;
   804→   804→
   805→   805→  // Ocupación (% de habitaciones ocupadas) — usa `habitaciones` selector.
   806→   806→  const totalHabitaciones = Object.keys(habitaciones).length;
   807→   807→  const ocupadas = Object.values(habitaciones).filter(h => h.estado === 'Ocupada').length;
   808→   808→  const ocupacionPct = totalHabitaciones === 0
   809→   809→    ? null
   810→   810→    : Math.round((ocupadas / totalHabitaciones) * 100);
   811→   811→
   812→   812→  const stats: Array<{
   813→   813→    key: string;
