    47→    47→function DatePickerInline({
    48→    48→ value,
    49→    49→ onChange,
    50→    50→ placeholder,
    51→    51→ label,
    52→    52→ minDate,
    53→    53→ disabled,
    54→    54→}: {
    55→    55→ value: string;
    56→    56→ onChange: (val: string) => void;
    57→    57→ placeholder: string;
    58→    58→ label?: string;
    59→    59→ minDate?: Date;
    60→    60→ disabled?: boolean;
    61→    61→}) {
    62→    62→ const [open, setOpen] = useState(false);
    63→    63→ const selectedDate = value ? new Date(value + 'T12:00:00') : undefined;
    64→    64→
    65→    65→ const handleSelect = (date: Date | undefined) => {
    66→    66→ if (date) {
    67→    67→ const yyyy = date.getFullYear();
    68→    68→ const mm = String(date.getMonth() + 1).padStart(2, '0');
    69→    69→ const dd = String(date.getDate()).padStart(2, '0');
    70→    70→ onChange(`${yyyy}-${mm}-${dd}`);
    71→    71→ }
    72→    72→ setOpen(false);
    73→    73→ };
    74→    74→
    75→    75→ return (
    76→    76→ <div className="grid gap-1.5">
    77→    77→ {label && <Label>{label} *</Label>}
    78→    78→ <Popover open={open} onOpenChange={setOpen}>
    79→    79→ <PopoverTrigger asChild>
    80→    80→ <Button
    81→    81→ variant="outline"
    82→    82→ className={cn(
    83→    83→ 'w-full justify-start text-left font-normal h-9',
    84→    84→ !value && 'text-muted-foreground'
    85→    85→ )}
    86→    86→ disabled={disabled}
    87→    87→ >
    88→    88→ {selectedDate
    89→    89→ ? format(selectedDate, "dd/MM/yyyy")
    90→    90→ : placeholder}
    91→    91→ </Button>
    92→    92→ </PopoverTrigger>
    93→    93→ <PopoverContent className="w-auto p-0" align="start">
    94→    94→ <Calendar
    95→    95→ mode="single"
    96→    96→ selected={selectedDate}
    97→    97→ onSelect={handleSelect}
    98→    98→ disabled={minDate ? (d) => d < minDate : undefined}

   109→   109→function DateRangePickerInline({
   110→   110→  checkin,
   111→   111→  checkout,
   112→   112→  onChangeCheckin,
   113→   113→  onChangeCheckout,
   114→   114→  label,
   115→   115→  disabled,
   116→   116→}: {
   117→   117→  checkin: string;
   118→   118→  checkout: string;
   119→   119→  onChangeCheckin: (val: string) => void;
   120→   120→  onChangeCheckout: (val: string) => void;
   121→   121→  label?: string;
   122→   122→  disabled?: boolean;
   123→   123→}) {
   124→   124→  const [open, setOpen] = useState(false);
   125→   125→  const [localRange, setLocalRange] = useState<{ from?: Date; to?: Date }>({});
   126→   126→  const syncedRef = useRef(false);
   127→   127→
   128→   128→  // Sync local state with the form when the popover opens (event-driven, not effect-driven).
   129→   129→  const handleOpenChange = (next: boolean) => {
   130→   130→    setOpen(next);
   131→   131→    if (next && !syncedRef.current) {
   132→   132→      syncedRef.current = true;
   133→   133→      const f = checkin ? new Date(checkin + 'T12:00:00') : undefined;
   134→   134→      const t = checkout ? new Date(checkout + 'T12:00:00') : undefined;
   135→   135→      setLocalRange(f ? { from: f, to: t } : {});
   136→   136→    }
   137→   137→    if (!next) syncedRef.current = false;
   138→   138→  };
   139→   139→
   140→   140→  const handleSelect = (range: { from?: Date; to?: Date } | undefined) => {
   141→   141→    if (!range?.from) {
   142→   142→      setLocalRange({});
   143→   143→      return;
   144→   144→    }
   145→   145→    setLocalRange(range);
   146→   146→  };
   147→   147→
   148→   148→  const fmt = (d: Date) => {
   149→   149→    const yyyy = d.getFullYear();
   150→   150→    const mm = String(d.getMonth() + 1).padStart(2, '0');
   151→   151→    const dd = String(d.getDate()).padStart(2, '0');
   152→   152→    return `${yyyy}-${mm}-${dd}`;
   153→   153→  };
   154→   154→
   155→   155→  const handleConfirm = () => {
   156→   156→    if (!localRange.from) return;
   157→   157→    onChangeCheckin(fmt(localRange.from));
   158→   158→    onChangeCheckout(localRange.to ? fmt(localRange.to) : '');
   159→   159→    handleOpenChange(false);
   160→   160→  };
   161→   161→
   162→   162→  const from = checkin ? new Date(checkin + 'T12:00:00') : undefined;
   163→   163→  const to = checkout ? new Date(checkout + 'T12:00:00') : undefined;
   164→   164→
   165→   165→  const displayText = from && to
   166→   166→    ? `${format(from, 'dd/MM/yyyy')} → ${format(to, 'dd/MM/yyyy')}`
   167→   167→    : from
   168→   168→      ? `${format(from, 'dd/MM/yyyy')} → ...`
   169→   169→      : 'Seleccionar fechas';
   170→   170→
   171→   171→  return (
   172→   172→    <div className="grid gap-1.5">
   173→   173→      {label && <Label>{label} *</Label>}
   174→   174→      <Button
   175→   175→        variant="outline"
   176→   176→        className={cn(
   177→   177→          'w-full justify-start text-left font-normal h-9',
   178→   178→          !checkin && 'text-muted-foreground'
   179→   179→        )}
   180→   180→        disabled={disabled}
   181→   181→        onClick={() => handleOpenChange(true)}
   182→   182→      >
   183→   183→        <CalendarDays className="w-4 h-4 mr-2 text-muted-foreground" />
   184→   184→        {displayText}
   185→   185→      </Button>
   186→   186→      {open
   187→   187→        ? createPortal(
   188→   188→            <div className="fixed inset-0 z-[200] flex items-center justify-center bg-black/40" onClick={() => handleOpenChange(false)}>
   189→   189→              <div
   190→   190→                className="bg-card rounded-xl shadow-2xl border border-border pointer-events-auto"
   191→   191→                onClick={e => e.stopPropagation()}
   192→   192→                onMouseDown={e => e.stopPropagation()}
   193→   193→              >
   194→   194→                <Calendar
   195→   195→                  mode="range"
   196→   196→                  selected={localRange.from ? { from: localRange.from, to: localRange.to } as { from: Date; to: Date } : undefined}
   197→   197→                  onSelect={handleSelect}
   198→   198→                  numberOfMonths={2}
   199→   199→                  autoFocus
   200→   200→                  classNames={{
   201→   201→                    range_start: 'rounded-l-md bg-primary [&>button]:!bg-primary [&>button]:!text-white',
   202→   202→                    range_end: 'rounded-r-md bg-primary [&>button]:!bg-primary [&>button]:!text-white',
   203→   203→                    range_middle: 'rounded-none bg-primary [&>button]:!bg-primary [&>button]:!text-white',
   204→   204→                    day: 'relative w-full h-full p-0 text-center [&:first-child[data-selected=true]_button]:rounded-l-md [&:last-child[data-selected=true]_button]:rounded-r-md group/day aspect-square select-none',
   205→   205→                  }}
   206→   206→                />
   207→   207→                <div className="flex justify-end border-t border-border p-3">
   208→   208→                  <Button
   209→   209→                    size="sm"
   210→   210→                    onClick={handleConfirm}
   211→   211→                    disabled={!localRange.from}
   212→   212→                    className="bg-primary hover:bg-primary/80 text-white text-xs h-8"
   213→   213→                  >
   214→   214→                    Confirmar fechas
   215→   215→                  </Button>
   216→   216→                </div>
   217→   217→              </div>
   218→   218→            </div>,
   219→   219→            document.body
   220→   220→          )
   221→   221→        : null}
   222→   222→    </div>
   223→   223→  );
   224→   224→}

   248→   248→function DesglosePrecio({ form, computed, formatMoney, s }: {
   249→   249→  form: { tipoTarifa: string; reservaMultiple: boolean; habitacion: string; habitacion2?: string; personas: string; personas2?: string };
   250→   250→  computed: { noches: number; subtotal: number; subtotal2: number; desglose: any };
   251→   251→  formatMoney: (n: number) => string;
   252→   252→  s: (n: number) => string;
   253→   253→}) {
   254→   254→  const d = computed.desglose;
   255→   255→
   256→   256→  const adultoLinea = (d: any) => {
   257→   257→    if (!d || d.totalAdultos === 0) return null;
   258→   258→    const modo = d.modoCobro;
   259→   259→    if (modo === 'porCama') {
   260→   260→      return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''} × ${formatMoney(d.precioRango)}/cama/noche`;
   261→   261→    }
   262→   262→    if (modo === 'porHabitacion') return 'Habitación';
   263→   263→    if (modo === 'porPersona') {
   264→   264→      return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''} × ${formatMoney(d.precioRango)}/noche`;
   265→   265→    }
   266→   266→    // porGrupo
   267→   267→    return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''}`;
   268→   268→  };
   269→   269→
   270→   270→  const renderLines = (sub: number, dLocal: any) => {
   271→   271→    if (!dLocal) return (
   272→   272→      <div className="space-y-1">
   273→   273→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-slate-400">Subtotal</span><span className="font-semibold text-foreground">{formatMoney(sub)}</span></div>
   274→   274→      </div>
   275→   275→    );
   276→   276→
   277→   277→    const lines: { key: string; label: string; value: string; dim?: boolean }[] = [];
   278→   278→
   279→   279→    const aLabel = adultoLinea(dLocal);
   280→   280→    if (aLabel) {
   281→   281→      lines.push({ key: 'adultos', label: aLabel, value: formatMoney(dLocal.totalAdultos) });
   282→   282→    }
   283→   283→
   284→   284→    if (dLocal.ninosCount > 0 && dLocal.totalNinos > 0) {
   285→   285→      lines.push({
   286→   286→        key: 'ninos',
   287→   287→        label: `${dLocal.ninosCount} niño${dLocal.ninosCount > 1 ? 's' : ''} × ${formatMoney(dLocal.precioNino)}/noche`,
   288→   288→        value: formatMoney(dLocal.totalNinos),
   289→   289→      });
   290→   290→    }
   291→   291→
   292→   292→    if (dLocal.nochesGratis > 0) {
   293→   293→      const totalBase = dLocal.totalAdultos + dLocal.totalNinos;
   294→   294→      const ahorro = totalBase > 0 && dLocal.nochesCobrables > 0
   295→   295→        ? Math.round(totalBase / dLocal.nochesCobrables * dLocal.nochesGratis)
   296→   296→        : 0;
   297→   297→      lines.push({
   298→   298→        key: 'cortesia',
   299→   299→        label: `${dLocal.nochesGratis} noche${dLocal.nochesGratis > 1 ? 's' : ''} de cortesía`,
   300→   300→        value: `- ${formatMoney(ahorro)}`,
   301→   301→        dim: true,
   302→   302→      });
   303→   303→    }
   304→   304→
   305→   305→    return (
   306→   306→      <div className="space-y-1">
   307→   307→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-slate-400">Tarifa</span><span className="font-semibold text-foreground capitalize">{form.tipoTarifa}</span></div>
   308→   308→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-slate-400">Noches</span><span className="font-semibold text-foreground">{computed.noches}{dLocal?.nochesGratis > 0 ? ` (${dLocal.nochesCobrables} cobrables)` : ''}</span></div>
   309→   309→        {lines.map(l => (
   310→   310→          <div key={l.key} className="flex justify-between items-center py-1 text-[13px]">
   311→   311→            <span className={l.dim ? 'text-slate-400 line-through' : 'text-slate-400'}>{l.label}</span>
   312→   312→            <span className={l.dim ? 'font-medium text-slate-400' : 'font-semibold text-foreground'}>{l.value}</span>
   313→   313→          </div>
   314→   314→        ))}
   315→   315→        <div className="flex justify-between items-center py-1.5 mt-1 border-t border-border text-[13px]">
   316→   316→          <span className="font-medium text-slate-600">Subtotal</span>
   317→   317→          <span className="font-bold text-foreground">{formatMoney(sub)}</span>
   318→   318→        </div>
   319→   319→      </div>
   320→   320→    );
   321→   321→  };
   322→   322→
   323→   323→  return (
   324→   324→    <>
   325→   325→      {form.reservaMultiple ? (
   326→   326→        <>
   327→   327→          <p className="text-[11px] font-semibold text-slate-400 uppercase tracking-wider">Habitación {form.habitacion}</p>
   328→   328→          {renderLines(computed.subtotal, d)}
   329→   329→          {computed.subtotal2 > 0 && (
   330→   330→            <>
   331→   331→              <p className="text-[11px] font-semibold text-slate-400 uppercase tracking-wider pt-2">Habitación {form.habitacion2}</p>
   332→   332→              {renderLines(computed.subtotal2, null)}
   333→   333→            </>
   334→   334→          )}
   335→   335→        </>
   336→   336→      ) : (
   337→   337→        renderLines(computed.subtotal, d)
   338→   338→      )}
   339→   339→    </>
   340→   340→  );
   341→   341→}
