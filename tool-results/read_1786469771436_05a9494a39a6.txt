   157→function WeatherIcon({ code }: { code: number }) {
   158→  if (code === 0) return <Sun className="w-5 h-5 text-[#F59E0B]" />;
   159→  if (code >= 1 && code <= 3) return <CloudSun className="w-5 h-5 text-[#94A3B8]" />;
   160→  if (code >= 45 && code <= 48) return <CloudFog className="w-5 h-5 text-[#94A3B8]" />;
   161→  if (code >= 51 && code <= 55) return <CloudDrizzle className="w-5 h-5 text-[#3B82F6]" />;
   162→  if (code >= 56 && code <= 57) return <CloudDrizzle className="w-5 h-5 text-[#3B82F6]" />;
   163→  if (code >= 61 && code <= 67) return <CloudRain className="w-5 h-5 text-[#3B82F6]" />;
   164→  if (code >= 71 && code <= 77) return <CloudSnow className="w-5 h-5 text-[#3B82F6]" />;
   165→  if (code >= 80 && code <= 82) return <CloudLightning className="w-5 h-5 text-[#92400E]" />;
   166→  if (code >= 95) return <CloudLightning className="w-5 h-5 text-[#5B21B6]" />;
   167→  return <Thermometer className="w-5 h-5 text-[#94A3B8]" />;
   168→}

   172→function LiveClockWeather() {
   173→  const [time, setTime] = useState('');
   174→  const [weather, setWeather] = useState<{ temp: number; code: number } | null>(null);
   175→
   176→  useEffect(() => {
   177→    const update = () => {
   178→      setTime(new Date().toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit', second: '2-digit' }));
   179→    };
   180→    update();
   181→    const id = setInterval(update, 1000);
   182→    return () => clearInterval(id);
   183→  }, []);
   184→
   185→  // TODO: Hacer coordenadas configurables desde la configuración del hotel
   186→  useEffect(() => {
   187→    const fetchWeather = async () => {
   188→      try {
   189→        const res = await fetch('https://api.open-meteo.com/v1/forecast?latitude=-27.65&longitude=-67.03&current_weather=true&timezone=America/Argentina/Catamarca');
   190→        if (!res.ok) return;
   191→        const data = await res.json();
   192→        setWeather({ temp: data.current_weather.temperature, code: data.current_weather.weathercode });
   193→      } catch { /* ignore */ }
   194→    };
   195→    fetchWeather();
   196→    const id = setInterval(fetchWeather, 30 * 60 * 1000);
   197→    return () => clearInterval(id);
   198→  }, []);
   199→
   200→  return (
   201→    <div className="flex items-center gap-4">
   202→      {weather && (
   203→        <div className="flex items-center gap-1.5 text-sm text-muted-foreground">
   204→          <WeatherIcon code={weather.code} />
   205→          <span className="font-semibold">{weather.temp}°C</span>
   206→        </div>
   207→      )}
   208→      <div className="font-mono text-lg font-semibold tabular-nums tracking-wide text-[#334155]">
   209→        {time}
   210→      </div>
   211→    </div>
   212→  );
   213→}

   359→function GraficoIngresosEgresos({ pagos, gastos }: { pagos: { fecha: string; monto: number }[]; gastos: { fecha: string; monto: number }[] }) {
   360→  const [mounted, setMounted] = useState(false);
   361→  useEffect(() => { const t = setTimeout(() => setMounted(true), 50); return () => clearTimeout(t); }, []);
   362→
   363→  const datos = useMemo(() => {
   364→    const hoy = new Date();
   365→    const dias: string[] = [];
   366→    for (let i = 6; i >= 0; i--) {
   367→      const d = new Date(hoy);
   368→      d.setDate(d.getDate() - i);
   369→      dias.push(toLocalDateStr(d));
   370→    }
   371→    return dias.map(dia => {
   372→      const ing = pagos.filter(p => p.fecha === dia).reduce((s, p) => s + p.monto, 0);
   373→      const egr = gastos.filter(g => g.fecha === dia).reduce((s, g) => s + g.monto, 0);
   374→      const d = new Date(dia + 'T12:00:00');
   375→      return { dia: d.toLocaleDateString('es-AR', { weekday: 'short', day: 'numeric' }).replace(/^\w/, c => c.toUpperCase()), ing, egr, fullDate: formatearFecha(dia) };
   376→    });
   377→  }, [pagos, gastos]);
   378→
   379→  const maxVal = Math.max(...datos.map(d => Math.max(d.ing, d.egr)), 1);
   380→
   381→  return (
   382→    <Card>
   383→      <CardHeader className="pb-3">
   384→        <CardTitle className="text-base flex items-center gap-2">
   385→          <BarChart3 className="w-4 h-4 text-[#3B82F6]" />
   386→          Ingresos vs Egresos (7 días)
   387→        </CardTitle>
   388→      </CardHeader>
   389→      <CardContent>
   390→        <div className="flex items-end gap-2" style={{ height: 160 }}>
   391→          {datos.map((d, i) => (
   392→            <div key={i} className="flex-1 flex flex-col items-center h-full">
   393→              <Tooltip text={`Ingresos: ${formatMoney(d.ing)}`}>
   394→                <div
   395→                  className="w-5 rounded-t-md transition-all duration-700 ease-out hover:opacity-80 cursor-default"
   396→                  style={{
   397→                    height: mounted ? `${(d.ing / maxVal) * 100}%` : '0%',
   398→                    background: 'linear-gradient(to top, #059669, #34d399)',
   399→                    boxShadow: '0 2px 8px rgba(5,150,105,0.3)',
   400→                    transitionDelay: `${i * 80}ms`,
   401→                  }}
   402→                />
   403→              </Tooltip>
   404→              <Tooltip text={`Egresos: ${formatMoney(d.egr)}`}>
   405→                <div
   406→                  className="w-5 rounded-t-md transition-all duration-700 ease-out hover:opacity-80 cursor-default mt-0.5"
   407→                  style={{
   408→                    height: mounted ? `${(d.egr / maxVal) * 100}%` : '0%',
   409→                    background: 'linear-gradient(to top, #dc2626, #fca5a5)',
   410→                    boxShadow: '0 2px 8px rgba(220,38,38,0.3)',
   411→                    transitionDelay: `${i * 80 + 40}ms`,
   412→                  }}
   413→                />
   414→              </Tooltip>
   415→              <span className="text-[9px] text-muted-foreground mt-1.5 font-medium">{d.dia.slice(0, 3)}</span>
   416→            </div>
   417→          ))}
   418→        </div>
   419→        <div className="flex justify-end gap-4 mt-3">
   420→          <span className="text-xs text-[#166534] flex items-center gap-1.5 font-medium">
   421→            <span className="inline-block w-3 h-3 rounded-sm" style={{ background: 'linear-gradient(to top, #059669, #34d399)' }} />
   422→            Ingresos
   423→          </span>

   426→            Egresos
   427→          </span>
   428→        </div>

   430→    </Card>

   461→      d.setDate(d.getDate() - i);

   465→      reservas.forEach(r => {
   466→        if (r.estado === 'Cancelada') return;
   467→        if (r.checkin <= diaStr && r.checkout > diaStr) {
   468→          ocupadasSet.add(r.habitacion);
   469→        }
   470→      });
   471→      const ocupadas = ocupadasSet.size;
   472→      const porcentaje = totalHabitaciones > 0 ? Math.round((ocupadas / totalHabitaciones) * 100) : 0;
   473→      const weekday = NOMBRES_DIAS[d.getDay()];
   474→      out.push({
   475→        fecha: diaStr,
   476→        ocupadas,
   477→        porcentaje,
   478→        label: `${weekday} ${d.getDate()}`,
   479→      });

   481→    return out;
   482→  }, [reservas, totalHabitaciones]);

   485→  const ayerData = datos[datos.length - 2];
   486→  const ultimos7 = datos.slice(-7);
   487→  const promedio7d = ultimos7.length > 0 ? ultimos7.reduce((s, d) => s + d.porcentaje, 0) / ultimos7.length : 0;
   488→
   489→  const diffVsAyer = hoyData && ayerData ? hoyData.porcentaje - ayerData.porcentaje : 0;
   490→  const trendUp = diffVsAyer > 0;
   491→  const trendDown = diffVsAyer < 0;

   495→  const renderTooltip = ({ active, payload }: TooltipProps<number, string>) => {

   526→              <span>{Math.abs(diffVsAyer)}%</span>
   527→              <span className="text-muted-foreground font-normal text-[10px]">vs ayer</span>
   528→            </div>
   529→          )}
   530→        </div>
   531→      </CardHeader>
   532→      <CardContent>
   533→        <div style={{ height: 220, minHeight: 200 }} className="w-full">
   534→          <ResponsiveContainer width="100%" height="100%">
   535→            <AreaChart data={datos} margin={{ top: 5, right: 10, bottom: 0, left: -10 }}>
   536→              <defs>
   537→                <linearGradient id="ocupGradient" x1="0" y1="0" x2="0" y2="1">
   538→                  <stop offset="0%" stopColor="#34d399" stopOpacity={0.85} />
   539→                  <stop offset="100%" stopColor="#059669" stopOpacity={0.15} />
   540→                </linearGradient>
   541→              </defs>
   542→              <XAxis
   543→                dataKey="label"
   544→                tick={{ fontSize: 10, fill: '#94A3B8' }}
   545→                tickLine={false}
   546→                axisLine={{ stroke: '#E2E8F0' }}
   547→                interval={0}
   548→              />
   549→              <YAxis
   550→                domain={[0, 100]}
   551→                tick={{ fontSize: 10, fill: '#94A3B8' }}
   552→                tickLine={false}
   553→                axisLine={false}
   554→                tickFormatter={(v) => `${v}%`}
   555→                width={36}
   556→              />
   557→              <RechartsTooltip
   558→                content={renderTooltip}
   559→                cursor={{ stroke: '#059669', strokeOpacity: 0.3, strokeWidth: 1 }}
   560→              />
   561→              <Area
   562→                type="monotone"
   563→                dataKey="porcentaje"
   564→                stroke="#059669"

   569→                isAnimationActive={mounted}
   570→                animationDuration={800}
   571→              />
   572→            </AreaChart>
   573→          </ResponsiveContainer>
   574→        </div>
   575→      </CardContent>
   576→    </Card>

   895→function calcularBarra(res: GanttReserva, colIdx: Record<string, number>, columnas: string[], DIAS: number) {
   896→  const COL_PCT = 100 / DIAS;
   897→  const MITAD_COL_PCT = COL_PCT / 2;
   898→  let startCol = Math.max(colIdx[res.checkin] ?? 0, 0);
   899→  let endCol = Math.min(colIdx[res.checkout] !== undefined ? colIdx[res.checkout] : DIAS - 1, DIAS - 1);
   900→
   901→  const esCheckinTarde = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckin && res.horaCheckin.getHours() >= 12;
   902→  const esCheckoutManana = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckout && res.horaCheckout.getHours() < 12;
   903→
   904→  let leftPct: number, widthPct: number;
   905→
   906→  if (esCheckinTarde && startCol >= 0 && !(res.checkin < columnas[0])) {
   907→    leftPct = startCol * COL_PCT + MITAD_COL_PCT;
   908→  } else {
   909→    leftPct = startCol * COL_PCT;
   910→  }
   911→
   912→  if (esCheckoutManana && endCol >= 0 && !(res.checkout > columnas[DIAS - 1])) {
   913→    widthPct = ((endCol - startCol) * COL_PCT) + (esCheckinTarde ? MITAD_COL_PCT : COL_PCT) - (esCheckoutManana ? MITAD_COL_PCT : 0);
   914→  } else {
   915→    widthPct = (endCol - startCol + 1) * COL_PCT;
   916→  }
   917→
   918→  widthPct = Math.max(widthPct, MITAD_COL_PCT);
   919→  return { leftPct, widthPct };
   920→}

   924→function RoomHeatmap({ habitaciones, reservas }: {

   963→                ? `${num} - ${hab.problema || 'Mantenimiento'}`
   964→                : `${num} - ${hab.estado} · Cap. ${hab.capacidad} · ${hab.tipo}`;

   970→                  {info && <div className="text-[9px] font-normal opacity-70 truncate mt-0.5">{info.huesped}</div>}

   978→            <span key={label} className="flex items-center gap-1.5"><span className={`w-2.5 h-2.5 rounded-full ${dot}`} />{label}</span>

   982→          </span>
   983→        </div>
