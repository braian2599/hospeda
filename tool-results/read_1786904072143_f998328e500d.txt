   149→function formatoRango(r: RangoPrecio): string {
   150→  if (r.maxPersonas === null) return `${r.minPersonas}+`;
   151→  if (r.minPersonas === r.maxPersonas) return `${r.minPersonas}`;
   152→  return `${r.minPersonas}-${r.maxPersonas}`;
   153→}

   181→// Count active promotions on a tariff
   182→function countPromos(promos?: PromocionesTarifa | null): number {
   183→  if (!promos) return 0;
   184→  let n = 0;
   185→  if (promos.acompananteSinCargo?.activo) n++;
   186→  if (promos.ninosDiferenciado?.activo) n++;
   187→  if (promos.nochesCortesia?.activo) n++;
   188→  return n;
   189→}

   191→// Compute "Desde" price (min positive price, fallback to first range)
   192→function precioDesde(rangos: RangoPrecio[]): number {
   193→  const preciosPositivos = rangos.map(r => r.precio).filter(p => p > 0);
   194→  return preciosPositivos.length > 0 ? Math.min(...preciosPositivos) : (rangos[0]?.precio || 0);
   195→}

   197→// Build a label describing a noches cortesía modalidad
   198→function describeNochesCortesia(mod: ModalidadNochesCortesia): string {
   199→  if (mod.tipo === 'cadaX') return `Cada ${mod.cada} noches, 1 gratis`;
   200→  if (mod.tipo === 'aPartirDe') return `Desde ${mod.minNoches} noches, ${mod.nochesGratis} gratis`;
   201→  const dias = ['Domingo', 'Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado'];
   202→  return `${dias[mod.dia]} gratis`;
   203→}

   205→// Export a single tariff as CSV (client-side download)
   206→function exportTariffCSV(tipo: string, t: TarifaPrecios) {
   207→  const rows: (string | number)[][] = [];
   208→  rows.push(['Campo', 'Valor']);
   209→  rows.push(['Nombre', tipo]);
   210→  rows.push(['Modo de cobro', t.modoCobro || 'porGrupo']);
   211→  const rangos = t.rangos || [];
   212→  rows.push(['Cantidad de rangos', rangos.length]);
   213→  rangos.forEach((r, i) => {
   214→    rows.push([`Rango ${i + 1} (${formatoRango(r)} pers.)`, r.precio]);
   215→  });
   216→  rows.push(['Acompañante sin cargo', t.promociones?.acompananteSinCargo?.activo ? 'Sí' : 'No']);
   217→  if (t.promociones?.acompananteSinCargo?.activo) {
   218→    rows.push(['  Etiqueta', t.promociones.acompananteSinCargo.etiqueta || '—']);
   219→    rows.push(['  Cantidad', t.promociones.acompananteSinCargo.cantidad ?? 1]);
   220→  }
   221→  rows.push(['Niños diferenciado', t.promociones?.ninosDiferenciado?.activo ? 'Sí' : 'No']);
   222→  if (t.promociones?.ninosDiferenciado?.activo) {
   223→    rows.push(['  Precio niño/noche', t.promociones.ninosDiferenciado.precioNino || 0]);
   224→  }
   225→  rows.push(['Noches cortesía', t.promociones?.nochesCortesia?.activo ? 'Sí' : 'No']);
   226→  if (t.promociones?.nochesCortesia?.activo) {
   227→    rows.push(['  Modalidad', describeNochesCortesia(t.promociones.nochesCortesia.modalidad)]);
   228→  }
   229→  const campos = t.camposPersonalizados || [];
   230→  rows.push(['Cantidad de campos personalizados', campos.length]);
   231→  campos.forEach((c, i) => {
   232→    rows.push([`  Campo ${i + 1}`, `${c.nombre} (${c.tipo}${c.requerido ? ', requerido' : ''})`]);
   233→  });
   234→
   235→  const csv = rows.map(r => r.map(cell => {
   236→    const s = String(cell);
   237→    return s.includes(',') || s.includes('"') ? `"${s.replace(/"/g, '""')}"` : s;
   238→  }).join(',')).join('\n');
   239→
   240→  const blob = new Blob([`\uFEFF${csv}`], { type: 'text/csv;charset=utf-8;' });
   241→  const url = URL.createObjectURL(blob);
   242→  const a = document.createElement('a');
   243→  a.href = url;
   244→  a.download = `tarifa_${tipo.toLowerCase().replace(/\s+/g, '_')}.csv`;
   245→  document.body.appendChild(a);
   246→  a.click();
   247→  document.body.removeChild(a);
   248→  URL.revokeObjectURL(url);
   249→}

   253→function WizardStepper({ current, onSelect }: { current: 1 | 2 | 3; onSelect: (s: 1 | 2 | 3) => void }) {
   254→  const steps: { n: 1 | 2 | 3; label: string; icon: typeof Info }[] = [
   255→    { n: 1, label: 'Información básica', icon: Info },
   256→    { n: 2, label: 'Rangos de precios', icon: Tags },
   257→    { n: 3, label: 'Promociones', icon: Sparkles },
   258→  ];
   259→  return (
   260→    <div className="space-y-2">
   261→      <div className="flex items-center gap-1">
   262→        {steps.map((s, i) => {
   263→          const Icon = s.icon;
   264→          const isDone = current > s.n;
   265→          const isActive = current === s.n;
   266→          return (
   267→            <div key={s.n} className="flex items-center flex-1 last:flex-none">
   268→              <button
   269→                type="button"
   270→                onClick={() => onSelect(s.n)}
   271→                className={`flex items-center gap-2 px-3 py-1.5 rounded-md text-xs font-semibold transition-all whitespace-nowrap ${
   272→                  isActive
   273→                    ? 'bg-primary text-white shadow-sm'
   274→                    : isDone
   275→                      ? 'bg-primary/10 text-primary hover:bg-primary/20'
   276→                      : 'bg-muted text-muted-foreground hover:bg-muted/80'
   277→                }`}
   278→                aria-current={isActive ? 'step' : undefined}
   279→              >
   280→                <span className={`w-5 h-5 rounded-full flex items-center justify-center text-[10px] ${
   281→                  isActive ? 'bg-white/20' : isDone ? 'bg-primary text-white' : 'bg-background'
   282→                }`}>
   283→                  {isDone ? <Check className="w-3 h-3" /> : s.n}
   284→                </span>
   285→                <span className="hidden sm:inline">{s.label}</span>
   286→                <Icon className="w-3.5 h-3.5 sm:hidden" />
   287→              </button>
   288→              {i < steps.length - 1 && (
   289→                <div className={`flex-1 h-0.5 mx-1 rounded ${isDone ? 'bg-primary' : 'bg-muted'}`} />
   290→              )}
   291→            </div>
   292→          );
   293→        })}
   294→      </div>
   295→      <Progress value={(current / 3) * 100} className="h-1 bg-muted" />
   296→    </div>
   297→  );
   298→}

   403→function ComparisonModal({ tariffs, onClose }: {
   404→  tariffs: { tipo: string; t: TarifaPrecios }[];
   405→  onClose: () => void;
   406→}) {
   407→  // Compute base price per person for each tariff (min positive price)
   408→  const basePrices = tariffs.map(({ t }) => {
   409→    const rangos = t.rangos || [];
   410→    const pos = rangos.map(r => r.precio).filter(p => p > 0);
   411→    return pos.length > 0 ? Math.min(...pos) : (rangos[0]?.precio || 0);
   412→  });
   413→
   414→  // Determine which rows have differing values for highlighting
   415→  const modos = tariffs.map(({ t }) => t.modoCobro || 'porGrupo');
   416→  const modoDiffers = new Set(modos).size > 1;
   417→  const priceDiffers = new Set(basePrices).size > 1;
   418→  const rangoCounts = tariffs.map(({ t }) => (t.rangos || []).length);
   419→  const rangoDiffers = new Set(rangoCounts).size > 1;
   420→  const promoCounts = tariffs.map(({ t }) => countPromos(t.promociones));
   421→  const promoDiffers = new Set(promoCounts).size > 1;
   422→  const campoCounts = tariffs.map(({ t }) => (t.camposPersonalizados || []).length);
   423→  const campoDiffers = new Set(campoCounts).size > 1;
   424→
   425→  return (
   426→    <DialogContent className="sm:max-w-4xl max-h-[90vh] overflow-y-auto">
   427→      <DialogHeader>
   428→        <DialogTitle className="flex items-center gap-2 text-xl">
   429→          <GitCompareArrows className="w-5 h-5 text-primary" />
   430→          Comparación de tarifas
   431→        </DialogTitle>
   432→        <p className="text-sm text-muted-foreground mt-1">
   433→          Comparando {tariffs.length} tarifas. Las filas con diferencias están resaltadas.
   434→        </p>
   435→      </DialogHeader>
