    40→    40→
    41→    41→function useCountUp(target: number, duration = 800, enabled = true) {
    42→    42→  const [value, setValue] = useState(0);
    43→    43→  const prevTarget = useRef(0);
    44→    44→
    45→    45→  useEffect(() => {
    46→    46→    if (!enabled) { setValue(target); return; }
    47→    47→    const startValue = prevTarget.current;
    48→    48→    const diff = target - startValue;
    49→    49→    if (Math.abs(diff) < 1) { setValue(target); prevTarget.current = target; return; }
    50→    50→    const startTime = performance.now();
    51→    51→    let raf: number;
    52→    52→    const animate = (now: number) => {
    53→    53→      const elapsed = now - startTime;
    54→    54→      const progress = Math.min(elapsed / duration, 1);
    55→    55→      const eased = 1 - Math.pow(1 - progress, 3); // ease-out cubic
    56→    56→      setValue(Math.round(startValue + diff * eased));
    57→    57→      if (progress < 1) { raf = requestAnimationFrame(animate); }
    58→    58→      else { prevTarget.current = target; }
    59→    59→    };
    60→    60→    raf = requestAnimationFrame(animate);
    61→    61→    return () => cancelAnimationFrame(raf);
    62→    62→  }, [target, duration, enabled]);
    63→    63→
    64→    64→  return value;
    65→    65→}

    69→    69→function KPIAnimated({ icon: Icon, label, value, sub, color, bgGradient, trend, numeric = false }: {

   688→   688→function calcularBarra(res: GanttReserva, colIdx: Record<string, number>, columnas: string[], DIAS: number) {
   689→   689→  let startCol = Math.max(colIdx[res.checkin] ?? 0, 0);
   690→   690→  let endCol = Math.min(colIdx[res.checkout] !== undefined ? colIdx[res.checkout] : DIAS - 1, DIAS - 1);
   691→   691→
   692→   692→  const esCheckinTarde = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckin && res.horaCheckin.getHours() >= 12;
   693→   693→  const esCheckoutManana = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckout && res.horaCheckout.getHours() < 12;
   694→   694→
   695→   695→  let leftPct: number, widthPct: number;
   696→   696→
   697→   697→  if (esCheckinTarde && startCol >= 0 && !(res.checkin < columnas[0])) {
   698→   698→    leftPct = startCol * COL_PCT + MITAD_COL_PCT;
   699→   699→  } else {
   700→   700→    leftPct = startCol * COL_PCT;
   701→   701→  }
   702→   702→
   703→   703→  if (esCheckoutManana && endCol >= 0 && !(res.checkout > columnas[DIAS - 1])) {
   704→   704→    widthPct = ((endCol - startCol) * COL_PCT) + (esCheckinTarde ? MITAD_COL_PCT : COL_PCT) - (esCheckoutManana ? MITAD_COL_PCT : 0);
   705→   705→  } else {
   706→   706→    widthPct = (endCol - startCol + 1) * COL_PCT;
   707→   707→  }
   708→   708→
   709→   709→  return { leftPct, widthPct };
   710→   710→}
