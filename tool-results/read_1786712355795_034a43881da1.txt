    31→

    33→  Pendiente: 'bg-[#FEF3C7] text-[#92400E] border-[#FDE68A]',

    52→
    53→export default function CheckInModule() {
    54→  // ── Granular Zustand selectors (no destructuring) ──
    55→  const reservas = useHotelStore(s => s.reservas);
    56→  const habitaciones = useHotelStore(s => s.habitaciones);
    57→  const realizarCheckIn = useHotelStore(s => s.realizarCheckIn);
    58→  const realizarCheckOut = useHotelStore(s => s.realizarCheckOut);
    59→  const calcularTotalReserva = useHotelStore(s => s.calcularTotalReserva);
    60→  const calcularTotalPagado = useHotelStore(s => s.calcularTotalPagado);
    61→  const nochesEntre = useHotelStore(s => s.nochesEntre);
    62→
    63→  const pendientesCheckIn = useMemo(
    64→    () => reservas.filter(r => r.estado === 'Confirmada'),
    65→    [reservas]
    66→  );
    67→  const pendientesCheckOut = useMemo(
    68→    () => reservas.filter(r => r.estado === 'Check-In realizado'),
    69→    [reservas]
    70→  );
    71→
    72→  // ── Local loading state (brief, for skeleton display) ──
    73→  // The store has no `loading` flag; we simulate a brief fetch on mount so
    74→  // skeleton placeholders are visible to the user.
    75→  const [loading, setLoading] = useState(true);
    76→  useEffect(() => {
    77→    const t = setTimeout(() => setLoading(false), 400);
    78→    return () => clearTimeout(t);
    79→  }, []);
    80→
    81→  // Modal state
    82→  const [modalTipo, setModalTipo] = useState<'checkin' | 'checkout' | null>(null);
    83→  const [selReserva, setSelReserva] = useState<Reserva | null>(null);
    84→  const [dialogKey, setDialogKey] = useState(0);
    85→
    86→  // Check-in form
    87→  const [docVerificado, setDocVerificado] = useState(false);
    88→  const [firmaConformidad, setFirmaConformidad] = useState(false);
    89→  const [llave, setLlave] = useState('');
    90→  const [emergenciaNombre, setEmergenciaNombre] = useState('');
    91→  const [emergenciaTelefono, setEmergenciaTelefono] = useState('');
    92→  const [observaciones, setObservaciones] = useState('');
    93→  const [acompanantes, setAcompanantes] = useState<Acompanante[]>([
    94→    { nombre: '', dni: '', celular: '' },
    95→  ]);
    96→
    97→  // Menores form state
    98→  const [menoresForms, setMenoresForms] = useState<MenorForm[]>([]);
    99→  const [menoresErrors, setMenoresErrors] = useState<string[]>([]);
   100→
   101→  const openCheckIn = (r: Reserva) => {
   102→    setSelReserva(r);
   103→    setDocVerificado(false);
   104→    setFirmaConformidad(false);
   105→    setLlave('');
   106→    setEmergenciaNombre(r.contactoEmergencia?.nombre || '');
   107→    setEmergenciaTelefono(r.contactoEmergencia?.telefono || '');
   108→    setObservaciones(r.observacionesHuesped || '');
   109→    setAcompanantes(r.acompanantes?.length ? [...r.acompanantes] : [{ nombre: '', dni: '', celular: '' }]);
   110→
   111→    // Si ya tiene menores registrados (check-in previo), precargar
   112→    if (r.menores && r.menores.length > 0) {
   113→      setMenoresForms(r.menores.map(m => ({
   114→        nombre: m.nombre,
   115→        documento: m.documento,
   116→        edad: String(m.edad),
   117→        parentesco: m.parentesco,
   118→      })));
   119→    } else if ((r.ninos || 0) > 0) {
   120→      // Generar formularios vacíos según la cantidad de menores declarados
   121→      const forms: MenorForm[] = [];
   122→      for (let i = 0; i < r.ninos!; i++) {
   123→        forms.push(emptyMenorForm());
   124→      }
   125→      setMenoresForms(forms);
   126→    } else {
   127→      setMenoresForms([]);
   128→    }
   129→    setMenoresErrors([]);
   130→
   131→    setModalTipo('checkin');
   132→  };
   133→
   134→  const openCheckOut = (r: Reserva) => {
   135→    setSelReserva(r);
   136→    setModalTipo('checkout');
   137→  };
   138→
   139→  const closeModal = () => {
   140→    setModalTipo(null);
   141→    setSelReserva(null);
   142→    setDialogKey(prev => prev + 1);
   143→  };
   144→
   145→  const validateMenores = (): boolean => {
   146→    const errs: string[] = [];
   147→    for (let i = 0; i < menoresForms.length; i++) {
   148→      const m = menoresForms[i];
   149→      if (!m.nombre.trim()) errs.push(`Menor ${i + 1}: nombre obligatorio`);
   150→      if (!m.documento.trim()) errs.push(`Menor ${i + 1}: documento obligatorio`);
   151→      const edad = parseInt(m.edad);
   152→      if (isNaN(edad) || edad < 0 || edad > 17) errs.push(`Menor ${i + 1}: edad debe ser entre 0 y 17`);
   153→      if (!m.parentesco.trim()) errs.push(`Menor ${i + 1}: parentesco obligatorio`);
   154→    }
   155→    setMenoresErrors(errs);
   156→    return errs.length === 0;
   157→  };
   158→
   159→  const [checkLoading, setCheckLoading] = useState(false);
   160→
   161→  const handleConfirmCheckIn = async () => {
   162→    if (!selReserva || !llave.trim()) return;
   163→
   164→    // Validar menores si la reserva tiene ninos > 0
   165→    const cantNinos = selReserva.ninos || 0;
   166→    if (cantNinos > 0) {
   167→      if (menoresForms.length !== cantNinos) {
   168→        toast.error(`Se requieren datos de ${cantNinos} menor${cantNinos > 1 ? 'es' : ''}`);
   169→        return;
   170→      }
   171→      if (!validateMenores()) {
   172→        toast.error('Completá todos los datos obligatorios de los menores');
   173→        return;
   174→      }
   175→    }
   176→
   177→    const datos: Parameters<typeof realizarCheckIn>[1] = {};
   178→    if (docVerificado) datos.documentoVerificado = true;
   179→    if (firmaConformidad) datos.firmaConformidad = true;
   180→    datos.llaveEntregada = llave.trim();
   181→    if (emergenciaNombre.trim() || emergenciaTelefono.trim()) {
   182→      datos.contactoEmergencia = { nombre: emergenciaNombre.trim(), telefono: emergenciaTelefono.trim() };
   183→    }
   184→    if (observaciones.trim()) datos.observacionesHuesped = observaciones.trim();
   185→    const validAcomp = acompanantes.filter(a => a.nombre.trim());
   186→    if (validAcomp.length > 0) datos.acompanantes = validAcomp;
   187→
   188→    // Pasar menores al store
   189→    if (cantNinos > 0 && menoresForms.length > 0) {
   190→      datos.menores = menoresForms.map(m => ({
   191→        nombre: m.nombre.trim(),
   192→        documento: m.documento.trim(),
   193→        edad: parseInt(m.edad) || 0,
   194→        parentesco: m.parentesco.trim(),
   195→      }));
   196→    }
   197→
   198→    try {
   199→      setCheckLoading(true);
   200→      const resultado = await realizarCheckIn(selReserva.id, datos);
   201→      if (resultado) {
   202→        closeModal();
   203→        const desc = `${selReserva.huesped} - Hab. ${selReserva.habitacion}${cantNinos > 0 ? ` (${cantNinos} menor${cantNinos > 1 ? 'es' : ''} registrados)` : ''}`;
   204→        notifySuccess('Check-in realizado', desc);
   205→      } else {
   206→        toast.error('No se pudo realizar el check-in', { description: 'La reserva ya no está en estado Confirmada, faltan datos de menores, o hubo un error de conexión.' });
   207→      }
   208→    } catch (err) {
   209→      console.error('Error en check-in:', err);
   210→      toast.error('Error al realizar check-in');
   211→    } finally {
   212→      setCheckLoading(false);
   213→    }
   214→  };
   215→
   216→  const handleConfirmCheckOut = async () => {
   217→    if (!selReserva) return;
   218→    try {
   219→      setCheckLoading(true);
   220→      const resultado = await realizarCheckOut(selReserva.id);
   221→      if (resultado) {
   222→        closeModal();
   223→        notifySuccess('Check-out realizado', `${selReserva.huesped} - Hab. ${selReserva.habitacion}`);
   224→      } else {
   225→        toast.error('No se pudo realizar el check-out', { description: 'La reserva ya no está en estado Check-In realizado o hubo un error de conexión.' });
   226→      }
   227→    } catch (err) {
   228→      console.error('Error en check-out:', err);
   229→      toast.error('Error al realizar check-out');
   230→    } finally {
   231→      setCheckLoading(false);
   232→    }
   233→  };
   234→
   235→  const addAcompanante = () => {
   236→    setAcompanantes(prev => [...prev, { nombre: '', dni: '', celular: '' }]);
   237→  };
   238→
   239→  const removeAcompanante = (idx: number) => {
   240→    setAcompanantes(prev => prev.filter((_, i) => i !== idx));
   241→  };
   242→
   243→  const updateAcompanante = (idx: number, field: keyof Acompanante, value: string) => {
   244→    setAcompanantes(prev => prev.map((a, i) => (i === idx ? { ...a, [field]: value } : a)));
   245→  };
   246→
   247→  const updateMenor = (idx: number, field: keyof MenorForm, value: string) => {
   248→    setMenoresForms(prev => prev.map((m, i) => (i === idx ? { ...m, [field]: value } : m)));
   249→  };
   250→
   251→  // Determinar si la reserva seleccionada requiere datos de menores
   252→  const requiereMenores = selReserva && (selReserva.ninos || 0) > 0;
   253→  const menoresCompletos = requiereMenores && menoresForms.length === (selReserva?.ninos || 0)
   254→    && menoresForms.every(m => m.nombre.trim() && m.documento.trim() && m.edad.trim() && m.parentesco.trim());
   255→
   256→  const puedeConfirmarCheckIn = llave.trim() && (!requiereMenores || menoresCompletos);
   257→
   258→  return (
   259→    <div className="space-y-6">
   260→      <ModuleHeader icon={LogIn} title="Check-In / Check-Out" subtitle="Gestioná ingresos y egresos de huéspedes" />
   261→
   262→      {/* ═══════════ TODAY'S ACTIVITY SUMMARY ═══════════ */}
   263→      {loading ? (
   264→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   265→          <StatCardSkeleton />
   266→          <StatCardSkeleton />
   267→          <StatCardSkeleton />
   268→        </div>
   269→      ) : (
   270→        <TodayActivitySummary />
   271→      )}
   272→
   273→      {/* ═══════════ CHECK-IN / CHECK-OUT CARDS ═══════════ */}
   274→      {loading ? (
   275→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   276→          <Card>
   277→            <CardHeader className="pb-3">
   278→              <Skeleton className="h-6 w-48" />
   279→            </CardHeader>
   280→            <CardContent className="space-y-3">
   281→              <ListItemSkeleton />
   282→              <ListItemSkeleton />
   283→              <ListItemSkeleton />
   284→            </CardContent>
   285→          </Card>
   286→          <Card>
   287→            <CardHeader className="pb-3">
   288→              <Skeleton className="h-6 w-48" />
   289→            </CardHeader>
   290→            <CardContent className="space-y-3">
   291→              <ListItemSkeleton />
   292→              <ListItemSkeleton />
   293→            </CardContent>
   294→          </Card>
   295→        </div>
   296→      ) : (
   297→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   298→          {/* Pendientes Check-In */}
   299→          <Card className="border-emerald-700/40/60 bg-emerald-950/20">
   300→            <CardHeader className="pb-3">
   301→              <CardTitle className="text-lg flex items-center gap-2">
   302→                <LogIn className="w-5 h-5 text-[#059669]" />
   303→                Check-Ins Pendientes
   304→                {pendientesCheckIn.length > 0 && <PulsingDot color="bg-emerald-500" />}
   305→                <Badge
   306→                  key={`cin-${pendientesCheckIn.length}`}
   307→                  variant="secondary"
   308→                  className="ml-auto count-pop bg-[#059669]/15 text-[#047857] border-emerald-700/40 hover:bg-[#059669]/20"
   309→                >
   310→                  {pendientesCheckIn.length}
   311→                </Badge>
   312→              </CardTitle>
   313→            </CardHeader>
   314→            <CardContent>
   315→              {pendientesCheckIn.length === 0 ? (
   316→                <CelebratoryEmptyState />
   317→              ) : (
   318→                <div className="space-y-3 max-h-[28rem] overflow-y-auto pr-1">
   319→                  {pendientesCheckIn.map(r => (
   320→                    <div key={r.id} className="border rounded-lg p-3 flex items-center justify-between gap-3 hover:bg-[#ECFDF5]/40 transition-colors duration-200 group">
   321→                      <div className="min-w-0 flex-1">
   322→                        <div className="flex items-center gap-2 flex-wrap">
   323→                          <span className="font-semibold text-sm group-hover:text-emerald-400 transition-colors">{r.huesped}</span>
   324→                          <Badge className={`font-semibold shadow-sm ${estadoPagoBadge[r.estadoPago] || ''}`}>{r.estadoPago}</Badge>
   325→                          {(r.ninos || 0) > 0 && (
   326→                            <Badge variant="outline" className="bg-[#F5F3FF] text-[#5B21B6] border-[#DDD6FE]">
   327→                              <Baby className="w-3 h-3 mr-1" />{r.ninos} menor{(r.ninos || 0) > 1 ? 'es' : ''}
   328→                            </Badge>
   329→                          )}
   330→                        </div>
   331→                        <div className="text-xs text-muted-foreground mt-1 flex flex-wrap gap-x-3 gap-y-0.5">
   332→                          <span className="flex items-center gap-1"><BedDouble className="w-3 h-3" /> Hab. {r.habitacion}</span>
   333→                          <span>{formatFecha(r.checkin)} → {formatFecha(r.checkout)}</span>
   334→                          <span>{nochesEntre(r.checkin, r.checkout)} noche{s(nochesEntre(r.checkin, r.checkout))}</span>
   335→                          <span>{r.personas} adulto{s(r.personas)}{(r.ninos || 0) > 0 ? ` + ${r.ninos} niño${(r.ninos || 0) > 1 ? 's' : ''}` : ''}</span>
   336→                        </div>
   337→                      </div>
   338→                      <Button size="sm" className="bg-[#059669] hover:bg-[#047857] text-white shadow-sm" onClick={() => openCheckIn(r)}>
   339→                        <KeyRound className="w-3.5 h-3.5 mr-1" />Check-In
   340→                      </Button>
   341→                    </div>
   342→                  ))}
   343→                </div>
   344→              )}
   345→            </CardContent>
   346→          </Card>
   347→
   348→          {/* Pendientes Check-Out */}
   349→          <Card className="border-[#FED7AA]/60 bg-amber-950/20">
   350→            <CardHeader className="pb-3">
   351→              <CardTitle className="text-lg flex items-center gap-2">
   352→                <LogOut className="w-5 h-5 text-[#EA580C]" />
   353→                Check-Outs Pendientes
   354→                {pendientesCheckOut.length > 0 && <PulsingDot color="bg-orange-500" />}
   355→                <Badge
   356→                  key={`cout-${pendientesCheckOut.length}`}
   357→                  variant="secondary"
   358→                  className="ml-auto count-pop bg-[#EA580C]/15 text-[#9A3412] border-[#FED7AA] hover:bg-[#EA580C]/20"
   359→                >
   360→                  {pendientesCheckOut.length}
   361→                </Badge>
   362→              </CardTitle>
   363→            </CardHeader>
   364→            <CardContent>
   365→              {pendientesCheckOut.length === 0 ? (
   366→                <CelebratoryEmptyState />
   367→              ) : (
   368→                <div className="space-y-3 max-h-[28rem] overflow-y-auto pr-1">
   369→                  {pendientesCheckOut.map(r => (
   370→                    <div key={r.id} className="border rounded-lg p-3 flex items-center justify-between gap-3 hover:bg-[#FFF7ED]/40 transition-colors duration-200 group">
   371→                      <div className="min-w-0 flex-1">
   372→                        <div className="flex items-center gap-2 flex-wrap">
   373→                          <span className="font-semibold text-sm group-hover:text-emerald-400 transition-colors">{r.huesped}</span>
   374→                          <Badge className={`font-semibold shadow-sm ${estadoPagoBadge[r.estadoPago] || ''}`}>{r.estadoPago}</Badge>
   375→                          {r.menores && r.menores.length > 0 && (
   376→                            <Badge variant="outline" className="bg-[#F5F3FF] text-[#5B21B6] border-[#DDD6FE]">
   377→                              <Baby className="w-3 h-3 mr-1" />{r.menores.length} menor{r.menores.length > 1 ? 'es' : ''}
   378→                            </Badge>
   379→                          )}
   380→                        </div>
   381→                        <div className="text-xs text-muted-foreground mt-1 flex flex-wrap gap-x-3 gap-y-0.5">
   382→                          <span className="flex items-center gap-1"><BedDouble className="w-3 h-3" /> Hab. {r.habitacion}</span>
   383→                          <span>{formatFecha(r.checkin)} → {formatFecha(r.checkout)}</span>
   384→                          <span>{nochesEntre(r.checkin, r.checkout)} noche{s(nochesEntre(r.checkin, r.checkout))}</span>
   385→                        </div>
   386→                      </div>
   387→                      <Button size="sm" variant="outline" onClick={() => openCheckOut(r)}>
   388→                        <LogOut className="w-3.5 h-3.5 mr-1" />Check-Out
   389→                      </Button>
   390→                    </div>
   391→                  ))}
   392→                </div>
   393→              )}
   394→            </CardContent>
   395→          </Card>
   396→        </div>
   397→      )}
   398→
   399→      {/* =================== MODAL CHECK-IN =================== */}
   400→      <Dialog key={`checkin-${dialogKey}`} open={modalTipo === 'checkin'} onOpenChange={closeModal}>
   401→        <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   402→          {selReserva && (
   403→            <>
   404→              <DialogHeader>
   405→                <DialogTitle className="flex items-center gap-2">
   406→                  <KeyRound className="w-5 h-5" />
   407→                  Check-In — {selReserva.huesped}
   408→                </DialogTitle>
   409→              </DialogHeader>
   410→
   411→              {/* Estado de cuenta */}
   412→              <CheckInAccountStatus reserva={selReserva} />
   413→
   414→              <Separator />
   415→
   416→              {/* ═══════════ DATOS DE MENORES (OBLIGATORIO si ninos > 0) ═══════════ */}
   417→              {requiereMenores && (
   418→                <div className="space-y-4">
   419→                  <div className="flex items-center gap-2">
   420→                    <Baby className="w-5 h-5 text-[#5B21B6]" />
   421→                    <h4 className="font-semibold text-sm">
   422→                      Datos de menores ({selReserva.ninos}) — <span className="text-[#EF4444]">Obligatorio</span>
   423→                    </h4>
   424→                  </div>
   425→                  <p className="text-xs text-muted-foreground">
   426→                    La reserva registra {selReserva.ninos} menor{selReserva.ninos! > 1 ? 'es' : ''}.
   427→                    Debes completar los datos de cada uno para poder realizar el check-in.
   428→                  </p>
   429→
   430→                  {menoresErrors.length > 0 && (
   431→                    <div className="rounded-lg border border-[#FECACA] bg-[#FEE2E2] p-3 space-y-1">
   432→                      {menoresErrors.map((err, i) => (
   433→                        <p key={i} className="text-xs text-[#991B1B]">{err}</p>
   434→                      ))}
   435→                    </div>
   436→                  )}
   437→
   438→                  <div className="space-y-4">
   439→                    {menoresForms.map((menor, idx) => (
   440→                      <div key={idx} className="rounded-lg border p-4 space-y-3 bg-muted/30">
   441→                        <div className="flex items-center justify-between">
   442→                          <span className="text-sm font-semibold flex items-center gap-1.5">
   443→                            <Baby className="w-4 h-4 text-[#7C3AED]" />
   444→                            Menor {idx + 1}
   445→                          </span>
   446→                        </div>
   447→                        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   448→                          <div className="grid gap-2">
   449→                            <Label htmlFor={`menor-nombre-${idx}`}>Nombre completo *</Label>
   450→                            <Input
   451→                              id={`menor-nombre-${idx}`}
   452→                              value={menor.nombre}
   453→                              onChange={e => updateMenor(idx, 'nombre', e.target.value)}
   454→                              placeholder="Nombre del menor"
   455→                            />

   752→/**
   753→ * TodayActivitySummary
   754→ *
   755→ * Banner con 3 stat cards mostrando la actividad de hoy:
   756→ *  - Check-ins completados hoy
   757→ *  - Check-outs completados hoy
   758→ *  - Estadías activas
   759→ *
   760→ * Usa selectores Zustand granulares (sin destructuring) y `todayLocal` / `daysAgo`
   761→ * de `@/lib/format` para evitar drift UTC.
   762→ */
   763→function TodayActivitySummary() {
   764→  // Granular selectors (no destructuring) — each subscribes to its own slice.
   765→  const reservas = useHotelStore(s => s.reservas);
   766→  const habitaciones = useHotelStore(s => s.habitaciones);
   767→
   768→  const hoyStr = todayLocal();
   769→  const ayerStr = daysAgo(1);
   770→
   771→  // ── Check-ins completados hoy (estado === 'Check-In realizado' AND checkin === hoy) ──
   772→  const checkinsHoy = reservas.filter(
   773→    r => r.estado === 'Check-In realizado' && r.checkin === hoyStr
   774→  ).length;
   775→  const checkinsAyer = reservas.filter(
   776→    r => r.estado === 'Check-In realizado' && r.checkin === ayerStr
   777→  ).length;
   778→
   779→  // ── Check-outs completados hoy (estado === 'Check-Out realizado' AND checkout === hoy) ──
   780→  const checkoutsHoy = reservas.filter(
   781→    r => r.estado === 'Check-Out realizado' && r.checkout === hoyStr
   782→  ).length;
   783→  const checkoutsAyer = reservas.filter(
   784→    r => r.estado === 'Check-Out realizado' && r.checkout === ayerStr
   785→  ).length;
   786→
   787→  // ── Estadías activas (snapshot: estado === 'Check-In realizado') ──
   788→  const estadiasActivas = reservas.filter(
   789→    r => r.estado === 'Check-In realizado'
   790→  ).length;
   791→  // Snapshot aproximado de "estadías activas antes de los check-outs de hoy"
   792→  // = activas actuales + las que hicieron check-out hoy.
   793→  const estadiasPrevias = reservas.filter(
   794→    r => r.estado === 'Check-In realizado'
   795→      || (r.estado === 'Check-Out realizado' && r.checkout === hoyStr)
   796→  ).length;
   797→
   798→  // Ocupación (% de habitaciones ocupadas) — usa `habitaciones` selector.
   799→  const totalHabitaciones = Object.keys(habitaciones).length;
   800→  const ocupadas = Object.values(habitaciones).filter(h => h.estado === 'Ocupada').length;
   801→  const ocupacionPct = totalHabitaciones === 0
   802→    ? null
   803→    : Math.round((ocupadas / totalHabitaciones) * 100);
   804→
   805→  const stats: Array<{
   806→    key: string;
   807→    label: string;
   808→    value: number;
   809→    icon: LucideIcon;
   810→    colorName: string;
   811→    trend?: number;
   812→    sublabel?: string;
   813→  }> = [
   814→    {
   815→      key: 'checkins',
   816→      label: 'Check-ins completados hoy',
   817→      value: checkinsHoy,
   818→      icon: LogIn,
   819→      colorName: 'emerald',
   820→      trend: checkinsHoy === 0 && checkinsAyer === 0 ? undefined : checkinsHoy - checkinsAyer,
   821→    },
   822→    {
   823→      key: 'checkouts',
   824→      label: 'Check-outs completados hoy',
   825→      value: checkoutsHoy,
   826→      icon: LogOut,
   827→      colorName: 'amber',
   828→      trend: checkoutsHoy === 0 && checkoutsAyer === 0 ? undefined : checkoutsHoy - checkoutsAyer,
   829→    },
   830→    {
   831→      key: 'estadias',
   832→      label: 'Estadías activas',
   833→      value: estadiasActivas,
   834→      icon: Bed,
   835→      colorName: 'teal',
   836→      trend: estadiasActivas === 0 && estadiasPrevias === 0 ? undefined : estadiasActivas - estadiasPrevias,
   837→      sublabel: ocupacionPct !== null ? `${ocupacionPct}% ocup.` : undefined,
   838→    },
   839→  ];
   840→
   841→  return (
   842→    <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   843→      {stats.map((s, i) => (

   845→          key={s.key}
   846→          label={s.label}
   847→          value={s.value}
   848→          icon={s.icon}
   849→          colorName={s.colorName}
   850→          trend={s.trend}
   851→          sublabel={s.sublabel}
   852→          delay={i * 80}
   853→        />
   854→      ))}
