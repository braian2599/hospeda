   106→function DateRangePickerInline({
   107→  checkin,
   108→  checkout,
   109→  onChangeCheckin,
   110→  onChangeCheckout,
   111→  label,
   112→  disabled,
   113→}: {
   114→  checkin: string;
   115→  checkout: string;
   116→  onChangeCheckin: (val: string) => void;
   117→  onChangeCheckout: (val: string) => void;
   118→  label?: string;
   119→  disabled?: boolean;
   120→}) {
   121→  const [open, setOpen] = useState(false);
   122→  const [localRange, setLocalRange] = useState<{ from?: Date; to?: Date }>({});
   123→  const syncedRef = useRef(false);
   124→
   125→  // Sync local state with the form when the popover opens (event-driven, not effect-driven).
   126→  const handleOpenChange = (next: boolean) => {
   127→    setOpen(next);
   128→    if (next && !syncedRef.current) {
   129→      syncedRef.current = true;
   130→      const f = checkin ? new Date(checkin + 'T12:00:00') : undefined;
   131→      const t = checkout ? new Date(checkout + 'T12:00:00') : undefined;
   132→      setLocalRange(f ? { from: f, to: t } : {});
   133→    }
   134→    if (!next) syncedRef.current = false;
   135→  };
   136→
   137→  const handleSelect = (range: { from?: Date; to?: Date } | undefined) => {
   138→    if (!range?.from) {
   139→      setLocalRange({});
   140→      return;
   141→    }
   142→    setLocalRange(range);
   143→  };
   144→
   145→  const fmt = (d: Date) => {
   146→    const yyyy = d.getFullYear();
   147→    const mm = String(d.getMonth() + 1).padStart(2, '0');
   148→    const dd = String(d.getDate()).padStart(2, '0');
   149→    return `${yyyy}-${mm}-${dd}`;
   150→  };
   151→
   152→  const handleConfirm = () => {
   153→    if (!localRange.from) return;
   154→    onChangeCheckin(fmt(localRange.from));
   155→    onChangeCheckout(localRange.to ? fmt(localRange.to) : '');
   156→    handleOpenChange(false);
   157→  };
   158→
   159→  const from = checkin ? new Date(checkin + 'T12:00:00') : undefined;
   160→  const to = checkout ? new Date(checkout + 'T12:00:00') : undefined;
   161→
   162→  const displayText = from && to
   163→    ? `${format(from, 'dd/MM/yyyy')} → ${format(to, 'dd/MM/yyyy')}`
   164→    : from
   165→      ? `${format(from, 'dd/MM/yyyy')} → ...`
   166→      : 'Seleccionar fechas';
   167→
   168→  return (
   169→    <div className="grid gap-1.5">
   170→      {label && <Label>{label} *</Label>}
   171→      <Button
   172→        variant="outline"
   173→        className={cn(
   174→          'w-full justify-start text-left font-normal h-9',
   175→          !checkin && 'text-muted-foreground'
   176→        )}
   177→        disabled={disabled}
   178→        onClick={() => handleOpenChange(true)}
   179→      >
   180→        <CalendarDays className="w-4 h-4 mr-2 text-muted-foreground" />
   181→        {displayText}
   182→      </Button>
   183→      {open
   184→        ? createPortal(
   185→            <div className="fixed inset-0 z-[200] flex items-center justify-center bg-black/40" onClick={() => handleOpenChange(false)}>
   186→              <div
   187→                className="bg-white rounded-xl shadow-2xl border border-[#E2E8F0] pointer-events-auto"
   188→                onClick={e => e.stopPropagation()}
   189→                onMouseDown={e => e.stopPropagation()}
   190→              >
   191→                <Calendar
   192→                  mode="range"
   193→                  selected={localRange.from ? { from: localRange.from, to: localRange.to } as { from: Date; to: Date } : undefined}
   194→                  onSelect={handleSelect}
   195→                  numberOfMonths={2}
   196→                  autoFocus
   197→                  classNames={{
   198→                    range_start: 'rounded-l-md bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   199→                    range_end: 'rounded-r-md bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   200→                    range_middle: 'rounded-none bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   201→                    day: 'relative w-full h-full p-0 text-center [&:first-child[data-selected=true]_button]:rounded-l-md [&:last-child[data-selected=true]_button]:rounded-r-md group/day aspect-square select-none',
   202→                  }}
   203→                />
   204→                <div className="flex justify-end border-t border-[#E2E8F0] p-3">
   205→                  <Button
   206→                    size="sm"
   207→                    onClick={handleConfirm}
   208→                    disabled={!localRange.from}
   209→                    className="bg-[#059669] hover:bg-[#047857] text-white text-xs h-8"
   210→                  >
   211→                    Confirmar fechas
   212→                  </Button>
   213→                </div>
   214→              </div>
   215→            </div>,
   216→            document.body
   217→          )
   218→        : null}
   219→    </div>
   220→  );
   221→}

   244→// ==================== DESGLOSE PRECIO COMPONENT ====================
   245→function DesglosePrecio({ form, computed, formatMoney, s }: {
   246→  form: { tipoTarifa: string; reservaMultiple: boolean; habitacion: string; habitacion2?: string; personas: string; personas2?: string };
   247→  computed: { noches: number; subtotal: number; subtotal2: number; desglose: any };
   248→  formatMoney: (n: number) => string;
   249→  s: (n: number) => string;
   250→}) {
   251→  const d = computed.desglose;
   252→
   253→  const adultoLinea = (d: any) => {
   254→    if (!d || d.totalAdultos === 0) return null;
   255→    const modo = d.modoCobro;
   256→    if (modo === 'porCama') {
   257→      return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''} × ${formatMoney(d.precioRango)}/cama/noche`;
   258→    }
   259→    if (modo === 'porHabitacion') return 'Habitación';
   260→    if (modo === 'porPersona') {
   261→      return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''} × ${formatMoney(d.precioRango)}/noche`;
   262→    }
   263→    // porGrupo
   264→    return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''}`;
   265→  };
   266→
   267→  const renderLines = (sub: number, dLocal: any) => {
   268→    if (!dLocal) return (
   269→      <div className="space-y-1">
   270→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Subtotal</span><span className="font-semibold text-[#1E293B]">{formatMoney(sub)}</span></div>
   271→      </div>
   272→    );
   273→
   274→    const lines: { key: string; label: string; value: string; dim?: boolean }[] = [];
   275→
   276→    const aLabel = adultoLinea(dLocal);
   277→    if (aLabel) {
   278→      lines.push({ key: 'adultos', label: aLabel, value: formatMoney(dLocal.totalAdultos) });
   279→    }
   280→
   281→    if (dLocal.ninosCount > 0 && dLocal.totalNinos > 0) {
   282→      lines.push({
   283→        key: 'ninos',
   284→        label: `${dLocal.ninosCount} niño${dLocal.ninosCount > 1 ? 's' : ''} × ${formatMoney(dLocal.precioNino)}/noche`,
   285→        value: formatMoney(dLocal.totalNinos),
   286→      });
   287→    }
   288→
   289→    if (dLocal.nochesGratis > 0) {
   290→      const totalBase = dLocal.totalAdultos + dLocal.totalNinos;
   291→      const ahorro = totalBase > 0 && dLocal.nochesCobrables > 0
   292→        ? Math.round(totalBase / dLocal.nochesCobrables * dLocal.nochesGratis)
   293→        : 0;
   294→      lines.push({
   295→        key: 'cortesia',
   296→        label: `${dLocal.nochesGratis} noche${dLocal.nochesGratis > 1 ? 's' : ''} de cortesía`,
   297→        value: `- ${formatMoney(ahorro)}`,
   298→        dim: true,
   299→      });
   300→    }
   301→
   302→    return (
   303→      <div className="space-y-1">
   304→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Tarifa</span><span className="font-semibold text-[#1E293B] capitalize">{form.tipoTarifa}</span></div>
   305→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Noches</span><span className="font-semibold text-[#1E293B]">{computed.noches}{dLocal?.nochesGratis > 0 ? ` (${dLocal.nochesCobrables} cobrables)` : ''}</span></div>
   306→        {lines.map(l => (
   307→          <div key={l.key} className="flex justify-between items-center py-1 text-[13px]">
   308→            <span className={l.dim ? 'text-[#94A3B8] line-through' : 'text-[#64748B]'}>{l.label}</span>
   309→            <span className={l.dim ? 'font-medium text-[#94A3B8]' : 'font-semibold text-[#1E293B]'}>{l.value}</span>
   310→          </div>
   311→        ))}
   312→        <div className="flex justify-between items-center py-1.5 mt-1 border-t border-[#E2E8F0] text-[13px]">
   313→          <span className="font-medium text-[#475569]">Subtotal</span>
   314→          <span className="font-bold text-[#1E293B]">{formatMoney(sub)}</span>
   315→        </div>
   316→      </div>
   317→    );
   318→  };
   319→
   320→  return (
   321→    <>
   322→      {form.reservaMultiple ? (
   323→        <>
   324→          <p className="text-[11px] font-semibold text-[#64748B] uppercase tracking-wider">Habitación {form.habitacion}</p>
   325→          {renderLines(computed.subtotal, d)}
   326→          {computed.subtotal2 > 0 && (
   327→            <>
   328→              <p className="text-[11px] font-semibold text-[#64748B] uppercase tracking-wider pt-2">Habitación {form.habitacion2}</p>
   329→              {renderLines(computed.subtotal2, null)}
   330→            </>
   331→          )}
   332→        </>
   333→      ) : (
   334→        renderLines(computed.subtotal, d)
   335→      )}
   336→    </>
   337→  );
   338→}
