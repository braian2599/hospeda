    47→function KPIAnimated({ icon: Icon, label, value, sub, color, bgGradient, trend, numericValue, suffix }: {
    48→  icon: ComponentType<{ className?: string }>;
    49→  label: string;
    50→  value: string;
    51→  sub?: string;
    52→  color: string;
    53→  bgGradient?: string;
    54→  trend?: { value: number; label: string };
    55→  numericValue?: number;
    56→  suffix?: string;
    57→}) {
    58→  const trendUp = trend && trend.value > 0;
    59→  const trendDown = trend && trend.value < 0;
    60→  const trendIcon = trendUp ? '\u2191' : trendDown ? '\u2193' : '';
    61→  const trendColor = trendUp ? 'text-[#4ADE80]' : trendDown ? 'text-[#EF4444]' : 'text-muted-foreground';
    62→
    63→  return (
    64→    <Card className={`overflow-hidden group hover:shadow-lg hover:-translate-y-0.5 transition-all duration-200 !py-0 !gap-0 border-0 ${bgGradient || 'bg-gradient-to-br from-slate-50 to-slate-100'}`}>
    65→      <CardContent className="p-4">
    66→        <div className="flex items-start justify-between">
    67→          <div className="min-w-0 flex-1">
    68→            <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-semibold">{label}</p>
    69→            <p className="text-2xl font-extrabold mt-1 text-[#0F172A]">
    70→              {numericValue !== undefined ? (
    71→                <><AnimatedNumber value={numericValue} duration={600} format={suffix === '%' ? (n: number) => `${Math.round(n)}%` : (n: number) => String(Math.round(n))} className="text-2xl font-extrabold text-[#0F172A]" /></>
    72→              ) : value}
    73→            </p>
    74→            {sub && <p className="text-xs text-muted-foreground mt-0.5">{sub}</p>}
    75→          </div>
    76→          <div className="p-2.5 rounded-xl bg-white/80 shadow-sm group-hover:scale-110 group-hover:shadow-md transition-all duration-200">
    77→            <Icon className={`w-5 h-5 ${color}`} />
    78→          </div>
    79→        </div>
    80→        {trend && (
    81→          <div className={`flex items-center gap-1 mt-2 text-xs font-medium ${trendColor}`}>
    82→            <span>{trendIcon}</span>
    83→            <span>{Math.abs(trend.value)}%</span>
    84→            <span className="text-muted-foreground font-normal">{trend.label}</span>
    85→          </div>
    86→        )}
    87→      </CardContent>
    88→    </Card>
    89→  );
    90→}

    94→function WeatherIcon({ code }: { code: number }) {
    95→  if (code === 0) return <Sun className="w-5 h-5 text-[#F59E0B]" />;
    96→  if (code >= 1 && code <= 3) return <CloudSun className="w-5 h-5 text-[#94A3B8]" />;
    97→  if (code >= 45 && code <= 48) return <CloudFog className="w-5 h-5 text-[#94A3B8]" />;
    98→  if (code >= 51 && code <= 55) return <CloudDrizzle className="w-5 h-5 text-[#3B82F6]" />;
    99→  if (code >= 56 && code <= 57) return <CloudDrizzle className="w-5 h-5 text-[#3B82F6]" />;
   100→  if (code >= 61 && code <= 67) return <CloudRain className="w-5 h-5 text-[#3B82F6]" />;
   101→  if (code >= 71 && code <= 77) return <CloudSnow className="w-5 h-5 text-[#3B82F6]" />;
   102→  if (code >= 80 && code <= 82) return <CloudLightning className="w-5 h-5 text-[#92400E]" />;
   103→  if (code >= 95) return <CloudLightning className="w-5 h-5 text-[#5B21B6]" />;
   104→  return <Thermometer className="w-5 h-5 text-[#94A3B8]" />;
   105→}

   296→function GraficoIngresosEgresos({ pagos, gastos }: { pagos: { fecha: string; monto: number }[]; gastos: { fecha: string; monto: number }[] }) {

   299→
   300→  const datos = useMemo(() => {
   301→    const hoy = new Date();
   302→    const dias: string[] = [];
   303→    for (let i = 6; i >= 0; i--) {
   304→      const d = new Date(hoy);
   305→      d.setDate(d.getDate() - i);
   306→      dias.push(toLocalDateStr(d));

   309→      const ing = pagos.filter(p => p.fecha === dia).reduce((s, p) => s + p.monto, 0);

   371→// ==================== TENDENCIA DE OCUPACIÓN (14 días) ====================
   372→
   373→interface OccupancyDayData {
   374→  fecha: string;
   375→  ocupadas: number;
   376→  porcentaje: number;
   377→  label: string;
   378→}
   379→
   380→function OccupancyTrendChart() {
   381→  // Granular Zustand selectors (no destructuring) — only re-renders when these slices change
   382→  const habitaciones = useHotelStore(s => s.habitaciones);
   383→  const reservas = useHotelStore(s => s.reservas);
   384→
   385→  const [mounted, setMounted] = useState(false);
   386→  useEffect(() => {
   387→    const t = setTimeout(() => setMounted(true), 60);
   388→    return () => clearTimeout(t);
   389→  }, []);
   390→
   391→  const totalHabitaciones = Object.keys(habitaciones).length;
   392→
   393→  const datos = useMemo<OccupancyDayData[]>(() => {
   394→    const hoy = new Date();
   395→    const out: OccupancyDayData[] = [];
   396→    for (let i = 13; i >= 0; i--) {
   397→      const d = new Date(hoy);
   398→      d.setDate(d.getDate() - i);
   399→      const diaStr = toLocalDateStr(d);
   400→      // Count unique rooms occupied on this day (checkin <= day < checkout, excluding cancelled)
   401→      const ocupadasSet = new Set<string>();
   402→      reservas.forEach(r => {
   403→        if (r.estado === 'Cancelada') return;
   404→        if (r.checkin <= diaStr && r.checkout > diaStr) {
   405→          ocupadasSet.add(r.habitacion);
   406→        }
   407→      });
   408→      const ocupadas = ocupadasSet.size;
   409→      const porcentaje = totalHabitaciones > 0 ? Math.round((ocupadas / totalHabitaciones) * 100) : 0;
   410→      const weekday = NOMBRES_DIAS[d.getDay()];
   411→      out.push({
   412→        fecha: diaStr,
   413→        ocupadas,
   414→        porcentaje,
   415→        label: `${weekday} ${d.getDate()}`,
   416→      });
   417→    }
   418→    return out;
   419→  }, [reservas, totalHabitaciones]);
   420→
   421→  const hoyData = datos[datos.length - 1];
   422→  const ayerData = datos[datos.length - 2];
   423→  const ultimos7 = datos.slice(-7);
   424→  const promedio7d = ultimos7.length > 0 ? ultimos7.reduce((s, d) => s + d.porcentaje, 0) / ultimos7.length : 0;
   425→
   426→  const diffVsAyer = hoyData && ayerData ? hoyData.porcentaje - ayerData.porcentaje : 0;
   427→  const trendUp = diffVsAyer > 0;
   428→  const trendDown = diffVsAyer < 0;
   429→  const trendIcon = trendUp ? '\u2191' : trendDown ? '\u2193' : '\u2192';
   430→  const trendColor = trendUp ? 'text-[#166534]' : trendDown ? 'text-[#991B1B]' : 'text-muted-foreground';
   431→
   432→  const renderTooltip = ({ active, payload }: TooltipProps<number, string>) => {
   433→    if (!active || !payload || !payload.length) return null;
   434→    const d = payload[0]?.payload as OccupancyDayData | undefined;
   435→    if (!d) return null;
   436→    return (
   437→      <div className="bg-[#0F2B28] text-white px-3 py-2 rounded-lg shadow-xl text-xs border border-[#059669]/30">
   438→        <p className="font-semibold text-[#34d399]">{formatearFecha(d.fecha)}</p>
   439→        <p className="mt-0.5">Ocupación: <span className="font-bold">{d.porcentaje}%</span></p>
   440→        <p className="text-white/70">{d.ocupadas} de {totalHabitaciones} hab.</p>
   441→      </div>
   442→    );
   443→  };
   444→
   445→  return (
   446→    <Card className={cn('transition-opacity duration-500', mounted ? 'opacity-100' : 'opacity-0')}>

   821→function getBarColorClass(tipo: string): string {
   822→  const map: Record<string, string> = {
   823→    Reservada: 'bg-[#3B82F6] shadow-[0_2px_6px_rgba(59,130,246,0.35)]',
   824→    Ocupada: 'bg-[#4ADE80] shadow-[0_2px_6px_rgba(74,222,128,0.35)]',
   825→    Finalizada: 'bg-[#94A3B8] opacity-50 border border-dashed border-[#CBD5E1]',
   826→    Limpieza: 'bg-[#F59E0B] shadow-[0_2px_6px_rgba(245,158,11,0.30)]',
   827→    Mantenimiento: 'bg-[#64748B] shadow-[0_2px_6px_rgba(100,116,139,0.25)]',
   828→  };
   829→  return map[tipo] || map.Reservada;
   830→}
   831→
   832→function calcularBarra(res: GanttReserva, colIdx: Record<string, number>, columnas: string[], DIAS: number) {
   833→  const COL_PCT = 100 / DIAS;
   834→  const MITAD_COL_PCT = COL_PCT / 2;
   835→  let startCol = Math.max(colIdx[res.checkin] ?? 0, 0);
   836→  let endCol = Math.min(colIdx[res.checkout] !== undefined ? colIdx[res.checkout] : DIAS - 1, DIAS - 1);
   837→
   838→  const esCheckinTarde = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckin && res.horaCheckin.getHours() >= 12;
   839→  const esCheckoutManana = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckout && res.horaCheckout.getHours() < 12;
   840→
   841→  let leftPct: number, widthPct: number;
   842→
   843→  if (esCheckinTarde && startCol >= 0 && !(res.checkin < columnas[0])) {
   844→    leftPct = startCol * COL_PCT + MITAD_COL_PCT;
   845→  } else {
   846→    leftPct = startCol * COL_PCT;
   847→  }
   848→
   849→  if (esCheckoutManana && endCol >= 0 && !(res.checkout > columnas[DIAS - 1])) {
   850→    widthPct = ((endCol - startCol) * COL_PCT) + (esCheckinTarde ? MITAD_COL_PCT : COL_PCT) - (esCheckoutManana ? MITAD_COL_PCT : 0);
   851→  } else {
   852→    widthPct = (endCol - startCol + 1) * COL_PCT;
   853→  }
   854→
   855→  widthPct = Math.max(widthPct, MITAD_COL_PCT);
   856→  return { leftPct, widthPct };
   857→}

   861→function RoomHeatmap({ habitaciones, reservas }: {
   862→  habitaciones: Record<string, { tipo: string; estado: string; capacidad: number; problema?: string }>;

   916→          ))}

   919→          </span>
