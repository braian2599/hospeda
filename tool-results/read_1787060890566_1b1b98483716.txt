    68→    68→/** Relative time since a date — "recién", "hace 5 min", "hace 2h 15min", "hace 3d" */
    69→    69→function formatRelative(dateStr: string, now: number = Date.now()): string {
    70→    70→  const t = safeDate(dateStr).getTime();
    71→    71→  if (isNaN(t)) return '';
    72→    72→  const diff = Math.max(0, now - t);
    73→    73→  const mins = Math.floor(diff / 60000);
    74→    74→  if (mins < 1) return 'recién';
    75→    75→  if (mins < 60) return `hace ${mins} min`;
    76→    76→  const hours = Math.floor(mins / 60);
    77→    77→  const restMins = mins % 60;
    78→    78→  if (hours < 24) return restMins > 0 ? `hace ${hours}h ${restMins}min` : `hace ${hours}h`;
    79→    79→  const days = Math.floor(hours / 24);
    80→    80→  return `hace ${days}d`;
    81→    81→}

    84→    84→function formatTimeSinceOpen(openedAt: string, now: number = Date.now()): string {

    87→    87→  const diff = Math.max(0, now - t);

    90→    90→  if (mins < 60) return `abierta hace ${mins}min`;

    93→    93→  return restMins > 0 ? `abierta hace ${hours}h ${restMins}min` : `abierta hace ${hours}h`;
    94→    94→}

   145→   145→/**
   146→   146→ * Suggest a category based on the description text.
   147→   147→ * Returns null if no keyword matches.
   148→   148→ */
   149→   149→function suggestCategory(descripcion: string): MovementCategory | null {
   150→   150→  const lower = descripcion.toLowerCase().trim();
   151→   151→  if (!lower) return null;
   152→   152→  for (const rule of CATEGORY_KEYWORDS) {
   153→   153→    if (rule.keywords.some(kw => lower.includes(kw))) {
   154→   154→      return rule.category;
   155→   155→    }
   156→   156→  }
   157→   157→  return null;
   158→   158→}
   159→   159→
   160→   160→/**
   161→   161→ * Determine the category for a movement based on its fields:
   162→   162→ * - ingresos → "Ingresos varios"
   163→   163→ * - egresos with desc/gasto.tipo containing "mantenimiento" → "Mantenimiento"
   164→   164→ * - egresos with desc containing "retiro" → "Retiros"
   165→   165→ * - egresos with gastoId (linked gasto) → "Gastos"
   166→   166→ * - otherwise → "Otros"
   167→   167→ */
   168→   168→function categorizeMovement(
   169→   169→  mov: { tipo: 'ingreso' | 'egreso'; descripcion: string; gastoId?: string | null; categoria?: string },
   170→   170→  gastoTipo?: string,
   171→   171→): MovementCategory {
   172→   172→  if (mov.tipo === 'ingreso') {
   173→   173→    // All ingresos fall under 'Ingresos varios'
   174→   174→    return 'Ingresos varios';
   175→   175→  }
   176→   176→  // egreso — try to classify by keywords first
   177→   177→  const text = `${mov.descripcion} ${gastoTipo || ''} ${mov.categoria || ''}`.toLowerCase();
   178→   178→  if (text.includes('retiro')) return 'Retiros';
   179→   179→  if (text.includes('mantenim') || text.includes('reparac') || text.includes('arreglo')) return 'Mantenimiento';
   180→   180→  if (mov.gastoId) return 'Gastos';
   181→   181→  // Try suggestion from keywords for more specific classification
   182→   182→  const suggested = suggestCategory(mov.descripcion);
   183→   183→  if (suggested === 'Retiros' || suggested === 'Mantenimiento' || suggested === 'Gastos') return suggested;
   184→   184→  return 'Otros';
   185→   185→}
