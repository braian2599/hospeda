    31→    31→/** Calculate stay duration in days (checkout - checkin). Uses safeDate to avoid UTC drift. */
    32→    32→const calcDias = (checkin: string, checkout: string): number => {
    33→    33→  const c = safeDate(checkin);
    34→    34→  const o = safeDate(checkout);
    35→    35→  const diff = Math.round((o.getTime() - c.getTime()) / (1000 * 60 * 60 * 24));
    36→    36→  return diff > 0 ? diff : 0;
    37→    37→};
    38→    38→
    39→    39→/** Get initials from a full name (first letter of first word + first letter of last word) */
    40→    40→const getInitials = (name: string): string => {
    41→    41→  const parts = name.trim().split(/\s+/).filter(Boolean);
    42→    42→  if (parts.length === 0) return '?';
    43→    43→  if (parts.length === 1) return parts[0][0].toUpperCase();
    44→    44→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
    45→    45→};
    46→    46→
    47→    47→/** Loyalty level based on stay count */
    48→    48→type LoyaltyLevel = 'Nuevo' | 'Habitual' | 'Frecuente' | 'VIP';
    49→    49→
    50→    50→interface LoyaltyInfo {
    51→    51→  level: LoyaltyLevel;
    52→    52→  color: string;       // badge bg class
    53→    53→  textColor: string;   // badge text class
    54→    54→  borderColor: string; // left border color
    55→    55→  icon?: React.ReactNode;
    56→    56→}
    57→    57→
    58→    58→const getLoyaltyInfo = (stayCount: number): LoyaltyInfo => {
    59→    59→  if (stayCount >= 7) return {
    60→    60→    level: 'VIP',
    61→    61→    color: 'bg-emerald-100',
    62→    62→    textColor: 'text-emerald-700',
    63→    63→    borderColor: 'border-l-emerald-500',
    64→    64→    icon: <Star className="w-3 h-3 fill-emerald-500 text-emerald-500" />,
    65→    65→  };
    66→    66→  if (stayCount >= 4) return {
    67→    67→    level: 'Frecuente',
    68→    68→    color: 'bg-amber-100',
    69→    69→    textColor: 'text-amber-700',
    70→    70→    borderColor: 'border-l-amber-400',
    71→    71→  };
    72→    72→  if (stayCount >= 2) return {
    73→    73→    level: 'Habitual',
    74→    74→    color: 'bg-sky-100',
    75→    75→    textColor: 'text-sky-700',
    76→    76→    borderColor: 'border-l-sky-400',

    79→    79→    level: 'Nuevo',
    80→    80→    color: 'bg-gray-100',
    81→    81→    textColor: 'text-gray-600',
    82→    82→    borderColor: 'border-l-gray-300',
    83→    83→  };
    84→    84→};

    86→    86→/** Format a date as relative time in Spanish */
    87→    87→const formatRelativeTime = (dateStr: string): string => {
    88→    88→  if (!dateStr) return '';
    89→    89→  const d = safeDate(dateStr);
    90→    90→  const now = new Date();
    91→    91→  const diffMs = now.getTime() - d.getTime();
    92→    92→  const diffDays = Math.floor(diffMs / (1000 * 60 * 60 * 24));
    93→    93→
    94→    94→  if (diffDays < 0) return 'próximo';
    95→    95→  if (diffDays === 0) return 'hoy';
    96→    96→  if (diffDays === 1) return 'ayer';
    97→    97→  if (diffDays < 7) return `hace ${diffDays} días`;
    98→    98→  if (diffDays < 14) return 'hace 1 semana';
    99→    99→  if (diffDays < 30) return `hace ${Math.floor(diffDays / 7)} semanas`;
   100→   100→  if (diffDays < 60) return 'hace 1 mes';
   101→   101→  if (diffDays < 365) return `hace ${Math.floor(diffDays / 30)} meses`;
   102→   102→  if (diffDays < 730) return 'hace 1 año';
   103→   103→  return `hace ${Math.floor(diffDays / 365)} años`;
   104→   104→};
