    83→function RangoFila({ rango, index, onRemove, onUpdate, modoCobro, totalRangos }: {
    84→  rango: RangoPrecio;
    85→  index: number;
    86→  onRemove: () => void;
    87→  onUpdate: (r: RangoPrecio) => void;
    88→  modoCobro: ModoCobro;
    89→  totalRangos: number;
    90→}) {
    91→  const esInfinito = rango.maxPersonas === null;
    92→  // Etiqueta legible del rango: "1 persona", "2 personas", "3+ personas", "2-4 personas"
    93→  const rangoLabel = esInfinito
    94→    ? `${rango.minPersonas}+ persona${rango.minPersonas > 1 ? 's' : ''}`
    95→    : rango.minPersonas === rango.maxPersonas
    96→      ? `${rango.minPersonas} persona${rango.minPersonas > 1 ? 's' : ''}`
    97→      : `${rango.minPersonas} a ${rango.maxPersonas} personas`;
    98→
    99→  return (
   100→    <div className="flex items-end gap-2">
   101→      <div className="w-28 shrink-0 flex items-center h-8 text-sm font-medium text-foreground">
   102→        {modoCobro === 'porHabitacion' ? 'Por noche' : modoCobro === 'porCama' ? 'Por cama/noche' : rangoLabel}
   103→      </div>
   104→      <div className="space-y-0.5 flex-1">
   105→        <Label className="text-[10px] text-muted-foreground">
   106→          {modoCobro === 'porPersona' || modoCobro === 'porCama' ? 'Precio c/u' : 'Precio'}
   107→        </Label>
   108→        <Input
   109→          type="number" min={0} step="0.01"
   110→          className="h-8 text-right"
   111→          value={rango.precio}
   112→          onChange={e => onUpdate({ ...rango, precio: parseFloat(e.target.value) || 0 })}
   113→        />
   114→      </div>
   115→      <Button
   116→        size="icon" variant="ghost"
   117→        className="h-8 w-8 text-destructive shrink-0 mb-0.5"
   118→        onClick={onRemove}
   119→        disabled={modoCobro === 'porHabitacion' || modoCobro === 'porCama' || totalRangos <= 1}
   120→      >
   121→        <Trash2 className="w-3.5 h-3.5" />
   122→      </Button>
   123→    </div>
   124→  );
   125→}

   136→function formatoRango(r: RangoPrecio): string {
   137→ if (r.maxPersonas === null) return `${r.minPersonas}+`;
   138→  if (r.minPersonas === r.maxPersonas) return `${r.minPersonas}`;
   139→  return `${r.minPersonas}-${r.maxPersonas}`;
   140→}
   141→
   142→function modoLabel(m: ModoCobro): string {
   143→  return MODO_OPTIONS.find(o => o.value === m)?.label || m;
   144→}
   145→
   146→function modoBadgeColor(m: ModoCobro): string {
   147→  switch (m) {
   148→    case 'porPersona': return 'bg-[#F5F3FF] text-[#6D28D9] border-0';
   149→    case 'porHabitacion': return 'bg-[#FEF3C7] text-[#92400E] border-0';
   150→    case 'porCama': return 'bg-[#DCFCE7] text-[#166534] border-0';
   151→    default: return 'bg-[#DBEAFE] text-[#1E40AF] border-0';
   152→  }
   153→}

   155→function crearRangoDefault(modo: ModoCobro, minPersonas: number = 1): RangoPrecio {
   156→  if (modo === 'porHabitacion') {
   157→    return { minPersonas: 1, maxPersonas: null, precio: 0 };
   158→  }
   159→  return { minPersonas, maxPersonas: minPersonas, precio: 0 };
