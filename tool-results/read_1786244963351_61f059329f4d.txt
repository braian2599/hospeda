   103→function DateRangePickerInline({
   104→  checkin,
   105→  checkout,
   106→  onChangeCheckin,
   107→  onChangeCheckout,
   108→  label,
   109→  disabled,
   110→}: {
   111→  checkin: string;
   112→  checkout: string;
   113→  onChangeCheckin: (val: string) => void;
   114→  onChangeCheckout: (val: string) => void;
   115→  label?: string;
   116→  disabled?: boolean;
   117→}) {
   118→  const [open, setOpen] = useState(false);
   119→  const [localRange, setLocalRange] = useState<{ from?: Date; to?: Date }>({});
   120→  const syncedRef = useRef(false);
   121→
   122→  // Sync local state with the form when the popover opens (event-driven, not effect-driven).
   123→  const handleOpenChange = (next: boolean) => {
   124→    setOpen(next);
   125→    if (next && !syncedRef.current) {
   126→      syncedRef.current = true;
   127→      const f = checkin ? new Date(checkin + 'T12:00:00') : undefined;
   128→      const t = checkout ? new Date(checkout + 'T12:00:00') : undefined;
   129→      setLocalRange(f ? { from: f, to: t } : {});
   130→    }
   131→    if (!next) syncedRef.current = false;
   132→  };
   133→
   134→  const handleSelect = (range: { from?: Date; to?: Date } | undefined) => {
   135→    if (!range?.from) {
   136→      setLocalRange({});
   137→      return;
   138→    }
   139→    setLocalRange(range);
   140→  };
   141→
   142→  const fmt = (d: Date) => {
   143→    const yyyy = d.getFullYear();
   144→    const mm = String(d.getMonth() + 1).padStart(2, '0');
   145→    const dd = String(d.getDate()).padStart(2, '0');
   146→    return `${yyyy}-${mm}-${dd}`;
   147→  };
   148→
   149→  const handleConfirm = () => {
   150→    if (!localRange.from) return;
   151→    onChangeCheckin(fmt(localRange.from));
   152→    onChangeCheckout(localRange.to ? fmt(localRange.to) : '');
   153→    handleOpenChange(false);
   154→  };
   155→
   156→  const from = checkin ? new Date(checkin + 'T12:00:00') : undefined;
   157→  const to = checkout ? new Date(checkout + 'T12:00:00') : undefined;
   158→
   159→  const displayText = from && to
   160→    ? `${format(from, 'dd/MM/yyyy')} → ${format(to, 'dd/MM/yyyy')}`
   161→    : from
   162→      ? `${format(from, 'dd/MM/yyyy')} → ...`
   163→      : 'Seleccionar fechas';
   164→
   165→  return (
   166→    <div className="grid gap-1.5">
   167→      {label && <Label>{label} *</Label>}
   168→      <Button
   169→        variant="outline"
   170→        className={cn(
   171→          'w-full justify-start text-left font-normal h-9',
   172→          !checkin && 'text-muted-foreground'
   173→        )}
   174→        disabled={disabled}
   175→        onClick={() => handleOpenChange(true)}
   176→      >
   177→        <CalendarDays className="w-4 h-4 mr-2 text-muted-foreground" />
   178→        {displayText}
   179→      </Button>
   180→      {open
   181→        ? createPortal(
   182→            <div className="fixed inset-0 z-[200] flex items-center justify-center bg-black/40" onClick={() => handleOpenChange(false)}>
   183→              <div
   184→                className="bg-white rounded-xl shadow-2xl border border-[#E2E8F0] pointer-events-auto"
   185→                onClick={e => e.stopPropagation()}
   186→                onMouseDown={e => e.stopPropagation()}
   187→              >
   188→                <Calendar
   189→                  mode="range"
   190→                  selected={localRange.from ? { from: localRange.from, to: localRange.to } as { from: Date; to: Date } : undefined}
   191→                  onSelect={handleSelect}
   192→                  numberOfMonths={2}
   193→                  autoFocus
   194→                  classNames={{
   195→                    range_start: 'rounded-l-md bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   196→                    range_end: 'rounded-r-md bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   197→                    range_middle: 'rounded-none bg-[#0D3D33] [&>button]:!bg-[#0D3D33] [&>button]:!text-white',
   198→                    day: 'relative w-full h-full p-0 text-center [&:first-child[data-selected=true]_button]:rounded-l-md [&:last-child[data-selected=true]_button]:rounded-r-md group/day aspect-square select-none',
   199→                  }}
   200→                />
   201→                <div className="flex justify-end border-t border-[#E2E8F0] p-3">
   202→                  <Button
   203→                    size="sm"
   204→                    onClick={handleConfirm}
   205→                    disabled={!localRange.from}
   206→                    className="bg-[#059669] hover:bg-[#047857] text-white text-xs h-8"
   207→                  >
   208→                    Confirmar fechas
   209→                  </Button>
   210→                </div>
   211→              </div>
   212→            </div>,
   213→            document.body
   214→          )
   215→        : null}
   216→    </div>
   217→  );
   218→}

   241→// ==================== DESGLOSE PRECIO COMPONENT ====================
   242→function DesglosePrecio({ form, computed, formatMoney, s }: {
   243→  form: { tipoTarifa: string; reservaMultiple: boolean; habitacion: string; habitacion2?: string; personas: string; personas2?: string };
   244→  computed: { noches: number; subtotal: number; subtotal2: number; desglose: any };
   245→  formatMoney: (n: number) => string;
   246→  s: (n: number) => string;
   247→}) {
   248→  const d = computed.desglose;
   249→
   250→  const adultoLinea = (d: any) => {
   251→    if (!d || d.totalAdultos === 0) return null;
   252→    const modo = d.modoCobro;
   253→    if (modo === 'porCama') {
   254→      return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''} × ${formatMoney(d.precioRango)}/cama/noche`;
   255→    }
   256→    if (modo === 'porHabitacion') return 'Habitación';
   257→    if (modo === 'porPersona') {
   258→      return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''} × ${formatMoney(d.precioRango)}/noche`;
   259→    }
   260→    // porGrupo
   261→    return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''}`;
   262→  };
   263→
   264→  const renderLines = (sub: number, dLocal: any) => {
   265→    if (!dLocal) return (
   266→      <div className="space-y-1">
   267→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Subtotal</span><span className="font-semibold text-[#1E293B]">{formatMoney(sub)}</span></div>
   268→      </div>
   269→    );
   270→
   271→    const lines: { key: string; label: string; value: string; dim?: boolean }[] = [];
   272→
   273→    const aLabel = adultoLinea(dLocal);
   274→    if (aLabel) {
   275→      lines.push({ key: 'adultos', label: aLabel, value: formatMoney(dLocal.totalAdultos) });
   276→    }
   277→
   278→    if (dLocal.ninosCount > 0 && dLocal.totalNinos > 0) {
   279→      lines.push({
   280→        key: 'ninos',
   281→        label: `${dLocal.ninosCount} niño${dLocal.ninosCount > 1 ? 's' : ''} × ${formatMoney(dLocal.precioNino)}/noche`,
   282→        value: formatMoney(dLocal.totalNinos),
   283→      });
   284→    }
   285→
   286→    if (dLocal.nochesGratis > 0) {
   287→      const totalBase = dLocal.totalAdultos + dLocal.totalNinos;
   288→      const ahorro = totalBase > 0 && dLocal.nochesCobrables > 0
   289→        ? Math.round(totalBase / dLocal.nochesCobrables * dLocal.nochesGratis)
   290→        : 0;
   291→      lines.push({
   292→        key: 'cortesia',
   293→        label: `${dLocal.nochesGratis} noche${dLocal.nochesGratis > 1 ? 's' : ''} de cortesía`,
   294→        value: `- ${formatMoney(ahorro)}`,
   295→        dim: true,
   296→      });
   297→    }
   298→
   299→    return (
   300→      <div className="space-y-1">
   301→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Tarifa</span><span className="font-semibold text-[#1E293B] capitalize">{form.tipoTarifa}</span></div>
   302→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-[#64748B]">Noches</span><span className="font-semibold text-[#1E293B]">{computed.noches}{dLocal?.nochesGratis > 0 ? ` (${dLocal.nochesCobrables} cobrables)` : ''}</span></div>
   303→        {lines.map(l => (
   304→          <div key={l.key} className="flex justify-between items-center py-1 text-[13px]">
   305→            <span className={l.dim ? 'text-[#94A3B8] line-through' : 'text-[#64748B]'}>{l.label}</span>
   306→            <span className={l.dim ? 'font-medium text-[#94A3B8]' : 'font-semibold text-[#1E293B]'}>{l.value}</span>
   307→          </div>
   308→        ))}
   309→        <div className="flex justify-between items-center py-1.5 mt-1 border-t border-[#E2E8F0] text-[13px]">
   310→          <span className="font-medium text-[#475569]">Subtotal</span>
   311→          <span className="font-bold text-[#1E293B]">{formatMoney(sub)}</span>
   312→        </div>
   313→      </div>
   314→    );
   315→  };
   316→
   317→  return (
   318→    <>
   319→      {form.reservaMultiple ? (
   320→        <>
   321→          <p className="text-[11px] font-semibold text-[#64748B] uppercase tracking-wider">Habitación {form.habitacion}</p>
   322→          {renderLines(computed.subtotal, d)}
   323→          {computed.subtotal2 > 0 && (
   324→            <>
   325→              <p className="text-[11px] font-semibold text-[#64748B] uppercase tracking-wider pt-2">Habitación {form.habitacion2}</p>
   326→              {renderLines(computed.subtotal2, null)}
   327→            </>
   328→          )}
   329→        </>
   330→      ) : (
   331→        renderLines(computed.subtotal, d)
   332→      )}
   333→    </>
   334→  );
   335→}
