    34→function escapeCSV(val: string | number | undefined): string {
    35→  const str = val == null ? '' : String(val);
    36→  if (str.includes(',') || str.includes('"') || str.includes('\n')) {
    37→    return `"${str.replace(/"/g, '""')}"`;
    38→  }
    39→  return str;
    40→}
    41→
    42→function downloadCSV(filename: string, headers: string[], rows: (string | number | undefined)[][]) {
    43→  const lines = [
    44→    headers.map(escapeCSV).join(','),
    45→    ...rows.map(r => r.map(escapeCSV).join(',')),
    46→  ];
    47→  const blob = new Blob([lines.join('\n')], { type: 'text/csv;charset=utf-8;' });
    48→  const url = URL.createObjectURL(blob);
    49→  const a = document.createElement('a');
    50→  a.href = url;
    51→  a.download = filename;
    52→  a.click();
    53→  URL.revokeObjectURL(url);
    54→}

    65→  if (!f) return '—';

    68→  return d.toLocaleDateString('es-AR') + ' ' + d.toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit' });

    81→interface KpiProps {
    82→  label: string;
    83→  value: string | number;
    84→  icon: React.ReactNode;

    92→
    93→function KpiCard({ label, value, icon, color, bgColor, tooltip, subtext, trend, onClick }: KpiProps) {
    94→  const card = (
    95→    <Card
    96→      className={`relative overflow-hidden transition-all duration-200 hover:shadow-md ${onClick ? 'cursor-pointer group' : ''}`}
    97→      onClick={onClick}
    98→    >
    99→      <CardContent className="p-3 sm:p-4">
   100→        <div className="flex items-start justify-between gap-2 sm:gap-3">
   101→          <div className="flex-1 min-w-0 text-center">
   102→            <p className="text-[10px] sm:text-[11px] font-medium uppercase tracking-wider text-muted-foreground mb-1 sm:mb-1.5">{label}</p>
   103→            <p className={`text-lg sm:text-2xl font-bold tracking-tight ${color} truncate`}>{value}</p>
   104→            {trend && (
   105→              <div className="flex items-center justify-center gap-1 mt-1 sm:mt-1.5">
   106→                {trend.value > 0 ? (
   107→                  <ArrowUpRight className="w-3 h-3 sm:w-3.5 sm:h-3.5 text-[#059669]" />
   108→                ) : trend.value < 0 ? (
   109→                  <ArrowDownRight className="w-3 h-3 sm:w-3.5 sm:h-3.5 text-[#EF4444]" />
   110→                ) : (
   111→                  <Minus className="w-3 h-3 sm:w-3.5 sm:h-3.5 text-muted-foreground" />
   112→                )}
   113→                <span className={`text-[11px] sm:text-xs font-medium ${trend.value > 0 ? 'text-[#059669]' : trend.value < 0 ? 'text-[#EF4444]' : 'text-muted-foreground'}`}>
   114→                  {Math.abs(trend.value)}%
   115→                </span>
   116→                <span className="text-[11px] sm:text-xs text-muted-foreground hidden sm:inline">{trend.label}</span>
   117→              </div>
   118→            )}
   119→            {subtext && !trend && <p className="text-[11px] sm:text-xs text-muted-foreground mt-1 truncate hidden sm:block">{subtext}</p>}
   120→          </div>
   121→          <div className={`shrink-0 w-8 h-8 sm:w-10 sm:h-10 rounded-lg sm:rounded-xl flex items-center justify-center ${bgColor}`}>
   122→            <div className="sm:scale-100 scale-90">{icon}</div>
   123→          </div>
   124→        </div>
   125→      </CardContent>
   126→      {onClick && (
   127→        <div className="absolute inset-x-0 bottom-0 h-0.5 bg-primary opacity-0 group-hover:opacity-100 transition-opacity" />
   128→      )}
   129→    </Card>
   130→  );
   131→
   132→  if (tooltip) {
   133→    return (
   134→      <TooltipProvider delayDuration={300}>
   135→        <Tooltip>
   136→          <TooltipTrigger asChild>{card}</TooltipTrigger>
   137→          <TooltipContent side="bottom" className="text-xs max-w-[220px]">{tooltip}</TooltipContent>
   138→        </Tooltip>
   139→      </TooltipProvider>
   140→    );
   141→  }
   142→  return card;
   143→}
