   107→// ==================== DATE RANGE PICKER (UNIFIED) ====================

   109→function DateRangePickerInline({
   110→  checkin,
   111→  checkout,
   112→  onChangeCheckin,
   113→  onChangeCheckout,
   114→  label,
   115→  disabled,
   116→}: {
   117→  checkin: string;
   118→  checkout: string;
   119→  onChangeCheckin: (val: string) => void;
   120→  onChangeCheckout: (val: string) => void;
   121→  label?: string;
   122→  disabled?: boolean;
   123→}) {
   124→  const [open, setOpen] = useState(false);
   125→  const [localRange, setLocalRange] = useState<{ from?: Date; to?: Date }>({});
   126→  const syncedRef = useRef(false);
   127→
   128→  // Sync local state with the form when the popover opens (event-driven, not effect-driven).
   129→  const handleOpenChange = (next: boolean) => {
   130→    setOpen(next);
   131→    if (next && !syncedRef.current) {
   132→      syncedRef.current = true;
   133→      const f = checkin ? new Date(checkin + 'T12:00:00') : undefined;
   134→      const t = checkout ? new Date(checkout + 'T12:00:00') : undefined;
   135→      setLocalRange(f ? { from: f, to: t } : {});
   136→    }
   137→    if (!next) syncedRef.current = false;
   138→  };
   139→
   140→  const handleSelect = (range: { from?: Date; to?: Date } | undefined) => {
   141→    if (!range?.from) {
   142→      setLocalRange({});
   143→      return;
   144→    }
   145→    setLocalRange(range);
   146→  };
   147→
   148→  const fmt = (d: Date) => {
   149→    const yyyy = d.getFullYear();
   150→    const mm = String(d.getMonth() + 1).padStart(2, '0');
   151→    const dd = String(d.getDate()).padStart(2, '0');
   152→    return `${yyyy}-${mm}-${dd}`;
   153→  };
   154→
   155→  const handleConfirm = () => {
   156→    if (!localRange.from) return;
   157→    onChangeCheckin(fmt(localRange.from));
   158→    onChangeCheckout(localRange.to ? fmt(localRange.to) : '');
   159→    handleOpenChange(false);
   160→  };
   161→
   162→  const from = checkin ? new Date(checkin + 'T12:00:00') : undefined;
   163→  const to = checkout ? new Date(checkout + 'T12:00:00') : undefined;
   164→
   165→  const displayText = from && to
   166→    ? `${format(from, 'dd/MM/yyyy')} → ${format(to, 'dd/MM/yyyy')}`
   167→    : from
   168→      ? `${format(from, 'dd/MM/yyyy')} → ...`
   169→      : 'Seleccionar fechas';
   170→
   171→  return (
   172→    <div className="grid gap-1.5">
   173→      {label && <Label>{label} *</Label>}
   174→      <Button
   175→        variant="outline"
   176→        className={cn(
   177→          'w-full justify-start text-left font-normal h-9',
   178→          !checkin && 'text-muted-foreground'
   179→        )}
   180→        disabled={disabled}
   181→        onClick={() => handleOpenChange(true)}
   182→      >
   183→        <CalendarDays className="w-4 h-4 mr-2 text-muted-foreground" />
   184→        {displayText}
   185→      </Button>
   186→      {open
   187→        ? createPortal(
   188→            <div className="fixed inset-0 z-[200] flex items-center justify-center bg-black/40" onClick={() => handleOpenChange(false)}>
   189→              <div
   190→                className="bg-card rounded-xl shadow-2xl border border-border pointer-events-auto"
   191→                onClick={e => e.stopPropagation()}
   192→                onMouseDown={e => e.stopPropagation()}
   193→              >
   194→                <Calendar
   195→                  mode="range"
   196→                  selected={localRange.from ? { from: localRange.from, to: localRange.to } as { from: Date; to: Date } : undefined}
   197→                  onSelect={handleSelect}
   198→                  numberOfMonths={2}
   199→                  autoFocus
   200→                  classNames={{
   201→                    range_start: 'rounded-l-md bg-primary [&>button]:!bg-primary [&>button]:!text-white',
   202→                    range_end: 'rounded-r-md bg-primary [&>button]:!bg-primary [&>button]:!text-white',
   203→                    range_middle: 'rounded-none bg-primary [&>button]:!bg-primary [&>button]:!text-white',
   204→                    day: 'relative w-full h-full p-0 text-center [&:first-child[data-selected=true]_button]:rounded-l-md [&:last-child[data-selected=true]_button]:rounded-r-md group/day aspect-square select-none',
   205→                  }}
   206→                />
   207→                <div className="flex justify-end border-t border-border p-3">
   208→                  <Button
   209→                    size="sm"
   210→                    onClick={handleConfirm}
   211→                    disabled={!localRange.from}
   212→                    className="bg-primary hover:bg-primary/80 text-white text-xs h-8"
   213→                  >
   214→                    Confirmar fechas
   215→                  </Button>
   216→                </div>
   217→              </div>
   218→            </div>,
   219→            document.body
   220→          )
   221→        : null}
   222→    </div>
   223→  );
   224→}

   247→// ==================== DESGLOSE PRECIO COMPONENT ====================
   248→function DesglosePrecio({ form, computed, formatMoney, s }: {
   249→  form: { tipoTarifa: string; reservaMultiple: boolean; habitacion: string; habitacion2?: string; personas: string; personas2?: string };
   250→  computed: { noches: number; subtotal: number; subtotal2: number; desglose: any };
   251→  formatMoney: (n: number) => string;
   252→  s: (n: number) => string;
   253→}) {
   254→  const d = computed.desglose;
   255→
   256→  const adultoLinea = (d: any) => {
   257→    if (!d || d.totalAdultos === 0) return null;
   258→    const modo = d.modoCobro;
   259→    if (modo === 'porCama') {
   260→      return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''} × ${formatMoney(d.precioRango)}/cama/noche`;
   261→    }
   262→    if (modo === 'porHabitacion') return 'Habitación';
   263→    if (modo === 'porPersona') {
   264→      return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''} × ${formatMoney(d.precioRango)}/noche`;
   265→    }
   266→    // porGrupo
   267→    return `${d.personasACobrar} adulto${d.personasACobrar > 1 ? 's' : ''}`;
   268→  };
   269→
   270→  const renderLines = (sub: number, dLocal: any) => {
   271→    if (!dLocal) return (
   272→      <div className="space-y-1">
   273→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-muted-foreground">Subtotal</span><span className="font-semibold text-foreground">{formatMoney(sub)}</span></div>
   274→      </div>
   275→    );
   276→
   277→    const lines: { key: string; label: string; value: string; dim?: boolean }[] = [];
   278→
   279→    const aLabel = adultoLinea(dLocal);
   280→    if (aLabel) {
   281→      lines.push({ key: 'adultos', label: aLabel, value: formatMoney(dLocal.totalAdultos) });
   282→    }
   283→
   284→    if (dLocal.ninosCount > 0 && dLocal.totalNinos > 0) {
   285→      lines.push({
   286→        key: 'ninos',
   287→        label: `${dLocal.ninosCount} niño${dLocal.ninosCount > 1 ? 's' : ''} × ${formatMoney(dLocal.precioNino)}/noche`,
   288→        value: formatMoney(dLocal.totalNinos),
   289→      });
   290→    }
   291→
   292→    if (dLocal.nochesGratis > 0) {
   293→      const totalBase = dLocal.totalAdultos + dLocal.totalNinos;
   294→      const ahorro = totalBase > 0 && dLocal.nochesCobrables > 0
   295→        ? Math.round(totalBase / dLocal.nochesCobrables * dLocal.nochesGratis)
   296→        : 0;
   297→      lines.push({
   298→        key: 'cortesia',
   299→        label: `${dLocal.nochesGratis} noche${dLocal.nochesGratis > 1 ? 's' : ''} de cortesía`,
   300→        value: `- ${formatMoney(ahorro)}`,
   301→        dim: true,
   302→      });
   303→    }
   304→
   305→    return (
   306→      <div className="space-y-1">
   307→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-muted-foreground">Tarifa</span><span className="font-semibold text-foreground capitalize">{form.tipoTarifa}</span></div>
   308→        <div className="flex justify-between items-center py-1 text-[13px]"><span className="text-muted-foreground">Noches</span><span className="font-semibold text-foreground">{computed.noches}{dLocal?.nochesGratis > 0 ? ` (${dLocal.nochesCobrables} cobrables)` : ''}</span></div>
   309→        {lines.map(l => (
   310→          <div key={l.key} className="flex justify-between items-center py-1 text-[13px]">
   311→            <span className={l.dim ? 'text-muted-foreground line-through' : 'text-muted-foreground'}>{l.label}</span>
   312→            <span className={l.dim ? 'font-medium text-muted-foreground' : 'font-semibold text-foreground'}>{l.value}</span>
   313→          </div>
   314→        ))}
   315→        <div className="flex justify-between items-center py-1.5 mt-1 border-t border-border text-[13px]">
   316→          <span className="font-medium text-muted-foreground">Subtotal</span>
   317→          <span className="font-bold text-foreground">{formatMoney(sub)}</span>
   318→        </div>
   319→      </div>
   320→    );
   321→  };
   322→
   323→  return (
   324→    <>
   325→      {form.reservaMultiple ? (
   326→        <>
   327→          <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wider">Habitación {form.habitacion}</p>
   328→          {renderLines(computed.subtotal, d)}
   329→          {computed.subtotal2 > 0 && (
   330→            <>
   331→              <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wider pt-2">Habitación {form.habitacion2}</p>
   332→              {renderLines(computed.subtotal2, null)}
   333→            </>
   334→          )}
   335→        </>
   336→      ) : (
   337→        renderLines(computed.subtotal, d)
   338→      )}
   339→    </>
   340→  );
   341→}
