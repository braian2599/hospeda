   139→   139→function WeatherIcon({ code }: { code: number }) {
   140→   140→  if (code === 0) return <Sun className="w-5 h-5 text-brand-amber" />;

   144→   144→  if (code >= 56 && code <= 57) return <CloudDrizzle className="w-5 h-5 text-info" />;
   145→   145→  if (code >= 61 && code <= 67) return <CloudRain className="w-5 h-5 text-info" />;
   146→   146→  if (code >= 71 && code <= 77) return <CloudSnow className="w-5 h-5 text-info" />;
   147→   147→  if (code >= 80 && code <= 82) return <CloudLightning className="w-5 h-5 text-amber-600" />;
   148→   148→  if (code >= 95) return <CloudLightning className="w-5 h-5 text-violet-600" />;

   643→   643→function getBarColorClass(tipo: string): string {
   644→   644→  const map: Record<string, string> = {
   645→   645→    Reservada: 'bg-status-reserved shadow-sm',
   646→   646→    Ocupada: 'bg-status-available shadow-sm',

   648→   648→    Limpieza: 'bg-status-cleaning shadow-sm',
   649→   649→    Mantenimiento: 'bg-status-maintenance shadow-sm',

   654→   654→function calcularBarra(res: GanttReserva, colIdx: Record<string, number>, columnas: string[], DIAS: number) {
   655→   655→  const COL_PCT = 100 / DIAS;
   656→   656→  const MITAD_COL_PCT = COL_PCT / 2;
   657→   657→  let startCol = Math.max(colIdx[res.checkin] ?? 0, 0);
   658→   658→  let endCol = Math.min(colIdx[res.checkout] !== undefined ? colIdx[res.checkout] : DIAS - 1, DIAS - 1);
   659→   659→
   660→   660→  const esCheckinTarde = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckin && res.horaCheckin.getHours() >= 12;
   661→   661→  const esCheckoutManana = res.tipo !== 'Limpieza' && res.tipo !== 'Mantenimiento' && res.horaCheckout && res.horaCheckout.getHours() < 12;
   662→   662→
   663→   663→  let leftPct: number, widthPct: number;
   664→   664→
   665→   665→  if (esCheckinTarde && startCol >= 0 && !(res.checkin < columnas[0])) {
   666→   666→    leftPct = startCol * COL_PCT + MITAD_COL_PCT;
   667→   667→  } else {
   668→   668→    leftPct = startCol * COL_PCT;
   669→   669→  }
   670→   670→
   671→   671→  if (esCheckoutManana && endCol >= 0 && !(res.checkout > columnas[DIAS - 1])) {
   672→   672→    widthPct = ((endCol - startCol) * COL_PCT) + (esCheckinTarde ? MITAD_COL_PCT : COL_PCT) - (esCheckoutManana ? MITAD_COL_PCT : 0);
   673→   673→  } else {
   674→   674→    widthPct = (endCol - startCol + 1) * COL_PCT;
   675→   675→  }
   676→   676→
   677→   677→  widthPct = Math.max(widthPct, MITAD_COL_PCT);
   678→   678→  return { leftPct, widthPct };
   679→   679→}
